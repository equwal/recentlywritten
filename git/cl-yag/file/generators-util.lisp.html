30 (defun load-file(path)
31   "Load a file as a string. We escape ~ to avoid failures with format."
32   (if (probe-file path)
33       (handler-case (with-open-file (stream path :direction :input)
34                       (let ((contents (make-string (file-length stream))))
35                         (read-sequence contents stream)
36                         contents))
37         (file-error (condition)
38           (cerror "ERROR : file ~a not found. Aborting~%" condition)))
39     ))

41 (defun save-file(path data)
42   "Save a string to a file."
43   (ensure-directories-exist path)
44   (with-open-file (stream path :direction :output :if-exists :supersede :if-does-not-exist :create)
45 		  (write-sequence data stream)))
