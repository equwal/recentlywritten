46 (defun get-day-of-week(day month year)
47   "Return the day of the week."
48   (multiple-value-bind
49    (second minute hour date month year day-of-week dst-p tz)
50    (decode-universal-time (encode-universal-time 0 0 0 day month year))
51    (declare (ignore second minute hour date month year dst-p tz))
52    day-of-week))

54 (defun date-parse(date)
55   "Parse a date into components."
56   (if (= 8 (length date))
57       (let* ((year     (parse-integer date :start 0 :end 4))
58              (monthnum (parse-integer date :start 4 :end 6))
59              (daynum   (parse-integer date :start 6 :end 8))
60              (day      (nth (get-day-of-week daynum monthnum year) *days*))
61              (month    (nth (- monthnum 1) *months*)))
62         (list
63          :dayname day
64          :daynumber daynum
65          :monthname month
66          :monthnumber monthnum
67          :year year))
68     nil))
