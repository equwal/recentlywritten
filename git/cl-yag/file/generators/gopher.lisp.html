12 (defun generate-gopher-index(articles)
13   (let ((output (load-file "templates/gopher_head.tpl")))
14     (dolist (article articles)
15       (setf output
16 	    (string
17 	     (concatenate 'string output
18                           (format nil (getf *config* :gopher-format)
19                                   0 ;;;; gopher type, 0 for text files
20 				  ;; here we create a 80 width char string with title on the left
21 				  ;; and date on the right
22 				  ;; we truncate the article title if it's too large
23 				  (let ((title (format nil "~80a"
24 						       (if (< 80 (length (article-title article)))
25 							   (subseq (article-title article) 0 80)
26 							   (article-title article)))))
27 				    (replace title (article-rawdate article) :start1 (- (length title) (length (article-rawdate article)))))
28 				  (concatenate 'string
29                                                (getf *config* :gopher-path) "/article-" (article-id article) ".txt")
30 				  (getf *config* :gopher-server)
31 				  (getf *config* :gopher-port)
32 				  )))))
33     output))
