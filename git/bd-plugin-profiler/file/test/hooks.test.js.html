143     // The same loop shape as BDFDB.PatchUtils.patch: one BetterDiscord patch calls all plugin callbacks.
144     const callbacks = [
145         Object.assign(() => { clock.t += 9; }, {pluginName: "spellcheck"}),
146         Object.assign(() => { clock.t += 4; throw new Error("plugin bug"); }, {pluginName: "Translator"})
147     ];
148     const dispatcher = () => {
149         clock.t += 1;
150         for (const callback of callbacks) bdfdb.TimeUtils.suppress(callback, "after callback", {name: callback.pluginName})({});
151         bdfdb.TimeUtils.suppress(() => { clock.t += 50; }, "originalMethod", {name: "Discord"})();
152     };
153     recorder.run("BDFDB", "patch after", dispatcher, null, []);
154     assert.equal(entry("SpellCheck").ms, 9);
155     assert.equal(entry("Translator").ms, 4);

157     assert.equal(recorder.unattributedMs, 50);
158     assert.equal(logged.length, 1, "BDFDB still logs the plugin error");

161     assert.equal(plain[ORIGINAL], undefined);
