18 /** Runs a call tree through the recorder. The clock moves only when a node does work. */
19 function execute(recorder, clock, node) {
20     return recorder.run(node.owner, "test", () => {
21         clock.t += node.before;
22         for (const child of node.children) execute(recorder, clock, child);
23         if (node.throws) throw new Error("planned failure");
24         clock.t += node.after;
25         return node;
26     }, null, []);
27 }

29 /** Slow reference: adds up the work of each owner directly. Returns true when the node throws. */
30 function reference(node, totals) {
31     const key = String(node.owner);
32     totals[key] = totals[key] || {ms: 0, calls: 0};
33     totals[key].calls += 1;
34     totals[key].ms += node.before;
35     for (const child of node.children) {
36         if (reference(child, totals)) return true;
37     }
38     if (node.throws) return true;
39     totals[key].ms += node.after;
40     return false;
41 }
