11 function readAsarFile(archive, name) {
12     const buffer = fs.readFileSync(archive);
13     const headerPickleSize = buffer.readUInt32LE(4);
14     const jsonLength = buffer.readUInt32LE(12);
15     const header = JSON.parse(buffer.toString("utf8", 16, 16 + jsonLength));
16     const entry = header.files[name];
17     const start = 8 + headerPickleSize + Number(entry.offset);
18     return buffer.toString("utf8", start, start + entry.size);
19 }

21 /** Returns the index of the brace that closes the brace at openIndex. */
22 function closingBrace(text, openIndex) {
23     let depth = 0;
24     for (let i = openIndex; i < text.length; i++) {
25         if (text[i] === "{") depth++;
26         else if (text[i] === "}" && --depth === 0) return i;
27     }
28     throw new Error("no closing brace");
29 }

31 /** Returns the internal Patcher class of the installed BetterDiscord. */
32 function loadBdPatcher(logger) {
33     const source = readAsarFile(BD_ASAR, "betterdiscord.js");
34     const marker = source.indexOf("static getPatchesByCaller(");
35     if (marker < 0) throw new Error("BetterDiscord Patcher not found");
36     const classStart = source.lastIndexOf("=class{", marker) + 1;
37     const body = source.slice(classStart, closingBrace(source, classStart + "class".length) + 1);
38     // The minified class uses g (logger), w (module cache), and ae (module search) from its bundle.
39     return new Function("g", "w", "ae", `return ${body};`)(logger, {}, () => null);
40 }
