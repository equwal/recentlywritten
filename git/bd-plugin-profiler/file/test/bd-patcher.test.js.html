7 function setup() {
8     const errors = [];
9     const Patcher = loadBdPatcher({err: (...args) => errors.push(args)});
10     const clock = {t: 0};
11     const recorder = new Recorder(() => clock.t);
12     const msByOwner = () => Object.fromEntries([...recorder.owners].map(([name, entry]) => [name, entry.ms]));
13     const wrapAll = callers => {
14         const children = callers.flatMap(caller => Patcher.getPatchesByCaller(caller));
15         for (const child of children) assert.equal(wrapPatchChild(recorder, child, child.caller), true);
16         return children;
17     };
18     return {Patcher, errors, clock, recorder, msByOwner, wrapAll};
19 }

21 test("patch time goes to the plugin, the original function time does not, and results stay the same", () => {

67 test("a patch that calls a patched function pays for the Discord work but not for the other plugin", () => {
68     const {Patcher, clock, msByOwner, wrapAll} = setup();
69     const inner = {step() { clock.t += 10; }};
