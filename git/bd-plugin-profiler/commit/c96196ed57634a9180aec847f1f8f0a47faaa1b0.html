Add PluginProfiler BetterDiscord plugin

The plugin measures the main-thread time of each BetterDiscord plugin.
It wraps BetterDiscord patch callbacks, the plugin callbacks that BDFDB
runs, the plugin hook methods, and the timers and observers that
plugins make. It also shows the CPU of all Discord processes, the
renderer JS heap, and long animation frames.

Property tests check the time accounting. Other tests check the hooks
against the patcher of the installed BetterDiscord and the suppress
function of the installed BDFDB.

+/**
+ * @name PluginProfiler
+ * @description Measures the main-thread time of each plugin and shows the CPU and memory use of Discord. Open the plugin settings to see the report.
+ * @version 1.0.0
+ */
+
+"use strict";
+
+const NAME = "PluginProfiler";
+const VERSION = "1.0.0";
+// BetterDiscord gives each plugin file this source URL. Old versions do not have "betterdiscord/".
+const PLUGIN_URL = /^betterdiscord:\/\/(?:betterdiscord\/)?plugins\/(.+)$/;
+// A timer that a library makes for a plugin belongs to that plugin, not to the library.
+const LIBRARY_FILES = new Set(["0BDFDB.plugin.js", "0PluginLibrary.plugin.js", "1XenoLib.plugin.js"]);
+// BetterDiscord and BDFDB call these plugin methods by name, so a wrapper on the instance sees each call.
+const INSTANCE_HOOKS = /^(observer|onSwitch|process[A-Z]\w*)$/;
+const OBSERVERS = [
+    ["MutationObserver", "mutation observer"],
+    ["ResizeObserver", "resize observer"],
+    ["IntersectionObserver", "intersection observer"]
+];
+const ORIGINAL = Symbol("PluginProfiler.original");
+const SAMPLE_MS = 1000;
+const RESCAN_EVERY = 5;
+const MAX_SCRIPTS = 500;
+const NOTE = "The table counts synchronous work in patches, plugin hooks, timers, and observers. "
+    + "Plugin time is wall time on the main thread. It includes waits for disk and for paged-out memory, so it can be more than the CPU time. "
+    + "It does not count event listeners, async work, CSS, or memory per plugin. "
+    + "Reload Discord (Ctrl+R) to also count the timers that plugins make when they start.";

+/** Adds up the self time of each owner. The time of a nested call goes to the nested owner only. */
+class Recorder {
+    constructor(now) {
+        this.now = now;
+        this.active = true;
+        this.reset();
+    }
+
+    reset() {
+        this.startedAt = this.now();
+        this.owners = new Map();
+        this.unattributedMs = 0;
+        this.stack = [];
+    }
+
+    /** Calls fn and adds its self time to owner. A null owner marks time that no plugin owns. */
+    run(owner, kind, fn, thisArg, args) {
+        if (!this.active) return Reflect.apply(fn, thisArg, args);
+        const stack = this.stack;
+        const frame = {start: this.now(), child: 0};
+        stack.push(frame);
+        try {
+            return Reflect.apply(fn, thisArg, args);
+        }
+        finally {
+            const total = this.now() - frame.start;
+            stack.pop();
+            if (stack.length) stack[stack.length - 1].child += total;
+            const self = total - frame.child;
+            if (owner === null) this.unattributedMs += self;
+            else this.add(owner, kind, self);
+        }
+    }
+
+    add(owner, kind, ms) {
+        let entry = this.owners.get(owner);
+        if (!entry) this.owners.set(owner, entry = {ms: 0, calls: 0, maxMs: 0, kinds: new Map()});
+        entry.ms += ms;
+        entry.calls += 1;
+        if (ms > entry.maxMs) entry.maxMs = ms;
+        entry.kinds.set(kind, (entry.kinds.get(kind) || 0) + ms);
+    }
+
+    /** Returns one row for each owner. The row with the most CPU time is first. */
+    rows() {
+        const elapsed = Math.max(1, this.now() - this.startedAt);
+        return [...this.owners].map(([name, entry]) => ({
+            name,
+            ms: entry.ms,
+            calls: entry.calls,
+            maxMs: entry.maxMs,
+            percent: entry.ms / elapsed * 100,
+            topKind: [...entry.kinds].sort((a, b) => b[1] - a[1])[0][0]
+        })).sort((a, b) => b.ms - a.ms);
+    }
+}

+function timed(recorder, owner, kind, fn) {
+    const wrapper = function (...args) {
+        return recorder.run(owner, kind, fn, this, args);
+    };
+    wrapper[ORIGINAL] = fn;
+    return wrapper;
+}

+/** Returns the file of each plugin frame on the stack, innermost first. */
+function pluginFilesOnStack() {
+    const prepare = Error.prepareStackTrace;
+    const limit = Error.stackTraceLimit;
+    const holder = {};
+    let sites;
+    try {
+        // Call sites cost less than a formatted stack string.
+        Error.prepareStackTrace = (_, callSites) => callSites;
+        Error.stackTraceLimit = 40;
+        Error.captureStackTrace(holder, pluginFilesOnStack);
+        sites = holder.stack;
+    }
+    finally {
+        Error.prepareStackTrace = prepare;
+        Error.stackTraceLimit = limit;
+    }
+    const files = [];
+    if (!Array.isArray(sites)) return files;
+    for (const site of sites) {
+        const match = PLUGIN_URL.exec(site.getScriptNameOrSourceURL() || "");
+        if (match) files.push(match[1]);
+    }
+    return files;
+}
+
+/** Picks the innermost plugin that is not a library. If only libraries are on the stack, picks the innermost library. */
+function ownerFile(files, ownFile) {
+    let library = null;
+    for (const file of files) {
+        if (file === ownFile) continue;
+        if (!LIBRARY_FILES.has(file)) return file;
+        if (!library) library = file;
+    }
+    return library;
+}

+function hookTimers(scope, recorder, resolveOwner, undo) {
+    for (const name of ["setTimeout", "setInterval"]) {
+        const original = scope[name];
+        if (typeof original !== "function") continue;
+        const hooked = function (handler, ...rest) {
+            if (typeof handler === "function" && recorder.active) {
+                const owner = recorder.run(NAME, "stack capture", resolveOwner, null, []);
+                if (owner) handler = timed(recorder, owner, "timer", handler);
+            }
+            return Reflect.apply(original, scope, [handler, ...rest]);
+        };
+        hooked[ORIGINAL] = original;
+        scope[name] = hooked;
+        undo.push(() => {
+            if (scope[name] === hooked) scope[name] = original;
+        });
+    }
+}
+
+function hookObservers(scope, recorder, resolveOwner, undo) {
+    for (const [name, kind] of OBSERVERS) {
+        const Original = scope[name];
+        if (typeof Original !== "function") continue;
+        const Hooked = class extends Original {
+            constructor(callback, ...rest) {
+                if (typeof callback === "function" && recorder.active) {
+                    const owner = recorder.run(NAME, "stack capture", resolveOwner, null, []);
+                    if (owner) callback = timed(recorder, owner, kind, callback);
+                }
+                super(callback, ...rest);
+            }
+        };
+        Object.defineProperty(Hooked, "name", {value: name});
+        scope[name] = Hooked;
+        undo.push(() => {
+            if (scope[name] === Hooked) scope[name] = Original;
+        });
+    }
+}
+
+/** Wraps the plugin methods that BetterDiscord and BDFDB call by name. */
+function hookInstance(recorder, instance, owner, undo) {
+    const seen = new Set();
+    for (let proto = instance; proto && proto !== Object.prototype; proto = Object.getPrototypeOf(proto)) {
+        for (const key of Object.getOwnPropertyNames(proto)) {
+            if (seen.has(key) || !INSTANCE_HOOKS.test(key)) continue;
+            seen.add(key);
+            const current = Object.getOwnPropertyDescriptor(proto, key).value;
+            if (typeof current !== "function" || current[ORIGINAL]) continue;
+            const wrapper = timed(recorder, owner, key.startsWith("process") ? "BDFDB process" : key, current);
+            const hadOwn = proto === instance;
+            try {
+                instance[key] = wrapper;
+            }
+            catch {
+                continue;
+            }
+            undo.push(() => {
+                if (instance[key] !== wrapper) return;
+                if (hadOwn) instance[key] = current;
+                else delete instance[key];
+            });
+        }
+    }
+}
+
+/** Wraps the callback of one BetterDiscord patch. BetterDiscord reads the callback on each call. */
+function wrapPatchChild(recorder, child, owner) {
+    const original = child.callback;
+    if (typeof original !== "function" || original[ORIGINAL]) return false;
+    const kind = `patch ${child.type}`;
+    const wrapper = child.type === "instead"
+        ? function (thisObject, args, next) {
+            // "next" runs the original function or the next plugin. That time is not the time of this plugin.
+            const timedNext = typeof next === "function" ? timed(recorder, null, "original", next) : next;
+            return recorder.run(owner, kind, original, this, [thisObject, args, timedNext]);
+        }
+        : function (...args) {
+            return recorder.run(owner, kind, original, this, args);
+        };
+    wrapper[ORIGINAL] = original;
+    child.callback = wrapper;
+    return true;
+}
+
+function unwrapPatchChild(child) {
+    const original = child.callback && child.callback[ORIGINAL];
+    if (original) child.callback = original;
+}
+
+/**
+ * BDFDB puts all its plugins behind one BetterDiscord patch. It calls each plugin callback
+ * through TimeUtils.suppress with the plugin name, so a hook there splits the time per plugin.
+ */
+function hookSuppress(recorder, bdfdb, canonicalName, undo) {
+    const timeUtils = bdfdb && bdfdb.TimeUtils;
+    const original = timeUtils && timeUtils.suppress;
+    if (typeof original !== "function" || original[ORIGINAL]) return false;
+    const hooked = function (callback, strings, config) {
+        const inner = Reflect.apply(original, this, [callback, strings, config]);
+        const name = config && typeof config.name === "string" ? config.name : "";
+        if (!name || typeof inner !== "function") return inner;
+        // BDFDB uses the name "Discord" when it calls the original Discord function.
+        return timed(recorder, name === "Discord" ? null : canonicalName(name), "BDFDB hook", inner);
+    };
+    hooked[ORIGINAL] = original;
+    timeUtils.suppress = hooked;
+    undo.push(() => {
+        if (timeUtils.suppress === hooked) timeUtils.suppress = original;
+    });
+    return true;
+}

+/** Returns the names that a plugin can use as the caller of its patches. */
+function callerNames(addon) {
+    const names = new Set([addon.name, addon.id]);
+    if (typeof addon.name === "string") names.add(addon.name.replace(/\s+/g, ""));
+    if (typeof addon.filename === "string") names.add(addon.filename.replace(/\.plugin\.js$/, ""));
+    try {
+        if (addon.instance && typeof addon.instance.getName === "function") names.add(addon.instance.getName());
+    }
+    catch {
+        // A broken getName does not stop the scan.
+    }
+    return [...names].filter(name => typeof name === "string" && name);
+}

+    const seconds = Math.floor(ms / 1000);
+    return `${Math.floor(seconds / 60)}m ${seconds % 60}s`;

+function percent(value, digits = 1) {
+    return `${value.toFixed(digits)}%`;

+function megabytes(bytes) {
+    return `${(bytes / 1048576).toFixed(0)} MB`;

+function summaryLines(snap) {
+    const samples = snap.samples;
+    const frames = snap.frames;
+    const lines = [`Session: ${formatDuration(snap.elapsedMs)}. All plugins: ${snap.pluginMs.toFixed(0)} ms on the main thread, ${percent(snap.pluginMs / snap.elapsedMs * 100, 2)} of the time.`];
+    if (samples.count) {
+        lines.push(`Discord CPU, all processes: now ${percent(samples.cpuNow)}, average ${percent(samples.cpuSum / samples.count)}, max ${percent(samples.cpuMax)} of one core.`);

+class PluginProfiler {
+    constructor(meta) {
+        this.meta = meta || {};
+        this.ownFile = this.meta.filename || pluginFilesOnStack()[0] || "";
+        this.recorder = null;
+    }
+
+    start() {
+        const recorder = this.recorder = new Recorder(() => performance.now());
+        this.undo = [];
+        this.callers = new Set();
+        this.fileOwners = new Map();
+        this.names = new Map();
+        this.bdfdbGlobal = null;
+        this.ticks = 0;
+        this.resetStats();
+        const resolveOwner = () => {
+            const file = ownerFile(pluginFilesOnStack(), this.ownFile);
+            return file ? this.fileOwners.get(file) || file : null;
+        };
+        try {
+            recorder.run(NAME, "setup", () => {
+                hookTimers(window, recorder, resolveOwner, this.undo);
+                hookObservers(window, recorder, resolveOwner, this.undo);
+                this.rescan();
+                this.observeFrames();
+            }, null, []);
+        }
+        catch (error) {
+            // BetterDiscord does not call stop() after a failed start(), so remove the hooks here.
+            this.stop();
+            throw error;
+        }
+        this.timer = setInterval(() => recorder.run(NAME, "sampling", () => this.tick(), null, []), SAMPLE_MS);
+    }
+
+    stop() {
+        if (!this.recorder) return;
+        clearInterval(this.timer);
+        if (this.frameObserver) this.frameObserver.disconnect();
+        this.frameObserver = null;
+        this.recorder.active = false;
+        for (const caller of this.callers) {
+            for (const child of BdApi.Patcher.getPatchesByCaller(caller)) unwrapPatchChild(child);
+        }
+        while (this.undo.length) {
+            try {
+                this.undo.pop()();
+            }
+            catch (error) {
+                console.error(NAME, "could not remove a hook", error);
+            }
+        }
+        this.recorder = null;
+    }
+
+    getSettingsPanel() {
+        return BdApi.React.createElement(Panel, {plugin: this});
+    }
+
+    resetStats() {
+        this.samples = {count: 0, cpuNow: 0, cpuSum: 0, cpuMax: 0, types: new Map(), heapNow: null, heapMax: 0, blinkKB: 0};
+        this.frames = {count: 0, totalMs: 0, blockingMs: 0, worstMs: 0, scripts: new Map()};
+    }
+
+    reset() {
+        if (!this.recorder) return;
+        this.recorder.reset();
+        this.resetStats();
+    }
+
+    tick() {
+        this.ticks += 1;
+        if (this.ticks % RESCAN_EVERY === 0) this.rescan();
+        this.sample();
+    }
+
+    /** Finds new patches, plugin instances, and a new BDFDB. Plugins can patch at any time. */
+    rescan() {
+        this.fileOwners = new Map();
+        this.names = new Map();
+        for (const addon of BdApi.Plugins.getAll()) {
+            if (!addon || addon.name === NAME) continue;
+            if (addon.filename) this.fileOwners.set(addon.filename, addon.name);
+            this.names.set(String(addon.name).toLowerCase(), addon.name);
+            for (const caller of callerNames(addon)) {
+                this.callers.add(caller);
+                for (const child of BdApi.Patcher.getPatchesByCaller(caller)) wrapPatchChild(this.recorder, child, addon.name);
+            }
+            if (addon.instance) hookInstance(this.recorder, addon.instance, addon.name, this.undo);
+        }
+        this.hookBdfdb();
+    }
+
+    hookBdfdb() {
+        const global = window.BDFDB_Global;
+        if (!global || !global.loaded || global === this.bdfdbGlobal) return;
+        this.bdfdbGlobal = global;
+        let bdfdb;
+        try {
+            // buildPlugin only makes a class and returns it with the library object.
+            bdfdb = global.PluginUtils.buildPlugin({})[1];
+        }
+        catch {
+            return;
+        }
+        hookSuppress(this.recorder, bdfdb, name => this.names.get(name.toLowerCase()) || name, this.undo);
+    }
+
+    sample() {
+        const samples = this.samples;
+        const utils = window.DiscordNative && window.DiscordNative.processUtils;
+        if (utils && typeof utils.getCurrentCPUUsagePercent === "function") {
+            // Discord adds up the CPU of all its processes. 100 means one full core.
+            const cpu = Number(utils.getCurrentCPUUsagePercent());
+            if (Number.isFinite(cpu)) {
+                samples.count += 1;
+                samples.cpuNow = cpu;
+                samples.cpuSum += cpu;
+                samples.cpuMax = Math.max(samples.cpuMax, cpu);
+                const types = typeof utils.getCpuUsageElectronProcessTypeDetails === "function" ? utils.getCpuUsageElectronProcessTypeDetails() : null;
+                for (const [type, detail] of Object.entries(types || {})) {
+                    const value = Number(detail && detail.percent_cpu);
+                    if (Number.isFinite(value)) samples.types.set(type, (samples.types.get(type) || 0) + value);
+                }
+            }
+        }
+        if (performance.memory) {
+            samples.heapNow = performance.memory.usedJSHeapSize;
+            samples.heapMax = Math.max(samples.heapMax, samples.heapNow);
+        }
+        if (utils && typeof utils.getBlinkMemoryInfo === "function") {
+            const blink = utils.getBlinkMemoryInfo();
+            if (blink && Number.isFinite(blink.allocated)) samples.blinkKB = blink.allocated;
+        }
+    }
+
+    observeFrames() {
+        const types = typeof PerformanceObserver === "function" ? PerformanceObserver.supportedEntryTypes || [] : [];
+        this.framesSupported = types.includes("long-animation-frame");
+        if (!this.framesSupported) return;
+        const recorder = this.recorder;
+        this.frameObserver = new PerformanceObserver(list => recorder.run(NAME, "frame stats", () => this.addFrames(list.getEntries()), null, []));
+        this.frameObserver.observe({type: "long-animation-frame"});
+    }
+
+    addFrames(entries) {
+        const frames = this.frames;
+        for (const entry of entries) {
+            frames.count += 1;
+            frames.totalMs += entry.duration;
+            frames.blockingMs += entry.blockingDuration || 0;
+            frames.worstMs = Math.max(frames.worstMs, entry.duration);
+            for (const script of entry.scripts || []) {
+                const label = this.scriptLabel(script);
+                if (frames.scripts.has(label) || frames.scripts.size < MAX_SCRIPTS) {
+                    frames.scripts.set(label, (frames.scripts.get(label) || 0) + script.duration);
+                }
+            }
+        }
+    }
+
+    scriptLabel(script) {
+        const url = script.sourceURL || "";
+        const match = PLUGIN_URL.exec(url);
+        const file = url.split(/[?#]/)[0].split("/").pop();
+        // A URL that is not a .js file is the page itself. It can hold a channel ID, so the label does not show it.
+        const where = match ? this.fileOwners.get(match[1]) || match[1] : /\.js$/.test(file) ? file : "page";
+        const invoker = script.sourceFunctionName || script.invoker || "";
+        const what = invoker.includes("://") ? "" : invoker;
+        return what ? `${where} (${what})` : where;
+    }
+
+    snapshot() {
+        if (!this.recorder) return null;
+        const elapsedMs = Math.max(1, this.recorder.now() - this.recorder.startedAt);
+        const rows = this.recorder.rows();
+        const pluginMs = rows.filter(row => row.name !== NAME).reduce((sum, row) => sum + row.ms, 0);
+        const scripts = [...this.frames.scripts].sort((a, b) => b[1] - a[1]).slice(0, 5);
+        return {elapsedMs, rows, pluginMs, samples: this.samples, frames: this.frames, framesSupported: this.framesSupported, scripts};

+# PluginProfiler
+
+PluginProfiler is a BetterDiscord plugin. It measures the main-thread time that each plugin uses. It also shows the CPU and memory use of Discord.
+
+## Install
+
+1. Copy `00PluginProfiler.plugin.js` into the BetterDiscord plugins folder (`%APPDATA%\BetterDiscord\plugins`).
+2. In Discord, open Settings > Plugins and turn on PluginProfiler.
+3. Press Ctrl+R. The file name starts with `00`, so the profiler starts before the other plugins. Then it also sees the timers that the other plugins make when they start.
+4. Open the settings of PluginProfiler to see the table. Click "Copy report" to copy the data as text.
+
+## What it measures
+
+- Time in BetterDiscord patch callbacks, per plugin. The time of the original Discord function does not count.
+- Time in the plugin callbacks that the BDFDB library runs, per plugin.
+- Time in the `observer`, `onSwitch`, and BDFDB `process*` methods of each plugin.
+- Time in the timers and observers that a plugin makes. The profiler finds the plugin from the call stack.
+- The CPU of all Discord processes, the renderer JS heap, and long animation frames.
+
+The time is wall time on the main thread. It includes waits for disk and for paged-out memory, so it can be more than the CPU time.
+
+The profiler does not measure event listeners, async work, CSS, or memory per plugin.
+
+## Tests
+
+Run `npm test` with Node 22 or later. The tests load the patcher of the installed BetterDiscord and `TimeUtils.suppress` of the installed BDFDB, so both must be installed.

+"use strict";
+// Loads code from the installed BetterDiscord and BDFDB, so the tests check the versions that Discord runs.
+const fs = require("node:fs");
+const path = require("node:path");
+
+const BD_FOLDER = path.join(process.env.APPDATA || "", "BetterDiscord");
+const BD_ASAR = path.join(BD_FOLDER, "data", "betterdiscord.asar");
+const BDFDB_FILE = path.join(BD_FOLDER, "plugins", "0BDFDB.plugin.js");
+
+/** Reads one top-level file from an asar archive. */
+function readAsarFile(archive, name) {
+    const buffer = fs.readFileSync(archive);
+    const headerPickleSize = buffer.readUInt32LE(4);
+    const jsonLength = buffer.readUInt32LE(12);
+    const header = JSON.parse(buffer.toString("utf8", 16, 16 + jsonLength));
+    const entry = header.files[name];
+    const start = 8 + headerPickleSize + Number(entry.offset);
+    return buffer.toString("utf8", start, start + entry.size);
+}
+
+/** Returns the index of the brace that closes the brace at openIndex. */
+function closingBrace(text, openIndex) {
+    let depth = 0;
+    for (let i = openIndex; i < text.length; i++) {
+        if (text[i] === "{") depth++;
+        else if (text[i] === "}" && --depth === 0) return i;
+    }
+    throw new Error("no closing brace");
+}
+
+/** Returns the internal Patcher class of the installed BetterDiscord. */
+function loadBdPatcher(logger) {
+    const source = readAsarFile(BD_ASAR, "betterdiscord.js");
+    const marker = source.indexOf("static getPatchesByCaller(");
+    if (marker < 0) throw new Error("BetterDiscord Patcher not found");
+    const classStart = source.lastIndexOf("=class{", marker) + 1;
+    const body = source.slice(classStart, closingBrace(source, classStart + "class".length) + 1);
+    // The minified class uses g (logger), w (module cache), and ae (module search) from its bundle.
+    return new Function("g", "w", "ae", `return ${body};`)(logger, {}, () => null);
+}
+
+/** Returns BDFDB.TimeUtils.suppress from the installed BDFDB library. */
+function loadBdfdbSuppress(logUtils) {
+    const source = fs.readFileSync(BDFDB_FILE, "utf8");
+    const match = /BDFDB\.TimeUtils\.suppress = (function \(callback, strings, config\) \{[\s\S]*?\n\t*\}\};)/.exec(source);
+    if (!match) throw new Error("BDFDB TimeUtils.suppress not found");
+    return new Function("BDFDB", `return ${match[1].slice(0, -1)}`)({LogUtils: logUtils});
+}
+
+module.exports = {BD_ASAR, BDFDB_FILE, loadBdPatcher, loadBdfdbSuppress};

+"use strict";
+const test = require("node:test");
+const assert = require("node:assert/strict");
+const fc = require("fast-check");
+const {Recorder} = require("../00PluginProfiler.plugin.js").internals;
+
+const owner = fc.constantFrom("A", "B", "C", null);
+const {node: callTree} = fc.letrec(tie => ({
+    node: fc.record({
+        owner,
+        before: fc.nat({max: 50}),
+        after: fc.nat({max: 50}),
+        throws: fc.nat({max: 9}).map(n => n === 0),
+        children: fc.oneof({depthSize: "small"}, fc.constant([]), fc.array(tie("node"), {maxLength: 3}))
+    })
+}));
+
+/** Runs a call tree through the recorder. The clock moves only when a node does work. */
+function execute(recorder, clock, node) {
+    return recorder.run(node.owner, "test", () => {
+        clock.t += node.before;
+        for (const child of node.children) execute(recorder, clock, child);
+        if (node.throws) throw new Error("planned failure");
+        clock.t += node.after;
+        return node;
+    }, null, []);
+}
+
+/** Slow reference: adds up the work of each owner directly. Returns true when the node throws. */
+function reference(node, totals) {
+    const key = String(node.owner);
+    totals[key] = totals[key] || {ms: 0, calls: 0};
+    totals[key].calls += 1;
+    totals[key].ms += node.before;
+    for (const child of node.children) {
+        if (reference(child, totals)) return true;
+    }
+    if (node.throws) return true;
+    totals[key].ms += node.after;
+    return false;
+}
+
+test("each owner gets exactly its own work, also when calls throw", () => {
+    fc.assert(fc.property(fc.array(callTree, {minLength: 1, maxLength: 4}), roots => {
+        const clock = {t: 1000};
+        const recorder = new Recorder(() => clock.t);
+        const expected = {};
+        for (const root of roots) {
+            let threw = false;
+            try {
+                assert.equal(execute(recorder, clock, root), root);
+            }
+            catch (error) {
+                assert.equal(error.message, "planned failure");
+                threw = true;
+            }
+            assert.equal(reference(root, expected), threw);
+            assert.equal(recorder.stack.length, 0);
+        }
+        const actual = {};
+        for (const [name, entry] of recorder.owners) actual[name] = {ms: entry.ms, calls: entry.calls};
+        const attributed = Object.fromEntries(Object.entries(expected).filter(([key]) => key !== "null"));
+        assert.deepEqual(actual, attributed);
+        assert.equal(recorder.unattributedMs, expected.null ? expected.null.ms : 0);
+    }), {numRuns: 500});
+});
