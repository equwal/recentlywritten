20 def setup(tmp: Path, deny: str = "") -> tuple[Path, Path, Path]:
21     home, source = tmp / "home", tmp / "src"
22     shutil.copytree(EXAMPLE, source)
23     if deny:
24         with (source / "aicr.toml").open("a", encoding="utf-8") as f:
25             f.write(f"\n[import]\ndeny = [{json.dumps(deny)}]\n")
26     machine = tmp / "machine.toml"
27     machine.write_text(
28         f'name = "box"\nstate = {json.dumps(str(tmp / "state.json"))}\n[vars]\nhome = {json.dumps(home.as_posix())}\n',
29         encoding="utf-8",
30     )
31     render(source, machine, dry=False)
32     return source, home, machine

95 def test_watch_runs_once_per_burst_and_ignores_its_own_writes(tmp_path: Path) -> None:
96     source, home, machine = setup(tmp_path)
97     runs: list[int] = []
98 
99     def run() -> None:
100         runs.append(1)
101         sync(source, machine)  # writes targets; must not trigger a second run
102 
103     def edit() -> None:
104         time.sleep(0.05)
105         for i in range(3):
106             (home / SKILL).write_text(f"burst {i}\n", encoding="utf-8")
107             time.sleep(0.01)
108 
109     t = threading.Thread(target=edit)
110     t.start()
111     watch(source, machine, run, interval=0.01, settle=0.1, rounds=80)
112     t.join()
113     assert runs == [1]
