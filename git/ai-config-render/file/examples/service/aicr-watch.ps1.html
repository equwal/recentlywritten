examples/service/aicr-watch.ps1 (706 bytes)

1 # Windows: register a logon task that runs `aicr watch` hidden and restarts it if it exits.
2 # Run once in PowerShell 7: pwsh -File aicr-watch.ps1
3 $source = Join-Path $HOME 'ai-config'
4 $cmd = "while (`$true) { aicr watch --source '$source' --remote-every 20 --exec 'sh commit-and-push.sh'; Start-Sleep 10 }"
5 $action = New-ScheduledTaskAction -Execute 'pwsh.exe' -Argument "-NoProfile -WindowStyle Hidden -Command $cmd"
6 $trigger = New-ScheduledTaskTrigger -AtLogOn -User $env:USERNAME
7 $settings = New-ScheduledTaskSettingsSet -ExecutionTimeLimit 0 -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries
8 Register-ScheduledTask -TaskName 'aicr watch' -Action $action -Trigger $trigger -Settings $settings -Force
