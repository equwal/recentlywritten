12     @Provide
13     fun bookmarks(): Arbitrary<Bookmark> {
14         val url = Arbitraries.strings().withCharRange('!', '~').ofMinLength(1).ofMaxLength(40)
15             .filter { !it.startsWith("#") }
16         val desc = Arbitraries.strings().ofMaxLength(40)
17             .filter { d -> d.none { it == '\t' || it == '\r' || it == '\n' } }
18         val tag = Arbitraries.strings().withCharRange('a', 'z').withChars('-', '0', '9')
19             .ofMinLength(1).ofMaxLength(10)
20         return Combinators.combine(url, desc, tag.list().ofMaxSize(5)).`as` { u, d, t -> Bookmark(u, d, t) }
21     }

42     fun hosts(): Arbitrary<String> {

44             .ofMinLength(1).ofMaxLength(10)
45         val tld = Arbitraries.strings().withCharRange('a', 'z').ofMinLength(2).ofMaxLength(4)
