11 class PreviewTest {
12     // The part of a web address after the scheme: a host, then a path.
13     @Provide
14     fun rests(): Arbitrary<String> {
15         val host = Arbitraries.strings().withCharRange('a', 'z').withChars('.', '-', '0', '9').ofMinLength(1).ofMaxLength(20)
16         val path = Arbitraries.strings().withCharRange('!', '~').ofMaxLength(40)
17             .map { if (it.isEmpty() || it[0] in "/?#") it else "/$it" }
18         return Combinators.combine(host, path).`as` { h, p -> h + p }
19     }
20

32     }
