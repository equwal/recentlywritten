35     /** Sign in, keep the token, then sync. [done] gets the result on the main thread. */
36     fun signIn(ctx: Context, server: String, email: String, password: String, done: (Result<Int>) -> Unit) {
37         val app = ctx.applicationContext
38         background(done) {
39             val site = address(server)
40             val form = "email=" + URLEncoder.encode(email.trim(), "UTF-8") +
41                 "&password=" + URLEncoder.encode(password, "UTF-8")
42             val r = post(app, "$site/api/login", form.toByteArray(), "application/x-www-form-urlencoded", null)
43             when (r.code) {
44                 200 -> {}
45                 401 -> throw Failure(app.getString(R.string.wrong_login))
46                 else -> throw Failure(r.text.trim())
47             }
48             prefs(app).edit().clear()
49                 .putString("server", site).putString("email", email.trim()).putString("token", r.text.trim())
50                 .apply()
51             syncNow(app)
52             count(app)
53         }
54     }

78     private fun syncNow(ctx: Context): Boolean {
79         val p = prefs(ctx)
80         val token = p.getString("token", null) ?: throw Failure(ctx.getString(R.string.sign_in_again), true)
81         val file = Store.file(ctx)?.toString() ?: throw Failure(ctx.getString(R.string.choose_first))
82         repeat(3) {
83             // The version belongs to the file that it came from.
84             val base = if (p.getString("file", null) == file) p.getString("version", null).orEmpty() else ""
85             val sent = Store.read(ctx)
86             val r = post(ctx, server(ctx) + "/api/sync?base=" + base, sent.toByteArray(Charsets.UTF_8),
87                 "text/plain; charset=utf-8", token)
88             when (r.code) {
89                 200 -> {}
90                 401 -> {
91                     p.edit().remove("token").apply()
92                     throw Failure(ctx.getString(R.string.sign_in_again), true)
93                 }
94                 402 -> throw Failure(ctx.getString(R.string.sync_paused), urgent = true)
95                 else -> throw Failure(r.text.trim())
96             }
97             val version = r.version ?: throw Failure(ctx.getString(R.string.no_version))
98             // A bookmark came in while the request ran: send the file again.
99             if (Store.read(ctx) != sent) return@repeat
100             val changed = r.text != sent
101             if (changed) Store.write(ctx, r.text)
102             p.edit().putString("file", file).putString("version", version).apply()
103             return changed
104         }
105         throw Failure(ctx.getString(R.string.file_busy))
106     }

116     private fun post(ctx: Context, url: String, body: ByteArray, type: String, token: String?): Reply {
117         val c = try {
118             URL(url).openConnection() as HttpURLConnection
119         } catch (e: Exception) {
120             throw Failure(ctx.getString(R.string.bad_server, url))
121         }
122         try {
123             c.requestMethod = "POST"
124             c.connectTimeout = 15_000
125             c.readTimeout = 60_000
126             c.doOutput = true
127             c.setRequestProperty("Content-Type", type)
128             if (token != null) c.setRequestProperty("Authorization", "Bearer $token")
129             c.setFixedLengthStreamingMode(body.size)
130             c.outputStream.use { it.write(body) }
131             val code = c.responseCode
132             val text = (if (code < 400) c.inputStream else c.errorStream)
133                 ?.use { it.readBytes().toString(Charsets.UTF_8) }.orEmpty()
134             return Reply(code, text, c.getHeaderField("Sbm-Version"))
135         } catch (e: Failure) {
136             throw e
137         } catch (e: IOException) {
138             throw Failure(ctx.getString(R.string.no_server, URL(url).host))
139         } finally {
140             c.disconnect()
141         }
142     }

144     private fun <T> background(done: (Result<T>) -> Unit, work: () -> T) {
145         worker.execute {
146             val result = try {
147                 Result.success(work())
148             } catch (e: Exception) {
149                 Result.failure(e)
150             }
151             main.post { done(result) }
152         }
