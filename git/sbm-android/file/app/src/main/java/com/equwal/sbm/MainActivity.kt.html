119     /** Sync, then show the file again. A quiet sync reports only what the user must act on. */
120     private fun sync(quiet: Boolean) {
121         Sync.run(this) { result ->
122             if (isDestroyed) return@run
123             result.onSuccess { changed ->
124                 if (changed) load()
125                 if (!quiet) toast(resources.getQuantityString(R.plurals.synced, all.size, all.size))
126             }.onFailure { e ->
127                 if (!quiet || (e is Sync.Failure && e.urgent)) toast(getString(R.string.sync_failed, e.message))
128                 if (e is Sync.Failure && e.signIn) invalidateOptionsMenu()
129             }
130         }
131     }

158     private fun load() {
159         all = try {
160             // Newest first: bm adds bookmarks at the end of the file.
161             Tsv.parse(Store.read(this)).asReversed()
162         } catch (e: Exception) {
163             toast(getString(R.string.cannot_read, e.message))
164             emptyList()
165         }
166         empty.setText(if (Store.file(this) == null) R.string.no_file else R.string.no_match)
167         show()
168     }

170     private fun show() {
171         rows.items = Fuzzy.filter(all, search.text.toString()) { "${it.desc} ${it.tags.joinToString(" ")} ${it.url}" }
172         rows.notifyDataSetChanged()
173         showPage(picked ?: if (search.text.isBlank()) null else rows.items.firstOrNull())
174     }

179     /**
180      * Show the page of [b] in the preview, or close the preview for null. The
181      * page loads only when [b] stays in the preview for a moment, so that
182      * fast typing does not load a page for each letter.
183      */
184     private fun showPage(b: Bookmark?) {
185         later.removeCallbacksAndMessages(null)
186         previewed = if (previewOn) b else null
187         val shown = previewed
188         if (shown == null) {
189             preview.visibility = View.GONE
190             loadPage(null)
191             return
192         }
193         preview.visibility = View.VISIBLE
194         val to = Preview.address(shown.url)
195         caption.text = to ?: getString(R.string.no_preview, shown.url)
196         if (to == null) loadPage(null) else later.postDelayed({ loadPage(to) }, PREVIEW_WAIT)
197     }
198 
199     /** Load the page at [to] in the preview, or an empty page for null. */
200     private fun loadPage(to: String?) {
201         if (to == framed) return
202         framed = to
203         val view = page ?: if (to == null) return else newPage()
204         view.loadUrl(to ?: "about:blank")
205     }

207     /**
208      * A web view for the preview, under the caption. The page in it is only
209      * to look at: it runs its scripts, as in a browser, but it cannot take
210      * the focus from the search, open other apps or read the files of the
211      * app. Its links stay in the preview, and an http link loads with https.
212      */
213     @SuppressLint("SetJavaScriptEnabled") // Most pages need scripts to show, and the page gets no interface to the app.
214     private fun newPage(): WebView {
215         val view = WebView(this)
216         view.isFocusable = false
217         view.isFocusableInTouchMode = false
218         view.settings.javaScriptEnabled = true
219         view.settings.domStorageEnabled = true
220         view.settings.allowFileAccess = false
221         view.settings.allowContentAccess = false
222         view.webViewClient = object : WebViewClient() {
223             override fun shouldOverrideUrlLoading(v: WebView, request: WebResourceRequest): Boolean {
224                 val url = request.url.toString()
225                 val to = Preview.address(url) ?: return true
226                 if (to == url) return false
227                 v.loadUrl(to)
228                 return true
229             }
230 
231             // A page that stops the renderer must not stop the app: a new
232             // web view takes the place of the old one.
233             override fun onRenderProcessGone(v: WebView, detail: RenderProcessGoneDetail): Boolean {
234                 preview.removeView(v)
235                 v.destroy()
236                 if (page === v) page = null
237                 framed = null
238                 return true
239             }
240         }
241         preview.addView(view, LinearLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, 0, 1f))
242         page = view
243         return view
244     }

371     private inner class Rows : BaseAdapter() {
372         var items: List<Bookmark> = emptyList()
373 
374         override fun getCount() = items.size

376         override fun getItemId(position: Int) = position.toLong()

378         override fun getView(position: Int, convertView: View?, parent: ViewGroup): View {
379             val view = convertView ?: layoutInflater.inflate(R.layout.item, parent, false)
380             val b = items[position]
381             view.findViewById<TextView>(R.id.title).text = b.desc.ifEmpty { b.url }
382             view.findViewById<TextView>(R.id.detail).text =
383                 listOf(Tsv.host(b.url), b.tags.joinToString(" ")).filter { it.isNotEmpty() }.joinToString("  ·  ")
