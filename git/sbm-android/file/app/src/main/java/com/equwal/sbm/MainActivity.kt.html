83     /** Sync, then show the file again. A quiet sync reports only what the user must act on. */
84     private fun sync(quiet: Boolean) {
85         Sync.run(this) { result ->
86             if (isDestroyed) return@run
87             result.onSuccess { changed ->
88                 if (changed) load()
89                 if (!quiet) toast(resources.getQuantityString(R.plurals.synced, all.size, all.size))
90             }.onFailure { e ->
91                 if (!quiet || (e is Sync.Failure && e.urgent)) toast(getString(R.string.sync_failed, e.message))
92                 if (e is Sync.Failure && e.signIn) invalidateOptionsMenu()
93             }
94         }
95     }

122     private fun load() {
123         all = try {
124             // Newest first: bm adds bookmarks at the end of the file.
125             Tsv.parse(Store.read(this)).asReversed()
126         } catch (e: Exception) {
127             toast(getString(R.string.cannot_read, e.message))
128             emptyList()
129         }
130         empty.setText(if (Store.file(this) == null) R.string.no_file else R.string.no_match)
131         show()
132     }

134     private fun show() {
135         rows.items = Fuzzy.filter(all, search.text.toString()) { "${it.desc} ${it.tags.joinToString(" ")} ${it.url}" }
136         rows.notifyDataSetChanged()

139     /** Go opens the first bookmark. Without one, the text is an address or a web search, as in bm. */

172         // Android 13 and later show their own message.
