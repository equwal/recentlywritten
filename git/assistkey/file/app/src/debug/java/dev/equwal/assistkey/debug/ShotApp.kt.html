14 /**
15  * Store screenshots, for a device that cannot take them.
16  *
17  * A second after any screen comes to the front, its view tree is drawn into a
18  * bitmap and written to the app's external files directory:
19  *
20  *   adb pull /sdcard/Android/data/dev.equwal.assistkey.debug/files/shots
21  */
22 class ShotApp : Application() {
23 
24     override fun onCreate() {
25         super.onCreate()
26         val main = Handler(Looper.getMainLooper())
27         registerActivityLifecycleCallbacks(object : ActivityLifecycleCallbacks {
28             override fun onActivityResumed(a: Activity) {
29                 main.postDelayed({ if (!a.isFinishing) shoot(a) }, 1200L)
30             }
31             override fun onActivityCreated(a: Activity, b: Bundle?) = Unit
32             override fun onActivityStarted(a: Activity) = Unit
33             override fun onActivityPaused(a: Activity) = Unit
34             override fun onActivityStopped(a: Activity) = Unit
35             override fun onActivitySaveInstanceState(a: Activity, b: Bundle) = Unit
36             override fun onActivityDestroyed(a: Activity) = Unit
37         })
38     }
39 
40     private fun shoot(a: Activity) {
41         val v = a.window.decorView
42         if (v.width == 0 || v.height == 0) return
43         val bmp = Bitmap.createBitmap(v.width, v.height, Bitmap.Config.ARGB_8888)
44         val canvas = Canvas(bmp)
45         canvas.drawColor(Color.WHITE)
46         v.draw(canvas)
47         val dir = File(getExternalFilesDir(null), "shots").apply { mkdirs() }
48         val out = File(dir, a.javaClass.simpleName + ".png")
49         out.outputStream().use { bmp.compress(Bitmap.CompressFormat.PNG, 100, it) }
50         Log.i("AssistKey", "shot " + out.absolutePath)
51     }
