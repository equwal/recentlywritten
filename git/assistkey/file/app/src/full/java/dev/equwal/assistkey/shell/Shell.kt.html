12 /**
13  * Shell access, by way of Shizuku.
14  *
15  * Shizuku is a separate, free app that a user starts from the device itself
16  * through Android's wireless debugging - no computer, no root. Once it is
17  * running and has said yes to us, this object can start processes under the
18  * shell uid, and everything Android keeps from ordinary apps is within reach:
19  * the kernel input nodes (so the Power key), `settings`, `cmd overlay`.
20  *
21  * It uses Shizuku's plain remote-process call and nothing grander. Shizuku's
22  * bound "user service" would be the tidier design, but its starter dies inside
23  * LoadedApk.makeApplication on Android 16 before our code is ever loaded, and a
24  * pipe from `getevent` needs none of it.
25  *
26  * All of this is optional. Without shell access the app falls back on what an
27  * installed app can do alone, and says what is missing.
28  */
29 object Shell {

76     /** Safe to call often. Registers for Shizuku's comings and goings, once. */
77     fun connect(c: Context) {
78         val app = c.applicationContext
79         if (!hooked) {
80             hooked = true
81             runCatching {
82                 Shizuku.addBinderReceivedListenerSticky { onUp(app) }
83                 Shizuku.addBinderDeadListener {
84                     stopWatching()
85                     changed()
86                 }

88             }
89         }

93     private fun onUp(app: Context) {
94         if (ready && !granted) {
95             granted = true
96             // With this the app can put the Power button back to firmware
97             // defaults by itself, even when Shizuku is gone after a restart.
98             run("pm grant " + app.packageName + " android.permission.WRITE_SECURE_SETTINGS")
99             Log.i(TAG, "shell access ready")
100         }
101         changed()
102     }

111      * Shizuku.newProcess is private in the v13 API but is the supported wire
112      * call underneath, and the one every shell-style client uses.
113      */
114     private val newProcess by lazy {
115         Shizuku::class.java.getDeclaredMethod(
116             "newProcess", Array<String>::class.java, Array<String>::class.java, String::class.java
117         ).apply { isAccessible = true }
118     }

120     private fun start(vararg argv: String): Process? = runCatching {
121         newProcess.invoke(null, argv, null, null) as Process
122     }.onFailure { Log.w(TAG, "shell process failed: " + argv.joinToString(" "), it) }.getOrNull()

124     data class Result(val exit: Int, val output: String) {
125         val ok: Boolean get() = exit == 0
126     }
127 
128     private fun execNow(command: String): Result {
129         val p = start("sh", "-c", "$command 2>&1") ?: return Result(-1, "no shell access")
130         return runCatching {
131             val out = p.inputStream.bufferedReader().readText()
132             Result(p.waitFor(), out)
133         }.getOrElse { Result(-1, it.toString()) }
134     }
135 
136     /** Runs off the main thread; [done] comes back on it. */
137     fun run(command: String, done: ((Result) -> Unit)? = null) {
138         worker.execute {
139             val r = if (ready) execNow(command) else Result(-1, "no shell access")
140             if (!r.ok) Log.i(TAG, "shell: " + command + " -> " + r.exit + " " + r.output.take(200))
141             done?.let { main.post { it(r) } }
142         }
143     }
144 
145     /** Commands in order, stopping at nothing; [done] gets true if every one succeeded. */
146     fun runAll(commands: List<String>, done: ((Boolean) -> Unit)? = null) {
147         worker.execute {
148             var allOk = true
149             for (c in commands) if (!ready || !execNow(c).ok) allOk = false
150             done?.let { main.post { it(allOk) } }
151         }
152     }

156     private val watchers = ArrayList<Process>()
157     @Volatile private var generation = 0
158 
159     /**
160      * Reports presses and releases of the given Linux key codes, read from the
161      * kernel input nodes by `getevent` running as the shell user. Only the
162      * nodes that declare one of the keys are opened, which keeps the
163      * touchscreen's firehose out of it.
164      */
165     fun watchKeys(scanCodes: Set<Int>, onKey: (code: Int, down: Boolean, at: Long) -> Unit, started: (Boolean) -> Unit) {
166         stopWatching()
167         val mine = ++generation
168         worker.execute {
169             val nodes = nodesDeclaring(scanCodes)
170             val procs = nodes.mapNotNull { node -> start("getevent", "-q", node)?.also { pump(it, scanCodes, mine, onKey) } }
171             synchronized(watchers) {
172                 if (generation == mine) watchers.addAll(procs) else procs.forEach { it.destroy() }
173             }
174             Log.i(TAG, "watching " + nodes.joinToString(" "))
175             main.post { started(procs.isNotEmpty()) }
176         }
177     }
178 
179     fun stopWatching() {
180         generation++
181         synchronized(watchers) {
182             watchers.forEach { p -> runCatching { p.destroy() } }
183             watchers.clear()
184         }
185     }

187     /** Lines look like `0001 0074 00000001`: type, code, value, in hex. */
188     private fun pump(p: Process, wanted: Set<Int>, mine: Int, onKey: (Int, Boolean, Long) -> Unit) {
189         Thread({
190             runCatching {
191                 p.inputStream.bufferedReader().forEachLine { line ->
192                     if (generation != mine) return@forEachLine
193                     val f = line.trim().split(' ').filter { it.isNotEmpty() }
194                     if (f.size != 3 || f[0] != "0001") return@forEachLine
195                     val code = f[1].toIntOrNull(16) ?: return@forEachLine
196                     val value = f[2].toLongOrNull(16) ?: return@forEachLine
197                     // value 2 is auto-repeat, which carries no news
198                     if (code !in wanted || value > 1L) return@forEachLine
199                     val at = SystemClock.uptimeMillis()
200                     main.post { if (generation == mine) onKey(code, value == 1L, at) }
201                 }
202             }
203             // The pipe closed: Shizuku went away, or we were told to stop.
204             if (generation == mine) main.post { changed() }
205         }, "keys").apply { isDaemon = true }.start()
206     }
207 
208     /** `getevent -p` prints, per device, the hex key codes it declares. */
209     private fun nodesDeclaring(codes: Set<Int>): List<String> {
210         val hex = codes.map { "%04x".format(it) }.toSet()
211         val found = LinkedHashSet<String>()
212         var node: String? = null
213         var inKeys = false
214         execNow("getevent -p").output.lineSequence().forEach { raw ->
215             val line = raw.trim()
216             when {
217                 line.startsWith("add device") -> {
218                     node = line.substringAfter(": ").trim()
219                     inKeys = false
220                 }
221                 line.startsWith("KEY (0001):") -> {
222                     inKeys = true
223                     if (line.substringAfter(":").split(' ').any { it in hex }) node?.let(found::add)
224                 }
225                 inKeys && line.contains("(00") -> inKeys = false // next event class
226                 inKeys -> if (line.split(' ').any { it in hex }) node?.let(found::add)
227             }
228         }
229         return found.toList()
230     }
