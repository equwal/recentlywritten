70     /** Hands the APK to the package installer of Android. Android then asks the user. */
71     fun install(a: Activity, item: Item) {
72         runCatching {
73             val installer = a.packageManager.packageInstaller
74             val params = PackageInstaller.SessionParams(PackageInstaller.SessionParams.MODE_FULL_INSTALL)
75             params.setAppPackageName(item.pkg)
76             val id = installer.createSession(params)
77             installer.openSession(id).use { session ->
78                 a.assets.open(item.asset).use { input ->
79                     session.openWrite("base.apk", 0, -1).use { out ->
80                         input.copyTo(out)
81                         session.fsync(out)
82                     }
83                 }
84                 val result = PendingIntent.getBroadcast(
85                     a, id, Intent(a, Result::class.java),
86                     PendingIntent.FLAG_MUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
87                 )
88                 session.commit(result.intentSender)
89             }
90         }.onFailure { Toast.makeText(a, "Could not start the install", Toast.LENGTH_LONG).show() }
91     }

95         override fun onReceive(c: Context, i: Intent) {
