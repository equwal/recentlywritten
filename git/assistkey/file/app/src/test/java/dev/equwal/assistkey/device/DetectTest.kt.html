112     private fun randomLine(r: Random): String {
113         val words = listOf(
114             "add device 4: /dev/input/event9", "name:", "KEY (0001):", "ABS (0003):",
115             "0072", "003b", "ffff", "0000", "zzzz", "-1", "0x74", "   ", "\t",
116             "input props:", "<none>", "value 0, min 0", "\u2603", "99999999", ":",
117             "KEY (0001): 0072 0073", "0247", "0210", "00e2"
118         )
119         return (0 until r.nextInt(6)).joinToString(" ") { words[r.nextInt(words.size)] }
120     }
