17     private fun randomText(r: Random): String {
18         val alphabet = "abcXYZ 09_-./:\"\\{}[]\n\u00e9\u4e2d\u2603"
19         return (0 until r.nextInt(12)).map { alphabet[r.nextInt(alphabet.length)] }.joinToString("")
20     }
21 
22     private fun randomValue(r: Random): Any = when (r.nextInt(6)) {
23         0 -> randomText(r)
24         1 -> r.nextBoolean()
25         2 -> r.nextInt()
26         // A long outside the int range, so that the two number types stay apart.
27         3 -> Int.MAX_VALUE.toLong() + 1 + r.nextInt(1000)
28         4 -> r.nextInt(1000) / 8f
29         else -> (0 until r.nextInt(4)).map { randomText(r) }.toSet()
30     }
31 
32     private fun randomSettings(r: Random): Map<String, Map<String, Any>> {
33         val out = LinkedHashMap<String, Map<String, Any>>()
34         SettingsFile.ALLOWED.forEach { (file, keys) ->
35             val names = keys?.filter { it != "bindings" } ?: listOf("accessibility", "assistant", "camera")
36             val values = names.filter { r.nextBoolean() }.associateWith { randomValue(r) }
37             if (values.isNotEmpty()) out[file] = values
38         }
39         return out
40     }
