42     @Test fun `every trigger and action that an old version stored still reads`() {
43         val root = JSONObject(bindings)
44         val kinds = HashSet<ActionKind>()
45         root.keys().forEach { id ->
46             val trigger = Trigger.parse(id)
47             assertNotNull("trigger " + id, trigger)
48             assertEquals(id, trigger!!.id)
49             val spec = ActionSpec.fromJson(root.getJSONObject(id))
50             assertEquals(root.getJSONObject(id).getString("kind"), spec.kind.name)
51             assertEquals(root.getJSONObject(id).getString("payload"), spec.payload)
52             kinds += spec.kind
53         }
54         // PASS_THROUGH is never stored: it means "no binding".
55         assertEquals(ActionKind.entries.toSet() - ActionKind.PASS_THROUGH, kinds)
56     }

58     @Test fun `the tokens of the keys do not change`() {
