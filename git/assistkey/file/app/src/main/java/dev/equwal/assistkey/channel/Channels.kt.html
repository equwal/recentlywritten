52     private fun applyComponents(c: Context, ch: Channel, on: Boolean) {
53         val pm = c.packageManager
54         val want = if (on) {
55             android.content.pm.PackageManager.COMPONENT_ENABLED_STATE_ENABLED
56         } else {
57             android.content.pm.PackageManager.COMPONENT_ENABLED_STATE_DISABLED
58         }
59         ch.components.forEach { cls ->
60             runCatching {
61                 val cn = ComponentName(c.packageName, cls)
62                 if (pm.getComponentEnabledSetting(cn) != want) {
63                     pm.setComponentEnabledSetting(
64                         cn, want, android.content.pm.PackageManager.DONT_KILL_APP
65                     )
66                 }
67             }
68         }
69     }

95      * The running service is the authority when it is there - it lives in this
96      * process and sets the holder itself. The settings string is only a
97      * fallback for the window between "enabled" and "bound", and is read
98      * defensively because restricted settings can make it unreadable.

100     fun isAccessibilityOn(c: Context): Boolean {
101         if (ServiceHolder.isRunning) return true
102         val want = ComponentName(c.packageName, KeyFilterService::class.java.name)
103         val list = runCatching {
104             Settings.Secure.getString(
105                 c.contentResolver, Settings.Secure.ENABLED_ACCESSIBILITY_SERVICES
106             )
107         }.getOrNull() ?: return false
108         return list.split(":").any {
109             ComponentName.unflattenFromString(it.trim()) == want
110         }
111     }

141      * Where to send the user to satisfy a channel. Role requests get a real
142      * system dialog; everything else falls back to the closest Settings screen,
143      * because no firmware exposes a direct intent for default camera.

145     fun claimIntent(c: Context, ch: Channel): Intent = when (ch) {
146         Channel.ACCESSIBILITY -> Intent(Settings.ACTION_ACCESSIBILITY_SETTINGS)
147 
148         // Android has no request dialog for the assistant role: the request
149         // intent closes at once and the user sees nothing. The settings page
150         // "Digital assistant app" is the only way in.
151         Channel.ASSISTANT -> Intent(Settings.ACTION_VOICE_INPUT_SETTINGS)
152 
153         Channel.WALLET -> roleRequest(c, ROLE_WALLET)
154             ?: defaultApps()
155 
156         Channel.CAMERA -> defaultApps()
157     }.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)

159     private fun roleRequest(c: Context, role: String): Intent? {
160         if (!isRoleAvailable(c, role)) return null
161         return runCatching { roleManager(c)?.createRequestRoleIntent(role) }.getOrNull()
162     }

164     private fun defaultApps(): Intent =

167     /**
168      * Not every firmware has the default-apps screen wired up; fall back to the
169      * top-level Settings rather than crashing on an unresolvable intent.
170      */
171     fun safeStart(c: Context, intent: Intent): Boolean {
172         if (intent.resolveActivity(c.packageManager) != null) {
173             c.startActivity(intent); return true
174         }
175         val fallback = Intent(Settings.ACTION_SETTINGS).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
176         if (fallback.resolveActivity(c.packageManager) != null) {
177             c.startActivity(fallback); return true
178         }
179         return false
180     }
