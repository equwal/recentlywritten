8 /**
9  * How the reader is navigated, as far as the system is concerned: whether the
10  * three-button bar is showing, and whether swipe gestures are live.
11  *
12  * Both can be read here and neither can be changed here. Measured on firmware
13  * 1.5.6:
14  *
15  *  - The bar is Android's navigation-mode overlay. `threebutton` shows it;
16  *    `gestural` removes it outright (no pill, no inset) and brings Android's
17  *    edge-swipe Back with it. Switching overlays is a shell-only operation.
18  *  - The swipe up from the bottom edge to Home is Viwoods' own, present in both
19  *    modes, and is switched by Settings.System `disable_gesture_bottom`. That
20  *    is an OEM key outside SettingsProvider's public list, so no ordinary app
21  *    may write it, whatever permissions it holds.
22  *  - In gestural mode the edge-swipe Back can only be removed by shrinking its
23  *    inset to nothing: Settings.Secure back_gesture_inset_scale_left/right = 0.
24  *
25  * Order matters when applying. Changing the overlay makes SystemUI forget
26  * `disable_gesture_bottom`, so the overlay goes first and the setting after.
27  */
28 object NavNative {
29 
30     private const val GESTURE_BOTTOM = "disable_gesture_bottom"
31     private const val OVERLAY = "com.android.internal.systemui.navbar."
32 
33     /** True if the three-button bar is on screen, null if it cannot be told yet. */
34     fun buttonsShowing(a: Activity): Boolean? {
35         val insets = a.window.decorView.rootWindowInsets ?: return null
36         val bar = insets.getInsetsIgnoringVisibility(WindowInsets.Type.navigationBars())
37         val dp = a.resources.displayMetrics.density
38         // The button bar is about 48dp tall. Gestural mode leaves nothing here
39         // on this firmware, and a thin pill strip on stock Android.
40         return maxOf(bar.bottom, bar.left, bar.right) >= 40 * dp
41     }
42 
43     /** True if the swipe-up gesture is live, null if the setting cannot be read. */
44     fun gesturesOn(c: Context): Boolean? {
45         val name = dev.equwal.assistkey.device.Device.bottomGestureSetting
46         if (name == null) {
47             // Stock Android: gestures exist exactly when the bar does not.
48             return null
49         }
50         val v = runCatching { Settings.System.getString(c.contentResolver, name) }
51             .getOrElse { return null }
52         return v?.trim() != "1"
53     }
54 
55     /**
56      * What a shell has to run to produce the wanted state, in the order it must
57      * run. The pause and the double write are not decoration: SystemUI rebuilds
58      * itself when the overlay changes and only notices the gesture setting if it
59      * changes afterwards.
60      */
61     fun shellCommands(buttons: Boolean, gestures: Boolean): List<String> {
62         val off = if (gestures) "0" else "1"
63         val nudge = if (gestures) "1" else "0"
64         // Only meaningful without the bar, but harmless with it, and setting it
65         // every time means a later switch to gestural mode cannot surprise.
66         val scale = if (gestures) "0.6" else "0"
67         val out = ArrayList<String>()
68         out += "cmd overlay enable-exclusive --category " + OVERLAY + (if (buttons) "threebutton" else "gestural")
69         if (dev.equwal.assistkey.device.Device.bottomGestureSetting != null) {
70             out += "sleep 4"
71             out += "settings put system $GESTURE_BOTTOM $nudge"
72             out += "sleep 1"
73             out += "settings put system $GESTURE_BOTTOM $off"
74         }
75         out += "settings put secure back_gesture_inset_scale_left $scale"
76         out += "settings put secure back_gesture_inset_scale_right $scale"
77         return out
78     }
79 
80     const val PREFS = "assistkey_nav"
81 
82     /**
83      * Shows the button bar when it is hidden, and hides it when it shows. The
84      * gestures stay as the user set them. It needs shell access. False where
85      * there is none.
86      */
87     fun toggleBar(c: Context): Boolean {
88         if (!dev.equwal.assistkey.shell.Shell.ready) return false
89         val p = c.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
90         val buttons = !p.getBoolean("buttons", true)
91         p.edit().putBoolean("buttons", buttons).apply()
92         dev.equwal.assistkey.shell.Shell.runAll(shellCommands(buttons, p.getBoolean("gestures", true)))
93         return true
94     }
95 
96     /** The same, for someone typing them at a computer. */
97     fun commands(buttons: Boolean, gestures: Boolean): List<String> =
98         shellCommands(buttons, gestures).filterNot { it.startsWith("sleep") }.distinct().map { "adb shell $it" }
