6 /**
7  * Settings as one JSON document, to move them to another device or to share.
8  *
9  * This file has no Android types, so a test can run it with no device. The
10  * caller reads and writes the preference files; this object turns their content
11  * into text and back, and decides what may pass.
12  *
13  * Only the names in [ALLOWED] pass, in both directions. The licence, the
14  * firmware values the app saved, and what is true of one device only (the keys
15  * it has, its light level) are not settings and never enter the file. A file
16  * from a stranger therefore cannot unlock the app or write a value this list
17  * does not name.
18  *
19  * Each value carries its type, because JSON has one number type and a
20  * preference file has three.
21  */
22 object SettingsFile {
23 
24     const val APP = "AssistKey"
25     const val FORMAT = 1
26 
27     /** Preference file name to the keys that may pass. `null` means every key in that file. */
28     val ALLOWED: Map<String, Set<String>?> = mapOf(
29         "assistkey" to setOf("bindings", "multitap_ms", "hold_ms", "chord_ms"),
30         "assistkey_channels" to null,
31         "assistkey_nav" to setOf("buttons", "gestures", "keys"),
32         "assistkey_voice" to setOf("engine", "language"),
33         "assistkey_device" to setOf("ai_key_returns", "ink_recents_for_system"),
34         "assistkey_power" to setOf("wanted")
35     )
36 
37     class BadFile(message: String) : Exception(message)
38 
39     fun allowed(file: String, key: String): Boolean =
40         ALLOWED.containsKey(file) && (ALLOWED[file]?.contains(key) ?: true)
41 
42     /** [settings] is preference file name to its key-value content. Entries that may not pass are dropped. */
43     fun encode(settings: Map<String, Map<String, Any?>>, appVersion: String, device: String): String {
44         val files = JSONObject()
45         settings.toSortedMap().forEach { (file, values) ->
46             val out = JSONObject()
47             values.toSortedMap().forEach { (key, value) ->
48                 if (allowed(file, key)) wrap(key, value)?.let { out.put(key, it) }
49             }
50             if (out.length() > 0) files.put(file, out)
51         }
52         return JSONObject()
53             .put("app", APP)
54             .put("format", FORMAT)
55             .put("version", appVersion)
56             .put("device", device)
57             .put("settings", files)
58             .toString(2)
59     }
60 
61     /** The opposite of [encode]. Throws [BadFile] when the text is not a settings file of this app. */
62     fun decode(text: String): Map<String, Map<String, Any>> {
63         val root = try {
64             JSONObject(text)
65         } catch (e: Exception) {
66             throw BadFile("This is not a settings file.")
67         }
68         if (root.optString("app") != APP) throw BadFile("This is not a Rebind settings file.")
69         if (root.optInt("format", -1) != FORMAT) {
70             throw BadFile("This settings file is from a newer version of Rebind.")
71         }
72         val files = root.optJSONObject("settings") ?: throw BadFile("This settings file is empty.")
73 
74         val result = LinkedHashMap<String, Map<String, Any>>()
75         files.keys().forEach { file ->
76             val content = files.optJSONObject(file) ?: return@forEach
77             val values = LinkedHashMap<String, Any>()
78             content.keys().forEach { key ->
79                 if (allowed(file, key)) unwrap(key, content.optJSONObject(key))?.let { values[key] = it }
80             }
81             if (values.isNotEmpty()) result[file] = values
82         }
83         return result
84     }

88     /** Bindings are JSON already; they go in as an object, so a person can read and edit them. */
89     private fun wrap(key: String, value: Any?): JSONObject? = when {
90         value == null -> null
91         key == "bindings" && value is String ->
92             runCatching { JSONObject().put("t", "json").put("v", JSONObject(value)) }.getOrNull()
93         value is String -> JSONObject().put("t", "s").put("v", value)
94         value is Boolean -> JSONObject().put("t", "b").put("v", value)
95         value is Int -> JSONObject().put("t", "i").put("v", value)
96         value is Long -> JSONObject().put("t", "l").put("v", value)
97         value is Float -> JSONObject().put("t", "f").put("v", value.toDouble())
98         value is Set<*> -> JSONObject().put("t", "set").put("v", JSONArray(value.map { it.toString() }.sorted()))
99         else -> null
100     }
101 
102     private fun unwrap(key: String, o: JSONObject?): Any? {
103         if (o == null || !o.has("v")) return null
104         return when (o.optString("t")) {
105             "json" -> if (key == "bindings") o.optJSONObject("v")?.toString() else null
106             "s" -> o.opt("v") as? String
107             "b" -> o.opt("v") as? Boolean
108             "i" -> (o.opt("v") as? Number)?.toInt()
109             "l" -> (o.opt("v") as? Number)?.toLong()
110             "f" -> (o.opt("v") as? Number)?.toFloat()
111             "set" -> o.optJSONArray("v")?.let { a -> (0 until a.length()).map { a.optString(it) }.toSet() }
112             else -> null
113         }
114     }
