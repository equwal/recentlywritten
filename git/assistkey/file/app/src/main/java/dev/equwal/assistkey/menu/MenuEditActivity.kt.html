19 /** Builds the menu that one gesture opens: add actions, order them, save. */
20 class MenuEditActivity : Activity() {

26         super.onCreate(savedInstanceState)

32     }

39     private fun build() {
40         val col = Ui.page(this, "Menu for " + trigger.label())
41 
42         col.header(if (items.isEmpty()) "No actions yet" else "Actions, in order")
43         items.forEachIndexed { i, spec ->
44             col.row(spec.describe(), "Tap to move or remove") {
45                 val choices = listOf("Move up", "Move down", "Remove")
46                 Ui.pick(this, spec.describe(), choices) { which ->
47                     when (which) {
48                         0 -> if (i > 0) items.add(i - 1, items.removeAt(i))
49                         1 -> if (i < items.size - 1) items.add(i + 1, items.removeAt(i))
50                         2 -> items.removeAt(i)
51                     }
52                     build()
53                 }
54             }
55         }
56         col.button("Add an action") {
57             startActivityForResult(ActionPickerActivity.pickIntent(this, trigger), PICK)
58         }
59 
60         col.primaryButton(if (items.isEmpty()) "Remove the menu" else "Save the menu") {
61             Store.bind(this, trigger, if (items.isEmpty()) ActionSpec.PASS else Menu.spec(items))
62             // Ask now for what the menu needs. The guided setup asks by itself.
63             if (items.isNotEmpty() && !intent.getBooleanExtra(ActionPickerActivity.EXTRA_NO_ASK, false)) {
64                 dev.equwal.assistkey.setup.GuidedSetupActivity.askIfMissing(this, trigger)
65             }
66             setResult(RESULT_OK)
67             finish()
68         }
69         col.button("Try it") { MenuActivity.open(this, Menu.encode(items)) }
70     }
71 
72     override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
73         super.onActivityResult(requestCode, resultCode, data)
74         if (requestCode != PICK || resultCode != RESULT_OK) return
75         val json = data?.getStringExtra(ActionPickerActivity.RESULT_SPEC) ?: return
76         runCatching { ActionSpec.fromJson(JSONObject(json)) }.getOrNull()?.let(items::add)
77         build()
