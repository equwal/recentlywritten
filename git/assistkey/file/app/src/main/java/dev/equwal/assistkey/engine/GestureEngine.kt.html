7 /**
8  * Turns a stream of key down/up events into tap-count / hold / chord gestures.
9  *
10  * Deliberately free of Android types: everything that touches the framework
11  * (timers, firing an action, emulating the native key) goes through [Host], so
12  * the whole state machine is unit-testable on the JVM.
13  *
14  * The one rule that matters for feel: a key with no bindings at all is never
15  * consumed, and a key whose highest bound tap count is 1 fires on key-up
16  * without waiting out the multi-tap window. We only pay the multi-tap latency
17  * on keys where the user actually asked for a double/triple tap.
18  */
19 class GestureEngine(
20     private val host: Host,
21     var cfg: Config = Config()
22 ) {
23 
24     data class Config(
25         /** How long to wait for another tap before committing the count. */
26         val multiTapMs: Long = 280,
27         /** How long a key must stay down to count as a hold. */
28         val holdMs: Long = 450,
29         /** Two keys going down within this window form a chord. */
30         val chordMs: Long = 140
31     )
32 
33     interface Host {
34         /** Highest bound tap count for this key set, or 0 if no tap is bound. */
35         fun maxTaps(keys: Set<HwKey>): Int
36 
37         /** True when a hold is bound for this key set. */
38         fun hasHold(keys: Set<HwKey>): Boolean
39 
40         /** True when this exact gesture has a binding. */
41         fun isBound(trigger: Trigger): Boolean
42 
43         /** Any key set that starts with this key and could still become a chord. */
44         fun chordPartners(key: HwKey): Set<HwKey>
45 
46         fun fire(trigger: Trigger)
47 
48         /** Reproduce what the system would have done, since we ate the event. */
49         fun passThrough(keys: Set<HwKey>, longPress: Boolean)
50 
51         fun schedule(token: String, delayMs: Long, action: () -> Unit)
52         fun cancel(token: String)
53     }
54 
55     private companion object {
56         const val T_HOLD = "hold"
57         const val T_TAP = "tap"
58     }
59 
60     private val down = LinkedHashSet<HwKey>()
61     private var seqKeys: Set<HwKey> = emptySet()
62     private var tapCount = 0
63     private var holdFired = false
64     private var consuming = false
65     private var firstDownAt = 0L
66 
67     /** @return true to swallow the event. */
68     fun onDown(key: HwKey, now: Long, repeatCount: Int): Boolean {
69         if (repeatCount > 0) return consuming   // our own timer drives holds
70 
71         down.add(key)
72 
73         val chordUpgrade = consuming &&
74             key !in seqKeys &&
75             (now - firstDownAt) <= cfg.chordMs &&
76             interesting(seqKeys + key)
77 
78         when {
79             chordUpgrade -> {
80                 host.cancel(T_HOLD); host.cancel(T_TAP)
81                 seqKeys = seqKeys + key
82                 tapCount = 0
83                 holdFired = false
84             }
85             !consuming || key !in seqKeys -> {
86                 host.cancel(T_HOLD); host.cancel(T_TAP)
87                 seqKeys = setOf(key)
88                 tapCount = 0
89                 holdFired = false
90                 firstDownAt = now
91             }
92             else -> {
93                 // Another tap of a sequence already in progress.
94                 host.cancel(T_TAP)
95             }
96         }
97 
98         // Only grab the key if this set, or a chord it could grow into, is bound.
99         if (!interesting(seqKeys) && !couldBecomeChord(seqKeys, now)) {
100             reset()
101             return false
102         }
103 
104         consuming = true
105         if (host.hasHold(seqKeys)) {
106             host.schedule(T_HOLD, cfg.holdMs) { onHoldElapsed() }
107         }
108         return true
109     }
110 
111     /** @return true to swallow the event. */
112     fun onUp(key: HwKey, now: Long): Boolean {
113         down.remove(key)
114         if (!consuming) return false
115 
116         host.cancel(T_HOLD)
117 
118         if (holdFired) {
119             if (down.isEmpty()) reset()
120             return true
121         }
122 
123         // For a chord, wait until every key of the chord is released.
124         if (down.any { it in seqKeys }) return true
125 
126         tapCount++
127         val max = host.maxTaps(seqKeys)
128 
129         if (max == 0) {
130             // Nothing bound to taps - only a hold was. Give the key back.
131             val keys = seqKeys
132             reset()
133             host.passThrough(keys, longPress = false)
134             return true
135         }
136 
137         if (tapCount >= max) {
138             commit()
139         } else {
140             host.schedule(T_TAP, cfg.multiTapMs) { commit() }
141         }
142         return true
143     }
144 
145     fun onCancel() {
146         host.cancel(T_HOLD); host.cancel(T_TAP)
147         down.clear()
148         reset()
149     }
150 
151     private fun onHoldElapsed() {
152         val t = Trigger(seqKeys, GestureType.HOLD)
153         holdFired = true
154         tapCount = 0
155         if (host.isBound(t)) host.fire(t)
156     }
157 
158     private fun commit() {
159         host.cancel(T_TAP)
160         val keys = seqKeys
161         val n = tapCount.coerceIn(1, Trigger.MAX_TAPS)
162         reset()
163         if (keys.isEmpty()) return
164 
165         val t = Trigger(keys, GestureType.TAP, n)
166         if (host.isBound(t)) {
167             host.fire(t)
168         } else if (n == 1) {
169             host.passThrough(keys, longPress = false)
170         }
171         // A 2+ tap with no binding is intentionally dropped rather than
172         // replayed - replaying would fire the native action N times.
173     }
174 
175     private fun interesting(keys: Set<HwKey>): Boolean =
176         host.maxTaps(keys) > 0 || host.hasHold(keys)
177 
178     /**
179      * A lone key with no bindings still has to be held briefly if it is the
180      * first half of a bound chord, otherwise the chord could never be entered.
181      */
182     private fun couldBecomeChord(keys: Set<HwKey>, now: Long): Boolean {
183         if (keys.size != 1) return false
184         val partners = host.chordPartners(keys.first())
185         return partners.isNotEmpty() && (now - firstDownAt) <= cfg.chordMs
186     }
187 
188     private fun reset() {
189         seqKeys = emptySet()
190         tapCount = 0
191         holdFired = false
192         consuming = false
193     }
