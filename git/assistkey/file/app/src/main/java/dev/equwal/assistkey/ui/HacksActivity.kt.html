37     private fun build() {
38         val col = Ui.page(this, "Hardware hacks")
39         fun door(title: String, hint: String, vararg ways: Channel) = col.row(
40             title, hint, state = if (ways.any { Channels.isSatisfied(this, it) }) "On" else "Off"
41         ) { startActivity(Intent(this, PowerActivity::class.java)) }
42         door("Hold Power", "Through the assistant role", Channel.ASSISTANT)
43         // One gesture, two ways in. A device sends the double tap to the camera or to the wallet.
44         door("Double tap Power", "Through the camera or the wallet intent", Channel.CAMERA, Channel.WALLET)
45         if (Shell.SUPPORTED) {
46             col.row(

49                 state = Shell.describe(this)

51             // No light node, no extra-dim. Detection says so; do not offer it.
52             if (Shell.ready && Detect.stored(this)?.capability(Detect.FRONTLIGHT) != Detect.State.NO) {
53                 col.row(
54                     "Extra-dim light",
55                     "Below the lowest the system slider allows"
56                 ) { startActivity(Intent(this, DisplayActivity::class.java)) }
57             }
58         }
59         if (Device.hasFirmwareKeyHooks) {
60             val hidden = ViwoodsBridge.keys().filter { ViwoodsBridge.hidesFromFilter(this, it) }
61             col.row(
62                 "Device button settings",
63                 if (hidden.isEmpty()) "What this device does with each button"
64                 else "Hiding " + hidden.joinToString(" and ") { it.label.lowercase() } +
65                     " from this app",
66                 state = if (hidden.isEmpty()) null else "Hiding a button"
67             ) { startActivity(Intent(this, ViwoodsActivity::class.java)) }
68         }
