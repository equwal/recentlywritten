97     override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
98         super.onActivityResult(requestCode, resultCode, data)
99         val uri: Uri = data?.data ?: return
100         if (resultCode != RESULT_OK) return
101         when (requestCode) {
102             SAVE -> runCatching {
103                 contentResolver.openOutputStream(uri)?.use { it.write(export().toByteArray()) }
104             }.onSuccess { toast("Saved") }.onFailure { toast("Could not write the file") }

106             OPEN -> runCatching {
107                 contentResolver.openInputStream(uri)?.use { it.readBytes().toString(Charsets.UTF_8) }
108             }.getOrNull()?.let(::offer) ?: toast("Could not read the file")
109         }

118     /** Reads the text, says what is in it, and applies it only after a yes. */
119     private fun offer(text: String) {
120         val settings = try {
121             SettingsFile.decode(text)
122         } catch (e: SettingsFile.BadFile) {
123             return toast(e.message ?: "This is not a settings file.")
124         }
125         if (settings.isEmpty()) return toast("That file has no settings in it")
126 
127         val bindings = (settings["assistkey"]?.get("bindings") as? String)
128             ?.let { runCatching { JSONObject(it) }.getOrNull() }
129         val count = bindings?.keys()?.asSequence()?.count { Trigger.parse(it) != null } ?: 0
130         AlertDialog.Builder(this)
131             .setTitle("Import these settings?")
132             .setMessage(
133                 count.toString() + " bindings, and settings for: " +
134                     settings.keys.joinToString(", ") { it.removePrefix("assistkey_").replace("assistkey", "keys") } +
135                     ".\n\nThey replace what you have now."
136             )
137             .setPositiveButton("Import") { _, _ -> apply(settings) }
138             .setNegativeButton(android.R.string.cancel, null)
139             .show()
140     }

142     private fun apply(settings: Map<String, Map<String, Any>>) {
143         settings.forEach { (file, values) ->
144             val e = getSharedPreferences(file, Context.MODE_PRIVATE).edit()
145             values.forEach { (key, value) ->
146                 when (value) {
147                     is String -> e.putString(key, if (key == "bindings") cleanBindings(value) else value)
148                     is Boolean -> e.putBoolean(key, value)
149                     is Int -> e.putInt(key, value)
150                     is Long -> e.putLong(key, value)
151                     is Float -> e.putFloat(key, value)
152                     is Set<*> -> e.putStringSet(key, value.map { it.toString() }.toSet())
153                 }
154             }
155             e.apply()
156         }
157         // Everything that holds a copy of the old settings has to look again.
158         Store.invalidate()
159         Channels.syncComponents(this)
160         (ServiceHolder.service as? KeyFilterService)?.let {
161             it.reloadTiming()
162             it.syncPower()
163         }
164         toast("Imported")
165         build()
166     }

168     /** Keeps only the bindings whose trigger this version understands. */
169     private fun cleanBindings(json: String): String {
170         val source = runCatching { JSONObject(json) }.getOrNull() ?: return "{}"
171         val out = JSONObject()
172         source.keys().forEach { id ->
173             val spec = source.optJSONObject(id)
174             if (Trigger.parse(id) != null && spec != null) out.put(id, spec)
175         }
176         return out.toString()
177     }
