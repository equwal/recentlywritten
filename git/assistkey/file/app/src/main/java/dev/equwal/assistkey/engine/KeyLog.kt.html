3 import android.view.KeyEvent

5 /**
6  * A short record of what the key filter actually saw.
7  *
8  * Worth having permanently rather than as a debugging aid: whether a given key
9  * reaches an input filter at all is a per-firmware question with no way to look
10  * it up. Keys consumed upstream by the window manager - Power everywhere, and
11  * on some builds an OEM key too - simply never appear here, and that absence is
12  * the answer.
13  *
14  * Injected events (adb shell input keyevent) bypass accessibility input filters
15  * entirely, so only a real press proves anything.
16  */
17 object KeyLog {
18 
19     data class Entry(
20         val at: Long,
21         val keyCode: Int,
22         val action: Int,
23         val repeat: Int,
24         val label: String,
25         val consumed: Boolean
26     ) {
27         fun describe(): String {
28             val act = when (action) {
29                 KeyEvent.ACTION_DOWN -> if (repeat > 0) "down x" + repeat else "down"
30                 KeyEvent.ACTION_UP -> "up"
31                 else -> "action " + action
32             }
33             return label + "  " + act + "  " + (if (consumed) "consumed" else "passed on")
34         }
35     }
36 
37     private const val CAPACITY = 60
38 
39     private val entries = ArrayDeque<Entry>(CAPACITY)
40 
41     /**
42      * Only the tester screen turns this on, and only while it is in front. A
43      * key filter is handed every key on the device, a Bluetooth keyboard's
44      * included, and the app's promise is that it keeps none of them - so
45      * outside that screen nothing is remembered, even in memory.
46      */
47     @Volatile
48     var recording = false
49 
50     @Synchronized
51     fun record(event: KeyEvent, label: String, consumed: Boolean) {
52         if (!recording) return
53         if (entries.size >= CAPACITY) entries.removeFirst()
54         entries.addLast(
55             Entry(
56                 at = System.currentTimeMillis(),
57                 keyCode = event.keyCode,
58                 action = event.action,
59                 repeat = event.repeatCount,
60                 label = label,
61                 consumed = consumed
62             )
63         )
64     }
65 
66     /** Newest first, which is the order the tester screen wants. */
67     @Synchronized
68     fun recent(): List<Entry> = entries.reversed()
69 
70     @Synchronized
71     fun clear() = entries.clear()
72 
73     @Synchronized
74     fun count(): Int = entries.size
75 }
