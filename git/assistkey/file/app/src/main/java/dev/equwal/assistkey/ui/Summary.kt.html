5 /**
6  * The short state lines the hub shows: one line for each tile, and one line
7  * for each binding in the "Your buttons" list.
8  *
9  * Every function here takes plain values and returns text. It holds no Android
10  * type, so the wording is tested on the computer.
11  */
12 object Summary {
13 
14     /** What a button does now, under its name in the drawing: "Back · Recent apps". */
15     fun caption(actions: List<String>): String =
16         if (actions.isEmpty()) "Not set" else actions.distinct().joinToString(" · ")
17 
18     /** How many gestures are bound, over how many buttons. */
19     fun keys(keyCount: Int, boundCount: Int): String {
20         if (boundCount == 0) return "Nothing bound yet"
21         return count(boundCount, "action") + " on " + count(keyCount, "button")
22     }
23 
24     /** The detection row: how many buttons the last detection found. */
25     fun detected(keyCount: Int?): String = when {
26         keyCount == null -> "Not run yet"
27         keyCount == 0 -> "The device declares no named keys"
28         else -> count(keyCount, "key") + " declared by the device"
29     }
30 
31     /** The Power button, which has its own screen. */
32     fun power(boundCount: Int): String =
33         if (boundCount == 0) "Default behaviour" else count(boundCount, "action") + " bound"
34 
35     /** Which ways of getting around are switched on. */
36     fun navigation(buttons: Boolean, gestures: Boolean, keys: Boolean): String {
37         val parts = ArrayList<String>(3)
38         if (buttons) parts.add("Bar")
39         if (gestures) parts.add("Gestures")
40         if (keys) parts.add("Power button")
41         if (parts.isEmpty()) return "Nothing switched on"
42         return parts.joinToString(" · ")
43     }
44 
45     /** Whether voice typing can work, and what is missing if it cannot. */
46     fun voice(microphone: Boolean, speechApps: Int, onDevice: Boolean): String = when {
47         speechApps == 0 -> "Needs a speech app"
48         !microphone -> "Needs the microphone"
49         onDevice -> "Ready, on this device"
50         else -> "Ready"
51     }
52 
53     /** The home screens and Ink Recents: how many of them are installed. */
54     fun homeAndRecents(installed: Int, offered: Int): String = when (installed) {
55         0 -> "Home screens, Ink Recents"
56         offered -> "All installed"
57         else -> installed.toString() + " of " + offered + " installed"
58     }
59 
60     /**
61      * Setup shows the blocker first. Button remapping being off stops everything
62      * else, so it is said before any count of permissions.
63      */
64     fun setup(keyRemapping: Boolean, granted: Int, total: Int): String = when {
65         !keyRemapping -> "Button remapping is off"
66         granted >= total -> "All set"

68     }
69 
70     /** The licence chip in the app bar. Short, because it sits beside the name. */
71     fun licenceChip(tier: License.Tier, trialDaysLeft: Int): String = when (tier) {
72         License.Tier.LICENSED -> "Unlocked"
73         License.Tier.NO_STORE -> "Free"
74         License.Tier.BETA -> "Beta"
75         License.Tier.TRIAL -> count(trialDaysLeft, "day") + " left"
76         License.Tier.LOCKED -> "Locked"
77     }

79     /** "1 key", "3 keys". */
