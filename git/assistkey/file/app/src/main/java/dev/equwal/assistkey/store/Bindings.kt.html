9 /**
10  * An immutable snapshot of every binding, with the lookups the key-event hot
11  * path needs precomputed. Rebuilt on save rather than queried per keystroke.
12  */
13 class Bindings(private val map: Map<Trigger, ActionSpec>) {
14 
15     private val active: Map<Trigger, ActionSpec> =
16         map.filterValues { it.kind != ActionKind.PASS_THROUGH }
17 
18     /** What the gesture engine is told, for one answer to "can it see Power?". */
19     private class View(triggers: Collection<Trigger>) {
20         val taps: Map<Set<HwKey>, Int> =
21             triggers.filter { it.type == GestureType.TAP }
22                 .groupBy { it.keys }
23                 .mapValues { (_, v) -> v.maxOf { it.count } }
24 
25         val holds: Set<Set<HwKey>> =
26             triggers.filter { it.type == GestureType.HOLD }.map { it.keys }.toSet()
27 
28         /** For each key, the other keys it forms a bound chord with. */
29         val partners: Map<HwKey, Set<HwKey>> = buildMap {
30             triggers.filter { it.isChord }.forEach { t ->
31                 t.keys.forEach { k -> merge(k, t.keys - k) { a, b -> a + b } }
32             }
33         }
34     }
35 
36     /**
37      * Two views, because whether the engine may know about Power depends on
38      * whether Power presses can reach it. Without shell access they cannot, and
39      * an engine that believed in a Power chord would hold every volume press
40      * back waiting for a partner that never arrives. With it, Power is a key
41      * like any other.
42      */
43     private val withPower = View(active.keys)
44     private val withoutPower = View(active.keys.filter { HwKey.POWER !in it.keys })
45 
46     private fun view(power: Boolean) = if (power) withPower else withoutPower
47 
48     fun maxTaps(keys: Set<HwKey>, power: Boolean = false): Int = view(power).taps[keys] ?: 0
49     fun hasHold(keys: Set<HwKey>, power: Boolean = false): Boolean = keys in view(power).holds
50     fun chordPartners(key: HwKey, power: Boolean = false): Set<HwKey> =
51         view(power).partners[key] ?: emptySet()
52 
53     operator fun get(trigger: Trigger): ActionSpec? = active[trigger]
54     fun isBound(trigger: Trigger): Boolean = trigger in active
55 
56     /** The action for "hold Power, then press [key]", if there is one. */
57     fun powerCombo(key: HwKey): ActionSpec? = active[Trigger.powerThen(key)]
58 
59     val hasPowerCombos: Boolean =
60         active.keys.any { it.keys.size == 2 && HwKey.POWER in it.keys }
61 
62     /** Including PASS_THROUGH entries, for the UI. */
63     fun raw(trigger: Trigger): ActionSpec = map[trigger] ?: ActionSpec.PASS
64     fun all(): Map<Trigger, ActionSpec> = map
65 
66     fun with(trigger: Trigger, spec: ActionSpec): Bindings =
67         Bindings(map.toMutableMap().apply {
68             if (spec.kind == ActionKind.PASS_THROUGH) remove(trigger) else put(trigger, spec)
69         })
70 
71     companion object {
72         val EMPTY = Bindings(emptyMap())
