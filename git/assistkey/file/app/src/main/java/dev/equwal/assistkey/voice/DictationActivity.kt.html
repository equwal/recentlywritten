79      * The other way to listen: the speech screen of the speech app, which

82     private fun listenOnScreen(): Boolean {
83         val engine = Dictation.engine(this) ?: return false
84         val ask = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).setPackage(engine.packageName)
85             .putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
86         Dictation.language(this).takeIf { it.isNotEmpty() }?.let { ask.putExtra(RecognizerIntent.EXTRA_LANGUAGE, it) }
87         if (ask.resolveActivity(packageManager) == null) return false
88         return runCatching { startActivityForResult(ask, SCREEN) }.isSuccess.also { onScreen = it }
89     }

103     private fun listen() {
104         val r = runCatching {
105             Dictation.engine(this)?.let { SpeechRecognizer.createSpeechRecognizer(this, it) }
106                 ?: SpeechRecognizer.createSpeechRecognizer(this)
107         }.getOrNull()
108         if (r == null) {
109             Dictation.toast(this, "The speech app did not start")
110             return done(null)
111         }
112         recognizer = r
113         Dictation.stop = { r.stopListening() }
114         r.setRecognitionListener(listener)
115         r.startListening(
116             Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
117                 putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
118                 putExtra(RecognizerIntent.EXTRA_PARTIAL_RESULTS, true)
119                 putExtra(RecognizerIntent.EXTRA_CALLING_PACKAGE, packageName)
120                 Dictation.language(this@DictationActivity).takeIf { it.isNotEmpty() }
121                     ?.let { putExtra(RecognizerIntent.EXTRA_LANGUAGE, it) }
122             }
123         )
124     }

126     private val listener = object : RecognitionListener {
127         override fun onResults(results: Bundle?) =
128             done(results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)?.firstOrNull())
129 
130         override fun onError(error: Int) {
131             Log.i("AssistKey", "speech recognition error " + error)
132             val engine = Dictation.engine(this@DictationActivity)
133             if (Dictation.fallBackToScreen(error, screenAvailable = engine != null)) {
134                 close()
135                 if (listenOnScreen()) {
136                     Dictation.rememberScreen(this@DictationActivity, engine)
137                     return
138                 }
139             }
140             Dictation.toast(
141                 this@DictationActivity,
142                 when (error) {
143                     SpeechRecognizer.ERROR_NO_MATCH, SpeechRecognizer.ERROR_SPEECH_TIMEOUT -> "Heard nothing"
144                     SpeechRecognizer.ERROR_INSUFFICIENT_PERMISSIONS -> "The microphone is not allowed"
145                     SpeechRecognizer.ERROR_NETWORK, SpeechRecognizer.ERROR_NETWORK_TIMEOUT ->
146                         "The speech app has no connection"
147                     else -> "Speech failed (" + error + ")"
148                 }
149             )
150             done(null)
151         }
152 
153         override fun onPartialResults(partial: Bundle?) {
154             partial?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)?.firstOrNull()
155                 ?.takeIf { it.isNotBlank() }?.let { label.text = it }
156         }

158         override fun onEndOfSpeech() { label.text = "Working..." }
159         override fun onReadyForSpeech(params: Bundle?) = Unit
160         override fun onBeginningOfSpeech() = Unit
161         override fun onRmsChanged(rmsdB: Float) = Unit
162         override fun onBufferReceived(buffer: ByteArray?) = Unit
163         override fun onEvent(eventType: Int, params: Bundle?) = Unit
164     }

172     /** Leaves first, then writes: the field must be the front window again. */
173     private fun done(text: String?) {
174         Log.i("AssistKey", "voice typing: " + if (text.isNullOrBlank()) "no text" else text.length.toString() + " characters")
175         close()
176         finish()
177         @Suppress("DEPRECATION")
178         overridePendingTransition(0, 0)
179         if (text.isNullOrBlank()) return
180         val app = applicationContext
181         Handler(Looper.getMainLooper()).postDelayed({
182             val svc = ServiceHolder.service ?: return@postDelayed
183             when (TextInsert.insert(svc, text)) {
184                 TextInsert.Result.DONE -> Unit
185                 TextInsert.Result.NO_FIELD -> Dictation.toast(app, "No text field has the cursor")
186                 TextInsert.Result.PASSWORD -> Dictation.toast(app, "Not into a password field")
187                 TextInsert.Result.REFUSED -> Dictation.toast(app, "That field did not accept the text")
188             }
189         }, INSERT_DELAY_MS)
190     }
