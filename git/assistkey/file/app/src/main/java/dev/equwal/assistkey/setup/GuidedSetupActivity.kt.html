40 /**
41  * Set up one button by saying what you want, in four short steps.
42  *
43  *  1. Which button: tap it in the drawing. Two buttons make a combination.
44  *     The main screen has the same drawing, and then this step is skipped.
45  *  2. How you press it: tap, double tap, hold, and the longer tap counts.
46  *  3. What it does.
47  *  4. Allow what that needs - and only that.
48  *
49  * The user never chooses a service, a role or a channel. [Route] works out
50  * which of them the wish needs, and step 4 asks for those by what they do.
51  * Everything here ends in an ordinary binding, so the full-control screens
52  * show and edit the same thing.
53  */

80         // From Android 13, the system does not call onBackPressed for an app
81         // with this target version. It calls this callback. Without it, Back
82         // always closed the screen, whatever the step.

141     private fun build() {
142         val heading = when (step) {
143             Step.BUTTON -> "Which button?"
144             Step.PRESS -> keys.joinToString(" + ") { it.label }.ifEmpty { "Which button?" }
145             Step.ACTION -> trigger?.label() ?: "What should it do?"
146             Step.ALLOW -> "Almost there"
147             Step.DONE -> "Done"
148         }
149         val col = Ui.page(this, heading)
150         when (step) {
151             Step.BUTTON -> whichButton(col)
152             Step.PRESS -> press(col)
153             Step.ACTION -> action(col)
154             Step.ALLOW -> allow(col)
155             Step.DONE -> done(col)
156         }

161     private fun whichButton(col: LinearLayout) {
162         col.note("Tap two for a combination.")
163         val view = DeviceView(this).apply {
164             keys = drawn(this@GuidedSetupActivity)
165             selected = this@GuidedSetupActivity.keys
166             onSelect = { this@GuidedSetupActivity.keys = it; build() }
167         }
168         col.addView(view)
169         // A device can have buttons the drawing does not know. The user names the button.
170         val others = HwKey.interceptable.filterNot { it in view.keys }
171         if (others.isNotEmpty()) {
172             col.row("My button is not in the drawing", keys.filter { it in others }.joinToString { it.label }.ifEmpty { null }) {
173                 Ui.pick(this, "Which button?", others.map { it.label }) { i ->
174                     keys = Route.toggle(keys, others[i])
175                     build()
176                 }
177             }
178         }
179         if (keys.isNotEmpty()) {
180             col.primaryButton("Next: " + keys.joinToString(" + ") { it.label }) { go(Step.PRESS) }
181         }

186     private fun press(col: LinearLayout) {
187         if (keys.isEmpty()) return go(Step.BUTTON)
188         val set = keys.toSet()
189         col.header("How do you press it?")
190         val b = Store.bindings(this)
191         val offered = Route.gestures(set, env)
192         offered.forEach { t ->
193             val state = b.raw(t).describe().takeIf { b.isBound(t) }?.let { "Now: $it" }
194                 ?: "Needs shell access".takeIf { Need.SHELL_POWER in Route.plan(t, false, env).needs }
195             col.row(gestureName(t), state) {
196                 trigger = t
197                 go(Step.ACTION)
198             }
199         }
200         when {
201             HwKey.SCREEN in set -> col.note("A small button floats over every app. Drag to move it.")
202             set == setOf(HwKey.POWER) && !env.powerIsDirect && !env.shellSupported ->
203                 col.note("A tap cannot reach any app. Hold and double tap work.")
204         }
205         if (HwKey.SCREEN !in set) {
206             col.row("Advanced", "Every gesture of this button") {
207                 startActivity(
208                     if (HwKey.POWER in set) Intent(this, PowerActivity::class.java)
209                     else TriggerListActivity.intent(this, set)
210                 )
211             }
212         }

286     private fun allow(col: LinearLayout) {
287         val t = trigger ?: return go(Step.PRESS)
288         val spec = Store.bindings(this)[t] ?: return go(Step.ACTION)
289         val needsFilter = ActionRouter.requiresAccessibility(spec) || spec.kind == ActionKind.MENU
290         val plan = Route.plan(t, needsFilter, env)
291 
292         // The switches that go with the route, set without a question.
293         if (HwKey.POWER in t.keys && env.powerIsDirect) (ServiceHolder.service as? KeyFilterService)?.syncPower()
294         Channels.setEnabled(this, Channel.ACCESSIBILITY, true)
295 
296         val missing = plan.needs.filterNot(::satisfied)
297         val microphone = spec.kind == ActionKind.VOICE && !Dictation.hasMicrophone(this)
298         val shell = needsShell(spec) && !Shell.ready
299         if (plan.possible && missing.isEmpty() && !microphone && !shell) return if (askOnly) finish() else go(Step.DONE)
300 
301         plan.blocked?.let {
302             col.note(it)
303             col.button("Press it another way") { go(Step.PRESS) }
304             return
305         }
306         col.note(t.label() + " will do: " + spec.describe() + ". Allow this first:")
307         missing.forEach { n -> col.row(n.title, n.why) { ask(n) } }
308         if (microphone) {
309             col.row("Allow the microphone", "For Voice typing") {
310                 requestPermissions(arrayOf(Manifest.permission.RECORD_AUDIO), 1)
311             }
312         }
313         if (shell) {
314             col.row("Turn on shell access", "This action works through Shizuku") {
315                 startActivity(Intent(this, ShellActivity::class.java))
316             }
317         }
318         if (Need.KEY_FILTER in missing) {
319             col.note("If the switch turns itself off, allow restricted settings in App info.")
320             col.button("Open App info") { Channels.safeStart(this, Channels.appInfoIntent(this)) }
321         }
322     }

331     private fun done(col: LinearLayout) {
332         val t = trigger
333         val spec = t?.let { Store.bindings(this)[it] }
334         if (t != null && spec != null) col.note(t.label() + " now does: " + spec.describe() + ". Try it.")
335         if (spec?.kind == ActionKind.VOICE && Dictation.engines(this).none(Dictation::isOnDevice)) {
336             col.note("See Voice typing for an on-device speech app.")
337         }
338         // Some firmware sends the double press of Power to the wallet app, not to the camera app.
339         val powerDouble = t == Trigger(setOf(HwKey.POWER), GestureType.TAP, 2) && !env.powerIsDirect
340         if (powerDouble && !Channels.isSatisfied(this, Channel.WALLET)) {
341             col.row("The double tap does nothing?", "Take the wallet place too") {
342                 Channels.setEnabled(this, Channel.WALLET, true)
343                 Channels.safeStart(this, Channels.claimIntent(this, Channel.WALLET))
344             }
345         }
346         col.button("Change what it does") { go(Step.ACTION) }
347         // The on-screen button has one gesture, so there is nothing to change here.
348         if (keys.singleOrNull() != HwKey.SCREEN) {
349             col.button("Change how you press it") { go(Step.PRESS) }
350         }
351         col.button("Set up another button") {
352             keys = emptyList()
353             trigger = null
354             fromHub = false
355             go(Step.BUTTON)
356         }
357         col.primaryButton("Finish") { finish() }
358     }

360     companion object {
361         private const val EXTRA_KEYS = "keys"
362         private const val EXTRA_ASK = "ask"

364         /**
365          * For any code that has just made a binding for [t], or has just tried
366          * to run it. Opens the allow step when the binding needs something the
367          * user has not allowed yet. True when it opened the step.
368          */
369         fun askIfMissing(c: android.content.Context, t: Trigger): Boolean {
370             if (!missing(c, t)) return false
371             val ask = Intent(c, GuidedSetupActivity::class.java).putExtra(EXTRA_ASK, t.id)
372             // A service or a channel entry is not a screen. It needs a task of its own.
373             if (c !is Activity) ask.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
374             return runCatching { c.startActivity(ask) }.isSuccess
375         }
376 
377         /** True when the binding of [t] needs something that the user has not allowed yet. */
378         fun missing(c: android.content.Context, t: Trigger): Boolean {
379             val spec = Store.bindings(c)[t] ?: return false
380             val needsFilter = ActionRouter.requiresAccessibility(spec) || spec.kind == ActionKind.MENU
381             val plan = Route.plan(t, needsFilter, env(c))
382             val microphone = spec.kind == ActionKind.VOICE && !Dictation.hasMicrophone(c)
383             val shell = needsShell(spec) && !Shell.ready
384             return !(plan.possible && plan.needs.all { satisfied(c, it) } && !microphone && !shell)
385         }
386 
387         /** The light and the button bar are set by shell commands. */
388         private fun needsShell(spec: ActionSpec): Boolean =
389             Shell.SUPPORTED && (spec.kind == ActionKind.NAV || spec.kind == ActionKind.DIM)
390 
391         private fun env(c: android.content.Context): Route.Env =
392             Route.Env(powerIsDirect = Shell.ready && PowerControl.wanted(c), shellSupported = Shell.SUPPORTED)
393 
394         private fun satisfied(c: android.content.Context, n: Need): Boolean = when (n) {
395             Need.KEY_FILTER -> Channels.isSatisfied(c, Channel.ACCESSIBILITY)
396             Need.ASSISTANT -> Channels.isSatisfied(c, Channel.ASSISTANT)
397             Need.CAMERA -> Channels.isSatisfied(c, Channel.CAMERA)
398             Need.SHELL_POWER -> env(c).powerIsDirect
399         }
