24 /**
25  * A description of this device, for getting it supported.
26  *
27  * Every device maker wires keys, gestures and lights differently, and the only
28  * way to support one is to know what it has. This screen gathers that - model,
29  * firmware, the input devices and the keys they declare, the navigation
30  * overlays, a handful of key-related settings - shows every line of it, and
31  * sends it nowhere. The user sends it, by email or however they like, if they
32  * choose to. The app has no internet permission and could not send it itself.
33  *
34  * What is deliberately not in it: installed apps, accounts, identifiers, the
35  * contents of anything, and any key that was actually pressed outside the key
36  * tester.
37  */
38 class ReportActivity : Activity() {
39 
40     private var report = ""
41     private val to = "truex@equwal.com"
42 
43     override fun onResume() {
44         super.onResume()
45         build("Gathering...")
46         gather { text ->
47             report = text
48             if (!isFinishing) build(text)
49         }

52     private fun build(text: String) {
53         val col = Ui.page(this, "Device report")
54         col.note("Nothing is sent unless you send it.")
55         col.primaryButton("Send by email") {
56             val i = Intent(Intent.ACTION_SENDTO, Uri.parse("mailto:" + to))
57                 .putExtra(Intent.EXTRA_SUBJECT, "Rebind device report: " + Device.name)
58                 .putExtra(Intent.EXTRA_TEXT, report)
59             if (runCatching { startActivity(i) }.isFailure) share()
60         }
61         col.button("Share another way") { share() }
62         col.button("Copy") {
63             getSystemService(ClipboardManager::class.java)
64                 ?.setPrimaryClip(ClipData.newPlainText("Rebind device report", report))
65             Toast.makeText(this, "Copied", Toast.LENGTH_SHORT).show()
66         }
67         col.header("The report")
68         col.code(text)
69         if (Shell.SUPPORTED && !Shell.ready) {
70             col.note("Shell access adds the input devices and their buttons.")
71         }
72     }

84         }

89     private fun gather(done: (String) -> Unit) {
90         val sb = StringBuilder()
91         fun line(k: String, v: Any?) { sb.append(k).append(": ").append(v).append('\n') }
92 
93         sb.append("# Rebind device report\n")
94         line("app", BuildConfig.VERSION_NAME + " (" + BuildConfig.VERSION_CODE + ")")
95         line("manufacturer", Build.MANUFACTURER)
96         line("brand", Build.BRAND)
97         line("model", Build.MODEL)
98         line("device", Build.DEVICE)
99         line("product", Build.PRODUCT)
100         line("android", Build.VERSION.RELEASE + " (SDK " + Build.VERSION.SDK_INT + ")")
101         line("firmware", Build.DISPLAY)
102         line("build type", Build.TYPE)
103         line("profile", Device.profile)
104         val dm = resources.displayMetrics
105         line("screen", dm.widthPixels.toString() + " x " + dm.heightPixels + " @ " + dm.densityDpi + " dpi")
106 
107         sb.append("\n# State\n")
108         line("key filter", Channels.status(this, Channel.ACCESSIBILITY))
109         line("shell access", Shell.state(this))
110         line("power button managed", PowerControl.active)
111         line("button bar showing", NavNative.buttonsShowing(this))
112         line("keys shown", Device.keys(this).joinToString { it.token })
113         line("keys seen in tester", Device.unknownSeen(this).ifEmpty { listOf("none") }.joinToString("; "))
114 
115         // What detection found by asking the device. No key press is in this.
116         sb.append("\n# Detection\n")
117         val found = Detect.stored(this)
118         if (found == null) {
119             line("detection", "not run yet")
120         } else {
121             line("keys found", found.keys.joinToString { it.token }.ifEmpty { "none" })
122             Detect.capabilityNames.forEach { name -> line(name.lowercase(), found.capability(name).label.lowercase()) }
123         }
124 
125         if (!Shell.ready) return done(sb.toString())
126 
127         // Names and shapes only. The settings query is a whitelist of key and
128         // navigation switches; per-app settings, which would name installed
129         // apps, are left out on purpose.
130         val wanted = "custom.*key|power_button|key_chord|navigation_mode|disable_gesture_bottom|" +
131             "double_tap_power|camera_double_tap|screen_brightness\$|back_gesture_inset"
132         val script = listOf(
133             "echo '# Input devices'; getevent -p 2>/dev/null | grep -E 'add device|name:|KEY \\(0001\\)|^ {20}[0-9a-f]' | head -80",
134             "echo; echo '# Navigation overlays'; cmd overlay list 2>/dev/null | grep -i navbar",
135             "echo; echo '# Settings'; for t in system secure global; do settings list \$t 2>/dev/null | grep -iE '$wanted' | grep -v 'disable_gesture_[a-z]*\\.' | sed \"s/^/[\$t] /\"; done",
136             "echo; echo '# Light'; for n in /sys/class/leds/lcd-backlight /sys/class/backlight/*; do [ -e \$n/brightness ] && echo \$n now=\$(cat \$n/brightness) max=\$(cat \$n/max_brightness 2>/dev/null) min=\$(cat \$n/min_brightness 2>/dev/null); done",
137             "echo; echo '# Shell'; id | cut -d' ' -f1"
138         ).joinToString("; ")
139         Shell.run(script) { r ->
140             sb.append('\n').append(r.output.trim()).append('\n')
141             done(sb.toString())
142         }
143     }
