25 /**
26  * The menu on screen: a box of actions over the app in front.
27  *
28  * A tap on an action closes the menu first and runs the action a moment later.
29  * The order matters: Back, a swipe or voice typing act on the app in front, and
30  * until the menu is gone the app in front is the menu. A tap outside the box
31  * closes the menu and does nothing.
32  */

35     override fun onCreate(savedInstanceState: Bundle?) {
36         super.onCreate(savedInstanceState)
37         val items = Menu.decode(intent.getStringExtra(EXTRA_ITEMS).orEmpty())
38         if (items.isEmpty()) return leave()

40         val list = LinearLayout(this).apply { orientation = LinearLayout.VERTICAL }
41         items.forEachIndexed { i, spec ->
42             if (i > 0) list.addView(View(this).apply { setBackgroundColor(Color.BLACK) }, LinearLayout.LayoutParams(MATCH_PARENT, dp(1)))
43             list.addView(TextView(this).apply {
44                 text = spec.describe()
45                 setTextColor(Color.BLACK)
46                 setTextSize(TypedValue.COMPLEX_UNIT_SP, 20f)
47                 setPadding(dp(20), dp(18), dp(20), dp(18))
48                 setOnClickListener { pick(spec) }
49             })
50         }
51         val box = ScrollView(this).apply {
52             background = GradientDrawable().apply {
53                 setColor(Color.WHITE)
54                 setStroke(dp(2), Color.BLACK)
55                 cornerRadius = dp(12).toFloat()
56             }
57             val p = dp(3)
58             setPadding(p, p, p, p)
59             addView(list, LinearLayout.LayoutParams(MATCH_PARENT, WRAP_CONTENT))
60         }
61         setContentView(
62             FrameLayout(this).apply {
63                 setOnClickListener { leave() }
64                 addView(

68             }

72     private fun pick(spec: ActionSpec) {
73         val app = applicationContext
