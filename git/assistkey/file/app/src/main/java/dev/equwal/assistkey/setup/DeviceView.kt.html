15 /**
16  * A drawing of the device with its buttons along one edge.
17  *
18  * It is the way to choose buttons: tap a button in the drawing to select it,
19  * tap it again to let it go. Two buttons can be selected together, which makes
20  * a combination. [Route.toggle] holds the rules. The buttons are drawn from the
21  * top down in the order of [keys], which the device profile gives. The
22  * on-screen button is drawn on the screen of the device, because that is
23  * where it is. A selected button is filled black and its name is bold.
24  *
25  * Drawn for e-ink: black lines on white, no shading, no animation.
26  */
27 class DeviceView(context: Context, private val heightDp: Int = 300) : View(context) {

30         set(value) { field = value; invalidate() }

33         set(value) { field = value; invalidate() }
34 
35     /** A short second line under the name of a button, for example what it does now. */
36     var captions: Map<HwKey, String> = emptyMap()
37         set(value) { field = value; invalidate() }

39     var onSelect: ((List<HwKey>) -> Unit)? = null

41     private val d = resources.displayMetrics.density
42     private val line = Paint(Paint.ANTI_ALIAS_FLAG).apply {
43         style = Paint.Style.STROKE
44         strokeWidth = 2.5f * d
45         color = Color.BLACK
46     }
47     private val fill = Paint(Paint.ANTI_ALIAS_FLAG).apply { style = Paint.Style.FILL; color = Color.BLACK }
48     private val paper = Paint(Paint.ANTI_ALIAS_FLAG).apply { style = Paint.Style.FILL; color = Color.WHITE }

50     private val caption = TextPaint(Paint.ANTI_ALIAS_FLAG).apply { color = 0xFF444444.toInt(); textSize = 13f * d }
51     private val rows = ArrayList<Pair<HwKey, RectF>>()

53     override fun onTouchEvent(event: MotionEvent): Boolean {
54         if (event.actionMasked == MotionEvent.ACTION_UP) {
55             rows.firstOrNull { it.second.contains(event.x, event.y) }?.let { (key, _) ->
56                 selected = Route.toggle(selected, key)
57                 onSelect?.invoke(selected)
58             }
59             performClick()
60         }
61         return true
62     }

71     override fun onDraw(canvas: Canvas) {
72         val w = width.toFloat()
73         val h = height.toFloat()
74         val body = RectF(w * 0.06f, 6 * d, w * 0.40f, h - 6 * d)
75         val corner = 18 * d
76         canvas.drawRoundRect(body, corner, corner, paper)
77         canvas.drawRoundRect(body, corner, corner, line)
78         val inset = 12 * d
79         canvas.drawRect(body.left + inset, body.top + inset * 1.6f, body.right - inset, body.bottom - inset * 1.6f, line)
80 
81         rows.clear()
82         if (keys.isEmpty()) return
83         val slot = (body.height() - 2 * corner) / keys.size
84         // A device with many buttons gives each name less room. Shrink the
85         // name to the slot so the list still reads.
86         val twoLines = captions.isNotEmpty()
87         label.textSize = minOf(17f * d, slot * (if (twoLines) 0.42f else 0.7f))
88         caption.textSize = minOf(13f * d, slot * 0.32f)
89         val textLeft = w * 0.50f
90         val textWidth = w - textLeft
91         keys.forEachIndexed { i, key ->
92             val cy = body.top + corner + slot * (i + 0.5f)
93             val chosen = key in selected
94             val leaderFrom: Float
95             if (key == HwKey.SCREEN) {
96                 // A round button on the screen of the device.
97                 val r = minOf(slot * 0.30f, 13 * d)
98                 val cx = body.right - inset - r - 6 * d
99                 canvas.drawCircle(cx, cy, r, if (chosen) fill else paper)
100                 canvas.drawCircle(cx, cy, r, line)
101                 leaderFrom = cx + r
102             } else {
103                 val half = minOf(slot * 0.32f, 20 * d)
104                 val nub = RectF(body.right - 1 * d, cy - half, body.right + 12 * d, cy + half)
105                 canvas.drawRoundRect(nub, 4 * d, 4 * d, if (chosen) fill else paper)
106                 canvas.drawRoundRect(nub, 4 * d, 4 * d, line)
107                 leaderFrom = nub.right
108             }
109             // A thin leader from the button to its name.
110             canvas.drawLine(leaderFrom, cy, textLeft - 8 * d, cy, line)
111             label.typeface = if (chosen) Typeface.DEFAULT_BOLD else Typeface.DEFAULT
112             val note = captions[key]
113             val base = if (note == null) cy + label.textSize * 0.35f else cy - 2 * d
114             canvas.drawText(key.label, textLeft, base, label)
115             if (note != null) {
116                 val cut = TextUtils.ellipsize(note, caption, textWidth, TextUtils.TruncateAt.END).toString()
117                 canvas.drawText(cut, textLeft, base + caption.textSize * 1.35f, caption)
118             }
119             rows.add(key to RectF(body.right - inset - 40 * d, cy - slot / 2, w, cy + slot / 2))
120         }
121     }
