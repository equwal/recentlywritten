14 /**
15  * Builds a key set, then hands it to the ordinary gesture list.

17  * Power is absent on purpose: it never reaches the input filter, so it can

29     private fun build() {
30         val col = Ui.page(this, "Button combinations")
31         col.note("Hold one, press the others.")
32 
33         col.header("Pick the buttons")
34         HwKey.entries.filter { it.interceptable }.forEach { key ->
35             col.check(key.label, null, key in picked) { on ->
36                 if (on) picked.add(key) else picked.remove(key)
37                 build()
38             }
39         }
40 
41         if (picked.size >= 2) {
42             col.primaryButton("Set up " + picked.joinToString(" + ") { it.label }) {
43                 startActivity(TriggerListActivity.intent(this, picked.toSet()))
44             }
45         } else {
46             col.note("Choose at least two buttons.")
47         }
48 
49         existing(col)
50 
51     }
52 
53     /** Anything already bound, so a chord is easy to find again. */
54     private fun existing(col: LinearLayout) {
55         val sets = Store.bindings(this).all().keys
56             .filter { it.isChord && dev.equwal.assistkey.model.HwKey.POWER !in it.keys }
57             .map { it.keys }
58             .distinct()
59         if (sets.isEmpty()) return
60 
61         col.header("Already configured")
62         sets.forEach { keys ->
63             val n = Trigger.allFor(keys).count { Store.bindings(this).isBound(it) }
64             col.row(
65                 keys.sortedBy { it.ordinal }.joinToString(" + ") { it.label },
66                 n.toString() + (if (n == 1) " gesture bound" else " gestures bound")
67             ) { startActivity(TriggerListActivity.intent(this, keys)) }
68         }
69     }
