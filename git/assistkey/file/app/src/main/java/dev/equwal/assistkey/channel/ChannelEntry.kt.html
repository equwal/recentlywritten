28     fun handle(activity: Activity, channel: Channel) {
29         val licensed = License.active(activity)
30 
31         // A Power hold with combinations bound does not act yet: it opens a
32         // short window in which the next key press decides what happens.
33         if (channel == Channel.ASSISTANT && Store.bindings(activity).hasPowerCombos) {
34             val svc = ServiceHolder.service as? KeyFilterService
35             if (svc != null) {
36                 val plain = channel.trigger?.let { Store.bindings(activity)[it] }
37                 svc.armPowerCombo(plain, lifeline = !licensed)
38                 activity.finishAndVanish()
39                 return
40             }
41         }
42 
43         if (!licensed) {
44             Toast.makeText(
45                 activity,
46                 "Rebind is locked. Open the app.",
47                 Toast.LENGTH_SHORT
48             ).show()
49             activity.finishAndVanish()
50             return
51         }
52 
53         val trigger = channel.trigger
54         val spec = trigger?.let { Store.bindings(activity)[it] }
55 
56         if (spec == null) {
57             Log.i(TAG, "channel " + channel.key + " fired with nothing bound")
58             activity.finishAndVanish()
59             return
60         }
61 
62         if (ActionRouter.requiresAccessibility(spec)) {
63             if (!ServiceHolder.isRunning) {
64                 // The user tried the button, and something is not allowed yet. Ask now.
65                 dev.equwal.assistkey.setup.GuidedSetupActivity.askIfMissing(activity.applicationContext, trigger)
66                 activity.finishAndVanish()
67                 return
68             }
69             val app = activity.applicationContext
70             activity.finishAndVanish()
71             Handler(Looper.getMainLooper()).postDelayed({ ActionRouter.run(app, spec) }, 150L)
72         } else {
73             ActionRouter.run(activity, spec)
74             activity.finishAndVanish()
75         }
76     }
77

79     private fun Activity.finishAndVanish() {
80         finish()
81         @Suppress("DEPRECATION")
