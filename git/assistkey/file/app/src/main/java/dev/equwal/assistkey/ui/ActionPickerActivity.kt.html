88     private fun choose(spec: ActionSpec) {
89         if (pickOnly) {
90             setResult(RESULT_OK, Intent().putExtra(RESULT_SPEC, spec.toJson().toString()))
91         } else {
92             Store.bind(this, trigger, spec)
93             // Ask now for what this binding needs. The screen closes itself when nothing is missing.
94             if (spec.kind != ActionKind.PASS_THROUGH && !intent.getBooleanExtra(EXTRA_NO_ASK, false)) {
95                 dev.equwal.assistkey.setup.GuidedSetupActivity.askIfMissing(this, trigger)
96             }
97         }
98         finish()
99     }

209     /**
210      * Only launchable apps, which is also exactly the set the manifest
211      * <queries> block makes visible to us.
212      */
213     private fun pickApp() {
214         val pm = packageManager
215         val main = Intent(Intent.ACTION_MAIN).addCategory(Intent.CATEGORY_LAUNCHER)
216         val apps = pm.queryIntentActivities(main, 0)
217             .map { it.loadLabel(pm).toString() to it.activityInfo.packageName }
218             .distinctBy { it.second }
219             .sortedBy { it.first.lowercase() }
220 
221         if (apps.isEmpty()) {
222             Ui.textInput(this, "Package name", "com.example.app") { v ->
223                 choose(ActionSpec(ActionKind.LAUNCH_APP, v, v))
224             }
225             return
226         }
227         Ui.pick(this, "Open an app", apps.map { it.first }) { i ->
228             val (label, pkg) = apps[i]
229             choose(ActionSpec(ActionKind.LAUNCH_APP, pkg, label))
230         }
