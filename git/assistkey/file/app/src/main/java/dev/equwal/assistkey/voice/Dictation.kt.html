13 /**
14  * Voice typing: speak, and the words go where the cursor is.
15  *
16  * This app has no speech engine and no network access. It uses the Android
17  * SpeechRecognizer interface, so the recognition is done by a speech app the
18  * user already has - the system one, or an offline Whisper app, or any other
19  * app that offers a RecognitionService. The words come back as text, and
20  * [TextInsert] puts them into the focused field.

57     /**
58      * The choice rule, on plain strings so that a test can run it with no
59      * device. [engines] are flattened component names, `package/class`.
60      * Order: the saved one if it is still installed, else the first on-device

63     fun pick(saved: String?, engines: List<String>): String? =

125     /**
126      * Starts listening. A second call while listening stops early.
127      *
128      * Android gives the microphone to the app in front only, and an accessibility
129      * service does not count. So the listening runs in [DictationActivity], which
130      * is see-through and cannot take focus: this app is in front, and the text
131      * field behind it keeps the cursor.
132      */
133     fun toggle(svc: AccessibilityService): Boolean {
134         stop?.let {
135             it()
136             return true
137         }
138         if (!hasMicrophone(svc)) {
139             // The user tried voice typing. Open the screen that asks for the microphone.
140             runCatching {
141                 svc.startActivity(
142                     Intent(svc, dev.equwal.assistkey.ui.VoiceActivity::class.java).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
143                 )
144             }
145             return false
146         }
147         if (!available(svc)) return fail(svc, "No speech app on this device")
148         return runCatching {
149             svc.startActivity(
150                 Intent(svc, DictationActivity::class.java)
151                     .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_NO_ANIMATION)
152             )
153             true
154         }.getOrDefault(false)
155     }
