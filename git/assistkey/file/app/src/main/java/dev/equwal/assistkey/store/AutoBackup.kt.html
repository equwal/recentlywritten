14 /**
15  * A copy of the settings outside the app, kept up to date by itself.
16  *
17  * An update keeps the settings: they are in the data of the app, and Android
18  * keeps that data when a new version goes over the old one. An uninstall
19  * removes the data. This copy is in the shared Documents folder, which an
20  * uninstall does not touch, so the user can import it after a new install.
21  *
22  * The copy holds what an export holds, and nothing else. See [SettingsFile].
23  * It needs no permission: an app may write its own file to Documents.
24  */
25 object AutoBackup {
26 
27     const val FOLDER = "Rebind"
28     const val FILE_NAME = "rebind-settings.json"
29 
30     /** For the user: where the copy is. */
31     val PLACE = Environment.DIRECTORY_DOCUMENTS + "/" + FOLDER + "/" + FILE_NAME
32 
33     private const val TAG = "AssistKey"
34     private const val DELAY_MS = 3000L
35 
36     private val thread by lazy { HandlerThread("rebind-backup").apply { start() } }
37     private val worker by lazy { Handler(thread.looper) }
38 
39     // SharedPreferences holds its listeners weakly. These references keep them alive.
40     private val listeners = ArrayList<SharedPreferences.OnSharedPreferenceChangeListener>()
41 
42     /** The settings as one JSON text, the same text that Export writes. */
43     fun export(c: Context): String {
44         val settings = SettingsFile.ALLOWED.keys.associateWith { file ->
45             c.getSharedPreferences(file, Context.MODE_PRIVATE).all
46         }
47         return SettingsFile.encode(settings, BuildConfig.VERSION_NAME, Device.name)
48     }
49 
50     /**
51      * Starts to watch the settings. Each change writes the copy again, a few
52      * seconds later, so that a run of changes makes one write. Safe to call
53      * more than once.
54      */
55     @Synchronized
56     fun watch(context: Context) {
57         if (listeners.isNotEmpty()) return
58         val app = context.applicationContext
59         SettingsFile.ALLOWED.keys.forEach { file ->
60             val l = SharedPreferences.OnSharedPreferenceChangeListener { _, _ -> schedule(app) }
61             listeners += l
62             app.getSharedPreferences(file, Context.MODE_PRIVATE).registerOnSharedPreferenceChangeListener(l)
63         }
64         // One write at the start, so that a user who only updates has a copy too.
65         schedule(app)
66     }
67 
68     private val token = Any()
69 
70     private fun schedule(app: Context) {
71         worker.removeCallbacksAndMessages(token)
72         worker.postDelayed({ write(app) }, token, DELAY_MS)
73     }
74 
75     /** Writes the copy now. False when the device refused. The app works without the copy. */
76     fun write(c: Context): Boolean = runCatching {
77         val resolver = c.contentResolver
78         val files = MediaStore.Files.getContentUri(MediaStore.VOLUME_EXTERNAL_PRIMARY)
79         val path = Environment.DIRECTORY_DOCUMENTS + "/" + FOLDER + "/"
80         // The file of this install, if there is one. After a new install the old
81         // file belongs to nobody, and Android gives the new file a name of its own.
82         val mine = resolver.query(
83             files, arrayOf(MediaStore.MediaColumns._ID),
84             MediaStore.MediaColumns.RELATIVE_PATH + "=? AND " + MediaStore.MediaColumns.DISPLAY_NAME + "=?",
85             arrayOf(path, FILE_NAME), null
86         )?.use { if (it.moveToFirst()) android.content.ContentUris.withAppendedId(files, it.getLong(0)) else null }
87         val target = mine ?: resolver.insert(
88             files,
89             ContentValues().apply {
90                 put(MediaStore.MediaColumns.DISPLAY_NAME, FILE_NAME)
91                 put(MediaStore.MediaColumns.MIME_TYPE, "application/json")
92                 put(MediaStore.MediaColumns.RELATIVE_PATH, path)
93             }
94         ) ?: return false
95         resolver.openOutputStream(target, "wt")?.use { it.write(export(c).toByteArray()) } ?: return false
96         true
97     }.onFailure { Log.w(TAG, "settings copy not written", it) }.getOrDefault(false)
