76     /**
77      * The name of a light action comes from its payload, not from the stored
78      * label, so that a binding made by an older version shows the current name.
79      */
80     private fun dimName(): String = when (payload) {
81         "darker" -> "Extra dim darker"
82         "brighter" -> "Extra dim brighter"
83         "system_up" -> "Brightness up"
84         "system_down" -> "Brightness down"

86     }

88     fun describe(): String = if (kind == ActionKind.DIM) dimName() else label.ifBlank {
89         when (kind) {
90             ActionKind.NONE -> "Disabled"
91             ActionKind.PASS_THROUGH -> "Default behaviour"
92             else -> if (payload.isBlank()) kind.name else "${kind.name}: $payload"
93         }
94     }
