24 /**
25  * Every permission the app can use, in one place: what it is for, whether it is
26  * granted, and a way to grant it.
27  *
28  * "Set up what is missing" walks through the missing ones in order. Android
29  * grants each of these on a screen of its own, so the walk opens one screen,
30  * waits for the user to come back, and opens the next. It can be run again at
31  * any time, which is the point: a reinstall or a system update takes grants
32  * away without saying so.
33  *
34  * Nothing here is required. The app works with whatever has been granted.
35  */

80      * What the apps that go with Rebind need. Each one is a switch in the

82      * it can apply: the install switch in the build that carries apps, the
83      * others when their app is installed.

85     private fun extensions(): List<Item> {
86         val list = ArrayList<Item>()
87         if (Bundled.SUPPORTED) {
88             list += Item(
89                 "Install apps from Rebind",
90                 "For the apps that Rebind carries",
91                 packageManager.canRequestPackageInstalls()
92             ) {
93                 start(Intent(Settings.ACTION_MANAGE_UNKNOWN_APP_SOURCES, Uri.parse("package:" + packageName)))
94             }
95         }
96         if (installed(INK_RECENTS)) {
97             list += Item(
98                 "Ink Recents: app usage data",
99                 "Puts the recent apps in order",
100                 usageAccess(INK_RECENTS)
101             ) {
102                 val page = Intent(Settings.ACTION_USAGE_ACCESS_SETTINGS)
103                 // Straight to the switch of that app, where the settings app can. Else the list.
104                 val direct = Intent(page).setData(Uri.parse("package:" + INK_RECENTS))
105                 if (!Channels.safeStart(this, direct)) start(page)
106             }
107         }
108         val homes = Bundled.items.filter { it.group == Bundled.HOME && it.pkg != INK_RECENTS }.map { it.pkg }
109         if (homes.any(::installed)) {
110             val home = packageManager.resolveActivity(
111                 Intent(Intent.ACTION_MAIN).addCategory(Intent.CATEGORY_HOME),
112                 PackageManager.MATCH_DEFAULT_ONLY
113             )?.activityInfo?.packageName
114             list += Item(
115                 "Home screen",
116                 "Choose inkOS or ThinkLauncher",
117                 home in homes
118             ) { start(Intent(Settings.ACTION_HOME_SETTINGS)) }
119         }
120         return list
121     }

126     /** Whether another app has usage access. False where Android does not tell. */
127     private fun usageAccess(pkg: String): Boolean = runCatching {
128         val uid = packageManager.getApplicationInfo(pkg, 0).uid
129         getSystemService(android.app.AppOpsManager::class.java)
130             .unsafeCheckOpNoThrow(android.app.AppOpsManager.OPSTR_GET_USAGE_STATS, uid, pkg) ==
131             android.app.AppOpsManager.MODE_ALLOWED
132     }.getOrDefault(false)

147     private var lastStep = 0L
148 
149     /**
150      * Opens the first missing grant that this walk has not asked for yet. Coming
151      * back from a permission dialog reports twice, as a result and as a resume,
152      * so a second call close behind the first is dropped.
153      */
154     private fun next() {
155         val now = android.os.SystemClock.uptimeMillis()
156         if (now - lastStep < 600) return
157         lastStep = now
158         val item = items().firstOrNull { !it.granted && it.title !in asked }
159         if (item == null) {
160             walking = false
161             build()
162             return
163         }
164         asked.add(item.title)
165         item.ask()
166     }
167 
168     private fun start(i: Intent) {
169         if (!Channels.safeStart(this, i)) next()
170     }
171 
172     private fun build() {
173         val all = items()
174         val missing = all.count { !it.granted }
175         val col = Ui.page(this, "Permissions")
176         col.note("Nothing here is required.")
177         if (missing > 0) {
178             col.primaryButton("Set up what is missing (" + missing + ")") {
179                 walking = true
180                 asked = HashSet()
181                 next()
182             }
183         }
184 
185         all.forEach { item -> entry(col, item) }
186 
187         col.note("If a switch turns itself off, allow restricted settings in App info.")
188         col.button("Open App info") { Channels.safeStart(this, Channels.appInfoIntent(this)) }
189 
190         if (!PowerNative.canWriteSecure(this)) {
191             col.header("From a computer, once")
192             col.note("It lets the app change the Power button and Emergency SOS settings.")
193             col.code(PowerNative.GRANT_COMMAND)
194         }
195     }
196 
197     private fun entry(col: LinearLayout, item: Item) {
198         col.row(item.title, item.why, state = if (item.granted) "Granted" else "Missing") {
199             walking = false
200             item.ask()

219         /** Opens this screen the first time the app runs, and never again by itself. */
220         fun showOnce(a: Activity) {
221             val p = a.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
222             if (p.getBoolean(K_SHOWN, false)) return
223             p.edit().putBoolean(K_SHOWN, true).apply()
