75     private fun choose(buttons: Boolean, gestures: Boolean, keys: Boolean) {
76         if (!buttons && !gestures && !keys) {
77             Toast.makeText(this, "Leaves no way to navigate", Toast.LENGTH_LONG).show()
78             build()
79             return
80         }
81         prefs().edit().putBoolean("buttons", buttons).putBoolean("gestures", gestures)
82             .putBoolean("keys", keys).apply()
83         applyKeys(keys)
84         if (Shell.ready) applySystem(buttons, gestures) else build()

102     /**
103      * On fills empty slots with the defaults and leaves anything already bound
104      * alone. Off clears exactly the slots that still hold a default, so a
105      * binding the user made by hand survives being toggled.
106      */
107     private fun applyKeys(on: Boolean) {
108         val b = Store.bindings(this)
109         defaults.forEach { (t, action) ->
110             val spec = ActionSpec(ActionKind.GLOBAL, action.name, action.label)
111             if (on && !b.isBound(t)) Store.bind(this, t, spec)
112             if (!on && b[t] == spec) Store.bind(this, t, ActionSpec.PASS)
113         }
114         if (on) {
115             Channels.setEnabled(this, Channel.ACCESSIBILITY, true)
116             PowerControl.setWanted(this, true)
117             // The side doors, for whenever shell access is not there.
118             Channels.setEnabled(this, Channel.ASSISTANT, true)
119         }
120         (ServiceHolder.service as? KeyFilterService)?.syncPower()
121         // One question at a time: the first of these bindings that needs something.
122         if (on) defaults.map { it.first }.firstOrNull { GuidedSetupActivity.missing(this, it) }
123             ?.let { GuidedSetupActivity.askIfMissing(this, it) }
124     }

141     private fun build() {
142         val buttonsNow = NavNative.buttonsShowing(this)
143         if (buttonsNow == null && !redrawnForInsets) {
144             redrawnForInsets = true
145             window.decorView.post { if (!isFinishing) build() } // insets need an attached window
146         }
147         // Stock Android has no separate gesture switch: gestures are on exactly when the bar is off.
148         val gesturesNow = NavNative.gesturesOn(this) ?: buttonsNow?.not()
149 
150         val wantButtons = wanted("buttons", buttonsNow ?: true)
151         val wantGestures = wanted("gestures", gesturesNow ?: true)
152         val wantKeys = wanted("keys", keysOn())
153 
154         val col = Ui.page(this, "Navigation")
155 
156         col.check("Button bar", "Back, Home and Recents at the bottom", wantButtons) {
157             choose(it, wantGestures, wantKeys)
158         }
159         col.check("Swipe gestures", "Swipe up for Home, in from a side for Back", wantGestures) {
160             choose(wantButtons, it, wantKeys)
161         }
162         col.check(
163             "Power button",
164             if (Shell.SUPPORTED) "Tap for Back, hold for Home, double tap for Recents"
165             else "Hold for Home, double press for Recents",
166             wantKeys
167         ) {
168             choose(wantButtons, wantGestures, it)
169         }
170 
171         val labels = setups.map { s ->
172             (if (s.buttons == wantButtons && s.gestures == wantGestures && s.keys == wantKeys) "* " else "   ") + s.name
173         }
174         col.button("Pick a setup") {
175             Ui.pick(this, "Navigation setup", labels) { i ->
176                 setups[i].let { choose(it.buttons, it.gestures, it.keys) }
177             }
178         }
179 
180         if (wantKeys) powerKey(col)
181         system(col, wantButtons, wantGestures, buttonsNow, gesturesNow)
182     }

184     private fun powerKey(col: LinearLayout) {
185         col.header("Power button")
186         val b = Store.bindings(this)
187         val direct = Shell.ready && PowerControl.wanted(this)
188         defaults.forEach { (t, _) ->
189             val label = when {
190                 t.type == GestureType.HOLD -> "Hold"
191                 t.count == 1 -> "Tap"
192                 t.count == 2 -> "Double tap"
193                 else -> "Triple tap"
194             }
195             // Without shell access only hold and double press have a way in.
196             val reachable = direct || t == Channel.POWER_HOLD || t == Channel.POWER_DOUBLE
197             if (!reachable && !Shell.SUPPORTED) return@forEach
198             col.row(label + ": " + b.raw(t).describe(), if (reachable) null else "Needs shell access", reachable) {
199                 startActivity(ActionPickerActivity.intent(this, t))
200             }
201         }
202         col.row(
203             if (Shell.SUPPORTED) "More Power button gestures" else "Set up the Power button",
204             if (Shell.SUPPORTED) "More taps and combinations" else "Hold and double press"
205         ) {
206             startActivity(Intent(this, PowerActivity::class.java))
207         }
208         if (!Channels.isSatisfied(this, Channel.ACCESSIBILITY)) {
209             col.row("Button remapping", "Nothing here can fire", state = "Off") {
210                 startActivity(Intent(this, SetupActivity::class.java))
211             }
212         }
213         if (!direct) {
214             col.note("A tap cannot reach any app. Hold and double press work.")
215         }
216     }

218     private fun system(
219         col: LinearLayout,
220         wantButtons: Boolean,
221         wantGestures: Boolean,
222         buttonsNow: Boolean?,
223         gesturesNow: Boolean?
224     ) {
225         col.header("Now")
226         col.row("Button bar", null, enabled = false, state = now(buttonsNow, "Showing", "Hidden"))
227         col.row("Swipe gestures", null, enabled = false, state = now(gesturesNow, "On", "Off"))
228 
229         if (applying) {
230             col.note("Applying...")
231             return
232         }
233         if (buttonsNow == wantButtons && gesturesNow == wantGestures) return
234 
235         if (Shell.ready) {
236             col.button("Apply") { applySystem(wantButtons, wantGestures) }
237             return
238         }
239         if (Shell.SUPPORTED) {
240             col.note("Shell access applies this from here:")
241             col.row(
242                 "Shell access",
243                 "No computer needed",
244                 state = Shell.describe(this)
245             ) { startActivity(Intent(this, ShellActivity::class.java)) }
246             col.note("Or from a computer with USB debugging:")
247         } else {
248             col.note("Run this once from a computer with USB debugging on.")
249         }
250         col.code(NavNative.commands(wantButtons, wantGestures).joinToString("\n"))
251     }
