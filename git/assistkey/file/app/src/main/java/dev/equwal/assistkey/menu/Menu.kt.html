12  * A menu of actions, bound to a gesture like any other action.
13  *
14  * The menu lives in the payload of its own binding, as a JSON array of action

23     fun decode(payload: String): List<ActionSpec> = runCatching {
24         val a = JSONArray(payload)
25         (0 until a.length()).mapNotNull { a.optJSONObject(it) }
26             .map(ActionSpec::fromJson)
27             .filter { it.kind != ActionKind.MENU && it.kind != ActionKind.PASS_THROUGH }
