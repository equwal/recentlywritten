21     fun run(ctx: Context, spec: ActionSpec): Boolean {
22         return try {
23             when (spec.kind) {
24                 ActionKind.NONE -> true
25                 ActionKind.PASS_THROUGH -> false
26                 ActionKind.GLOBAL -> global(spec.payload)
27                 ActionKind.VOLUME -> volume(ctx, spec.payload)
28                 ActionKind.MEDIA -> media(ctx, spec.payload)
29                 ActionKind.LAUNCH_APP -> launchApp(ctx, spec.payload)
30                 ActionKind.LAUNCH_COMPONENT -> launchComponent(ctx, spec.payload)
31                 ActionKind.LAUNCH_ACTION -> launchAction(ctx, spec.payload)
32                 ActionKind.BROADCAST -> broadcast(ctx, spec.payload)
33                 ActionKind.SWIPE -> swipe(spec.payload)
34                 ActionKind.SCROLL -> scroll(spec.payload)
35                 ActionKind.MENU -> dev.equwal.assistkey.menu.MenuActivity.open(ctx, spec.payload)
36                 ActionKind.DIM -> {
37                     dev.equwal.assistkey.display.ExtraDim.act(ctx.applicationContext, spec.payload)
38                     true
39                 }
40                 ActionKind.NAV -> dev.equwal.assistkey.native.NavNative.toggleBar(ctx.applicationContext)
41                 ActionKind.VOICE -> ServiceHolder.service
42                     ?.let { dev.equwal.assistkey.voice.Dictation.toggle(it) } ?: false
43             }
44         } catch (e: Exception) {
45             Log.w(TAG, "action " + spec.kind + "/" + spec.payload + " failed", e)
46             false
47         }
48     }

50     private fun global(name: String): Boolean {
51         val svc = ServiceHolder.service ?: return false
52         val action = GlobalAction.fromName(name) ?: return false
53         if (!action.available) return false
54         if (action == GlobalAction.HOME_CLOSE_IME) return homeClosingKeyboard(svc)
55         return svc.performGlobalAction(action.id)
56     }

58     /**
59      * Home alone leaves a keyboard floating over the launcher on some firmware,
60      * so an open keyboard is dismissed first. Seeing the keyboard's window needs
61      * flagRetrieveInteractiveWindows; if the list is unavailable this is Home.
62      */
63     private fun homeClosingKeyboard(svc: android.accessibilityservice.AccessibilityService): Boolean {
64         val keyboardUp = runCatching {
65             svc.windows.any { it.type == android.view.accessibility.AccessibilityWindowInfo.TYPE_INPUT_METHOD }
66         }.getOrDefault(false)
67         if (!keyboardUp) return svc.performGlobalAction(GlobalAction.HOME.id)
68         svc.performGlobalAction(GlobalAction.BACK.id)
69         android.os.Handler(android.os.Looper.getMainLooper()).postDelayed(
70             { svc.performGlobalAction(GlobalAction.HOME.id) }, 180L
71         )
72         return true
73     }

92     private fun media(ctx: Context, what: String): Boolean {
93         val code = when (what) {
94             "play_pause" -> KeyEvent.KEYCODE_MEDIA_PLAY_PAUSE
95             "next" -> KeyEvent.KEYCODE_MEDIA_NEXT
96             "previous" -> KeyEvent.KEYCODE_MEDIA_PREVIOUS
97             "stop" -> KeyEvent.KEYCODE_MEDIA_STOP
98             else -> return false
99         }
100         val am = ctx.getSystemService(AudioManager::class.java) ?: return false
101         val now = System.currentTimeMillis()
102         am.dispatchMediaKeyEvent(KeyEvent(now, now, KeyEvent.ACTION_DOWN, code, 0))
103         am.dispatchMediaKeyEvent(KeyEvent(now, now, KeyEvent.ACTION_UP, code, 0))
104         return true
105     }

120         val cn = ComponentName.unflattenFromString(parts.component) ?: return false
121         val i = Intent()
122         i.component = cn
123         parts.extras.forEach { (key, value) -> i.putExtra(key, value) }
124         i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
125         ctx.startActivity(i)

141     /**
142      * Synthetic swipe - the page-turn action for reader apps that only respond
143      * to touch. Uses display bounds so it scales to any screen.
144      */
145     private fun swipe(dir: String): Boolean {
146         val svc = ServiceHolder.service ?: return false
147         val dm = svc.resources.displayMetrics
148         val w = dm.widthPixels.toFloat()
149         val h = dm.heightPixels.toFloat()
150         val cx = w / 2f
151         val cy = h / 2f
152         val dx = w * 0.35f
153         val dy = h * 0.35f
154 
155         var sx = cx; var sy = cy; var ex = cx; var ey = cy
156         when (dir) {
157             "left" -> { sx = cx + dx; ex = cx - dx }
158             "right" -> { sx = cx - dx; ex = cx + dx }
159             "up" -> { sy = cy + dy; ey = cy - dy }
160             "down" -> { sy = cy - dy; ey = cy + dy }
161             else -> return false
162         }
163 
164         val path = Path()
165         path.moveTo(sx, sy)
166         path.lineTo(ex, ey)
167         val stroke = GestureDescription.StrokeDescription(path, 0L, 120L)
168         return svc.dispatchGesture(
169             GestureDescription.Builder().addStroke(stroke).build(), null, null
170         )
171     }

173     private fun scroll(dir: String): Boolean {
174         val svc = ServiceHolder.service ?: return false
175         val action = if (dir == "backward") {
176             AccessibilityNodeInfo.ACTION_SCROLL_BACKWARD
177         } else {
178             AccessibilityNodeInfo.ACTION_SCROLL_FORWARD
179         }
180         val root = svc.rootInActiveWindow ?: return false
181         val target = findScrollable(root) ?: return false
182         return target.performAction(action)
183     }

185     private fun findScrollable(node: AccessibilityNodeInfo?): AccessibilityNodeInfo? {
186         if (node == null) return null
187         if (node.isScrollable) return node
188         for (i in 0 until node.childCount) {
189             val hit = findScrollable(node.getChild(i))
190             if (hit != null) return hit
191         }
192         return null
193     }
