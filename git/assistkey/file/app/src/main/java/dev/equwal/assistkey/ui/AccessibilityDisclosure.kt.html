6 /**
7  * What the accessibility service does, said before the user is sent to switch
8  * it on.
9  *
10  * Google Play requires this of an app that uses the accessibility interface for
11  * a purpose other than assistive technology: in the app, in plain words, with a
12  * yes before anything else happens. It is one text in one place, so that every
13  * screen which leads to the switch shows the same words, and so that the words
14  * change when the service learns to do something new.
15  */
16 object AccessibilityDisclosure {
17 
18     const val TEXT =
19         "Rebind uses the Android AccessibilityService API to remap the buttons " +
20             "of this device.\n\n" +
21             "With it on, Rebind:\n\n" +
22             "- sees button presses;\n" +
23             "- does the action you chose;\n" +
24             "- reads the front window only for the Scroll action;\n" +
25             "- puts dictated words in the focused text field, never a password " +
26             "field;\n" +
27             "- shows a small on-screen button only when you bind one;\n" +
28             "- remembers the front app's name, in memory only;\n" +
29             "- sees the recent apps of the system open, to open Ink Recents in " +
30             "their place, if you have it.\n\n" +
31             "Nothing is collected, stored or sent. The app has no internet permission."
32 
33     fun show(a: Activity, onAgree: () -> Unit, onDecline: () -> Unit = {}) {
34         AlertDialog.Builder(a)
35             .setTitle("Accessibility service")
36             .setMessage(TEXT)
37             .setPositiveButton("Agree") { _, _ -> onAgree() }
38             .setNegativeButton("Not now") { _, _ -> onDecline() }
39             .setOnCancelListener { onDecline() }
40             .show()
