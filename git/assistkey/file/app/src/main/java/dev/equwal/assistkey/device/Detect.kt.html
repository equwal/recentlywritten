19 /**
20  * What this device has, found by asking, not by making the user press anything.
21  *
22  * AssistKey is a key remap tool. It must never say "press the button you want
23  * to find". So every source here is passive: Android's own key tables, the
24  * input devices the framework lists, and - with shell access - the key codes
25  * each kernel input node declares. No press is recorded, and no sequence of
26  * presses is ever part of detection.
27  *
28  * The result is held in a preference file of its own. It describes one device
29  * and one build, so it is not a setting and never enters an export - see
30  * [dev.equwal.assistkey.store.SettingsFile].
31  *
32  * The parsing and the storage format are pure functions over text, so a test
33  * runs them with no device.
34  */

114     /**
115      * The Linux key codes the input nodes declare, from `getevent -p`. A node
116      * prints its codes in hex after `KEY (0001):` and continues on indented
117      * lines until the next event class. Junk lines add nothing.
118      */
119     fun scanCodesFromGetevent(text: String): Set<Int> {
120         val out = LinkedHashSet<Int>()
121         var inKeys = false
122         text.lineSequence().forEach { raw ->
123             val line = raw.trim()
124             when {
125                 line.startsWith("KEY (0001):") -> {
126                     inKeys = true
127                     hexInto(line.substringAfter(':'), out)
128                 }
129                 line.isEmpty() || line.startsWith("add device") ||
130                     line.startsWith("name:") || line.contains("(00") -> inKeys = false
131                 inKeys -> hexInto(line, out)
132             }
133         }
134         return out
135     }

137     /** The keys of [scanCodesFromGetevent] this app has a name for. */
138     fun keysFromGetevent(text: String): List<HwKey> =
139         ordered(scanCodesFromGetevent(text).mapNotNull { byScanCode[it] })

141     /** Adds every short hex word in [part]. Anything else is ignored. */
142     private fun hexInto(part: String, out: MutableCollection<Int>) {
143         part.split(' ', '\t').forEach { word ->
144             if (word.isNotEmpty() && word.length <= 4 && word.all(::isHex)) {
145                 word.toIntOrNull(16)?.let(out::add)
146             }
147         }
148     }

150     private fun isHex(c: Char): Boolean =
151         c in '0'..'9' || c in 'a'..'f' || c in 'A'..'F'

153     /**
154      * The text under [marker] in the shell probe output, up to the next marker.
155      * Empty when the marker is not there.
156      */
157     fun section(text: String, marker: String): String {
158         val at = text.indexOf(MARK + marker)
159         if (at < 0) return ""
160         val from = at + MARK.length + marker.length
161         val end = text.indexOf(MARK, from)
162         return if (end < 0) text.substring(from) else text.substring(from, end)
163     }

165     fun encode(r: Result): String = JSONObject()
166         .put("format", FORMAT)
167         .put("fingerprint", r.fingerprint)
168         .put("app", r.appVersion)
169         .put("name", r.name)
170         .put("profile", r.profile)
171         .put("keys", JSONArray(r.keys.map { it.token }))
172         .put("capabilities", JSONObject().also { o ->

174         })
175         .toString()

177     /**
178      * The opposite of [encode]. Null for text this build cannot read. A key or
179      * a capability this build does not know is dropped, so an older build can
180      * still read a newer file.
181      */
182     fun decode(text: String): Result? {
183         val root = runCatching { JSONObject(text) }.getOrNull() ?: return null
184         if (root.optInt("format", -1) != FORMAT) return null
185         val keys = root.optJSONArray("keys") ?: JSONArray()
186         val caps = root.optJSONObject("capabilities") ?: JSONObject()
187         val states = LinkedHashMap<String, State>()
188         capabilityNames.forEach { name ->
189             if (!caps.has(name)) return@forEach
190             val value = runCatching { State.valueOf(caps.optString(name)) }.getOrNull()
191             if (value != null) states[name] = value
192         }
193         return Result(
194             fingerprint = root.optString("fingerprint"),
195             appVersion = root.optString("app"),
196             name = root.optString("name"),
197             profile = root.optString("profile"),
198             keys = ordered((0 until keys.length()).mapNotNull { HwKey.fromToken(keys.optString(it)) }),
199             capabilities = states
200         )
201     }

228     /**
229      * Detects once at first start, and again after a firmware or app change.
230      * A stored result that still matches costs one preference read.
231      *
232      * A result that was found without shell access is also stale as soon as
233      * there is shell access, because the shell sees input nodes the framework
234      * does not report. The other way round it stands: detection only adds.
235      */
236     fun refreshIfStale(c: Context) {
237         val have = stored(c)
238         if (have != null &&
239             have.fingerprint == Build.FINGERPRINT &&
240             have.appVersion == BuildConfig.VERSION_NAME &&
241             !(have.capability(SHELL_ACCESS) != State.YES && Shell.ready)
242         ) {
243             return
244         }
245         val app = c.applicationContext
246         run(app) { save(app, it) }
247     }

251     /**
252      * Asks the device what it has. [done] comes back on the main thread. The
253      * framework part is a handful of table lookups; only the shell part, which
254      * runs off the main thread, takes any time.
255      */
256     fun run(c: Context, done: (Result) -> Unit) {
257         val app = c.applicationContext
258         val caps = LinkedHashMap<String, State>()
259         caps[KEY_FILTER] = yesNo(Channels.isAccessibilityOn(app))
260         caps[ASSISTANT_ROLE] = yesNo(Channels.isRoleAvailable(app, RoleManager.ROLE_ASSISTANT))
261         caps[CAMERA_ROUTE] = maybe(PowerNative.cameraDoubleTapEnabled(app))
262         caps[SECURE_SETTINGS] = yesNo(PowerNative.canWriteSecure(app))
263         caps[SHELL_ACCESS] = yesNo(Shell.ready)
264         caps[EMERGENCY_SOS] = yesNo(PowerNative.canWriteSecure(app) || Shell.ready)
265 
266         val framework = frameworkKeys()
267         if (!Shell.ready) {
268             // Without the shell the light node is usually unreadable, and the
269             // overlays cannot be listed at all. Say so rather than guess.
270             caps[FRONTLIGHT] = if (anyNode()) State.YES else State.UNKNOWN
271             caps[NAV_OVERLAYS] = State.UNKNOWN
272             return done(result(framework, caps))
273         }
274         Shell.run(probeScript()) { r ->
275             val keys = if (r.ok) keysFromGetevent(r.output.substringBefore(MARK)) else emptyList()
276             caps[FRONTLIGHT] = when {
277                 !r.ok -> State.UNKNOWN
278                 section(r.output, "light").contains("present") -> State.YES
279                 else -> State.NO
280             }
281             val navbars = section(r.output, "nav").trim().toIntOrNull()
282             caps[NAV_OVERLAYS] = when {
283                 !r.ok || navbars == null -> State.UNKNOWN
284                 navbars > 0 -> State.YES
285                 else -> State.NO
286             }
287             done(result(ordered(framework + keys), caps))
288         }
289     }

300     private fun yesNo(on: Boolean): State = if (on) State.YES else State.NO

302     private fun maybe(on: Boolean?): State = if (on == null) State.UNKNOWN else yesNo(on)

304     /**
305      * What the framework says this device has. [KeyCharacterMap.deviceHasKey]
306      * asks every input device; the second pass asks each device by itself and
307      * leaves out full keyboards, because a typing keyboard is not a button on
308      * this device.
309      */
310     private fun frameworkKeys(): List<HwKey> {
311         val found = LinkedHashSet<HwKey>()
312         HwKey.entries.forEach { key ->
313             if (runCatching { KeyCharacterMap.deviceHasKey(key.code) }.getOrDefault(false)) {
314                 found += key
315             }
316         }
317         val codes = HwKey.entries.map { it.code }.toIntArray()
318         runCatching { InputDevice.getDeviceIds() }.getOrDefault(IntArray(0)).forEach { id ->
319             val device = runCatching { InputDevice.getDevice(id) }.getOrNull() ?: return@forEach
320             if (device.keyboardType == InputDevice.KEYBOARD_TYPE_ALPHABETIC) return@forEach
321             val has = runCatching { device.hasKeys(*codes) }.getOrNull() ?: return@forEach
322             HwKey.entries.forEachIndexed { i, key -> if (has.getOrNull(i) == true) found += key }
323         }
324         return ordered(found)
325     }

330     /**
331      * One shell command for the three things only the shell can see: the key
332      * codes of every input node, the navigation overlays, and the light node.
333      */
334     private fun probeScript(): String {
335         val nodes = ExtraDim.nodes.joinToString(" ") { "'" + it + "'" }
336         return listOf(
337             "getevent -p 2>/dev/null",
338             "echo '" + MARK + "nav'; cmd overlay list 2>/dev/null | grep -ci navbar",
339             "echo '" + MARK + "light'; for n in " + nodes +
340                 "; do [ -e \$n ] && echo present; done"
341         ).joinToString("; ")
342     }
