7 /**
8  * How a gesture can be made to work, and what that needs from the user.
9  *
10  * The user says what he wants: this button, pressed this way, does that. This
11  * object turns the wish into the list of things that must be allowed for it,
12  * so that no screen has to talk about services, roles or channels first.
13  *
14  * It has no Android types, so a test can run it with no device.
15  */
16 object Route {
17 
18     /** A step of the guided setup screen. The single definition; the activity uses this one. */
19     enum class Step { BUTTON, PRESS, ACTION, ALLOW, DONE }
20 
21     /**
22      * Where hardware Back leads from [step]. Null means leave the screen.
23      *
24      * [fromHub] is true when the main screen chose the button, so Back from
25      * PRESS leaves instead of going to BUTTON. [askOnly] is true when another
26      * screen made the binding and this screen only asks, so Back from ALLOW
27      * or DONE leaves instead of going back into the flow.
28      */
29     fun back(step: Step, fromHub: Boolean, askOnly: Boolean): Step? = when (step) {
30         Step.BUTTON -> null
31         Step.PRESS -> if (fromHub) null else Step.BUTTON
32         Step.ACTION -> Step.PRESS
33         Step.ALLOW -> if (askOnly) null else Step.ACTION
34         Step.DONE -> if (askOnly) null else Step.ACTION
35     }
36 
37     /** One thing the user must allow. The order of the entries is the order to ask in. */
38     enum class Need(val title: String, val why: String) {
39         KEY_FILTER(
40             "Let Rebind see the buttons",
41             "Turn on Button remapping"
42         ),
43         ASSISTANT(
44             "Let Rebind get the Power hold",
45             "Choose Rebind as assistant app"
46         ),
47         CAMERA(
48             "Let Rebind get the Power double tap",
49             "Choose Rebind as camera app"
50         ),
51         SHELL_POWER(
52             "Open the full Power button",
53             "Needs shell access"
54         )
55     }
56 
57     /** What is true of the device and the build at this moment. */
58     data class Env(
59         /** The app reads the Power button directly (shell access is on and in use). */
60         val powerIsDirect: Boolean,
61         /** This build can have shell access. The Google Play build cannot. */
62         val shellSupported: Boolean = false
63     )
64 
65     data class Plan(
66         val needs: List<Need>,
67         /** Null when the gesture can work. Otherwise the reason, in words for the user. */
68         val blocked: String? = null
69     ) {
70         val possible: Boolean get() = blocked == null
71     }
72 
73     /**
74      * [actionNeedsFilter] is true for an action that the key filter carries out:
75      * Back, Home, a swipe, a scroll, voice typing.
76      */
77     fun plan(trigger: Trigger, actionNeedsFilter: Boolean, env: Env): Plan {
78         if (HwKey.SCREEN in trigger.keys) {
79             val plainTap = trigger.keys.size == 1 && trigger.type == GestureType.TAP && trigger.count == 1
80             // The key filter draws the button and carries out the action.
81             return if (plainTap) Plan(listOf(Need.KEY_FILTER))
82             else Plan(emptyList(), "The on-screen button has one gesture: a tap.")
83         }
84         val power = HwKey.POWER in trigger.keys
85         if (power && trigger.keys.size > 1 && trigger != Trigger.powerThen((trigger.keys - HwKey.POWER).first())) {
86             return Plan(emptyList(), "Hold Power, then press the other button.")
87         }
88         if (!power || env.powerIsDirect) return Plan(listOf(Need.KEY_FILTER))
89 
90         val alone = trigger.keys.size == 1
91         val door = when {
92             alone && trigger.type == GestureType.HOLD -> Need.ASSISTANT
93             alone && trigger.type == GestureType.TAP && trigger.count == 2 -> Need.CAMERA
94             // Power held, then one other button: the hold opens the assistant door.
95             !alone && trigger.keys.size == 2 && trigger.type == GestureType.TAP && trigger.count == 1 -> Need.ASSISTANT
96             // No door of Android leads here. Shell access does, where the build has it.
97             else -> return if (env.shellSupported) Plan(listOf(Need.KEY_FILTER, Need.SHELL_POWER)) else Plan(
98                 emptyList(),
99                 "Android hides this Power press. Hold and double tap work."
100             )
101         }
102         val second = !alone
103         return Plan(if (actionNeedsFilter || second) listOf(Need.KEY_FILTER, door) else listOf(door))
104     }

106     /** The most buttons that one combination can have. */
107     const val MAX_TOGETHER = 2
108 
109     /**
110      * The selection after a tap on [key] in the drawing. A tap on a selected
111      * button lets it go. The on-screen button cannot be part of a combination,
112      * so it is always alone. A tap past [MAX_TOGETHER] drops the oldest button.
113      */
114     fun toggle(selection: List<HwKey>, key: HwKey): List<HwKey> = when {
115         key in selection -> selection - key
116         key == HwKey.SCREEN || HwKey.SCREEN in selection -> listOf(key)
117         else -> (selection + key).takeLast(MAX_TOGETHER)
118     }
119 
120     /** The ways to press [keys] that can work, for the choice the user is shown. */
121     fun gestures(keys: Set<HwKey>, env: Env): List<Trigger> {
122         // Tap, double tap, triple tap and hold come first. The long tap counts come last.
123         val all = Trigger.allFor(keys).sortedBy { if (it.type == GestureType.TAP && it.count > 3) 1 else 0 }
124         return all.filter { plan(it, false, env).possible }
125     }
