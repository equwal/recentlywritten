12 /**
13  * Shows what the firmware itself has bound to each key.
14  *
15  * Read-only: Android does not let an ordinary app write these settings, however
16  * many permissions it holds. The screen earns its place anyway, because a set
17  * volume hook is the one thing that makes a key silently unmappable, and
18  * nothing else on the device will tell you that is what happened.
19  */
20 class ViwoodsActivity : Activity() {
21 
22     override fun onResume() {
23         super.onResume()
24         build()
25     }
26 
27     private fun build() {
28         val col = Ui.page(this, "Device button settings")
29         col.note("Apps cannot change these. Run the command from a computer.")
30         ViwoodsBridge.keys().forEach { keyBlock(col, it) }
31     }
32 
33     private fun keyBlock(col: LinearLayout, key: HwKey) {
34         col.header(key.label)
35         col.row(
36             "Device setting",
37             ViwoodsBridge.describe(this, key),
38             enabled = false,
39             state = if (ViwoodsBridge.hidesFromFilter(this, key)) "Hiding" else "Clear"
40         )
41 
42         if (key == HwKey.AI) {
43             if (ViwoodsBridge.aiHookedToUs(this)) {
44                 col.note("Taps work; hold and combinations do not. To give the button back:")
45                 col.code(ViwoodsBridge.aiUnhookCommand())
46             } else {
47                 col.note("The AI screen opens on every press. To stop it:")
48                 col.code(ViwoodsBridge.aiHookCommand(this))
49                 col.note("Then taps work, but hold and combinations stop.")
50             }
51             return
52         }
53 
54         if (ViwoodsBridge.hidesFromFilter(this, key)) {
55             col.note(
56                 "The device keeps " + key.label.lowercase() + " to itself. To clear it:"
57             )
58             ViwoodsBridge.unsetCommand(key)?.let { col.code(it) }
59             col.note("The device's own button settings set it again.")
60         }
61     }
