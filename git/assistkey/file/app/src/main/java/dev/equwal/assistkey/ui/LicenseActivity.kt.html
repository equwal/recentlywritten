40     private fun build() {
41         val state = License.state(this)
42         val col = Ui.page(this, "Licence")
43         status(col, state)
44         // With no Google Play there is nothing to buy and nothing to restore.
45         if (state.tier != License.Tier.NO_STORE) {
46             if (state.tier != License.Tier.LICENSED) buy(col, state)
47             restore(col)
48         }
49         tip(col)
50         moreApps(col)
51         col.note("Version " + BuildConfig.VERSION_NAME)
52     }

54     /** Links to the other sites and apps of the same author. A tap opens the page in the browser. */
55     private fun moreApps(col: LinearLayout) {
56         if (!MoreApps.shown) return
57         col.header("More apps")
58         for (app in MoreApps.ALL) {
59             col.row(app.name, app.line) {
60                 runCatching {
61                     startActivity(Intent(Intent.ACTION_VIEW, Uri.parse(app.url)))
62                 }.onFailure { Toast.makeText(this, "No browser on this device", Toast.LENGTH_LONG).show() }
63             }
64         }
65     }

84     private fun status(col: LinearLayout, state: License.State) {
85         when (state.tier) {
86             License.Tier.LICENSED -> col.header("Unlocked")
87             License.Tier.NO_STORE -> {
88                 col.header("Free on this device")
89                 col.note("No Google Play here, so everything is unlocked.")
90             }
91             License.Tier.BETA -> {
92                 col.header("Beta - free for now")
93                 col.note("When the beta closes, remapping stops until you buy.")
94                 col.note(
95                     if (state.betaConfirmedByPlay) {
96                         "Google Play confirms the beta is open."
97                     } else {
98                         "Free until " + BuildConfig.BETA_EXPIRES_DATE + " at the latest."
99                     }
100                 )
101             }
102             License.Tier.TRIAL -> {
103                 col.header("Trial - " + days(state.trialDaysLeft) + " left")
104                 col.note("After it, remapping stops until you buy. Your bindings are kept.")
105             }
106             License.Tier.LOCKED -> {
107                 col.header("Locked")
108                 col.note("Remapping is off. Your bindings are kept.")
109             }
110         }
111     }

117     private fun buy(col: LinearLayout, state: License.State) {
118         col.header("Buy")
119 
120         if (PlayBilling.purchasePending) {
121             col.note("A payment is being processed. The app unlocks by itself.")
122         }
123 
124         val tester = License.testerEligible(this)
125         val fullPrice = PlayBilling.price(Sku.PRO)
126         val testerPrice = PlayBilling.price(Sku.PRO_TESTER)
127 
128         when {
129             fullPrice == null && testerPrice == null -> notOnSale(col, state)
130             else -> {
131                 if (tester && testerPrice != null) {
132                     col.note(
133                         "Tester price" +
134                             (if (fullPrice != null) ", instead of " + fullPrice else "") + "."
135                     )
136                     col.primaryButton("Buy - " + testerPrice) { purchase(Sku.PRO_TESTER) }
137                 } else if (fullPrice != null) {
138                     col.primaryButton("Buy - " + fullPrice) { purchase(Sku.PRO) }
139                 }
140                 col.note("One payment. No subscription.")
141             }
142         }
143 
144         if (!tester) {
145             col.row("I was a beta tester", "For the tester price") { askForCode() }
146         }
147     }

149     /** No prices means Play gave us nothing to sell - say which kind of nothing. */
150     private fun notOnSale(col: LinearLayout, state: License.State) {
151         val problem = PlayBilling.lastProblem
152         col.note(
153             when {
154                 PlayBilling.availability == PlayBilling.Availability.UNKNOWN ->
155                     "Asking Google Play..."
156                 problem != null -> problem
157                 state.tier == License.Tier.BETA -> "Not on sale yet."
158                 else ->
159                     "Google Play is not offering it to this account. Install the app " +
160                         "from Google Play."
161             }
162         )
163     }

165     private fun purchase(productId: String) {
166         if (!PlayBilling.buy(this, productId)) {

175     private fun askForCode() {
176         Ui.textInput(this, "Tester code", "TESTER-XXXX-XXXX") { code ->
177             val ok = License.redeemTesterCode(this, code)
178             Toast.makeText(
179                 this,
180                 if (ok) "Tester price unlocked" else "That code was not recognised",
181                 Toast.LENGTH_LONG
182             ).show()
183             if (ok) build()

199                     Toast.LENGTH_LONG

201             }
202         }
