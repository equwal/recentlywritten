40     private fun build() {
41         val state = License.state(this)
42         val col = Ui.page(this, "Licence")
43         status(col, state)
44         // With no Google Play there is nothing to buy and nothing to restore.
45         if (state.tier != License.Tier.NO_STORE) {
46             if (state.tier != License.Tier.LICENSED) buy(col, state)
47             restore(col)
48         }
49         tip(col)
50         col.note("Version " + BuildConfig.VERSION_NAME)
51     }

53     /**
54      * A link to give a tip. It unlocks nothing. It is only in the build for
55      * direct install: Google Play does not allow a link to another way to pay
56      * in an app that it distributes.
57      */
58     private fun tip(col: LinearLayout) {
59         if (Tip.URL.isEmpty()) return
60         col.header("Say thanks")
61         col.row("Buy me a coffee", Tip.URL.removePrefix("https://") + " · unlocks nothing") {
62             runCatching {
63                 startActivity(Intent(Intent.ACTION_VIEW, Uri.parse(Tip.URL)))
64             }.onFailure { Toast.makeText(this, "No browser on this device", Toast.LENGTH_LONG).show() }
65         }
66     }

70     private fun status(col: LinearLayout, state: License.State) {
71         when (state.tier) {
72             License.Tier.LICENSED -> col.header("Unlocked")
73             License.Tier.NO_STORE -> {
74                 col.header("Free on this device")
75                 col.note("No Google Play here, so everything is unlocked.")
76             }
77             License.Tier.BETA -> {
78                 col.header("Beta - free for now")
79                 col.note("When the beta closes, remapping stops until you buy.")
80                 col.note(
81                     if (state.betaConfirmedByPlay) {
82                         "Google Play confirms the beta is open."
83                     } else {
84                         "Free until " + BuildConfig.BETA_EXPIRES_DATE + " at the latest."
85                     }
86                 )
87             }
88             License.Tier.TRIAL -> {
89                 col.header("Trial - " + days(state.trialDaysLeft) + " left")
90                 col.note("After it, remapping stops until you buy. Your bindings are kept.")
91             }
92             License.Tier.LOCKED -> {
93                 col.header("Locked")
94                 col.note("Remapping is off. Your bindings are kept.")
95             }
96         }
97     }

103     private fun buy(col: LinearLayout, state: License.State) {
104         col.header("Buy")
105 
106         if (PlayBilling.purchasePending) {
107             col.note("A payment is being processed. The app unlocks by itself.")
108         }
109 
110         val tester = License.testerEligible(this)
111         val fullPrice = PlayBilling.price(Sku.PRO)
112         val testerPrice = PlayBilling.price(Sku.PRO_TESTER)
113 
114         when {
115             fullPrice == null && testerPrice == null -> notOnSale(col, state)
116             else -> {
117                 if (tester && testerPrice != null) {
118                     col.note(
119                         "Tester price" +
120                             (if (fullPrice != null) ", instead of " + fullPrice else "") + "."
121                     )
122                     col.primaryButton("Buy - " + testerPrice) { purchase(Sku.PRO_TESTER) }
123                 } else if (fullPrice != null) {
124                     col.primaryButton("Buy - " + fullPrice) { purchase(Sku.PRO) }
125                 }
126                 col.note("One payment. No subscription.")
127             }
128         }
129 
130         if (!tester) {
131             col.row("I was a beta tester", "For the tester price") { askForCode() }
132         }
133     }
134 
135     /** No prices means Play gave us nothing to sell - say which kind of nothing. */
136     private fun notOnSale(col: LinearLayout, state: License.State) {
137         val problem = PlayBilling.lastProblem
138         col.note(
139             when {
140                 PlayBilling.availability == PlayBilling.Availability.UNKNOWN ->
141                     "Asking Google Play..."
142                 problem != null -> problem
143                 state.tier == License.Tier.BETA -> "Not on sale yet."
144                 else ->
145                     "Google Play is not offering it to this account. Install the app " +
146                         "from Google Play."
147             }
148         )
149     }

151     private fun purchase(productId: String) {
152         if (!PlayBilling.buy(this, productId)) {
153             Toast.makeText(

161     private fun askForCode() {
162         Ui.textInput(this, "Tester code", "TESTER-XXXX-XXXX") { code ->
163             val ok = License.redeemTesterCode(this, code)
164             Toast.makeText(
165                 this,
166                 if (ok) "Tester price unlocked" else "That code was not recognised",
