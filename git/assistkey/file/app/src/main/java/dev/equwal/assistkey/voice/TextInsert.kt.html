9 /**
10  * Puts text where the cursor is, in the app in front.
11  *
12  * The accessibility service does this, not a keyboard. It finds the field that
13  * has input focus and replaces the selection with the new text. A field that
14  * refuses a direct edit gets the text through the clipboard and a paste.
15  *
16  * Password fields are never written to.
17  */
18 object TextInsert {
19 
20     enum class Result { DONE, NO_FIELD, PASSWORD, REFUSED }
21 
22     fun insert(svc: AccessibilityService, text: String): Result {
23         val node = svc.findFocus(AccessibilityNodeInfo.FOCUS_INPUT)
24             ?: svc.rootInActiveWindow?.findFocus(AccessibilityNodeInfo.FOCUS_INPUT)
25             ?: return Result.NO_FIELD
26         if (node.isPassword) return Result.PASSWORD
27         if (!node.isEditable) return Result.NO_FIELD
28 
29         // A field that shows its hint reports the hint as its text.
30         val old = if (node.isShowingHintText) "" else node.text?.toString().orEmpty()
31         val start = node.textSelectionStart.takeIf { it in 0..old.length } ?: old.length
32         val end = node.textSelectionEnd.takeIf { it in start..old.length } ?: start
33 
34         // A space between the old text and the new words, when there is none.
35         val lead = if (start > 0 && !old[start - 1].isWhitespace()) " " else ""
36         val piece = lead + text
37         val merged = old.substring(0, start) + piece + old.substring(end)
38 
39         val set = Bundle().apply {
40             putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, merged)
41         }
42         if (node.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, set)) {
43             val caret = start + piece.length
44             node.performAction(
45                 AccessibilityNodeInfo.ACTION_SET_SELECTION,
46                 Bundle().apply {
47                     putInt(AccessibilityNodeInfo.ACTION_ARGUMENT_SELECTION_START_INT, caret)
48                     putInt(AccessibilityNodeInfo.ACTION_ARGUMENT_SELECTION_END_INT, caret)
49                 }
50             )
51             return Result.DONE
52         }
53 
54         // Web pages and some custom fields accept a paste but not a direct edit.
55         val clipboard = svc.getSystemService(ClipboardManager::class.java) ?: return Result.REFUSED
56         clipboard.setPrimaryClip(ClipData.newPlainText("Rebind voice typing", piece))
57         return if (node.performAction(AccessibilityNodeInfo.ACTION_PASTE)) Result.DONE else Result.REFUSED
58     }
