25 /**
26  * The Power button.
27  *
28  * It has two lives. With shell access the app reads the button straight from
29  * the kernel and it becomes a key like any other: any number of taps, hold, and
30  * combinations. Without shell access no app is ever shown the key, and what is
31  * left is two side doors - a hold arrives as an assistant request, a double
32  * press as a camera launch - plus whatever the firmware's own switches offer.
33  *
34  * The bindings are the same in both lives. "Hold" is one binding whichever door
35  * it comes through, so gaining or losing shell access changes what can fire,
36  * never what is configured.
37  */

62      * Android starts Emergency SOS on five quick Power presses. That collides
63      * with a 5-tap binding, and with fast tapping in general. It is a safety
64      * feature, so the app never turns it off by itself: this is the user's switch.
65      */
66     private fun emergencySos(col: LinearLayout) {
67         if (!Shell.ready && !PowerNative.canWriteSecure(this)) return
68         col.header("Emergency SOS")
69         col.check(
70             "Start it on five presses",
71             "Untick to use 5 taps for an action",
72             PowerNative.emergencySos(this)
73         ) { on ->
74             PowerNative.setEmergencySos(this, on) { ok ->
75                 if (!ok) Toast.makeText(this, "The system refused", Toast.LENGTH_LONG).show()
76                 build()
77             }
78         }

156     private fun door(col: LinearLayout, ch: Channel) {
157         val on = Channels.isEnabled(this, ch)
158         col.check(ch.title, ch.summary, on) { checked ->
159             Channels.setEnabled(this, ch, checked)
160             if (checked && !Channels.isSatisfied(this, ch)) claim(ch)
161             build()
162         }
163         if (on) {
164             col.row("Status", null, enabled = false, state = Channels.status(this, ch))
165             if (!Channels.isSatisfied(this, ch)) col.button("Set up") { claim(ch) }
166         }
167     }

169     private fun claim(ch: Channel) {
170         if (ch == Channel.CAMERA) {
171             Toast.makeText(this, "Choose Camera app, then pick Rebind", Toast.LENGTH_LONG).show()
172         }
173         if (!Channels.safeStart(this, Channels.claimIntent(this, ch))) {

180     private fun firmware(col: LinearLayout) {
181         if (!PowerNative.canWriteSecure(this)) {
182             col.note("Needs shell access, or this once from a computer:")
183             col.code(PowerNative.GRANT_COMMAND)
184             return
185         }
186         val short = PowerNative.shortPressValue(this)
187         col.row("Tap: " + PowerNative.describe(PowerNative.shortPress, short), "Handled by the system") {
188             pick("Tap", PowerNative.shortPress, short) { v ->
189                 applied(PowerNative.setShortPress(this, v))
190             }
191         }
192         val long = PowerNative.longPressValue(this)
193         col.row(
194             "Hold: " + PowerNative.describe(PowerNative.longPress, long),
195             "Must be Digital assistant"
196         ) {
197             pick("Hold", PowerNative.longPress, long) { v -> applied(PowerNative.setLongPress(this, v)) }
198         }
199         val ms = PowerNative.longPressMs(this)
200         val times = listOf(250, 350, 500, 650, 800, 1000).map { PowerNative.Option(it, it.toString() + " ms") }
201         col.row("Hold time: " + PowerNative.describe(times, ms), null) {
202             pick("Hold time", times, ms) { v -> applied(PowerNative.setLongPressMs(this, v)) }
203         }
204     }

206     private fun pick(title: String, options: List<PowerNative.Option>, current: Int?, onPick: (Int) -> Unit) {
207         val labels = options.map { o ->

209                 if (o.note.isBlank()) "" else "  -  " + o.note

211         Ui.pick(this, title, labels) { i -> onPick(options[i].value) }
