9 /**
10  * The firmware's own key remapper: a handful of Settings.System strings. The
11  * names and the token vocabulary were read out of setting_se08 1.3.7 on
12  * firmware 1.5.6.
13  *
14  * This is read-only, and not by choice. An ordinary app cannot write these
15  * keys. SettingsProvider refuses any Settings.System name outside its public
16  * list unless the caller is a privileged system app - WRITE_SETTINGS does not
17  * help and neither does an adb-granted WRITE_SECURE_SETTINGS; the call dies
18  * with "You cannot keep your settings in the secure settings". Only the shell
19  * can change them, so the most this app can do is show them and hand over the
20  * command.
21  *
22  * They still matter, because of one interaction measured on the device:
23  *
24  *   While a volume key's hook holds ANY value - even its "default" token,
25  *   volume_up or volume_down - the firmware handles that key before the input
26  *   filter stage, and no app ever sees the press. Unset, the key reaches the
27  *   accessibility filter normally.
28  *
29  * The AI key is not like that: it reaches the filter whatever its hook says,
30  * and the hook only decides what happens when nothing consumes the press.
31  *
32  * The two families are not symmetric. A volume key holds a token, with a
33  * component kept alongside in *AppShortcut; the AI key holds the flat component
34  * itself.
35  */
36 object ViwoodsBridge {
37 
38     private const val AI_KEY = "CustomAiKey"
39 
40     private val VOLUME_SLOT = mapOf(
41         HwKey.VOL_UP to Slot("CustomVolumeUpKey", "VolumeUpAppShortcut"),
42         HwKey.VOL_DOWN to Slot("CustomVolumeDownKey", "VolumeDownAppShortcut")
43     )
44 
45     data class Slot(val key: String, val shortcut: String)
46 
47     private const val TOKEN_APP = "shortcut_app"
48 
49     /** Every value the stock settings screen writes for a volume key. */
50     private val volumeTokens = mapOf(
51         "volume_up" to "Volume up",
52         "volume_down" to "Volume down",
53         "screenshot" to "Screenshot",
54         TOKEN_APP to "Open an app"
55     )
56 
57     fun keys(): List<HwKey> = listOf(HwKey.AI, HwKey.VOL_UP, HwKey.VOL_DOWN)
58 
59     /** Defensive: an OEM key that a later firmware hides must not take the app down. */
60     private fun get(c: Context, name: String): String? =
61         runCatching { Settings.System.getString(c.contentResolver, name) }
62             .getOrNull()?.takeIf { it.isNotBlank() && it != "null" }
63 
64     fun read(c: Context, key: HwKey): String? = when (key) {
65         HwKey.AI -> get(c, AI_KEY)
66         else -> VOLUME_SLOT[key]?.let { get(c, it.key) }
67     }
68 
69     private fun readShortcut(c: Context, key: HwKey): String? =
70         VOLUME_SLOT[key]?.let { get(c, it.shortcut) }
71 
72     /**
73      * True when the firmware is keeping this key to itself, so that bindings
74      * made for it in this app can never fire.
75      */
76     fun hidesFromFilter(c: Context, key: HwKey): Boolean =
77         dev.equwal.assistkey.device.Device.hasFirmwareKeyHooks &&
78             VOLUME_SLOT.containsKey(key) && read(c, key) != null
79 
80     private const val AI_ENTRY = "dev.equwal.assistkey.channel.AiKeyActivity"
81 
82     /**
83      * What should open when an AI press is handed back to the firmware. Never
84      * our own entry point: that would be a loop.
85      */
86     fun aiTarget(c: Context): String =
87         read(c, HwKey.AI)?.takeUnless { aiHookedToUs(c) } ?: Presets.STOCK_AI_KEY
88 
89     /** True when the firmware sends the AI key straight to this app. */
90     fun aiHookedToUs(c: Context): Boolean =
91         dev.equwal.assistkey.device.Device.hasFirmwareKeyHooks && read(c, HwKey.AI)?.let { ComponentName.unflattenFromString(it) } ==
92             ComponentName(c.packageName, AI_ENTRY)
93 
94     fun aiHookCommand(c: Context): String =
95         "adb shell settings put system " + AI_KEY + " " + c.packageName + "/" + AI_ENTRY
96 
97     fun aiUnhookCommand(): String =
98         "adb shell settings put system " + AI_KEY + " " + Presets.STOCK_AI_KEY
99 
100     /** The command that hands a volume key back to the input pipeline. */
101     fun unsetCommand(key: HwKey): String? =
102         VOLUME_SLOT[key]?.let { "adb shell settings delete system " + it.key }
103 
104     /** Human description of whatever the firmware currently has bound. */
105     fun describe(c: Context, key: HwKey): String {
106         val v = read(c, key) ?: return "Not set"
107         if (key == HwKey.AI) return if (aiHookedToUs(c)) "Rebind" else shortName(c, v)
108 
109         val label = volumeTokens[v] ?: return shortName(c, v)
110         if (v != TOKEN_APP) return label
111         val target = readShortcut(c, key)
112         return if (target == null) "Open an app (none chosen)" else "Open " + shortName(c, target)
113     }
114 
115     private fun shortName(c: Context, flat: String): String {
116         val cn = ComponentName.unflattenFromString(flat) ?: return flat
117         Presets.viwoods.firstOrNull { it.component == flat }?.let { return it.label }
118         return runCatching {
119             c.packageManager.getActivityInfo(cn, 0).loadLabel(c.packageManager).toString()
120         }.getOrElse { cn.shortClassName.substringAfterLast('.') }
121     }
