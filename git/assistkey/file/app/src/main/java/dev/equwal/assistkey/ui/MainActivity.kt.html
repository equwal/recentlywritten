107      * The drawing of the device is the way in. Each button shows what it does

111     private fun buttons(col: LinearLayout) {
112         val bound = Store.bindings(this).all()
113         val drawn = GuidedSetupActivity.drawn(this)
114         col.addView(DeviceView(this, heightDp = 270).apply {
115             keys = drawn
116             selected = this@MainActivity.selected
117             captions = drawn.associateWith { key ->
118                 Summary.caption(bound.filterKeys { key in it.keys }.values.map { it.describe() })
119             }
120             onSelect = { this@MainActivity.selected = it; build() }
121         })
122         if (selected.isEmpty()) {
123             col.note("Tap a button. Tap two for a combination.")
124         } else {
125             col.primaryButton("Set up " + selected.joinToString(" + ") { it.label }) {
126                 startActivity(GuidedSetupActivity.intent(this, selected))
127             }
128         }
129     }

158     private fun navigationSummary(): String {
159         val p = getSharedPreferences("assistkey_nav", Context.MODE_PRIVATE)
160         val buttonsNow = NavNative.buttonsShowing(this)
161         val gesturesNow = NavNative.gesturesOn(this) ?: buttonsNow?.not()
162         val power = setOf(HwKey.POWER)
163         return Summary.navigation(
164             p.getBoolean("buttons", buttonsNow ?: true),
165             p.getBoolean("gestures", gesturesNow ?: true),
166             p.getBoolean("keys", Store.bindings(this).all().keys.any { it.keys == power })
167         )
168     }
169 
170     private fun voiceSummary(): String {
171         val engines = Dictation.engines(this)
172         return Summary.voice(
173             Dictation.hasMicrophone(this),
174             engines.size,
175             engines.any(Dictation::isOnDevice)
176         )
177     }
