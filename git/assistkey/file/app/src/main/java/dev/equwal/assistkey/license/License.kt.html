8 /**
9  * Whether the app is allowed to do its job, and why.
10  *
11  * There are four tiers, tried in order:
12  *
13  *  1. LICENSED - Google Play says this account owns one of the two paid
14  *     products. Cached, so it survives being offline.
15  *  2. BETA     - the free beta is still open. Everything works.
16  *  3. TRIAL    - the beta is over; seven days from first launch to decide.
17  *  4. LOCKED   - nothing is remapped until a licence is bought.
18  *
19  * "Is the beta open" is answered by Google Play, not by a server of ours. The
20  * catalogue holds a third product, [Sku.BETA_OPEN], that is never sold; it is
21  * only a flag. While it is active the beta is open. Deactivating it in Play
22  * Console ends the beta for every install at once, sideloaded or not. The
23  * state "the paid product is visible but the flag is gone" is the only one read
24  * as closed, so an install that cannot reach Play - or a build that predates
25  * the Play listing entirely - stays open, up to the date compiled into it.
26  *
27  * Anyone who ran the app while the beta was open is remembered as a tester and
28  * is offered [Sku.PRO_TESTER], the same licence at a lower price.
29  */
30 object License {

32     /**
33      * NO_STORE: the device has no Google Play, so there is no way to buy. The
34      * app is then free and complete. A user can get this tier on purpose by
35      * turning Google Play off. The owner accepts that: the users who can pay
36      * have Google Play, and most of them do not do this.
37      */
38     enum class Tier { LICENSED, NO_STORE, BETA, TRIAL, LOCKED }
39 
40     data class State(
41         val tier: Tier,
42         /** Whole days of trial left, rounded up; zero outside TRIAL. */
43         val trialDaysLeft: Int = 0,
44         /** True when Play has positively confirmed the beta flag, not just a date. */
45         val betaConfirmedByPlay: Boolean = false
46     ) {
47         val active: Boolean get() = tier != Tier.LOCKED
48     }

86     fun state(c: Context): State {
87         val now = System.currentTimeMillis()
88         cached?.let { if (now - cachedAt < STATE_TTL_MS) return it }
89         return compute(c, now).also {
90             cached = it
91             cachedAt = now
92         }
93     }
94 
95     fun active(c: Context): Boolean = state(c).active

97     fun invalidate() {
98         cached = null
99     }

101     private fun compute(c: Context, now: Long): State {
102         val p = prefs(c)
103         val firstRun = firstRunAt(p, now)
104 
105         val owned = p.getStringSet(K_OWNED, emptySet()) ?: emptySet()
106         val fresh = now - p.getLong(K_CAT_AT, 0L) < CATALOG_FRESH_MS
107         val pro = if (fresh) p.getInt(K_CAT_PRO, Seen.UNKNOWN) else Seen.UNKNOWN
108         val beta = if (fresh) p.getInt(K_CAT_BETA, Seen.UNKNOWN) else Seen.UNKNOWN
109 
110         // The F-Droid build has no licence check: it is always the free tier.
111         val store = BuildConfig.LICENCE_CHECK && storePresent(c)
112         val state = decide(owned, store, pro, beta, now, firstRun, BuildConfig.BETA_EXPIRES_MS)
113         if (state.tier == Tier.BETA && !p.getBoolean(K_TESTER, false)) p.edit().putBoolean(K_TESTER, true).apply()
114         return state
115     }
116 
117     /** The rules, with no Android types, so that a test can run them. */
118     fun decide(
119         owned: Set<String>,
120         storePresent: Boolean,
121         pro: Int,
122         beta: Int,
123         now: Long,
124         firstRun: Long,
125         betaExpires: Long
126     ): State {
127         if (owned.any { it in Sku.LICENCES }) return State(Tier.LICENSED)
128         if (!storePresent) return State(Tier.NO_STORE)
129 
130         val open = when {
131             beta == Seen.FOUND -> true
132             pro == Seen.FOUND && beta == Seen.NOT_FOUND -> false
133             else -> now < betaExpires
134         }
135         if (open) return State(Tier.BETA, betaConfirmedByPlay = beta == Seen.FOUND)
136 
137         val left = firstRun + TRIAL_DAYS * DAY_MS - now
138         if (left > 0) {
139             return State(Tier.TRIAL, trialDaysLeft = ((left + DAY_MS - 1) / DAY_MS).toInt())
140         }
141         return State(Tier.LOCKED)
142     }

150     private fun firstRunAt(p: SharedPreferences, now: Long): Long {
151         val at = p.getLong(K_FIRST_RUN, 0L)
152         if (at in 1..now) return at
153         p.edit().putLong(K_FIRST_RUN, now).apply()
154         return now

182      * For testers who cannot carry the marker over - a new device, or a Play

185     fun redeemTesterCode(c: Context, code: String): Boolean {
186         val digest = MessageDigest.getInstance("SHA-256")
187             .digest(code.trim().uppercase().toByteArray(Charsets.UTF_8))
188             .joinToString("") { "%02x".format(it) }
189         val ok = MessageDigest.isEqual(
190             digest.toByteArray(), BuildConfig.TESTER_CODE_SHA256.toByteArray()
191         )
192         if (ok) prefs(c).edit().putBoolean(K_CODE_OK, true).apply()
193         return ok
194     }
