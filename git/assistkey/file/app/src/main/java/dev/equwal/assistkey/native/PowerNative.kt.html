8 /**
9  * The firmware's own power-button behaviour, which lives in Settings.Global /
10  * Settings.Secure rather than in anything an app can intercept.
11  *
12  * This is the only way to touch short-press Power at all: no app ever sees that
13  * event.
14  *
15  * Two separate restrictions apply, and they are not symmetric. Writing needs
16  * WRITE_SECURE_SETTINGS, which is not grantable from a settings screen - it has
17  * to come over adb. Reading is blocked outright: since Android 12 a settings
18  * key annotated @hide throws SecurityException for any non-system caller, and
19  * holding WRITE_SECURE_SETTINGS does not exempt you. So every read here returns
20  * null on refusal and the UI says "unknown" rather than inventing a value.
21  */

85     /**
86      * Switches the system's "press Power five times" Emergency SOS. It writes
87      * the setting directly when the app holds WRITE_SECURE_SETTINGS, and asks
88      * the shell otherwise. [done] gets false when neither way is open.
89      */
90     fun setEmergencySos(c: Context, on: Boolean, done: (Boolean) -> Unit) {
91         val value = if (on) "1" else "0"
92         fun remember(ok: Boolean) {
93             if (ok) c.getSharedPreferences(PREFS, Context.MODE_PRIVATE).edit().putBoolean(K_SOS, on).apply()
94             done(ok)
95         }
96         val direct = runCatching { Settings.Secure.putString(c.contentResolver, EMERGENCY_GESTURE, value) }
97             .getOrDefault(false)
98         if (direct) return remember(true)
99         if (!dev.equwal.assistkey.shell.Shell.ready) return remember(false)
100         dev.equwal.assistkey.shell.Shell.run("settings put secure $EMERGENCY_GESTURE $value") { remember(it.ok) }
101     }
