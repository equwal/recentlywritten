3 /** Tap (possibly repeated) or a sustained hold. */
4 enum class GestureType { TAP, HOLD }

6 /**
7  * One bindable gesture: a set of keys pressed together (one key = a plain
8  * press, several = a chord) performed as N taps or as a hold.
9  *
10  * Serialised as "ai+vol_up:tap:2" - keys sorted by enum ordinal so the same
11  * chord always produces the same id regardless of press order.
12  */
13 data class Trigger(
14     val keys: Set<HwKey>,
15     val type: GestureType,
16     val count: Int = 1
17 ) {
18     init {
19         require(keys.isNotEmpty()) { "trigger needs at least one key" }
20         require(count in 1..MAX_TAPS) { "count out of range: $count" }
21         require(type == GestureType.TAP || count == 1) { "hold cannot repeat" }
22     }
23 
24     val isChord: Boolean get() = keys.size > 1
25 
26     val id: String
27         get() = keys.sortedBy { it.ordinal }.joinToString("+") { it.token } +
28             ":" + type.name.lowercase() + ":" + count
29 
30     /** Human label, e.g. "AI key + Volume up - double tap". */
31     fun label(): String {
32         val k = keys.sortedBy { it.ordinal }.joinToString(" + ") { it.label }
33         val g = when {
34             type == GestureType.HOLD -> "hold"
35             count == 1 -> "tap"
36             count == 2 -> "double tap"
37             count == 3 -> "triple tap"
38             else -> "$count taps"
39         }
40         return "$k — $g"
41     }
42 
43     companion object {
44         const val MAX_TAPS = 5
45 
46         /**
47          * "Hold Power, then press [key]". Power cannot take part in an ordinary
48          * combination, but a held Power announces itself - the firmware fires
49          * the assistant at us - and a key pressed while it is still down
50          * reaches the filter. See KeyFilterService.armPowerCombo.
51          */
52         fun powerThen(key: HwKey): Trigger = Trigger(setOf(HwKey.POWER, key), GestureType.TAP, 1)
53 
54         fun parse(id: String): Trigger? {
55             val parts = id.split(":")
56             if (parts.size != 3) return null
57             val keys = parts[0].split("+").mapNotNull { HwKey.fromToken(it) }.toSet()
58             if (keys.isEmpty()) return null
59             val type = when (parts[1]) {
60                 "tap" -> GestureType.TAP
61                 "hold" -> GestureType.HOLD
62                 else -> return null
63             }
64             val count = parts[2].toIntOrNull() ?: return null
65             if (count !in 1..MAX_TAPS) return null
66             if (type == GestureType.HOLD && count != 1) return null
67             return Trigger(keys, type, count)
68         }
69 
70         /**
71          * Every gesture the UI offers for a given key set: taps 1..MAX_TAPS
72          * plus a hold.
73          */
74         fun allFor(keys: Set<HwKey>): List<Trigger> =
75             (1..MAX_TAPS).map { Trigger(keys, GestureType.TAP, it) } +
76                 Trigger(keys, GestureType.HOLD)
77     }
