7 /**
8  * What this particular device is, and what that changes.
9  *
10  * AssistKey runs on any Android 12+ device. Most of it is the same everywhere:
11  * the key filter, the Power button, navigation, shell access. A profile only
12  * adds what a maker did differently - extra keys, its own key settings, a
13  * gesture switch nobody else has, a brightness floor. Viwoods is the first
14  * profile because it is where the app was built; an unknown device is simply
15  * the generic profile plus whatever keys the key tester has seen.
16  *
17  * New profiles come from the device reports beta testers send in.
18  */

118     /**
119      * Keys this app has no name for yet. Only recorded while the key tester is
120      * open, so an ordinary keyboard being typed on never lands here; and only
121      * as "this key exists", never as a sequence.
122      */
123     fun noteUnknown(c: Context, keyCode: Int, scanCode: Int, device: String?) {
124         val p = c.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
125         val seen = p.getStringSet(K_UNKNOWN, emptySet()).orEmpty()
126         val entry = "keycode " + keyCode + " scancode " + scanCode + " on " + (device ?: "?")
127         if (entry !in seen && seen.size < 40) p.edit().putStringSet(K_UNKNOWN, seen + entry).apply()
128     }

130     fun unknownSeen(c: Context): List<String> =

132             .getStringSet(K_UNKNOWN, emptySet()).orEmpty().sorted()

134     /** Called by the key filter for every supported key it is handed. */
135     fun noteSeen(c: Context, key: HwKey) {
136         if (key in builtIn) return
137         val p = c.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
138         val seen = p.getStringSet(K_SEEN, emptySet()).orEmpty()
139         if (key.token !in seen) p.edit().putStringSet(K_SEEN, seen + key.token).apply()
