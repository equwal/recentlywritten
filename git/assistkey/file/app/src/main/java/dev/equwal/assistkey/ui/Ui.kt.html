29 /**
30  * View construction, in code, with no support library.
31  *
32  * The screen is e-ink: no colour, slow refresh, and a hard time with subtle
33  * greys. So everything here is black on white at generous sizes, outlines are
34  * real black lines, and nothing animates or ripples.
35  *
36  * The rules this file keeps, so that every screen looks the same:
37  *
38  *  - 16 dp side margins, and spacing on an 8 dp grid.
39  *  - Rows and checks are at least 56 dp high, and the whole row is the
40  *    touch target.
41  *  - Titles are 17 sp, summaries 14 sp, section headers 13 sp small caps.
42  *  - Secondary text is #444444. Nothing lighter carries a fact.
43  *  - Buttons are 48 dp high and outlined. The one primary button on a screen
44  *    is 56 dp high and filled black.
45  */

65      * Root of a screen; returns the column that callers fill.
66      *
67      * With a [title] the screen gets the standard top bar: a back arrow with a
68      * 48 dp touch target, the title, and a rule under it. The content below

71     fun page(a: Activity, title: String? = null): LinearLayout {
72         val root = LinearLayout(a).apply {
73             orientation = LinearLayout.VERTICAL
74             setBackgroundColor(Color.WHITE)
75         }
76         if (title != null) {
77             root.addView(topBar(a, title))
78             root.addView(View(a).apply {
79                 setBackgroundColor(RULE)
80                 layoutParams = LinearLayout.LayoutParams(MATCH_PARENT, 1)
81             })
82         }
83 
84         val col = LinearLayout(a).apply {
85             orientation = LinearLayout.VERTICAL
86             setPadding(a.dp(16), a.dp(if (title == null) 8 else 12), a.dp(16), a.dp(24))
87         }
88         val scroll = ScrollView(a).apply {
89             isFillViewport = true
90             clipToPadding = false
91             addView(col, LinearLayout.LayoutParams(MATCH_PARENT, WRAP_CONTENT))
92         }
93         root.addView(scroll, LinearLayout.LayoutParams(MATCH_PARENT, 0, 1f))
94 
95         // Edge-to-edge is enforced from Android 15, so the system bars overlap
96         // the window. The top bar moves down by whatever the status bar covers,
97         // and the scrolling content keeps clear of the navigation bar.
98         root.setOnApplyWindowInsetsListener { v, insets ->
99             val bars = insets.getInsets(
100                 WindowInsets.Type.systemBars() or WindowInsets.Type.displayCutout()
101             )
102             v.setPadding(bars.left, bars.top, bars.right, 0)
103             col.setPadding(
104                 col.paddingLeft, col.paddingTop, col.paddingRight, a.dp(24) + bars.bottom
105             )
106             insets
107         }
108         a.setContentView(root)
109         root.requestApplyInsets()
110         return col
111     }

292      * Two columns of outlined tiles. A tile left over at the end takes the

295     fun LinearLayout.tiles(items: List<Tile>) {
296         var i = 0
297         while (i < items.size) {
298             if (items.size - i == 1) {
299                 add(tile(context, items[i]).apply {
300                     layoutParams = LinearLayout.LayoutParams(MATCH_PARENT, WRAP_CONTENT).apply {
301                         topMargin = context.dp(8)
302                     }
303                 })
304                 i += 1
305                 continue
306             }
307             val pair = LinearLayout(context).apply {
308                 orientation = LinearLayout.HORIZONTAL
309                 layoutParams = LinearLayout.LayoutParams(MATCH_PARENT, WRAP_CONTENT).apply {
310                     topMargin = context.dp(8)
311                 }
312             }
313             pair.addView(tile(context, items[i]), LinearLayout.LayoutParams(0, WRAP_CONTENT, 1f).apply {
314                 marginEnd = context.dp(8)
315             })
316             pair.addView(tile(context, items[i + 1]), LinearLayout.LayoutParams(0, WRAP_CONTENT, 1f))
317             add(pair)
318             i += 2
319         }
320     }

434     /** Empty square, or square with a solid black block in it. */
435     private fun checkBox(c: Context): StateListDrawable {
436         val size = c.dp(24)
437         val empty = GradientDrawable().apply {
438             shape = GradientDrawable.RECTANGLE
439             setColor(Color.WHITE)
440             cornerRadius = c.dp(2).toFloat()
441             setStroke(c.px(2), INK)
442             setSize(size, size)
443         }
444         val frame = GradientDrawable().apply {
445             shape = GradientDrawable.RECTANGLE
446             setColor(Color.WHITE)
447             cornerRadius = c.dp(2).toFloat()
448             setStroke(c.px(2), INK)
449             setSize(size, size)
450         }
451         val block = GradientDrawable().apply {
452             shape = GradientDrawable.RECTANGLE
453             setColor(INK)
454         }
455         val ticked = LayerDrawable(arrayOf(frame, block)).apply {
456             val inset = c.dp(6)
457             setLayerInset(1, inset, inset, inset, inset)
458         }
459         return StateListDrawable().apply {
460             addState(intArrayOf(android.R.attr.state_checked), ticked)
461             addState(intArrayOf(), empty)
462         }
463     }

465     /**
466      * The back arrow and the row chevron, drawn rather than typed, because a
467      * firmware font may not carry the characters.
468      */
469     private class Glyph(c: Context, private val kind: Int) : View(c) {
470 
471         private val paint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
472             color = INK
473             style = Paint.Style.STROKE
474             strokeWidth = c.resources.displayMetrics.density * 2f
475             strokeCap = Paint.Cap.ROUND
476             strokeJoin = Paint.Join.ROUND
477         }
478         private val path = Path()
479 
480         override fun onDraw(canvas: Canvas) {
481             val cx = width / 2f
482             val cy = height / 2f
483             path.reset()
484             if (kind == BACK) {
485                 val r = minOf(width, height) * 0.24f
486                 val h = r * 0.62f
487                 path.moveTo(cx + r, cy)
488                 path.lineTo(cx - r, cy)
489                 path.moveTo(cx - r + h, cy - h)
490                 path.lineTo(cx - r, cy)
491                 path.lineTo(cx - r + h, cy + h)
492             } else {
493                 val w = minOf(width, height) * 0.17f
494                 val h = w * 1.7f
495                 path.moveTo(cx - w, cy - h)
496                 path.lineTo(cx + w, cy)
497                 path.lineTo(cx - w, cy + h)
498             }
499             canvas.drawPath(path, paint)
500         }
501 
502         companion object {
503             const val BACK = 0
504             const val CHEVRON = 1
505         }
506     }
