10 /**
11  * Persistence. Bindings live as one JSON blob so adding an action kind never
12  * needs a migration, and the parsed result is cached because the key-event
13  * path must not touch disk.

21     private const val K_CHORD = "chord_ms"

25     private fun prefs(c: Context): SharedPreferences =
26         c.applicationContext.getSharedPreferences(PREFS, Context.MODE_PRIVATE)

28     fun bindings(c: Context): Bindings = cache ?: load(c).also { cache = it }

30     private fun load(c: Context): Bindings {
31         val raw = prefs(c).getString(K_BINDINGS, null) ?: return Bindings.EMPTY
32         return runCatching {
33             val o = JSONObject(raw)
34             val m = HashMap<Trigger, ActionSpec>()
35             o.keys().forEach { id ->
36                 val t = Trigger.parse(id) ?: return@forEach
37                 m[t] = ActionSpec.fromJson(o.getJSONObject(id))
38             }
39             Bindings(m)
40         }.getOrDefault(Bindings.EMPTY)
41     }
42 
43     fun save(c: Context, b: Bindings) {
44         val o = JSONObject()
45         b.all().forEach { (t, spec) -> o.put(t.id, spec.toJson()) }
46         prefs(c).edit().putString(K_BINDINGS, o.toString()).apply()
47         cache = b
48         onBindingsChanged?.invoke()
49     }
50 
51     /** Set by the key service, which has things to re-decide when bindings move. */
52     @Volatile var onBindingsChanged: (() -> Unit)? = null
53 
54     fun bind(c: Context, trigger: Trigger, spec: ActionSpec) {
55         save(c, bindings(c).with(trigger, spec))
56     }
57 
58     fun timing(c: Context): GestureEngine.Config {
59         val p = prefs(c)
60         val d = GestureEngine.Config()
61         return GestureEngine.Config(

64             chordMs = p.getLong(K_CHORD, d.chordMs)
65         )
66     }

68     fun setTiming(c: Context, cfg: GestureEngine.Config) {
69         prefs(c).edit()

71             .putLong(K_HOLD, cfg.holdMs)
72             .putLong(K_CHORD, cfg.chordMs)
73             .apply()
74     }
