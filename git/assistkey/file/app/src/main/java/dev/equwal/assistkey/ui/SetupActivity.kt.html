34             "Button remapping",

40             build()
41         }
42         if (on && !ok) {
43             col.button("Turn it on") { claim(ch) }
44             col.note("If the switch turns itself off, allow restricted settings in App info.")

46                 Channels.safeStart(this, Channels.appInfoIntent(this))
47             }
48         }
49 
50         val granted = PermissionsActivity.granted(this)
51         col.row(
52             "Permissions",
53             state = if (granted.all { it }) "All granted"
54             else (granted.size - granted.count { it }).toString() + " missing"
55         ) { startActivity(Intent(this, PermissionsActivity::class.java)) }
56 
57         val s = License.state(this)
58         col.row(
59             "Licence",

62 
63         // Rebind has no internet permission, so it cannot look for a new version.

68     private fun claim(ch: Channel) {
