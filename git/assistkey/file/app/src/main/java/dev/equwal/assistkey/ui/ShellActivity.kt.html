15 /**
16  * Getting shell access, from the device itself.
17  *
18  * Shizuku is the piece that makes it possible without a computer: it pairs with
19  * Android's own wireless debugging over the loopback interface and then runs a
20  * small server as the shell user. This screen walks through that and then asks
21  * Shizuku for permission.
22  */
23 class ShellActivity : Activity() {
24 
25     private val redraw: () -> Unit = { if (!isFinishing) build() }
26 
27     override fun onResume() {
28         super.onResume()
29         Shell.onChange(redraw)
30         Shell.connect(this)
31         build()
32     }
33 
34     override fun onPause() {
35         super.onPause()
36         Shell.removeOnChange(redraw)
37     }
38 
39     private fun build() {
40         val state = Shell.state(this)
41         val col = Ui.page(this, "Shell access")
42         col.row("Status", null, enabled = false, state = Shell.describe(this))
43         col.note("Shizuku, a free app, sets this up with no computer.")
44 
45         if (state == Shell.State.READY) {
46             col.header("What this unlocks")
47             col.note("Every Power button gesture, the button bar and system gestures.")
48             return
49         }
50 
51         steps(col, state)
52     }
53 
54     private fun steps(col: LinearLayout, state: Shell.State) {
55         col.header("1. Install Shizuku")
56         if (state == Shell.State.NOT_INSTALLED) {
57             col.button("Get Shizuku") {
58                 val market = Intent(Intent.ACTION_VIEW, Uri.parse("market://details?id=" + Shell.SHIZUKU_PACKAGE))
59                 val web = Intent(Intent.ACTION_VIEW, Uri.parse("https://shizuku.rikka.app/download/"))
60                 if (!tryStart(market)) tryStart(web)
61             }
62         } else {
63             col.note("Done.")
64         }
65 
66         col.header("2. Turn on wireless debugging")
67         col.note(
68             "Tap Build number seven times in Settings > About. Then switch on " +
69                 "Wireless debugging. It needs Wi-Fi."
70         )
71         col.button("Open Developer options") {
72             if (!tryStart(Intent(Settings.ACTION_APPLICATION_DEVELOPMENT_SETTINGS))) {
73                 tryStart(Intent(Settings.ACTION_DEVICE_INFO_SETTINGS))
74             }
75         }
76 
77         col.header("3. Start Shizuku")
78         col.note("In Shizuku choose Pairing, pair with the code, then press Start.")
79         if (state != Shell.State.NOT_INSTALLED) {
80             col.button("Open Shizuku") {
81                 packageManager.getLaunchIntentForPackage(Shell.SHIZUKU_PACKAGE)?.let { tryStart(it) }
82             }
83         }
84 
85         col.header("4. Allow Rebind")
86         when (state) {
87             Shell.State.NO_PERMISSION -> col.button("Ask Shizuku for permission") {
88                 Shell.requestPermission()
89             }
90             else -> col.note("Once Shizuku is running, come back here.")
91         }
92     }
93 
94     private fun tryStart(i: Intent): Boolean =
95         runCatching { startActivity(i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)); true }.getOrDefault(false)
