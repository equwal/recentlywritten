105     /**
106      * A Power press on a sleeping or locked device has one job: wake it and
107      * unlock it. It is never a gesture. The firmware normally wakes the device
108      * by itself, but the button is ours now, so this does not rely on that: the
109      * wake and the keyguard dismissal are sent through the shell as well.
110      */
111     private fun onRawPower(down: Boolean, at: Long) {
112         if (!::engine.isInitialized) return
113         if (!down) {
114             if (powerIgnoredUntilUp) powerIgnoredUntilUp = false else engine.onUp(HwKey.POWER, at)
115             return
116         }
117         val asleep = screenOff ||
118             getSystemService(android.os.PowerManager::class.java)?.isInteractive == false
119         val locked = getSystemService(android.app.KeyguardManager::class.java)?.isKeyguardLocked == true
120         val justWoke = SystemClock.uptimeMillis() - wokeAt < WAKE_GRACE_MS
121         if (asleep || locked || justWoke) {
122             powerIgnoredUntilUp = true
123             engine.onCancel()
124             if (asleep || locked) Shell.run("input keyevent KEYCODE_WAKEUP; wm dismiss-keyguard")
125             return
126         }
127         powerIgnoredUntilUp = false
128         engine.onDown(HwKey.POWER, at, 0)
129     }

159     /**
160      * Called when the firmware reports a Power hold. For [POWER_COMBO_WINDOW_MS]
161      * the next key press is read as a Power combination. If none comes, the
162      * plain hold action runs - late by exactly that window, which is the price
163      * of having combinations at all and is only paid when one is bound.
164      *
165      * [lifeline] is the locked state. A reader set up with no button bar and no
166      * gestures is navigated entirely from here, and must not become a brick
167      * because a trial ran out: Back, Home and Recents always work.
168      */
169     fun armPowerCombo(plainHold: ActionSpec?, lifeline: Boolean) {
170         holdFallback?.let { handler.removeCallbacks(it) }
171         armedUntil = SystemClock.uptimeMillis() + POWER_COMBO_WINDOW_MS
172         lifelineOnly = lifeline
173         val r = Runnable {
174             armedUntil = 0L
175             holdFallback = null
176             if (!lifeline) plainHold?.let { ActionRouter.run(this, it) }
177         }
178         holdFallback = r
179         handler.postDelayed(r, POWER_COMBO_WINDOW_MS)
180     }

182     private fun powerCombo(event: KeyEvent, key: HwKey): Boolean {
183         if (event.action == KeyEvent.ACTION_UP) return swallowed.remove(key)
184         if (event.action != KeyEvent.ACTION_DOWN) return false
185         if (key in swallowed) return true // auto-repeat of a press already taken
186         if (!takePowerCombo(key)) return false
187         swallowed.add(key)
188         return true
189     }
190 
191     /**
192      * If a Power hold is waiting for its second key and [key] is bound as one,
193      * run it and report true. Public because the AI key, once its firmware hook
194      * points at this app, arrives as an activity launch rather than a key event.
195      */
196     fun takePowerCombo(key: HwKey): Boolean {
197         if (SystemClock.uptimeMillis() >= armedUntil) return false
198         val spec = bindings().powerCombo(key) ?: return false
199         if (lifelineOnly && !isLifeline(spec)) return false
200 
201         armedUntil = 0L
202         holdFallback?.let { handler.removeCallbacks(it) }
203         holdFallback = null
204         // A beat, so that whichever invisible window of ours is in front has
205         // gone before Back or Home lands on it.
206         handler.postDelayed({ ActionRouter.run(this, spec) }, 150L)
207         return true
208     }

210     private fun isLifeline(spec: ActionSpec): Boolean =
211         spec.kind == ActionKind.GLOBAL && spec.payload in LIFELINE

213     override fun onKeyEvent(event: KeyEvent): Boolean {
214         val key = HwKey.fromCode(event.keyCode)
215         if (key != null && event.action == KeyEvent.ACTION_DOWN && event.repeatCount == 0) {
216             dev.equwal.assistkey.device.Device.noteSeen(this, key)
217         }
218         val consumed = when {
219             !Channels.isEnabled(this, Channel.ACCESSIBILITY) -> false
220             key != null && License.active(this) && aiKeyReturn(event, key) -> true
221             // Ahead of the licence check on purpose - see armPowerCombo.
222             key != null && key.interceptable && powerCombo(event, key) -> true
223             // Locked means inert, not broken: every key goes to the firmware.
224             !License.active(this) -> false
225             key == null || !key.interceptable -> false
226             event.action == KeyEvent.ACTION_DOWN ->
227                 engine.onDown(key, event.eventTime, event.repeatCount)
228             event.action == KeyEvent.ACTION_UP -> engine.onUp(key, event.eventTime)
229             else -> false
230         }
231         // Every event, not just the ones we act on: the tester screen is the
232         // only way to find out whether a key reaches a filter on this firmware.
233         // KeyLog drops it on the floor unless that screen is open.
234         KeyLog.record(event, key?.label ?: ("Key " + event.keyCode), consumed)
235         if (key == null && KeyLog.recording && event.action == KeyEvent.ACTION_DOWN) {
236             dev.equwal.assistkey.device.Device.noteUnknown(
237                 this, event.keyCode, event.scanCode, event.device?.name
238             )
239         }
240         return consumed
241     }

254      * Window changes are watched for one purpose: to know which app to go back

258     override fun onAccessibilityEvent(event: AccessibilityEvent?) {
259         if (event?.eventType != AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED) return
260         val pkg = event.packageName?.toString() ?: return
261         val cls = event.className?.toString()
262         when {
263             // Before the line that skips the system UI: stock Android has its recents there.
264             dev.equwal.assistkey.device.Device.isSystemRecents(pkg, cls) -> swapInInkRecents()
265             pkg == packageName || pkg == "android" || pkg == "com.android.systemui" -> Unit
266             dev.equwal.assistkey.device.Device.isAiScreen(pkg, cls) -> inAiScreen = true
267             // Dialogs and keyboards report window changes too; only activities count.
268             cls != null && isActivity(pkg, cls) -> {
269                 inAiScreen = false
270                 lastApp = pkg
271             }
272         }
273     }

275     /**
276      * The Recents button of the bar, the swipe, and the Recents action all open
277      * the recent-apps screen of the system. Where the user has Ink Recents, this
278      * closes that screen and opens Ink Recents. Android has no other way to put
279      * an app in that place.
280      */
281     private fun swapInInkRecents() {
282         if (!License.active(this) || !dev.equwal.assistkey.device.Device.inkRecentsForSystem(this)) return
283         val open = android.content.Intent(dev.equwal.assistkey.home.RecentsActivity.ACTION_OPEN)
284             .setPackage(dev.equwal.assistkey.home.RecentsActivity.INK_RECENTS)
285             .addFlags(android.content.Intent.FLAG_ACTIVITY_NEW_TASK)
286         if (open.resolveActivity(packageManager) == null) return
287         // Back first, so that the screen of the system is not left under Ink Recents.
288         performGlobalAction(GLOBAL_ACTION_BACK)
289         handler.postDelayed({ runCatching { startActivity(open) } }, 120L)
290     }

296     /**
297      * Leaves the AI screen for the app that was in use before it. The firmware
298      * reacts to the AI key by itself and may open its AI screen again, so this
299      * waits for that to land first.
300      */
301     fun returnFromAiScreen() {
302         handler.postDelayed({
303             val pkg = lastApp
304             val launch = pkg?.let { packageManager.getLaunchIntentForPackage(it) }
305             val isHome = pkg != null && packageManager.resolveActivity(
306                 android.content.Intent(android.content.Intent.ACTION_MAIN)
307                     .addCategory(android.content.Intent.CATEGORY_HOME),
308                 0
309             )?.activityInfo?.packageName == pkg
310             if (launch == null || isHome) {
311                 performGlobalAction(GLOBAL_ACTION_HOME)
312             } else {
313                 // The same intent the launcher sends: it brings the task back as it was.
314                 startActivity(
315                     launch.addFlags(
316                         android.content.Intent.FLAG_ACTIVITY_NEW_TASK or
317                             android.content.Intent.FLAG_ACTIVITY_RESET_TASK_IF_NEEDED
318                     )
319                 )
320             }
321         }, AI_RETURN_DELAY_MS)
322     }

324     private fun aiKeyReturn(event: KeyEvent, key: HwKey): Boolean {
325         if (key != HwKey.AI) return false
326         if (event.action == KeyEvent.ACTION_UP) return aiSwallowed.remove(key)
327         if (key in aiSwallowed) return true
328         if (!inAiScreen || !dev.equwal.assistkey.device.Device.aiKeyReturns(this)) return false
329         aiSwallowed.add(key)
330         returnFromAiScreen()
331         return true
332     }

380     /**
381      * We swallowed the key-down, so anything the system would normally have
382      * done has to be reproduced by hand.
383      */
384     override fun passThrough(keys: Set<HwKey>, longPress: Boolean) {
385         if (keys.size != 1) return
386         when (keys.first()) {
387             HwKey.VOL_UP -> adjust(AudioManager.ADJUST_RAISE)
388             HwKey.VOL_DOWN -> adjust(AudioManager.ADJUST_LOWER)
389             // Whatever the firmware would have opened, not a guess at it.
390             HwKey.AI -> ActionRouter.run(
391                 this,
392                 ActionSpec(ActionKind.LAUNCH_COMPONENT, ViwoodsBridge.aiTarget(this))
393             )
394             // An unbound Power gesture does what Power always did.
395             HwKey.POWER -> performGlobalAction(GLOBAL_ACTION_LOCK_SCREEN)
396             // A swallowed press cannot be replayed to the app in front. Keys
397             // with a system-wide meaning are handled above; the rest are lost,
398             // which is why a key with nothing bound is never swallowed at all.
399             else -> Unit
400         }
401     }
