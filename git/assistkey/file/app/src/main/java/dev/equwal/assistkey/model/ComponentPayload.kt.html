3 /**
4  * The payload of [ActionKind.LAUNCH_COMPONENT]: "pkg/class", and after a "?"
5  * the text extras of the intent as "key=value" with "&" between them.
6  *
7  * The extras exist for a screen that does something else when it gets one.
8  * The Viwoods AI screen starts its voice prompt with recodeKey=recode_key_start.
9  * A payload from before has no "?" and reads as it always did.
10  */
11 object ComponentPayload {
12 
13     data class Parts(val component: String, val extras: Map<String, String>)
14 
15     fun parse(payload: String): Parts {
16         val component = payload.substringBefore('?')
17         val extras = LinkedHashMap<String, String>()
18         if ('?' in payload) {
19             payload.substringAfter('?').split('&').forEach { pair ->
20                 val key = pair.substringBefore('=')
21                 if (key.isNotEmpty() && '=' in pair) extras[key] = pair.substringAfter('=')
22             }
23         }
24         return Parts(component, extras)
25     }
26 
27     fun format(component: String, extras: Map<String, String>): String =
28         if (extras.isEmpty()) component
29         else component + "?" + extras.entries.joinToString("&") { it.key + "=" + it.value }
