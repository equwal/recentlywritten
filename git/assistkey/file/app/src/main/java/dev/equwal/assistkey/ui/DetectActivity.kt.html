15 /**
16  * What AssistKey found on this device: the buttons, and what the device lets
17  * the app do.
18  *
19  * Detection is passive. The app reads Android's key tables, the input devices
20  * the framework lists, and - with shell access - the key codes each input node
21  * declares. It never asks anybody to press a button, and it records no press.
22  *
23  * "Unknown" is an answer. Android hides some settings from apps, so a state the
24  * app may not read says so instead of guessing.
25  */
26 class DetectActivity : Activity() {
27 
28     private var busy = false
29 
30     override fun onResume() {
31         super.onResume()
32         Shell.connect(this)
33         build()
34         if (Detect.stored(this) == null) detect()
35     }
36 
37     private fun detect() {
38         if (busy) return
39         busy = true
40         build()
41         Detect.run(this) { r ->
42             Detect.save(this, r)
43             busy = false
44             if (!isFinishing) build()
45         }
46     }
47 
48     private fun build() {
49         val col = Ui.page(this, "Detect this device")
50 
51         val found = Detect.stored(this)
52         if (found == null) {
53             col.note(if (busy) "Detecting..." else "Nothing detected yet.")
54             col.primaryButton("Detect again") { detect() }
55             return
56         }
57 
58         col.header("This device")
59         col.row(found.name, found.profile + " · Android " + Build.VERSION.RELEASE)
60 
61         buttons(col, found)
62         capabilities(col, found)
63 
64         if (Shell.SUPPORTED && !Shell.ready) {
65             col.note("Shell access finds buttons Android does not declare.")
66         }
67         col.primaryButton(if (busy) "Detecting..." else "Detect again") { detect() }
68     }
69 
70     private fun buttons(col: LinearLayout, found: Detect.Result) {
71         col.header("Keys the device declares")
72         if (found.keys.isEmpty()) {
73             col.note("None found.")
74             return
75         }
76         found.keys.forEach { key ->
77             val real = key == HwKey.POWER || key in Device.keys(this)
78             col.row(
79                 key.label,
80                 "Code " + key.code + (if (key.interceptable) "" else " · kept by the system"),
81                 state = if (real) null else "No such button seen"
82             )
83         }
84     }
85 
86     private fun capabilities(col: LinearLayout, found: Detect.Result) {
87         col.header("What it can do")
88         Detect.capabilityNames.forEach { name ->
89             col.row(name, state = found.capability(name).label)
90         }
91     }
