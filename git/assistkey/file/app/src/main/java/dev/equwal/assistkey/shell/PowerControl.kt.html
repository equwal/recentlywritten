9 /**
10  * Taking the Power button over, and giving it back.
11  *
12  * No app is ever shown the Power key: the window manager acts on it before
13  * input reaches anybody. But the shell user can read the kernel input nodes,
14  * so with shell access the button can be watched from outside the framework.
15  * To make that useful the firmware has to stop reacting first - short press,
16  * long press and the double-tap camera gesture are all switched to "nothing" -
17  * and from then on every tap, double tap and hold is ours to time and to bind,
18  * through the same gesture engine that serves the other keys.
19  *
20  * The firmware's own values are saved before they are touched and restored the
21  * moment any of this stops being true: shell access lost, the app locked, the
22  * user switching it off. Restoring does not need shell access; the app grants
23  * itself WRITE_SECURE_SETTINGS through the shell the first time it connects,
24  * for exactly this reason. A restart that leaves Shizuku stopped therefore
25  * leaves a Power button that works normally, not one that does nothing.
26  *
27  * Power + Volume up is always left opening the power menu.
28  */
29 object PowerControl {
30 
31     private const val TAG = "AssistKey"
32     private const val PREFS = "assistkey_power"
33     private const val K_WANTED = "wanted"
34     private const val K_TAKEN = "taken"
35     private const val K_SAVED = "saved_"
36 
37     const val KEY_POWER = 116
38 
39     private class Knob(val scope: String, val name: String, val managed: String)
40 
41     private val knobs = listOf(
42         Knob("global", "power_button_short_press", "0"),
43         Knob("global", "power_button_long_press", "0"),
44         Knob("secure", "camera_double_tap_power_gesture_disabled", "1"),
45         Knob("secure", "double_tap_power_button_gesture_enabled", "0"),
46         // The way out, whatever else happens.
47         Knob("global", "key_chord_power_volume_up", "2")
48     )

50     private val main = android.os.Handler(android.os.Looper.getMainLooper())

52     /** True while raw Power presses are reaching the gesture engine. */
53     @Volatile var active = false
54         private set
55 
56     private fun prefs(c: Context) = c.applicationContext.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
57 
58     /** The user's switch. On by default: it only takes effect with shell access. */
59     fun wanted(c: Context): Boolean = prefs(c).getBoolean(K_WANTED, true)
60 
61     fun setWanted(c: Context, on: Boolean) {
62         prefs(c).edit().putBoolean(K_WANTED, on).apply()
63     }

65     private fun anyPowerBinding(c: Context): Boolean =
66         Store.bindings(c).all().any { (t, _) -> dev.equwal.assistkey.model.HwKey.POWER in t.keys }
67 
68     /**
69      * Brings reality into line with what should be true right now. Cheap and
70      * idempotent; call it whenever shell access, the licence, the bindings or
71      * the service may have changed. [onKey] is where presses are delivered.
72      */
73     fun sync(c: Context, serviceRunning: Boolean, onKey: (down: Boolean, at: Long) -> Unit) {
74         val should = serviceRunning && Shell.ready && wanted(c) &&
75             License.active(c) && anyPowerBinding(c)
76         when {
77             should && !active -> take(c, onKey)
78             !should && (active || prefs(c).getBoolean(K_TAKEN, false)) -> release(c)
79         }
80     }
81 
82     private var taking = false
83 
84     private fun take(c: Context, onKey: (Boolean, Long) -> Unit) {
85         if (taking) return
86         taking = true
87         main.postDelayed({ taking = false }, 8000L)
88         val p = prefs(c)
89         val read = knobs.joinToString("; ") { "settings get " + it.scope + " " + it.name }
90         Shell.run(read) { r ->
91             if (!r.ok) return@run
92             // Save once. A second take() after a crash must not record our own
93             // values as if they were the firmware's.
94             if (!p.getBoolean(K_TAKEN, false)) {
95                 val lines = r.output.trim().lines()
96                 val e = p.edit()
97                 knobs.forEachIndexed { i, k -> e.putString(K_SAVED + k.name, lines.getOrNull(i)?.trim() ?: "null") }
98                 e.putBoolean(K_TAKEN, true).apply()
99             }
100             Shell.runAll(knobs.map { "settings put " + it.scope + " " + it.name + " " + it.managed }) {
101                 Shell.watchKeys(setOf(KEY_POWER), { _, down, at -> onKey(down, at) }) { ok ->
102                     active = ok
103                     taking = false
104                     Log.i(TAG, "power button: " + if (ok) "managed" else "watch failed")
105                     if (!ok) release(c)
106                 }
107             }
108         }
109     }
110 
111     private fun release(c: Context) {
112         active = false
113         Shell.stopWatching()
114         val p = prefs(c)
115         if (!p.getBoolean(K_TAKEN, false)) return
116 
117         val saved = knobs.associateWith { p.getString(K_SAVED + it.name, "null") ?: "null" }
118         if (Shell.ready) {
119             Shell.runAll(saved.map { (k, v) ->
120                 if (v == "null") "settings delete " + k.scope + " " + k.name
121                 else "settings put " + k.scope + " " + k.name + " " + v
122             }) { ok -> if (ok) p.edit().putBoolean(K_TAKEN, false).apply() }
123             return
124         }
125         // No shell: fall back on the permission the shell granted us earlier.
126         val ok = saved.all { (k, v) ->
127             runCatching {
128                 val value = if (v == "null") null else v
129                 if (k.scope == "global") Settings.Global.putString(c.contentResolver, k.name, value)
130                 else Settings.Secure.putString(c.contentResolver, k.name, value)
131             }.getOrDefault(false)
132         }
133         if (ok) p.edit().putBoolean(K_TAKEN, false).apply()
134         Log.i(TAG, "power button: handed back to the firmware" + if (ok) "" else " (incomplete)")
135     }
