51     private fun keyRow(col: LinearLayout, key: HwKey) {
52         val hidden = ViwoodsBridge.hidesFromFilter(this, key)
53         col.row(
54             key.label,
55             if (hidden) "See Device button settings"
56             else boundSummary(setOf(key)),
57             state = if (hidden) "Hidden" else null
58         ) { startActivity(TriggerListActivity.intent(this, setOf(key))) }
59     }

61     private fun boundSummary(keys: Set<HwKey>): String {
62         val b = Store.bindings(this)
63         val bound = Trigger.allFor(keys).filter { b.isBound(it) }
64         if (bound.isEmpty()) return "Default behaviour"
65         return bound.joinToString(", ") { t -> shortGesture(t) + " → " + b.raw(t).describe() }
66     }
67 
68     private fun shortGesture(t: Trigger): String = when {
69         t.type == GestureType.HOLD -> "hold"
70         t.count == 1 -> "tap"
71         t.count == 2 -> "double"
72         t.count == 3 -> "triple"
73         else -> t.count.toString() + " taps"
74     }
75 
76     private fun powerBindings(): Int =
