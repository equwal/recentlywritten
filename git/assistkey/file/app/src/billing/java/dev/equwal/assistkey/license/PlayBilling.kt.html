23 /**
24  * Everything that touches Google Play Billing.
25  *
26  * The rest of the app never sees a billing type: this object turns what Play
27  * says into [License.recordOwned] and [License.recordCatalog], and the licence
28  * is computed from those. That keeps the key-event hot path free of IPC, and
29  * means a device with no Play Store at all simply never writes anything.
30  *
31  * All state here is confined to the main thread. Billing callbacks are
32  * re-posted onto it rather than trusted to arrive there.
33  */
34 object PlayBilling {

69     /**
70      * Connects if need be, then re-reads the catalogue and the purchases.
71      * Callers that arrive while one is running share its result.
72      */
73     fun refresh(c: Context, done: (() -> Unit)? = null) {
74         val app = c.applicationContext
75         main.post {
76             done?.let { waiters.add(it) }
77             if (inFlight) return@post
78             inFlight = true
79             connect(app) { ok ->
80                 if (!ok) finish() else queryCatalog(app) { queryOwned(app) { finish() } }
81             }
82         }
83     }
84 
85     private fun finish() {
86         inFlight = false
87         val run = ArrayList(waiters)
88         waiters.clear()
89         run.forEach { it() }
90         onChange?.invoke()
91     }

93     private fun connect(app: Context, then: (Boolean) -> Unit) {
94         val existing = client
95         if (existing != null && existing.isReady) {
96             then(true)
97             return
98         }
99         val fresh = existing ?: try {
100             BillingClient.newBuilder(app)
101                 .setListener(purchaseListener(app))
102                 .enablePendingPurchases(
103                     PendingPurchasesParams.newBuilder().enableOneTimeProducts().build()
104                 )
105                 .build()
106                 .also { client = it }
107         } catch (e: Exception) {
108             // No Play Store, or one too old to speak this protocol.
109             Log.w(TAG, "billing client could not be built", e)
110             unavailable("Google Play billing is not available on this device.")
111             then(false)
112             return
113         }
114 
115         var answered = false
116         fresh.startConnection(object : BillingClientStateListener {
117             override fun onBillingSetupFinished(result: BillingResult) {
118                 main.post {
119                     if (answered) return@post
120                     answered = true
121                     Log.i(TAG, "billing setup: " + result.responseCode + " " + result.debugMessage)
122                     if (result.responseCode == BillingResponseCode.OK) {
123                         availability = Availability.READY
124                         then(true)
125                     } else {
126                         unavailable(explain(result))
127                         then(false)
128                     }
129                 }
130             }
131 
132             override fun onBillingServiceDisconnected() {
133                 // Nothing to do here. Every refresh checks isReady and connects
134                 // again if need be, which is one attempt with a prompt answer.
135                 // The library's own auto-reconnection was tried and dropped: on
136                 // a device with the Play Store disabled - how this reader ships
137                 // - it retries three times per call and holds each answer back
138                 // for seconds.
139             }
140         })
141     }

143     private fun unavailable(why: String) {
144         availability = Availability.UNAVAILABLE

146     }

148     private fun queryCatalog(app: Context, then: () -> Unit) {
149         val c = client ?: return then()
150         val params = QueryProductDetailsParams.newBuilder()
151             .setProductList(
152                 Sku.ALL.map {
153                     QueryProductDetailsParams.Product.newBuilder()
154                         .setProductId(it)
155                         .setProductType(ProductType.INAPP)
156                         .build()
157                 }
158             )
159             .build()
160 
161         c.queryProductDetailsAsync(params) { result, response ->
162             main.post {
163                 if (result.responseCode == BillingResponseCode.OK) {
164                     details.clear()
165                     response.productDetailsList.forEach { details[it.productId] = it }
166                     val missing = response.unfetchedProductList
167                         .associate { it.productId to it.statusCode }
168                     License.recordCatalog(
169                         app,
170                         pro = seen(Sku.PRO, missing),
171                         beta = seen(Sku.BETA_OPEN, missing)
172                     )
173                     lastProblem = null
174                 } else {
175                     // Keep whatever was cached; one bad answer proves nothing.
176                     lastProblem = explain(result)
177                     Log.i(TAG, "catalogue query: " + result.responseCode + " " + result.debugMessage)
178                 }
179                 then()
180             }
181         }
182     }

184     /**
185      * Present means found. Absent means not found - unless Play said it could
186      * not tell, in which case neither can we. A product whose only purchase
187      * option has been switched off comes back as NO_ELIGIBLE_OFFER rather than
188      * PRODUCT_NOT_FOUND; for a flag that means the same thing.
189      */
190     private fun seen(productId: String, missing: Map<String, Int>): Int = when {
191         details.containsKey(productId) -> License.Seen.FOUND
192         missing[productId] == UnfetchedProduct.StatusCode.UNKNOWN -> License.Seen.UNKNOWN
193         else -> License.Seen.NOT_FOUND
194     }

196     private fun queryOwned(app: Context, then: () -> Unit) {
197         val c = client ?: return then()
198         val params = QueryPurchasesParams.newBuilder().setProductType(ProductType.INAPP).build()
199         c.queryPurchasesAsync(params) { result, purchases ->
200             main.post {
201                 if (result.responseCode == BillingResponseCode.OK) {
202                     absorb(app, purchases, replace = true)
203                 } else {
204                     Log.i(TAG, "purchase query: " + result.responseCode + " " + result.debugMessage)
205                 }
206                 then()
207             }
208         }
209     }

213     /** Returns false, with [lastProblem] set, if the purchase sheet could not open. */
214     fun buy(activity: Activity, productId: String): Boolean {
215         val product = details[productId]
216         val c = client
217         if (product == null || c == null || !c.isReady) {
218             lastProblem = "Google Play has not answered yet. Try again in a moment."
219             return false
220         }
221         val item = BillingFlowParams.ProductDetailsParams.newBuilder().setProductDetails(product)
222         offerOf(product)?.offerToken?.takeIf { it.isNotEmpty() }?.let { item.setOfferToken(it) }
223 
224         val result = c.launchBillingFlow(
225             activity,
226             BillingFlowParams.newBuilder().setProductDetailsParamsList(listOf(item.build())).build()
227         )
228         if (result.responseCode == BillingResponseCode.OK) return true
229         if (result.responseCode == BillingResponseCode.ITEM_ALREADY_OWNED) {
230             refresh(activity)
231             return true
232         }
233         lastProblem = explain(result)
234         return false
235     }

237     private fun purchaseListener(app: Context) = PurchasesUpdatedListener { result, purchases ->
238         main.post {
239             when (result.responseCode) {
240                 BillingResponseCode.OK -> {
241                     absorb(app, purchases ?: emptyList(), replace = false)
242                     lastProblem = null
243                 }
244                 BillingResponseCode.USER_CANCELED -> Unit
245                 BillingResponseCode.ITEM_ALREADY_OWNED -> refresh(app)
246                 else -> lastProblem = explain(result)
247             }
248             onChange?.invoke()
249         }
250     }
251 
252     /**
253      * Records what is owned and acknowledges anything new. Acknowledging is not
254      * optional: Play refunds a purchase that goes three days without it.
255      */
256     private fun absorb(app: Context, purchases: List<Purchase>, replace: Boolean) {
257         val owned = HashSet<String>()
258         var pending = false
259         purchases.forEach { p ->
260             when (p.purchaseState) {
261                 Purchase.PurchaseState.PURCHASED -> {
262                     owned.addAll(p.products)
263                     if (!p.isAcknowledged) acknowledge(p)
264                 }
265                 Purchase.PurchaseState.PENDING -> pending = true
266                 else -> Unit
267             }
268         }
269         purchasePending = pending
270         if (replace || owned.isNotEmpty()) License.recordOwned(app, owned)
271     }
272 
273     private fun acknowledge(p: Purchase) {
274         val params = AcknowledgePurchaseParams.newBuilder().setPurchaseToken(p.purchaseToken).build()
275         client?.acknowledgePurchase(params) { result ->
276             if (result.responseCode != BillingResponseCode.OK) {
277                 // Not fatal: the next refresh sees it unacknowledged and retries.
278                 Log.w(TAG, "acknowledge failed: " + result.responseCode + " " + result.debugMessage)
279             }
280         }
281     }

285     /** The cheapest way to buy a product, should it ever carry more than one offer. */
286     private fun offerOf(p: ProductDetails): ProductDetails.OneTimePurchaseOfferDetails? =
287         p.oneTimePurchaseOfferDetailsList?.minByOrNull { it.priceAmountMicros }

290     private fun explain(r: BillingResult): String = when (r.responseCode) {
291         BillingResponseCode.BILLING_UNAVAILABLE ->
292             "Google Play billing is not available for this device or account."
293         BillingResponseCode.SERVICE_UNAVAILABLE,
294         BillingResponseCode.SERVICE_DISCONNECTED,
295         BillingResponseCode.NETWORK_ERROR ->
296             "Could not reach Google Play. Check the connection and try again."
297         BillingResponseCode.ITEM_UNAVAILABLE ->
298             "That product is not available for this account."
299         BillingResponseCode.FEATURE_NOT_SUPPORTED ->
300             "The Play Store app on this device is too old. Update it and try again."
301         else -> "Google Play error " + r.responseCode +
302             (if (r.debugMessage.isNullOrBlank()) "" else ": " + r.debugMessage)
303     }
