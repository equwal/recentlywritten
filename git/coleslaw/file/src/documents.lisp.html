36 (defun compute-url (document unique-id &optional class)
37   "Compute the relative URL for a DOCUMENT based on its UNIQUE-ID. If CLASS
38 is provided, it overrides the route used."
39   (let* ((class-name (or class (class-name (class-of document))))
40          (route (get-route class-name)))
41     (unless route
42       (error "No routing method found for: ~A" class-name))
43     (let* ((result (format nil route unique-id))
44            (type (or (pathname-type result) (page-ext *config*))))
45       (make-pathname :name (funcall (name-fn *config*) (pathname-name result))
46                      :type type
47                      :defaults result))))

53 (defun add-document (document)
54   "Add DOCUMENT to the in-memory database. Error if a matching entry is present."
55   (let ((url (page-url document)))
56     (if (gethash url *site*)
57         (error "There is already an existing document with the url ~a" url)
58         (setf (gethash url *site*) document))))

60 (defun delete-document (document)
61   "Given a DOCUMENT, delete it from the in-memory database."

64 (defun write-document (document &optional theme-fn &rest render-args)
65   "Write the given DOCUMENT to disk as HTML. If THEME-FN is present,
66 use it as the template passing any RENDER-ARGS."
67   (let ((html (if (or theme-fn render-args)
68                   (apply #'render-page document theme-fn render-args)
69                   (render-page document nil)))
70         (url (namestring (page-url document))))
71     (write-file (rel-path (staging-dir *config*) url) html)))
