6 (defun add-injection (injection location)
7   "Adds an INJECTION to a given LOCATION for rendering. The INJECTION should be a
8 function that takes a DOCUMENT and returns NIL or a STRING for template insertion."
9   (push injection (getf *injections* location)))

11 (defun find-injections (content)
12   "Iterate over *INJECTIONS* collecting any that should be added to CONTENT."
13   (flet ((injections-for (location)
14            (loop for injection in (getf *injections* location)
15               collecting (funcall injection content))))
16     (list :head (injections-for :head)
17           :body (injections-for :body))))
