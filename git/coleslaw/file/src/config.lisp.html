53 (defun enable-plugin (name args)
54   "Given a plugin, NAME, compile+load it and call its ENABLE function with ARGS."
55   (flet ((plugin-path (sym)
56            (if (probe-file (repo-path "plugins/~(~A~).lisp" sym))
57                (repo-path "plugins/~(~A~)" sym)
58                (app-path "plugins/~(~A~)" sym)))
59          (plugin-package (sym)
60            (format nil "~:@(coleslaw-~A~)" sym)))
61     (let ((file (plugin-path name)))
62       (multiple-value-bind (output-file error)
63           (ignore-errors (compile-file file :verbose nil :print nil))
64         (when error
65           (warn "Error while compiling plugin ~A: ~A.~%" name error))
66         (load (or output-file file) :verbose t)))
67     (let ((package (find-package (plugin-package name))))
68       (apply (find-symbol "ENABLE" package) args))))

70 (defun load-plugins (plugins)
71   "Compile and load the listed PLUGINS. It is expected that matching *.lisp files
72 are in the plugins folder in coleslaw's source directory."
73   (setf *injections* nil)
74   (dolist (plugin plugins)
