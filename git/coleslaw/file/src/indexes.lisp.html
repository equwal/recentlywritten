69 (defclass numeric-index (index) ())

71 (defmethod discover ((doc-type (eql (find-class 'numeric-index))))
72   (let ((content (by-date (find-all 'post))))
73     (dotimes (i (ceiling (length content) *pages-listings-on-index*))
74       (add-document (index-by-n i content)))))

76 (defun index-by-n (i content)
77   "Return the index for the Ith page of CONTENT in reverse chronological order."
78   (let ((content (subseq content (* *pages-listings-on-index* i))))
79     (make-instance 'numeric-index :slug (1+ i) :name (1+ i)
80                    :content (take-up-to *pages-listings-on-index* content)
81                    :title "Recent Content")))

96 (defun all-months ()
97   "Retrieve a list of all months with published content."
98   (let ((months (loop :for post :in (find-all 'post)
99                       :for content-date := (content-date post)
100                       :when content-date
101                       :collect (subseq content-date 0
102                                        (min 7 (length content-date))))))
103     (sort (remove-duplicates months :test #'string=) #'string>)))
104 
105 (defun all-tags ()
106   "Retrieve a list of all tags used in content."
107   (let* ((dupes (mappend #'content-tags (find-all 'post)))
108          (tags (remove-duplicates dupes :test #'tag-slug=)))
109     (sort tags #'string< :key #'tag-name)))
