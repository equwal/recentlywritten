18 (defun node-val (name post)
19   (flet ((value (node)
20            (let ((child (dom:last-child node)))
21              (when child (dom:data child)))))
22     (let ((nodes (dom:get-elements-by-tag-name post name)))
23       (if (string= "category" name)
24           (loop for node across nodes collecting (value node))
25           (when (plusp (length nodes)) (value (elt nodes 0)))))))

33 (defun import-post (post output &optional since)
34   (when (and (string= "publish" (node-val "wp:status" post)) ; is it public?
35              (string= "post" (node-val "wp:post_type" post)) ; is it a post?
36              (or (null since) (string>= (get-timestamp post) since)))
37     (let ((slug (slugify (node-val "title" post))))
38       (when (string= "" slug)
39         (error "No valid slug-title for post ~a." (get-timestamp post)))
40       (export-post (node-val "title" post) (node-val "category" post)
41                    (get-timestamp post) (node-val "content:encoded" post)
42                    (format nil "~a.post" slug) output))))

59 (defun import-posts (filepath output &optional since)
60   (when (probe-file filepath)
61     (ensure-directories-exist (or output (repo-dir *config*)))
62     (let* ((xml (cxml:parse-file filepath (cxml-dom:make-dom-builder)))
63            (posts (dom:get-elements-by-tag-name xml "item")))
64       (loop for post across posts do (import-post post output since))
