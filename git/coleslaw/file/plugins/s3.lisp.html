31 (defun s3-sync (filepath dir)
32   (let ((etag (zs3:file-etag filepath))
33         (key (enough-namestring filepath dir)))
34     (if (gethash etag *cache*)
35         (remhash etag *cache*)
36         (zs3:put-file filepath *bucket* key :public t
37                       :content-type (content-type (pathname-type filepath))))))

39 (defun dir->s3 (dir)
40   (flet ((upload (file) (s3-sync file dir)))
41     (cl-fad:walk-directory dir #'upload)))

43 (defmethod deploy :after (staging)
44   (let ((blog (deploy-dir *config*)))
45     (loop for key across (zs3:all-keys *bucket*)
46        do (setf (gethash (zs3:etag key) *cache*) key))
47     (dir->s3 blog)
