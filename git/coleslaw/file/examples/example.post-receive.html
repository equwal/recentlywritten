1 ########## CONFIGURATION VALUES ##########
2 
3 TMP_GIT_CLONE=$HOME/tmp/improvedmeans/
4 
5 # Set LISP to your preferred implementation. The following
6 # implementations are currently supported:
7 #   * sbcl
8 #   * ccl
9 LISP=sbcl
10 
11 ########## DON'T EDIT ANYTHING BELOW THIS LINE ##########
12 
13 if cd `dirname "$0"`/..; then
14     GIT_REPO=`pwd`
15     cd $OLDPWD || exit 1
16 else
17     exit 1
18 fi
19 
20 git clone $GIT_REPO $TMP_GIT_CLONE || exit 1
21 
22 while read oldrev newrev refname; do
23     if [ $refname = "refs/heads/master" ]; then
24         echo -e "\n  Master updated. Running coleslaw...\n"
25         if [ $LISP = sbcl ]; then
26             sbcl --eval "(ql:quickload 'coleslaw)" \
27                  --eval "(coleslaw:main \"$TMP_GIT_CLONE\" \"$oldrev\")" \
28                  --eval "(uiop:quit)"
29         elif [ $LISP = ccl ]; then
30             ccl -e "(ql:quickload 'coleslaw)" \
31                 -e "(coleslaw:main \"$TMP_GIT_CLONE\" \"$oldrev\")" \
32                 -e "(uiop:quit)"
33         else
34             echo -e "$LISP is not a supported lisp dialect at this time. Exiting.\n"
35             exit 1
36         fi
37     fi
38 done
39 
40 rm -rf $TMP_GIT_CLONE
41 exit
