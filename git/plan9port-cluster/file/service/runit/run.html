1 #!/bin/sh
2 # runit service for p9c-server. Install as /etc/sv/p9c-server/run, replace
3 # USER with the account that owns the data, and link the folder into the
4 # runit service directory. The script runs in the foreground and exits when
5 # venti dies, so runit restarts both venti and fossil.
6 exec chpst -u USER:USER env HOME=/home/USER \
7 	P9C_FOSSIL=/home/USER/.local/plan9/bin/fossil/fossil \
8 	/home/USER/.local/bin/p9c-server 2>&1
