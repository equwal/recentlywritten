48 --- Returns the list as plain text: the name, the line and the address of
49 --- each app, with an empty line between two apps.
50 function MoreApps.text()
51     local blocks = {}
52     for _, app in ipairs(MoreApps.list) do
53         blocks[#blocks + 1] = app.name .. "\n" .. app.line .. "\n" .. app.url
54     end
55     return table.concat(blocks, "\n\n")
56 end
57 
58 return MoreApps
