4 local function build(triples)
5     local list = {}
6     for _, t in ipairs(triples) do
7         list[#list + 1] = {
8             start = t[1], stop = t[2], text = t[3],
9             norm = t[3], no_place = false,
10         }
11     end
12     return Cues.new(list)
13 end
