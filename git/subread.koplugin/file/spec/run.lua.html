11 local function deepEqual(a, b)
12     if a == b then return true end
13     if type(a) ~= "table" or type(b) ~= "table" then return false end
14     for k, v in pairs(a) do if not deepEqual(v, b[k]) then return false end end
15     for k in pairs(b) do if a[k] == nil then return false end end
16     return true
17 end
