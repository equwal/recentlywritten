1 # Alpine: apk --root from the live system. OpenRC, eudev, seatd, dbus.
2 # shellcheck shell=sh disable=SC2034
3 bootstrap() {
4     mkdir -p $M/etc/apk/keys $M/etc/mkinitfs
5     cp /etc/apk/keys/* $M/etc/apk/keys/
6     grep '^http' /etc/apk/repositories > $M/etc/apk/repositories
7     apk add -q --root $M --initdb alpine-base
8     chroot_mounts
9     echo 'features="ata base cryptsetup ext4 keymap mmc nvme scsi usb virtio"' > $M/etc/mkinitfs/mkinitfs.conf
10     fw=linux-firmware; [ "${FIRMWARE:-}" = no ] && fw=linux-firmware-none
11     ch "apk add -q linux-lts $fw mkinitfs cryptsetup e2fsprogs eudev seatd dbus shadow \
12         mesa-dri-gallium font-dejavu fontconfig"
13     for s in devfs dmesg udev udev-trigger udev-settle; do ch "rc-update -q add $s sysinit"; done
14     for s in modules sysctl hostname bootmisc syslog hwclock; do ch "rc-update -q add $s boot"; done
15     for s in networking seatd dbus; do ch "rc-update -q add $s default"; done
16     for s in killprocs savecache mount-ro; do ch "rc-update -q add $s shutdown"; done
17     printf 'auto lo\niface lo inet loopback\n\nauto eth0\niface eth0 inet dhcp\n' > $M/etc/network/interfaces
18 }
