1 #!/bin/sh
2 # waybase installer: install KISS, Gentoo, Arch or Alpine with Nix and a
3 # Wayland compositor onto a blank disk. Runs as root from the waybase live
4 # image (Alpine), with network:   sh /media/*/waybase/install.sh [ANSWERS]
5 #
6 # ANSWERS is an optional shell file that presets any of the variables below
7 # (see answers.example). Unset variables are asked for interactively.
8 #   BASE        kiss | gentoo | arch | alpine
9 #   COMPOSITOR  sway | river (river-classic 0.3) | labwc | dwl  (a list installs several; the
10 #               first one starts on tty1 login)
11 #   DISK        target disk, for example /dev/sda. ALL DATA ON IT IS ERASED.
12 #   ENCRYPT     yes | no (LUKS2 on the root partition)
13 #   LUKS_PASS, USERNAME, USER_PASS, ROOT_PASS, HOSTNAME_NEW
14 #   YES=1       skip the "type the disk name again" check
15 #   SERIAL=1    login getty on ttyS0 and kernel console there (for tests)
16 #   NIXPKGS     flake ref for Nix-supplied packages (default: nixpkgs)
17 #   FIRMWARE=no skip linux-firmware (VMs; real hardware usually needs it)
18 #   FIRMWARE_TAG, FIRMWARE_LIST   install the pinned files of LIST (SHA256 PATH lines, see firmware.sh)
19 #               from linux-firmware tag FIRMWARE_TAG into /lib/firmware
20 set -eu

28 log() { printf '\n=== %s\n' "$*"; }

30 ask() { # ask VAR "prompt" [secret]: read VAR unless the answers file set it
31     eval "[ -n \"\${$1:-}\" ]" && return 0
32     printf '%s: ' "$2"
33     [ -n "${3:-}" ] && stty -echo
34     read -r v

38 menu() { # menu VAR "prompt" choice...: numbered menu unless VAR is set
39     eval "[ -n \"\${$1:-}\" ]" && return 0
40     var=$1 prompt=$2; shift 2
41     i=0; for o; do i=$((i + 1)); echo "  $i) $o"; done
42     while :; do
43         printf '%s [1-%d]: ' "$prompt" $#; read -r n
44         case $n in '' | *[!0-9]*) continue ;; esac
45         [ "$n" -ge 1 ] && [ "$n" -le $# ] && break
46     done
47     eval "$var=\${$n}"
48 }

125 chroot_mounts() {
126     mkdir -p $M/proc $M/sys $M/dev $M/run $M/tmp
127     for f in proc sys dev; do mount --rbind /$f $M/$f; done

129     rm -f $M/etc/resolv.conf; cp /etc/resolv.conf $M/etc/resolv.conf
