1 #!/usr/bin/env python3
2 """Unattended QEMU test of waybase.img for one base.
3 
4 Usage: test-qemu.py IMG WORKDIR BASE "COMPOSITORS" [encrypt|plain]
5 Stdlib only. Env: QEMU, QEMU_IMG, OVMF (UEFI firmware), MEM (MiB, 2048),
6 ACCEL (kvm|tcg), CPU (default Nehalem = x86-64-v2, so no binary may need AVX).
7 
8 1. UEFI boot of the live image as a USB stick; install.sh with an answers file.
9 2. BIOS boot of the installed disk to the passphrase or login prompt.
10 3. UEFI boot of the installed disk: unlock, log in as the user, then check
11    `nix --version`, the nix daemon, and that each compositor starts headless
12    and creates a Wayland socket.
13 Exit 0 only if every check passes. The serial log goes to WORKDIR/BASE.log.
14 """

31 class VM:
32     def __init__(self, uefi, live, port=45560):
33         args = [QEMU, "-accel", ACCEL, "-cpu", CPU, "-m", os.environ.get("MEM", "2048"), "-smp", "2",
34                 "-display", "none", "-monitor", "none",
35                 "-chardev", f"socket,id=s0,host=127.0.0.1,port={port},server=on,wait=on",
36                 "-serial", "chardev:s0", "-nic", "user,model=virtio-net-pci",
37                 "-drive", f"file={disk},if=virtio,format=qcow2,cache=unsafe"]
38         if uefi:
39             args += ["-bios", OVMF]
40         if live:
41             args += ["-device", "qemu-xhci", "-drive", f"if=none,id=usb,file={img},format=raw,readonly=on",
42                      "-device", "usb-storage,drive=usb,bootindex=0"]
43         self.p = subprocess.Popen(args)
44         for _ in range(150):
45             try:
46                 self.s = socket.create_connection(("127.0.0.1", port)); break
47             except OSError:
48                 time.sleep(0.2)
49         else:
50             sys.exit("FAIL: no QEMU serial socket")
51         self.buf = ""

53     def expect(self, pat, timeout):
54         end = time.time() + timeout
55         while True:
56             m = re.search(pat, self.buf, re.S)
57             if m:
58                 self.buf = self.buf[m.end():]
59                 return m
60             if time.time() > end or self.p.poll() is not None:
61                 self.kill()
62                 sys.exit(f"FAIL: {base}: timeout or QEMU exit waiting for {pat!r}")
63             self.s.settimeout(5)
64             try:
65                 d = self.s.recv(65536)
66             except socket.timeout:
67                 continue
68             if not d:
69                 sys.exit(f"FAIL: {base}: serial closed waiting for {pat!r}")
70             t = d.decode("utf-8", "replace")
71             if "\x1b[6n" in t:  # busybox ash asks for the cursor position
72                 self.s.sendall(b"\x1b[50;200R")
73             LOG.write(t); LOG.flush()
74             self.buf = (self.buf + t)[-200000:]
75 
76     def send(self, line):
77         self.s.sendall((line + "\r").encode())
78 
79     def run(self, script, timeout=600):
80         """Run script with sh and return the output between markers."""
81         self.buf = ""
82         self.send("sh <<'EOS'\recho @@BEG\"\"@@\r" + script.strip().replace("\n", "\r") + "\recho @@END\"\"@@\rEOS")
83         return self.expect(r"@@BEG@@(.*?)@@END@@", timeout).group(1)
84 
85     def kill(self):
86         self.p.kill(); self.p.wait()
