155     private val arbTerm = Arb.list(
156         Arb.stringPattern("[ぁ-ん一-十a-z]{1,6}"),
157         1..40,
158     ).map { words ->
159         words.mapIndexed { i, word ->
160             Term(
161                 expression = word,
162                 reading = if (i % 2 == 0) "" else "よみ$i",
163                 definitionTags = if (i % 3 == 0) "" else "n vs",
164                 rules = if (i % 4 == 0) "v1" else "",
165                 score = (i * 7 % 11).toDouble(),
166                 glossary = JsonArray().apply { add(JsonPrimitive("gloss $i \"quoted\" <b>")); add(JsonPrimitive("second")) }.toString(),
167                 sequence = i.toLong() * 100,
168                 termTags = "P",
169             )
170         }
171     }
172 
173     private fun zip(index: String, terms: List<Term>, metas: List<TermMeta>, tags: List<Tag>): ByteArray {
174         val bytes = ByteArrayOutputStream()
175         ZipOutputStream(bytes).use { out ->
176             fun file(name: String, body: String) {
177                 out.putNextEntry(ZipEntry(name))
178                 out.write(body.toByteArray())
179                 out.closeEntry()
180             }
181             file("term_bank_1.json", JsonArray().apply {
182                 for (t in terms) add(JsonArray().apply {
183                     add(t.expression); add(t.reading)
184                     if (t.definitionTags.isEmpty()) add(com.google.gson.JsonNull.INSTANCE) else add(t.definitionTags)
185                     add(t.rules); add(t.score); add(com.google.gson.JsonParser.parseString(t.glossary)); add(t.sequence); add(t.termTags)
186                 })
187             }.toString())
188             file("term_meta_bank_1.json", JsonArray().apply {
189                 for (m in metas) add(JsonArray().apply { add(m.expression); add(m.mode); add(com.google.gson.JsonParser.parseString(m.data)) })
190             }.toString())
191             file("tag_bank_1.json", JsonArray().apply {
192                 for (t in tags) add(JsonArray().apply { add(t.name); add(t.category); add(t.order); add(t.notes); add(t.score) })
193             }.toString())
194             file("kanji_bank_1.json", "[[\"日\",\"\",\"\",\"\",[],{}]]")
195             // The index last: a reader that stops at the first file would miss it.
196             file("index.json", index)
197         }
198         return bytes.toByteArray()
199     }
