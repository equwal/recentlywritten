4  * A dictionary form that an inflected text can come from.
5  *
6  * @param text the dictionary form.
7  * @param conditions bit mask of the word classes that the last rule produced, from [Conditions].
8  *   Zero: the text is not deinflected, any term matches.

13     /** True when a term with these rule identifiers (`v1`, `v5`, `vs`, `vk`, `vz`, `adj-i`) can be this form. */
14     fun accepts(termRules: String): Boolean {
15         if (conditions == 0) return true
16         return Conditions.maskOf(termRules) and conditions != 0
17     }

20 /** The word classes of the Japanese rules of Yomitan. Each leaf is one bit; a group is the bits of its leaves. */
21 object Conditions {
22     private val leaves = listOf(
23         "v1d", "v1p", "v5d", "v5s", "v5ss", "v5sp", "vk", "vs", "vz", "adj-i",
24         "-ます", "-ません", "-て", "-ば", "-く", "-た", "-ん", "-なさい", "-ゃ",
25     )
26     private val bits: Map<String, Int> = buildMap {
27         leaves.forEachIndexed { i, name -> put(name, 1 shl i) }
28         put("v1", getValue("v1d") or getValue("v1p"))
29         put("v5", getValue("v5d") or getValue("v5s") or getValue("v5ss") or getValue("v5sp"))
30         put("v", getValue("v1") or getValue("v5") or getValue("vk") or getValue("vs") or getValue("vz"))
31     }
32 
33     fun maskOf(names: Collection<String>): Int = names.fold(0) { acc, name -> acc or (bits[name] ?: 0) }
34 
35     /** The mask of a space-separated rule string of a term. */
36     fun maskOf(rules: String): Int = maskOf(rules.split(' ').filter { it.isNotEmpty() })
37 }

83     /** Rules for a suffix that follows the te/ta sound change (音便): て, た, たら, たり, ちゃう. */
84     private fun teSuffix(reason: String, te: String, conditionsIn: List<String>, vkOut: String = "vk") {
85         // The voiced form after ん and after the い of ぐ: 読んで, 泳いだ, 死んじゃう.
86         val de = when (te.first()) { 'て' -> "で"; 'た' -> "だ"; 'ち' -> "じ"; else -> te.take(1) } + te.drop(1)
87         rule(reason, te, "る", conditionsIn, listOf("v1"))
88         rule(reason, "い$te", "く", conditionsIn, listOf("v5"))
89         rule(reason, "い$de", "ぐ", conditionsIn, listOf("v5"))
90         rule(reason, "し$te", "す", conditionsIn, listOf("v5"))
91         rule(reason, "っ$te", "う", conditionsIn, listOf("v5"))
92         rule(reason, "っ$te", "つ", conditionsIn, listOf("v5"))
93         rule(reason, "っ$te", "る", conditionsIn, listOf("v5"))
94         // 行く, 往く, 逝く and いく sound like う-verbs in this form: 行った, not 行いた.
95         for (iku in listOf("行", "往", "逝", "い")) rule(reason, "${iku}っ$te", "${iku}く", conditionsIn, listOf("v5"))
96         rule(reason, "ん$de", "ぬ", conditionsIn, listOf("v5"))
97         rule(reason, "ん$de", "ぶ", conditionsIn, listOf("v5"))
98         rule(reason, "ん$de", "む", conditionsIn, listOf("v5"))
99         rule(reason, "じ$te", "ずる", conditionsIn, listOf("vz"))
100         rule(reason, "し$te", "する", conditionsIn, listOf("vs"))
101         rule(reason, "為$te", "為る", conditionsIn, listOf("vs"))
102         rule(reason, "き$te", "くる", conditionsIn, listOf(vkOut))
103         rule(reason, "来$te", "来る", conditionsIn, listOf(vkOut))
104         rule(reason, "來$te", "來る", conditionsIn, listOf(vkOut))
105     }

256     /** The maximum number of forms tried for one text. A text of normal length is far below it. */
257     private const val LIMIT = 2000
258 
259     /**
260      * Each dictionary form that the text can come from. The first result is the text itself, with
261      * no conditions. The others come in breadth-first order: the fewest rules first.
262      */
263     fun deinflect(text: String): List<Deinflection> {
264         if (text.isEmpty()) return emptyList()
265         val results = ArrayList<Deinflection>()
266         val seen = HashSet<Pair<String, Int>>()
267         results.add(Deinflection(text, 0, emptyList()))
268         seen.add(text to 0)
269         var i = 0
270         while (i < results.size && results.size < LIMIT) {
271             val current = results[i++]
272             for (rule in rules) {
273                 if (rule.conditionsIn != 0 && current.conditions != 0 && rule.conditionsIn and current.conditions == 0) continue
274                 if (!current.text.endsWith(rule.inflected)) continue
275                 val base = current.text.dropLast(rule.inflected.length) + rule.base
276                 if (base.isEmpty()) continue
277                 if (!seen.add(base to rule.conditionsOut)) continue
278                 results.add(Deinflection(base, rule.conditionsOut, current.reasons + rule.reason))
279             }
280         }
281         return results
282     }
