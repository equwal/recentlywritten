6     /** The file extension: mp3, ogg, wav, flac or m4a. Bytes that say nothing count as mp3. */
7     fun extension(bytes: ByteArray): String {
8         fun startsWith(prefix: String, at: Int = 0) = bytes.size >= at + prefix.length &&
9             prefix.indices.all { bytes[at + it] == prefix[it].code.toByte() }
10         return when {
11             startsWith("OggS") -> "ogg"
12             startsWith("RIFF") && startsWith("WAVE", 8) -> "wav"
13             startsWith("fLaC") -> "flac"
14             startsWith("ftyp", 4) -> "m4a"
15             else -> "mp3"
16         }
17     }
