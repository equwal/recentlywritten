8 /**
9  * Turns the glossary of a term, and the meta of a term, into simple HTML: `b`, `i`, `u`, `br`,
10  * `ul`, `li`, `a`. A phone shows it with `Html.fromHtml`, which knows these tags and no table.
11  */

14     /** The glossary array of a term as HTML. Each definition is one line. */
15     fun toHtml(glossaryJson: String): String {
16         val array = runCatching { JsonParser.parseString(glossaryJson) }.getOrNull()
17         if (array == null || !array.isJsonArray) return escape(glossaryJson)
18         val lines = array.asJsonArray.map { definition(it).trim() }.filter { it.isNotEmpty() }
19         return lines.joinToString("<br>")
20     }

22     /** The glossary as plain text, one definition per line. For a search result list, or a test. */
23     fun toText(glossaryJson: String): String = toHtml(glossaryJson)
24         .replace("<br>", "\n")
25         .replace(Regex("<[^>]+>"), "")
26         .let(::unescape)

28     private fun definition(element: JsonElement): String = when {
29         element.isJsonPrimitive -> escape(element.asString)
30         element.isJsonArray -> element.asJsonArray.joinToString("") { definition(it) }
31         element.isJsonObject -> {
32             val obj = element.asJsonObject
33             when (obj.get("type")?.asString) {
34                 "text" -> escape(obj.get("text")?.asString ?: "")
35                 "image" -> image(obj)
36                 "structured-content" -> node(obj.get("content"))
37                 else -> node(element)
38             }
39         }
40         else -> ""
41     }
42 
43     /** A node of structured content: a string, an array of nodes, or an element with a tag. */
44     private fun node(element: JsonElement?): String {
45         if (element == null || element.isJsonNull) return ""
46         if (element.isJsonPrimitive) return escape(element.asString)
47         if (element.isJsonArray) return element.asJsonArray.joinToString("") { node(it) }
48         val obj = element.asJsonObject
49         val tag = obj.get("tag")?.asString ?: return ""
50         val inner = node(obj.get("content"))
51         return when (tag) {
52             "br" -> "<br>"
53             "img" -> image(obj)
54             "a" -> {
55                 val href = obj.get("href")?.asString
56                 if (href != null && href.startsWith("http")) "<a href=\"${escape(href)}\">$inner</a>" else inner
57             }
58             "ruby" -> inner
59             "rt" -> "<small>($inner)</small>"
60             "rp" -> ""
61             "ol", "ul" -> "<ul>$inner</ul>"
62             "li" -> "<li>${styled(obj, inner)}</li>"
63             "table", "thead", "tbody", "tfoot" -> inner
64             "tr" -> "$inner<br>"
65             "td", "th" -> "$inner&nbsp;| "
66             "div", "details", "summary" -> block(styled(obj, inner))
67             else -> styled(obj, inner)
68         }
69     }
70 
71     /** A block starts on its own line. Two blocks in a row keep one line break between them. */
72     private fun block(inner: String): String =
73         if (inner.isEmpty()) "" else if (inner.endsWith("<br>")) inner else "$inner<br>"

75     private fun styled(obj: JsonObject, inner: String): String {
76         val style = obj.get("style")?.takeIf { it.isJsonObject }?.asJsonObject ?: return inner
77         var out = inner
78         if (style.get("fontWeight")?.asString?.let { it == "bold" || (it.toIntOrNull() ?: 0) >= 600 } == true) out = "<b>$out</b>"
79         if (style.get("fontStyle")?.asString == "italic") out = "<i>$out</i>"
80         if (style.get("textDecorationLine")?.let { it.toString().contains("underline") } == true) out = "<u>$out</u>"
81         if (style.get("verticalAlign")?.asString == "super") out = "<sup>$out</sup>"
82         if (style.get("verticalAlign")?.asString == "sub") out = "<sub>$out</sub>"
83         return out
84     }

98     private fun frequency(data: JsonElement): String = when {
99         data.isJsonPrimitive -> data.asString
100         data.isJsonObject -> {
101             val obj = data.asJsonObject
102             when {
103                 obj.has("frequency") -> frequency(obj.get("frequency"))
104                 obj.has("displayValue") -> obj.get("displayValue").asString
105                 obj.has("value") -> obj.get("value").asString
106                 else -> ""
107             }
108         }
109         else -> ""
110     }

118     /** Pitch meta as text: the reading and the accent positions, `はし [2]`. */
119     fun pitchText(dataJson: String): String {
120         val obj = runCatching { JsonParser.parseString(dataJson) }.getOrNull()?.takeIf { it.isJsonObject }?.asJsonObject ?: return ""
121         val reading = obj.get("reading")?.asString ?: ""
122         val pitches = obj.get("pitches")?.takeIf { it.isJsonArray }?.asJsonArray ?: JsonArray()
123         val positions = pitches.mapNotNull { it.asJsonObject.get("position")?.asString }
124         return if (positions.isEmpty()) reading else "$reading [${positions.joinToString(", ")}]"
125     }

130     fun escape(text: String): String = buildString(text.length) {
131         for (c in text) when (c) {
132             '&' -> append("&amp;")
133             '<' -> append("&lt;")
134             '>' -> append("&gt;")
135             '"' -> append("&quot;")
136             else -> append(c)
137         }
138     }
139 
140     private fun unescape(text: String): String =
141         text.replace("&nbsp;", " ").replace("&lt;", "<").replace("&gt;", ">").replace("&quot;", "\"").replace("&amp;", "&")
