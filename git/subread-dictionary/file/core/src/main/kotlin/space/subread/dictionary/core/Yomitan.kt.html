61 /**
62  * Reads a Yomitan dictionary zip. The JSON files are in the root of the zip. Each file is read as
63  * a stream, so a term bank of many megabytes takes little memory.
64  */

67     /** The index alone. The zip can have the index after the term banks, so the caller opens the zip twice. */
68     fun readIndex(zip: InputStream): DictionaryIndex {
69         ZipInputStream(zip).use { stream ->
70             while (true) {
71                 val entry = stream.nextEntry ?: break
72                 if (entry.name == "index.json") return parseIndex(stream.readBytes().decodeToString())
73             }
74         }
75         throw DictionaryFormatException("No index.json in the zip.")
76     }
77 
78     fun parseIndex(json: String): DictionaryIndex {
79         val root = JsonParser.parseString(json).asJsonObject
80         val format = root.get("format")?.asInt ?: root.get("version")?.asInt
81             ?: throw DictionaryFormatException("index.json has no format or version.")
82         return DictionaryIndex(
83             title = root.get("title")?.asString ?: throw DictionaryFormatException("index.json has no title."),
84             revision = root.get("revision")?.asString ?: "",
85             format = format,
86             sequenced = root.get("sequenced")?.asBoolean ?: false,
87         )
88     }
89 
90     /** Reads the term, term meta and tag banks. Kanji banks are skipped. */
91     fun readBanks(zip: InputStream, sink: DictionarySink) {
92         ZipInputStream(zip).use { stream ->
93             while (true) {
94                 val entry = stream.nextEntry ?: break
95                 val name = entry.name.substringAfterLast('/')
96                 val reader = JsonReader(InputStreamReader(stream, Charsets.UTF_8))
97                 when {
98                     name.startsWith("term_meta_bank_") -> readTermMetaBank(reader, sink)
99                     name.startsWith("term_bank_") -> readTermBank(reader, sink)
100                     name.startsWith("tag_bank_") -> readTagBank(reader, sink)
101                 }
102             }
103         }
104     }
105 
106     fun readTermBank(reader: JsonReader, sink: DictionarySink) {
107         reader.beginArray()
108         while (reader.hasNext()) {
109             reader.beginArray()
110             val term = Term(
111                 expression = reader.nextString(),
112                 reading = reader.nextString(),
113                 definitionTags = reader.nextStringOrNull() ?: "",
114                 rules = reader.nextString(),
115                 score = reader.nextDouble(),
116                 glossary = JsonParser.parseReader(reader).toString(),
117                 sequence = reader.nextDouble().toLong(),
118                 termTags = reader.nextString(),
119             )
120             // A dictionary of another format has more fields, or fewer. Skip what is left of the row.
121             while (reader.hasNext()) reader.skipValue()
122             reader.endArray()
123             sink.term(term)
124         }
125         reader.endArray()
126     }
127 
128     fun readTermMetaBank(reader: JsonReader, sink: DictionarySink) {
129         reader.beginArray()
130         while (reader.hasNext()) {
131             reader.beginArray()
132             val meta = TermMeta(
133                 expression = reader.nextString(),
134                 mode = reader.nextString(),
135                 data = JsonParser.parseReader(reader).toString(),
136             )
137             while (reader.hasNext()) reader.skipValue()
138             reader.endArray()
139             sink.termMeta(meta)
140         }
141         reader.endArray()
142     }
143 
144     fun readTagBank(reader: JsonReader, sink: DictionarySink) {
145         reader.beginArray()
146         while (reader.hasNext()) {
147             reader.beginArray()
148             val tag = Tag(
149                 name = reader.nextString(),
150                 category = reader.nextString(),
151                 order = reader.nextDouble().toInt(),
152                 notes = reader.nextString(),
153                 score = reader.nextDouble(),
154             )
155             while (reader.hasNext()) reader.skipValue()
156             reader.endArray()
157             sink.tag(tag)
158         }
159         reader.endArray()
160     }

162     private fun JsonReader.nextStringOrNull(): String? =
163         if (peek() == JsonToken.NULL) {
164             nextNull()
165             null
166         } else {
167             nextString()
168         }
