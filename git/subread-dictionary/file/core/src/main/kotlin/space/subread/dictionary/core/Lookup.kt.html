23     /** The most characters a term can cover. Longer terms are rare, and each character costs a scan. */

25 
26     /**
27      * Each term that starts at `offset` in `text`, longest first. Within one length, the
28      * dictionaries in their order, and within one dictionary the higher score first.
29      */
30     fun find(text: String, offset: Int, source: TermSource): List<Match> {
31         val end = minOf(text.length, offset + MAX_LENGTH)
32         if (offset < 0 || offset >= end) return emptyList()
33 
34         // Every dictionary form for every prefix, then one query for all of them.
35         data class Candidate(val form: Deinflection, val length: Int)
36         val candidates = ArrayList<Candidate>()
37         for (stop in end downTo offset + 1) {
38             val slice = text.substring(offset, stop)
39             val variants = linkedSetOf(slice, Kana.toHiragana(slice))
40             for (variant in variants) {
41                 for (form in Deinflector.deinflect(variant)) candidates.add(Candidate(form, stop - offset))
42             }
43         }
44         val byText = candidates.groupBy { it.form.text }
45         val found = source.terms(byText.keys)
46 
47         val matches = ArrayList<Match>()
48         // One match per term: the longest one, because the candidates come longest first.
49         // 食べる is both the text itself and the continuative 食べ plus る; the text itself wins.
50         val seen = HashSet<Pair<Long, Term>>()
51         for (stored in found) {
52             val term = stored.term
53             // A kana text matches the reading; a text with kanji matches the expression.
54             val forms = (byText[term.expression].orEmpty() +
55                 (if (term.reading.isNotEmpty() && term.reading != term.expression) byText[term.reading].orEmpty() else emptyList()))
56                 .sortedByDescending { it.length }
57             for (candidate in forms) {
58                 if (!candidate.form.accepts(term.rules)) continue
59                 if (!seen.add(stored.dictionaryId to term)) continue
60                 matches.add(Match(stored, candidate.length, candidate.form.reasons))
61             }
62         }
63         return matches.sortedWith(
64             compareByDescending<Match> { it.length }
65                 .thenBy { it.reasons.size }
66                 .thenBy { it.stored.dictionaryPosition }
67                 .thenByDescending { it.stored.term.score },
68         )
69     }
