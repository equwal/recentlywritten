191     /** Reads the index, then the banks, on a thread. A dialog shows the count. */
192     private fun importDictionary(uri: Uri) {
193         val dialog = AlertDialog.Builder(this).setMessage(displayName(uri)).setCancelable(false).show()
194         thread {
195             val result = runCatching {
196                 val index = contentResolver.openInputStream(uri)!!.use { YomitanZip.readIndex(it) }
197                 if (index.format != 3) throw DictionaryFormatException(getString(R.string.import_format, index.format))
198                 if (dictionaries.hasTitle(index.title)) throw DictionaryFormatException(getString(R.string.import_exists))
199                 contentResolver.openInputStream(uri)!!.use { zip ->
200                     dictionaries.import(index, zip) { count -> runOnUiThread { dialog.setMessage(getString(R.string.importing, index.title, count)) } }
201                 }
202             }
203             runOnUiThread {
204                 dialog.dismiss()
205                 result.onFailure { toast(getString(R.string.import_failed, it.message ?: it.javaClass.simpleName)) }
206                 draw()
207             }
208         }
209     }

211     /** Copies the android.db into the app folder. It can be gigabytes, so the dialog shows the size so far. */
212     private fun copyAudio(uri: Uri) {
213         val target = store.localAudioFile
214         val dialog = AlertDialog.Builder(this).setMessage(getString(R.string.local_audio_copying, "0 MB")).setCancelable(false).show()
215         thread {
216             val result = runCatching {
217                 val partial = File(target.path + ".part")
218                 contentResolver.openInputStream(uri)!!.use { input ->
219                     partial.outputStream().use { output ->
220                         val buffer = ByteArray(1 shl 20)
221                         var total = 0L
222                         while (true) {
223                             val n = input.read(buffer)
224                             if (n < 0) break
225                             output.write(buffer, 0, n)
226                             total += n
227                             if (total % (32L shl 20) < (1 shl 20)) runOnUiThread { dialog.setMessage(getString(R.string.local_audio_copying, size(total))) }
228                         }
229                     }
230                 }
231                 check(partial.renameTo(target)) { "rename" }
232             }
233             runOnUiThread {
234                 dialog.dismiss()
235                 result.onFailure { toast(getString(R.string.import_failed, it.message ?: it.javaClass.simpleName)) }
236                 draw()
237             }
238         }
239     }
