163     /** Reads the index, then the banks, on a thread. A dialog shows the count. */
164     private fun importDictionary(uri: Uri) {
165         val dialog = AlertDialog.Builder(this).setMessage(displayName(uri)).setCancelable(false).show()
166         thread {
167             val result = runCatching {
168                 val index = contentResolver.openInputStream(uri)!!.use { YomitanZip.readIndex(it) }
169                 if (index.format != 3) throw DictionaryFormatException(getString(R.string.import_format, index.format))
170                 if (dictionaries.hasTitle(index.title)) throw DictionaryFormatException(getString(R.string.import_exists))
171                 contentResolver.openInputStream(uri)!!.use { zip ->
172                     dictionaries.import(index, zip) { count -> runOnUiThread { dialog.setMessage(getString(R.string.importing, index.title, count)) } }
173                 }
174             }
175             runOnUiThread {
176                 dialog.dismiss()
177                 result.onFailure { toast(getString(R.string.import_failed, it.message ?: it.javaClass.simpleName)) }
178                 draw()
179             }
180         }
181     }

183     /** Copies the android.db into the app folder. It can be gigabytes, so the dialog shows the size so far. */
184     private fun copyAudio(uri: Uri) {
185         val target = store.localAudioFile
186         val dialog = AlertDialog.Builder(this).setMessage(getString(R.string.local_audio_copying, "0 MB")).setCancelable(false).show()
187         thread {
188             val result = runCatching {
189                 val partial = File(target.path + ".part")
190                 contentResolver.openInputStream(uri)!!.use { input ->
191                     partial.outputStream().use { output ->
192                         val buffer = ByteArray(1 shl 20)
193                         var total = 0L
194                         while (true) {
195                             val n = input.read(buffer)
196                             if (n < 0) break
197                             output.write(buffer, 0, n)
198                             total += n
199                             if (total % (32L shl 20) < (1 shl 20)) runOnUiThread { dialog.setMessage(getString(R.string.local_audio_copying, size(total))) }
200                         }
201                     }
202                 }
203                 check(partial.renameTo(target)) { "rename" }
204             }
205             runOnUiThread {
206                 dialog.dismiss()
207                 result.onFailure { toast(getString(R.string.import_failed, it.message ?: it.javaClass.simpleName)) }
208                 draw()
209             }
210         }
211     }
