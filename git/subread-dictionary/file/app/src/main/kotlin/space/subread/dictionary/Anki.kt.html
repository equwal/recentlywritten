30     /**
31      * The intent for one card. The audio file, when there is one, goes through the file provider
32      * of this app; the clip data of the intent carries the read permission for it.
33      */
34     fun intent(context: Context, word: String, reading: String, definition: String, sentence: String, source: String, audio: File?): Intent {
35         val intent = Intent(ACTION_ADD)
36             .setPackage(PACKAGE)
37             .putExtra(EXTRA_WORD, word)
38             .putExtra(EXTRA_READING, reading)
39             .putExtra(EXTRA_DEFINITION, definition)
40             .putExtra(EXTRA_SENTENCE, sentence)
41             .putExtra(EXTRA_SOURCE, source)
42             .putExtra(EXTRA_SCREENSHOT, true)
43         if (audio != null) {
44             val uri: Uri = FileProvider.getUriForFile(context, context.packageName + ".files", audio)
45             intent.putExtra(EXTRA_AUDIO, uri.toString())
46             intent.clipData = ClipData.newUri(context.contentResolver, "audio", uri)
47             intent.addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
48         }
49         return intent
50     }
