119     private fun take(intent: Intent) {
120         text = (intent.getCharSequenceExtra(Intent.EXTRA_PROCESS_TEXT) ?: intent.getCharSequenceExtra(Intent.EXTRA_TEXT) ?: "")
121             .toString().trim()
122         if (text.isEmpty()) {
123             Toast.makeText(this, R.string.no_text, Toast.LENGTH_SHORT).show()
124             finish()
125             return
126         }
127         scan(0)

130     private fun scan(at: Int) {
131         offset = at
132         val shown = SpannableString(text)
133         textView.text = shown
134         val run = ++generation
135         thread {
136             val found = if (dictionaries.anyEnabled()) Lookup.find(text, at, dictionaries) else null
137             val metas = found.orEmpty().map { it.stored.term.expression }.distinct().associateWith { dictionaries.meta(it) }
138             runOnUiThread { if (run == generation) show(found, metas) }
139         }
140     }

142     private fun show(found: List<Match>?, metas: Map<String, List<StoredMeta>>) {
143         results.removeAllViews()
144         if (found == null) {
145             note(getString(R.string.no_dictionaries))
146             return
147         }
148         if (found.isEmpty()) {
149             note(getString(R.string.no_results))
150             return
151         }
152         // The longest match is marked in the text.
153         val shown = SpannableString(text)
154         shown.setSpan(BackgroundColorSpan(0xFFFFE082.toInt()), offset, offset + found.first().length, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
155         textView.text = shown
156 
157         // One card per expression and reading, with the entries of each dictionary inside it.
158         val groups = found.groupBy { it.stored.term.expression to it.stored.term.reading }
159         for ((key, matches) in groups) {
160             val (expression, reading) = key
161             results.addView(View(this).apply { setBackgroundColor(Color.LTGRAY) }, LinearLayout.LayoutParams(-1, dp(1)).apply { topMargin = dp(10) })
162             results.addView(LinearLayout(this).apply {
163                 gravity = Gravity.CENTER_VERTICAL
164                 addView(TextView(this@LookupActivity).apply {
165                     text = if (reading.isEmpty() || reading == expression) expression else "$expression【$reading】"
166                     textSize = store.textSizeSp + 5
167                     setTypeface(typeface, Typeface.BOLD)
168                     setTextColor(Color.BLACK)
169                 }, LinearLayout.LayoutParams(0, -2, 1f))
170                 addView(smallButton("▶ " + getString(R.string.audio_play)) { play(expression, reading, choose = false) }.apply {
171                     setOnLongClickListener { play(expression, reading, choose = true); true }
172                 })
173                 if (anki) addView(smallButton(getString(R.string.anki)) { sendToAnki(expression, reading, matches) })
174             }, LinearLayout.LayoutParams(-1, -2).apply { topMargin = dp(6) })
175 
176             val reasons = matches.first().reasons
177             val meta = metas[expression].orEmpty().filter { m ->
178                 val forReading = when (m.meta.mode) { "freq" -> Glossary.frequencyReading(m.meta.data); "pitch" -> Glossary.pitchReading(m.meta.data); else -> null }
179                 forReading == null || forReading == reading || reading.isEmpty()
180             }
181             val line = buildList {
182                 if (reasons.isNotEmpty()) add(reasons.joinToString(" ‹ "))
183                 for (m in meta) when (m.meta.mode) {
184                     "freq" -> add("${m.dictionaryTitle}: ${Glossary.frequencyText(m.meta.data)}")
185                     "pitch" -> add(Glossary.pitchText(m.meta.data))
186                 }
187             }
188             if (line.isNotEmpty()) note(line.joinToString("   "), color = Color.DKGRAY, size = store.textSizeSp - 3)
189 
190             for (match in matches) {
191                 val term = match.stored.term
192                 val tags = (term.definitionTags.split(' ') + term.termTags.split(' ')).filter { it.isNotEmpty() }.distinct()
193                 note(

195                     color = Color.GRAY,

197                     top = 6,
198                 )
199                 results.addView(TextView(this).apply {

201                     textSize = store.textSizeSp
202                     setTextColor(Color.BLACK)
203                     movementMethod = LinkMovementMethod.getInstance()
204                 }, LinearLayout.LayoutParams(-1, -2).apply { topMargin = dp(2) })
205             }
206         }

214     /** Plays the first source that has audio. */
215     private fun play(expression: String, reading: String, choose: Boolean) {
216         thread {
217             val sources = sources(expression, reading)
218             if (choose) {
219                 runOnUiThread {
220                     if (sources.isEmpty()) toast(getString(R.string.no_audio)) else {
221                         AlertDialog.Builder(this).setItems(sources.map { it.name }.toTypedArray()) { _, i -> thread { playFirst(listOf(sources[i])) } }.show()
222                     }
223                 }
224             } else {
225                 playFirst(sources)
226             }
227         }
228     }

230     private fun playFirst(sources: List<AudioSource>) {
231         val bytes = loadFirst(sources)
232         runOnUiThread {
233             if (bytes == null) toast(getString(R.string.no_audio))
234             else runCatching { player.play(bytes) }.onFailure { toast(getString(R.string.audio_failed, it.message ?: "")) }
235         }
236     }

238     /** The bytes of the first source that has them. */
239     private fun loadFirst(sources: List<AudioSource>): ByteArray? {
240         for (source in sources) {
241             val bytes = runCatching { source.load() }.getOrNull() ?: continue
242             if (bytes.isNotEmpty()) return bytes
243         }
244         return null
245     }

247     /**
248      * One tap: the card goes to SubRead Anki with the word, the reading, the definitions of
249      * each dictionary, the text as the sentence, and the first audio. The pop-up hides for a
250      * moment, so that the screenshot of SubRead Anki shows the app under it.
251      */
252     private fun sendToAnki(expression: String, reading: String, matches: List<Match>) {
253         val definition = matches.joinToString("<br><br>") {
254             "<i>${Glossary.escape(it.stored.dictionaryTitle)}</i><br>${Glossary.toHtml(it.stored.term.glossary)}"
255         }
256         val source = referrer?.takeIf { it.scheme == "android-app" }?.host?.let { pkg ->
257             runCatching { packageManager.getApplicationLabel(packageManager.getApplicationInfo(pkg, 0)).toString() }.getOrNull()
258         } ?: ""
259         val decor = window.decorView
260         decor.visibility = View.INVISIBLE
261         thread {
262             val audio = loadFirst(sources(expression, reading))?.let { bytes ->
263                 val dir = File(cacheDir, "anki").apply { mkdirs() }
264                 File(dir, "word-${System.nanoTime()}.${AudioType.extension(bytes)}").also { it.writeBytes(bytes) }
265             }
266             runOnUiThread {
267                 runCatching { startActivity(Anki.intent(this, expression, reading, definition, text, source, audio)) }
268                     .onFailure { toast(getString(R.string.anki_failed)) }
269                 decor.postDelayed({ decor.visibility = View.VISIBLE }, HIDE_MS)
270             }
271         }
272     }
