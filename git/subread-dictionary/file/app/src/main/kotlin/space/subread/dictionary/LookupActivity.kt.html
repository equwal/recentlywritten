113     /** The text of the intent, from whichever of the three ways in. */
114     private fun take(intent: Intent) {
115         text = (intent.getCharSequenceExtra(Intent.EXTRA_PROCESS_TEXT) ?: intent.getCharSequenceExtra(Intent.EXTRA_TEXT) ?: "")
116             .toString().trim()
117         if (text.isEmpty()) {
118             Toast.makeText(this, R.string.no_text, Toast.LENGTH_SHORT).show()
119             finish()
120             return
121         }
122         scan(0)
123     }

125     private fun scan(at: Int) {
126         offset = at
127         val shown = SpannableString(text)
128         textView.text = shown
129         val run = ++generation
130         thread {
131             val found = if (dictionaries.anyEnabled()) Lookup.find(text, at, dictionaries) else null
132             val metas = found.orEmpty().map { it.stored.term.expression }.distinct().associateWith { dictionaries.meta(it) }
133             runOnUiThread { if (run == generation) show(found, metas) }
134         }
135     }

137     private fun show(found: List<Match>?, metas: Map<String, List<StoredMeta>>) {
138         results.removeAllViews()
139         if (found == null) {
140             note(getString(R.string.no_dictionaries))
141             return
142         }
143         if (found.isEmpty()) {
144             note(getString(R.string.no_results))
145             return
146         }
147         // The longest match is marked in the text.
148         val shown = SpannableString(text)
149         shown.setSpan(BackgroundColorSpan(0xFFFFE082.toInt()), offset, offset + found.first().length, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
150         textView.text = shown
151 
152         // One card per expression and reading, with the entries of each dictionary inside it.
153         val groups = found.groupBy { it.stored.term.expression to it.stored.term.reading }
154         for ((key, matches) in groups) {
155             val (expression, reading) = key
156             results.addView(View(this).apply { setBackgroundColor(Color.LTGRAY) }, LinearLayout.LayoutParams(-1, dp(1)).apply { topMargin = dp(10) })
157             results.addView(LinearLayout(this).apply {
158                 gravity = Gravity.CENTER_VERTICAL
159                 addView(TextView(this@LookupActivity).apply {
160                     text = if (reading.isEmpty() || reading == expression) expression else "$expression【$reading】"
161                     textSize = store.textSizeSp + 5
162                     setTypeface(typeface, Typeface.BOLD)
163                     setTextColor(Color.BLACK)
164                 }, LinearLayout.LayoutParams(0, -2, 1f))
165                 addView(smallButton("▶ " + getString(R.string.audio_play)) { play(expression, reading, choose = false) }.apply {
166                     setOnLongClickListener { play(expression, reading, choose = true); true }
167                 })
168             }, LinearLayout.LayoutParams(-1, -2).apply { topMargin = dp(6) })
169 
170             val reasons = matches.first().reasons
171             val meta = metas[expression].orEmpty().filter { m ->
172                 val forReading = when (m.meta.mode) { "freq" -> Glossary.frequencyReading(m.meta.data); "pitch" -> Glossary.pitchReading(m.meta.data); else -> null }
173                 forReading == null || forReading == reading || reading.isEmpty()
174             }
175             val line = buildList {
176                 if (reasons.isNotEmpty()) add(reasons.joinToString(" ‹ "))
177                 for (m in meta) when (m.meta.mode) {
178                     "freq" -> add("${m.dictionaryTitle}: ${Glossary.frequencyText(m.meta.data)}")
179                     "pitch" -> add(Glossary.pitchText(m.meta.data))
180                 }
181             }
182             if (line.isNotEmpty()) note(line.joinToString("   "), color = Color.DKGRAY, size = store.textSizeSp - 3)
183 
184             for (match in matches) {
185                 val term = match.stored.term
186                 val tags = (term.definitionTags.split(' ') + term.termTags.split(' ')).filter { it.isNotEmpty() }.distinct()
187                 note(
188                     listOf(match.stored.dictionaryTitle).plus(tags).joinToString("  "),
189                     color = Color.GRAY,
190                     size = store.textSizeSp - 4,
191                     top = 6,
192                 )
193                 results.addView(TextView(this).apply {
194                     text = Html.fromHtml(Glossary.toHtml(term.glossary), Html.FROM_HTML_MODE_COMPACT)
195                     textSize = store.textSizeSp
196                     setTextColor(Color.BLACK)
197                     movementMethod = LinkMovementMethod.getInstance()
198                 }, LinearLayout.LayoutParams(-1, -2).apply { topMargin = dp(2) })
199             }
200         }
201     }

203     /** Plays the first source that has audio: the local ones in their order, then the remote ones. */
204     private fun play(expression: String, reading: String, choose: Boolean) {
205         thread {
206             val sources = LocalAudio(store.localAudioFile, store.localSources.split(',').map { it.trim() }).sources(expression, reading) +
207                 if (store.remoteEnabled) RemoteAudio.sources(store.remoteUrls.lines().map { it.trim() }.filter { it.isNotEmpty() }, expression, reading) else emptyList()
208             if (choose) {
209                 runOnUiThread {
210                     if (sources.isEmpty()) toast(getString(R.string.no_audio)) else {
211                         AlertDialog.Builder(this).setItems(sources.map { it.name }.toTypedArray()) { _, i -> thread { playFirst(listOf(sources[i])) } }.show()
212                     }
213                 }
214             } else {
215                 playFirst(sources)
216             }
217         }
218     }

220     private fun playFirst(sources: List<AudioSource>) {
221         for (source in sources) {
222             val bytes = runCatching { source.load() }.getOrNull() ?: continue
223             runOnUiThread {
224                 runCatching { player.play(bytes) }.onFailure { toast(getString(R.string.audio_failed, it.message ?: "")) }
225             }
226             return
227         }
228         runOnUiThread { toast(getString(R.string.no_audio)) }
229     }
