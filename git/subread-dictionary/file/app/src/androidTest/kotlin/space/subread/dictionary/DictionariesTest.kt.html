37     private fun zip(vararg files: Pair<String, String>): ByteArray {
38         val bytes = ByteArrayOutputStream()
39         ZipOutputStream(bytes).use { out ->
40             for ((name, body) in files) {
41                 out.putNextEntry(ZipEntry(name))
42                 out.write(body.toByteArray())
43                 out.closeEntry()
44             }
45         }
46         return bytes.toByteArray()
47     }
