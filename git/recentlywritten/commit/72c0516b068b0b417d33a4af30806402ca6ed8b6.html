+<span class="a">+</span>
+<span class="a">+# ---- the bookmark file ----</span>
+<span class="a">+</span>
+<span class="a">+# Bookmarks restricted to $tagfilter and ordered by $sortorder, each followed</span>
+<span class="a">+# by a fourth field: how often it was used.</span>
+<span class="a">+rows () {</span>
+<span class="a">+    SBM_TAG=$tagfilter SBM_ORDER=$sortorder SBM_USAGE=$USAGE awk -F&#x27;\t&#x27; &#x27;</span>
+<span class="a">+        BEGIN {</span>
+<span class="a">+            while ((getline line &lt; ENVIRON[&quot;SBM_USAGE&quot;]) &gt; 0) {</span>
+<span class="a">+                split(line, f, &quot;\t&quot;)</span>
+<span class="a">+                used[f[1]] = f[2]</span>
+<span class="a">+            }</span>
+<span class="a">+            order = ENVIRON[&quot;SBM_ORDER&quot;]</span>
+<span class="a">+        }</span>
+<span class="a">+        NF == 0 || /^#/ { next }</span>
+<span class="a">+        ENVIRON[&quot;SBM_TAG&quot;] != &quot;&quot; {</span>
+<span class="a">+            hit = 0</span>
+<span class="a">+            n = split($3, tags, &quot; &quot;)</span>
+<span class="a">+            for (i = 1; i &lt;= n; i++)</span>
+<span class="a">+                if (tags[i] == ENVIRON[&quot;SBM_TAG&quot;]) hit = 1</span>
+<span class="a">+            if (!hit) next</span>
+<span class="a">+        }</span>
+<span class="a">+        {</span>
+<span class="a">+            if (order == &quot;used&quot;)        key = sprintf(&quot;%09d&quot;, 999999999 - used[$1])</span>
+<span class="a">+            else if (order == &quot;recent&quot;) key = sprintf(&quot;%09d&quot;, 999999999 - NR)</span>
+<span class="a">+            else if (order == &quot;desc&quot;)   key = tolower($2)</span>
+<span class="a">+            else if (order == &quot;tag&quot;)    key = tolower($3)</span>
+<span class="a">+            else                        key = &quot;&quot;</span>
+<span class="a">+            printf &quot;%s\t%s\t%s\t%s\t%s\t%d\n&quot;, key, tolower($1), $1, $2, $3, used[$1]</span>
+<span class="a">+        }&#x27; &quot;$BOOKMARKS&quot; | LC_ALL=C sort -t&quot;$TAB&quot; -k1,1 -k2,2 | cut -f3-</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# details &lt;url&gt;: a bookmark in full, one field per line.</span>
+<span class="a">+details () {</span>
+<span class="a">+    rows | SBM_URL=$1 awk -F&#x27;\t&#x27; &#x27;$1 == ENVIRON[&quot;SBM_URL&quot;] {</span>
+<span class="a">+        printf &quot;%s\n%s\ntags: %s\nused: %d\n&quot;, $1, $2, $3, $4</span>
+<span class="a">+        exit</span>
+<span class="a">+    }&#x27;</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+is_bookmark () {</span>
+<span class="a">+    SBM_URL=$1 awk -F&#x27;\t&#x27; &#x27;$1 == ENVIRON[&quot;SBM_URL&quot;] { found = 1; exit }</span>
+<span class="a">+        END { exit !found }&#x27; &quot;$BOOKMARKS&quot;</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# For fzf a fifth field is added, and it is all that is listed. The first</span>
+<span class="a">+# four feed the preview, which follows the cursor: the bookmark in full, then</span>
+<span class="a">+# the text of its page when bm-page is installed. fzf stops the preview when</span>
+<span class="a">+# the cursor moves on, so the short sleep keeps a fast scroll from fetching</span>
+<span class="a">+# each page on the way. A sleep that knows no fractions fails and skips it.</span>
+<span class="a">+PREVIEW=&quot;printf &#x27;%s\n%s\ntags: %s\nused: %s\n\n&#x27; {1} {2} {3} {4}; sleep 0.3 2&gt;/dev/null; bm-page {1} 2&gt;/dev/null&quot;</span>
+<span class="a">+fzfrows () {</span>
+<span class="a">+    awk -F&#x27;\t&#x27; &#x27;NF {</span>
+<span class="a">+        desc = $2</span>
+<span class="a">+        if (length(desc) &gt; 44) desc = substr(desc, 1, 42) &quot;..&quot;</span>
+<span class="a">+        printf &quot;%s\t%-44s  %-18s  %s\n&quot;, $0, desc, $3, $1</span>
+<span class="a">+    }&#x27;</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# withpreview &lt;fzf command and options&gt;: for the output of fzfrows.</span>
+<span class="a">+withpreview () {</span>
+<span class="a">+    &quot;$@&quot; --delimiter=&quot;$TAB&quot; --with-nth=5.. \</span>
+<span class="a">+         --preview-window=right:50%:wrap --preview=&quot;$PREVIEW&quot;</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# choose &lt;prompt&gt;: a bookmark&#x27;s URL, or whatever else the user typed.</span>
+<span class="a">+choose () {</span>
+<span class="a">+    if [ &quot;$SBM_MENU&quot; = fzf ]; then</span>
+<span class="a">+        rows | fzfrows | withpreview fzfpick --prompt=&quot;$1 &quot;</span>
+<span class="a">+    else</span>
+<span class="a">+        rows | cut -f1-3 | pick &quot;$1&quot;</span>
+<span class="a">+    fi | cut -f1</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# choose_url &lt;prompt&gt;: like choose, but only a bookmark will do.</span>
+<span class="a">+choose_url () {</span>
+<span class="a">+    if [ -z &quot;$(rows)&quot; ]; then</span>
+<span class="a">+        [ -n &quot;$tagfilter&quot; ] &amp;&amp; die &quot;no bookmarks tagged &#x27;$tagfilter&#x27;&quot;</span>
+<span class="a">+        die &#x27;no bookmarks yet; add one with: bm --add&#x27;</span>
+<span class="a">+    fi</span>
+<span class="a">+    url=$(choose &quot;$1&quot;)</span>
+<span class="a">+    [ -n &quot;$url&quot; ] || exit 1</span>
+<span class="a">+    is_bookmark &quot;$url&quot; || die &quot;not a bookmark: $url&quot;</span>
+<span class="a">+    printf &#x27;%s\n&#x27; &quot;$url&quot;</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# find_dup &lt;url&gt;: print the bookmark that is the same page as &lt;url&gt;, if any.</span>
+<span class="a">+find_dup () {</span>
+<span class="a">+    SBM_URL=$1 awk -F&#x27;\t&#x27; &quot;$AWK_NORM&quot;&#x27;</span>
+<span class="a">+        BEGIN { want = norm(ENVIRON[&quot;SBM_URL&quot;]) }</span>
+<span class="a">+        NF &amp;&amp; !/^#/ &amp;&amp; norm($1) == want { print; exit }&#x27; &quot;$BOOKMARKS&quot;</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# tmpfile &lt;file&gt;: create an empty temporary file and print its name. Beside</span>
+<span class="a">+# &lt;file&gt; if that directory can be written to, else in $TMPDIR. noclobber makes</span>
+<span class="a">+# the shell create it with O_EXCL, so a file or symlink planted under that</span>
+<span class="a">+# name in a shared /tmp is refused rather than followed: mktemp, in POSIX.</span>
+<span class="a">+tmpfile () {</span>
+<span class="a">+    for candidate in &quot;$1.tmp.$$&quot; &quot;${TMPDIR:-/tmp}/sbm.$$.${1##*/}&quot;; do</span>
+<span class="a">+        if (umask 077; set -C; : &gt; &quot;$candidate&quot;) 2&gt;/dev/null; then</span>
+<span class="a">+            printf &#x27;%s\n&#x27; &quot;$candidate&quot;</span>
+<span class="a">+            return 0</span>
+<span class="a">+        fi</span>
+<span class="a">+    done</span>
+<span class="a">+    return 1</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# rewrite &lt;file&gt;: replace &lt;file&gt; with stdin. Through cat, not mv, so that a</span>
+<span class="a">+# symlinked file stays a symlink and keeps its mode.</span>
+<span class="a">+rewrite () {</span>
+<span class="a">+    tmp=$(tmpfile &quot;$1&quot;) || die &quot;cannot create a temporary file for $1&quot;</span>
+<span class="a">+    cat &gt; &quot;$tmp&quot; &amp;&amp; cat &quot;$tmp&quot; &gt; &quot;$1&quot;</span>
+<span class="a">+    rc=$?</span>
+<span class="a">+    rm -f &quot;$tmp&quot;</span>
+<span class="a">+    return $rc</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# lock: take the write lock of the bookmark file, a directory beside it. bm</span>
+<span class="a">+# holds it while it changes the file, and bm-sync while it reads or writes</span>
+<span class="a">+# the file, so that neither writes over a change of the other. A lock whose</span>
+<span class="a">+# process is gone is taken over. After $SBM_LOCK_WAIT seconds (default 10)</span>
+<span class="a">+# bm stops.</span>
+<span class="a">+lock () {</span>
+<span class="a">+    waited=0</span>
+<span class="a">+    until mkdir &quot;$WLOCK&quot; 2&gt;/dev/null; do</span>
+<span class="a">+        pid=$(cat &quot;$WLOCK/pid&quot; 2&gt;/dev/null)</span>
+<span class="a">+        if [ -n &quot;$pid&quot; ] &amp;&amp; ! kill -0 &quot;$pid&quot; 2&gt;/dev/null; then</span>
+<span class="a">+            [ &quot;$(cat &quot;$WLOCK/pid&quot; 2&gt;/dev/null)&quot; != &quot;$pid&quot; ] || rm -rf &quot;$WLOCK&quot;</span>
+<span class="a">+            continue</span>
+<span class="a">+        fi</span>
+<span class="a">+        [ &quot;$waited&quot; -lt &quot;${SBM_LOCK_WAIT:-10}&quot; ] \</span>
+<span class="a">+            || die &quot;$BOOKMARKS is in use (bm -e or bm-sync). If neither runs, remove $WLOCK&quot;</span>
+<span class="a">+        sleep 1</span>
+<span class="a">+        waited=$((waited + 1))</span>
+<span class="a">+    done</span>
+<span class="a">+    echo $$ &gt; &quot;$WLOCK/pid&quot;</span>
+<span class="a">+    locked=1</span>
+<span class="a">+    # Also in the subshell of add_inside, which does not keep these traps.</span>
+<span class="a">+    trap unlock EXIT</span>
+<span class="a">+    trap &#x27;exit 1&#x27; HUP INT TERM</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+unlock () {</span>
+<span class="a">+    [ -z &quot;$locked&quot; ] || rm -rf &quot;$WLOCK&quot;</span>
+<span class="a">+    locked=</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# bump &lt;url&gt;: count a use, for the &quot;used&quot; order. The counts live beside the</span>
+<span class="a">+# bookmark file, which stays a plain list: url, count, time of last use.</span>
+<span class="a">+# srand() returns the previous seed, and the first seed is the time of day:</span>
+<span class="a">+# that is &quot;date +%s&quot; for a POSIX that has no %s.</span>
+<span class="a">+bump () {</span>
+<span class="a">+    [ -e &quot;$USAGE&quot; ] || : &gt; &quot;$USAGE&quot; || return 0</span>
+<span class="a">+    SBM_URL=$1 awk -F&#x27;\t&#x27; &#x27;</span>
+<span class="a">+        BEGIN { OFS = &quot;\t&quot;; srand(); now = srand() }</span>
+<span class="a">+        $1 == ENVIRON[&quot;SBM_URL&quot;] { $2 = $2 + 1; $3 = now; seen = 1 }</span>
+<span class="a">+        { print }</span>
+<span class="a">+        END { if (!seen) print ENVIRON[&quot;SBM_URL&quot;], 1, now }</span>
+<span class="a">+    &#x27; &quot;$USAGE&quot; | rewrite &quot;$USAGE&quot;</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# learn_tags: tag names on stdin; the ones $USERTAGS lacks are appended to it.</span>
+<span class="a">+learn_tags () {</span>
+<span class="a">+    [ -e &quot;$USERTAGS&quot; ] || : &gt; &quot;$USERTAGS&quot;</span>
+<span class="a">+    while IFS= read -r name; do</span>
+<span class="a">+        [ -n &quot;$name&quot; ] || continue</span>
+<span class="a">+        sed &#x27;s/[ |].*//&#x27; &quot;$USERTAGS&quot; | grep -Fxq -- &quot;$name&quot; \</span>
+<span class="a">+            || printf &#x27;%s | \n&#x27; &quot;$name&quot; &gt;&gt; &quot;$USERTAGS&quot;</span>
+<span class="a">+    done</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# changed &lt;message&gt;: the file was written. bm-commit, where installed, keeps</span>
+<span class="a">+# its history in git. bm-sync, where installed, tells your other machines; it</span>
+<span class="a">+# runs in the background, so that a slow network never holds the menu up.</span>
+<span class="a">+changed () {</span>
+<span class="a">+    if have bm-commit; then bm-commit &quot;$1&quot;; fi</span>
+<span class="a">+    if have bm-sync; then bm-sync -q &gt;/dev/null 2&gt;&amp;1 &amp; fi</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# ---- actions ----</span>
+<span class="a">+</span>
+<span class="a">+copy_url () {</span>
+<span class="a">+    bump &quot;$1&quot;</span>
+<span class="a">+    printf &#x27;%s&#x27; &quot;$1&quot; | clip_out</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+open_url () {</span>
+<span class="a">+    bump &quot;$1&quot;</span>
+<span class="a">+    opener &quot;$1&quot;</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# The editor reads the file when it starts and writes it when you save, so</span>
+<span class="a">+# bm holds the lock until the editor ends. bm-sync then waits, and syncs the</span>
+<span class="a">+# edit after it.</span>
+<span class="a">+edit_url () {</span>
+<span class="a">+    lock</span>
+<span class="a">+    line=$(SBM_URL=$1 awk -F&#x27;\t&#x27; \</span>
+<span class="a">+           &#x27;$1 == ENVIRON[&quot;SBM_URL&quot;] { print NR; exit }&#x27; &quot;$BOOKMARKS&quot;)</span>
+<span class="a">+    ${VISUAL:-${EDITOR:-vi}} &quot;+${line:-1}&quot; &quot;$BOOKMARKS&quot;</span>
+<span class="a">+    unlock</span>
+<span class="a">+    changed &#x27;bm: edit&#x27;</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+delete_url () {</span>
+<span class="a">+    lock</span>
+<span class="a">+    for file in &quot;$BOOKMARKS&quot; &quot;$USAGE&quot;; do</span>
+<span class="a">+        [ -e &quot;$file&quot; ] || continue</span>
+<span class="a">+        SBM_URL=$1 awk -F&#x27;\t&#x27; &#x27;$1 != ENVIRON[&quot;SBM_URL&quot;]&#x27; &quot;$file&quot; \</span>
+<span class="a">+            | rewrite &quot;$file&quot; || die &quot;could not rewrite $file&quot;</span>
+<span class="a">+    done</span>
+<span class="a">+    unlock</span>
+<span class="a">+    changed &quot;bm: delete $1&quot;</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# urlencode &lt;text&gt;: for a query string. awk has no ord(), hence the table.</span>
+<span class="a">+urlencode () {</span>
+<span class="a">+    printf &#x27;%s&#x27; &quot;$1&quot; | LC_ALL=C awk &#x27;</span>
+<span class="a">+        BEGIN { for (i = 1; i &lt; 256; i++) ord[sprintf(&quot;%c&quot;, i)] = i }</span>
+<span class="a">+        {</span>
+<span class="a">+            for (i = 1; i &lt;= length($0); i++) {</span>
+<span class="a">+                c = substr($0, i, 1)</span>
+<span class="a">+                if (c ~ /[A-Za-z0-9._~-]/) out = out c</span>
+<span class="a">+                else if (c == &quot; &quot;)         out = out &quot;+&quot;</span>
+<span class="a">+                else                       out = out sprintf(&quot;%%%02X&quot;, ord[c])</span>
+<span class="a">+            }</span>
+<span class="a">+        }</span>
+<span class="a">+        END { printf &quot;%s&quot;, out }&#x27;</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# go &lt;text&gt;: open text that is not a bookmark. One word with &quot;://&quot; or a dot</span>
+<span class="a">+# in it is an address. Anything else is a web search: by the engine whose</span>
+<span class="a">+# keyword is the first word ($ENGINES holds &quot;keyword | url with %s&quot; lines),</span>
+<span class="a">+# or else by $SEARCH.</span>
+<span class="a">+go () {</span>
+<span class="a">+    case $1 in</span>
+<span class="a">+        *&#x27; &#x27;*) ;;</span>
+<span class="a">+        *://*) opener &quot;$1&quot;; return ;;</span>
+<span class="a">+        *.*)   opener &quot;https://$1&quot;; return ;;</span>
+<span class="a">+    esac</span>
+<span class="a">+    engine=$SEARCH</span>
+<span class="a">+    query=$1</span>
+<span class="a">+    if [ &quot;${1%% *}&quot; != &quot;$1&quot; ] &amp;&amp; [ -r &quot;$ENGINES&quot; ]; then</span>
+<span class="a">+        found=$(SBM_KEY=${1%% *} awk -F&#x27; *[|] *&#x27; \</span>
+<span class="a">+                &#x27;$1 == ENVIRON[&quot;SBM_KEY&quot;] { print $2; exit }&#x27; &quot;$ENGINES&quot;)</span>
+<span class="a">+        if [ -n &quot;$found&quot; ]; then</span>
+<span class="a">+            engine=$found</span>
+<span class="a">+            query=${1#* }</span>
+<span class="a">+        fi</span>
+<span class="a">+    fi</span>
+<span class="a">+    opener &quot;$(printf &#x27;%s\n&#x27; &quot;$engine&quot; | sed &quot;s|%s|$(urlencode &quot;$query&quot;)|&quot;)&quot;</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+plumb () {</span>
+<span class="a">+    url=$(choose &#x27;open:&#x27;)</span>
+<span class="a">+    [ -n &quot;$url&quot; ] || exit 1</span>
+<span class="a">+    if is_bookmark &quot;$url&quot;; then open_url &quot;$url&quot;; else go &quot;$url&quot;; fi</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# Exit status 3 means &quot;already bookmarked&quot;.</span>
+<span class="a">+add () {</span>
+<span class="a">+    url=$1</span>
+<span class="a">+    if [ -z &quot;$url&quot; ]; then</span>
+<span class="a">+        url=$(ask &#x27;url&#x27; &quot;$(clip_in | grep &#x27;://&#x27; | head -n 1)&quot;)</span>
+<span class="a">+    fi</span>
+<span class="a">+    url=$(printf &#x27;%s&#x27; &quot;$url&quot; | tr -d &#x27;[:space:]&#x27;)</span>
+<span class="a">+    [ -n &quot;$url&quot; ] || exit 1</span>
+<span class="a">+</span>
+<span class="a">+    dup=$(find_dup &quot;$url&quot;)</span>
+<span class="a">+    if [ -n &quot;$dup&quot; ]; then</span>
+<span class="a">+        printf &#x27;bm: already bookmarked:\n%s\n&#x27; &quot;$dup&quot; &gt;&amp;2</span>
+<span class="a">+        exit 3</span>
+<span class="a">+    fi</span>
+<span class="a">+</span>
+<span class="a">+    # bm-title, where installed, fetches the page&#x27;s title.</span>
+<span class="a">+    title=</span>
+<span class="a">+    if have bm-title; then title=$(bm-title &quot;$url&quot;); fi</span>
+<span class="a">+    desc=$(ask &#x27;description&#x27; &quot;$title&quot; | tr &#x27;\t&#x27; &#x27; &#x27;)</span>
+<span class="a">+</span>
+<span class="a">+    [ -e &quot;$USERTAGS&quot; ] || : &gt; &quot;$USERTAGS&quot;</span>
+<span class="a">+    names=$(pickmulti &#x27;tags:&#x27; &lt; &quot;$USERTAGS&quot; | sed &#x27;s/[ |].*//&#x27; | sed &#x27;/^$/d&#x27; \</span>
+<span class="a">+            | sort -u)</span>
+<span class="a">+    printf &#x27;%s\n&#x27; &quot;$names&quot; | learn_tags</span>
+<span class="a">+    tags=$(printf &#x27;%s\n&#x27; &quot;$names&quot; | paste -sd &#x27; &#x27; -)</span>
+<span class="a">+</span>
+<span class="a">+    lock</span>
+<span class="a">+    printf &#x27;%s\t%s\t%s\n&#x27; &quot;$url&quot; &quot;$desc&quot; &quot;$tags&quot; &gt;&gt; &quot;$BOOKMARKS&quot;</span>
+<span class="a">+    unlock</span>
+<span class="a">+    changed &quot;bm: add $url&quot;</span>
+<span class="a">+}</span>
+<span class="a">+</span>
+<span class="a">+# merge: bookmark lines on stdin (a bare URL per line will do); the ones not</span>
+<span class="a">+# known yet are appended. This is how the output of bm-import gets in.</span>
+<span class="a">+merge () {</span>
