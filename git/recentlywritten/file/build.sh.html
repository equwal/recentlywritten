1 #!/bin/sh
2 # build.sh — converts posts/*.md to site/ using pandoc
3 # Usage: sh build.sh
4 
5 set -e
6 
7 POSTS_DIR="posts"
8 SITE_DIR="site"
9 CSS="../style.css"   # relative path from inside site/
10 
11 # ── setup ──────────────────────────────────────────────────
12 rm -rf "$SITE_DIR"
13 mkdir -p "$SITE_DIR"
14 cp style.css "$SITE_DIR/style.css"
15 
16 # pandoc HTML template for individual posts
17 TEMPLATE="template.html"
18 cat > "$TEMPLATE" << 'TMPL'
19 <!DOCTYPE html>
20 <html lang="en">
21 <head>
22   <meta charset="UTF-8" />
23   <meta name="viewport" content="width=device-width, initial-scale=1" />
24   <title>$title$ — Recently Written</title>
25   <link rel="stylesheet" href="style.css" />
26 </head>
27 <body>
28 <div id="container">
29 
30   <div id="top">
31     <a class="site-title" href="index.html">Recently Written</a>
32     <nav>
33       <a href="index.html">Home</a>
34     </nav>
35   </div>
36 
37   <h1>$title$</h1>
38   <div class="post-meta">$date$</div>
39 
40   $body$
41 
42   <div id="footer">
43     <a href="index.html">← Home</a> ·
44     Powered by <a href="https://neuron.zettel.page">Neuron</a>
45   </div>
46 
47 </div>
48 </body>
49 </html>
50 TMPL
51 
52 # ── build each post ─────────────────────────────────────────
53 LIST=""   # accumulate post-list items (newest first)
54 
55 # collect posts and sort by date descending
56 for md in "$POSTS_DIR"/*.md; do
57     slug=$(basename "$md" .md)
58     title=$(grep '^title:' "$md" | sed 's/title: *//')
59     date=$(grep '^date:'  "$md" | sed 's/date: *//')
60 
61     pandoc \
62         --from markdown \
63         --to html5 \
64         --template "$TEMPLATE" \
65         --metadata title="$title" \
66         --metadata date="$date" \
67         --output "$SITE_DIR/${slug}.html" \
68         "$md"
69 
70     # prepend to list so newest-first after sort
71     LIST="$(printf '%s\n' "${date}|${slug}|${title}")\n${LIST}"
72 done
73 
74 # ── build index ─────────────────────────────────────────────
75 # sort descending by date, emit list items
76 POST_ITEMS=""
77 printf '%b' "$LIST" | sort -r | while IFS='|' read -r date slug title; do
78     [ -z "$slug" ] && continue
79     printf '<li><span class="post-date">%s</span><a href="%s.html">%s</a></li>\n' \
80         "$date" "$slug" "$title"
81 done > /tmp/rw-items.txt
82 
83 ITEMS=$(cat /tmp/rw-items.txt)
84 
85 # inject post list into index.html using awk
86 awk -v items="<ul class=\"post-list\">$(cat /tmp/rw-items.txt)</ul>" \
87     '{ gsub(/<!-- POST_LIST -->/, items); print }' \
88     index.html > "$SITE_DIR/index.html"
89 
90 # ── cleanup ──────────────────────────────────────────────────
91 rm -f "$TEMPLATE" /tmp/rw-items.txt
92 
93 echo "Built $(ls "$SITE_DIR"/*.html | wc -l | tr -d ' ') pages → $SITE_DIR/"
94 rsync -avzP --delete site/ root@honjimaku.com:/var/www/recentlywritten/
