118 # Read one `key: value` line out of a file's front matter, dropping the
119 # surrounding quotes that titles containing a colon have to be written with.
120 meta() {
121     sed -n "s/^$2: *//p" "$1" | head -1 | sed -e 's/^"//' -e 's/"$//' -e 's/\\"/"/g'
122 }
123 
124 # Escape the characters that must not appear raw in generated HTML.
125 esc() {
126     sed -e 's/&/\&amp;/g' -e 's/</\&lt;/g' -e 's/>/\&gt;/g'
127 }

129 # render <markdown> <output> <title>
130 render() {
131     pandoc \
132         --from markdown \
133         --to html5 \
134         --template "$TEMPLATE" \
135         --metadata title="$3" \
136         --metadata url="$SITE_URL/$(basename "$2")" \
137         --output "$2" \
138         "$1"
139 }

141 # ── build each post ─────────────────────────────────────────
142 : > "$WORK/posts.tsv"
143 for md in "$POSTS_DIR"/*.md; do
144     [ -e "$md" ] || continue
145     slug=$(basename "$md" .md)
146     title=$(meta "$md" title)
147     date=$(meta "$md" date)
148     # Recovered posts carry `order`: their position on the old front page,
149     # the only surviving record of their sequence. Everything else sorts by
150     # date. The two never interleave, since an order is three digits ("051")
151     # and a date leads with its year ("2026-..."), so new posts land on top.
152     order=$(meta "$md" order)
153     [ -z "$order" ] && order="$date"
154     render "$md" "$SITE_DIR/${slug}.html" "$title"
155     printf '%s\t%s\t%s\t%s\n' "$order" "$date" "$slug" "$title" >> "$WORK/posts.tsv"
156 done

212 # ── feed ─────────────────────────────────────────────────────
213 # Only the newest FEED_SIZE posts go in, and that is deliberate as well as
214 # conventional: a feed reader shows each item's date, and the newest posts are
215 # the ones whose dates are exact rather than recovered from a crawl.
216 
217 # RFC 822, as RSS requires. The C locale matters: without it the day and
218 # month names come out in the system language. `date -d` is GNU; where it is
219 # missing the item simply goes out without a pubDate. A post has only a
220 # date, so its time is midnight; a podcast episode has an exact time.
221 feed_date() {
222     LC_ALL=C date -u -d "$1" '+%a, %d %b %Y %H:%M:%S +0000' 2>/dev/null
223 }
224 
225 # A feed reader shows the HTML away from the site, so links relative to it
226 # need the site put in front of them.
227 absolute() {
228     sed -E \
229         -e 's@(href|src)="/([^/"][^"]*)"@\1="'"$SITE_URL"'/\2"@g' \
230         -e 's@(href|src)="([^"/#][^":]*)"@\1="'"$SITE_URL"'/\2"@g'
231 }

233 {
234     printf '<?xml version="1.0" encoding="UTF-8"?>\n'
235     printf '<rss version="2.0" xmlns:atom="http://www.w3.org/2005/Atom">\n<channel>\n'
236     printf '  <title>Recently Written</title>\n'
237     printf '  <link>%s/</link>\n' "$SITE_URL"
238     printf '  <description>A site full of things which I have recently written.</description>\n'
239     printf '  <language>en-us</language>\n'
240     printf '  <atom:link href="%s/rss.xml" rel="self" type="application/rss+xml" />\n' "$SITE_URL"
241     head -n "$FEED_SIZE" "$WORK/sorted.tsv" | while IFS="$TAB" read -r order date slug title; do
242         url="$SITE_URL/$slug.html"
243         printf '  <item>\n'
244         printf '    <title>%s</title>\n' "$(printf '%s' "$title" | esc)"
245         printf '    <link>%s</link>\n' "$url"
246         printf '    <guid isPermaLink="true">%s</guid>\n' "$url"
247         pub=$(feed_date "$date") && [ -n "$pub" ] &&

267 # podcast.tsv lists the podcast episodes, newest first, one on each line:
268 #   date <TAB> file <TAB> bytes <TAB> seconds <TAB> youtube <TAB> title
269 # date is the UTC time of the upload in ISO 8601. file is the name of the

366 # ── deploy ───────────────────────────────────────────────────
367 # git/ and git.html share this web root but are published by deploy-git.sh,
368 # not built here. Excluding them keeps --delete from treating them as stale:
369 # rsync never deletes an excluded path on the receiving side.
370 # static/book/ holds book files that stay out of the repo (see .gitignore).
371 # They exist only on the server, so the same rule protects them.
372 # static/language-podcast/ holds the podcast audio, which also stays out of
373 # the repo. The P (protect) rule stops --delete from removing the server
374 # copy when the working tree has no audio, but the audio that the working
375 # tree has still uploads.
376 # sitemap.xml and robots.txt were written on the server and are not built here.
377 if [ "$DEPLOY" = yes ]; then
378     rsync -avzP --delete --exclude=/git/ --exclude=/git.html \
379         --exclude=/static/book/ --filter='P /static/language-podcast/***' \
380         --exclude=/sitemap.xml --exclude=/robots.txt \
381         "$SITE_DIR/" "$DEPLOY_HOST:$DEPLOY_PATH"
382     ssh "$DEPLOY_HOST" "chmod -R a+rX $DEPLOY_PATH"
383 fi
