31 func (b *browser) do(method, path string, form url.Values) (int, string) {
32 	b.t.Helper()
33 	var body io.Reader
34 	if form != nil {
35 		body = strings.NewReader(form.Encode())
36 	}
37 	req, err := http.NewRequest(method, b.base+path, body)
38 	if err != nil {
39 		b.t.Fatal(err)
40 	}
41 	if form != nil {
42 		req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
43 	}
44 	resp, err := b.client.Do(req)
45 	if err != nil {
46 		b.t.Fatal(err)
47 	}
48 	defer resp.Body.Close()
49 	text, _ := io.ReadAll(resp.Body)
50 	return resp.StatusCode, string(text)
51 }

55 // signIn does the sign-in steps of a person in a browser and returns the
56 // settings page.
57 func (b *browser) signIn(e *testEnv, email string) string {
58 	b.t.Helper()
59 	n := len(e.mail.sent)
60 	if code, page := b.do("POST", "/account/signin", url.Values{"email": {email}}); code != 200 || !strings.Contains(page, "Check your email") {
61 		b.t.Fatalf("signin: %d\n%s", code, page)
62 	}
63 	if len(e.mail.sent) != n+1 {
64 		b.t.Fatalf("sent %d sign-in messages, want 1", len(e.mail.sent)-n)
65 	}
66 	m := linkToken.FindStringSubmatch(e.mail.sent[n].Body)
67 	if m == nil {
68 		b.t.Fatalf("no link in:\n%s", e.mail.sent[n].Body)
69 	}
70 	// A GET request, as from a mail scanner, must not use the token.
71 	if code, page := b.do("GET", "/account/verify?t="+m[1], nil); code != 200 || !strings.Contains(page, `value="`+m[1]+`"`) {
72 		b.t.Fatalf("verify page: %d\n%s", code, page)
73 	}
74 	code, page := b.do("POST", "/account/verify", url.Values{"t": {m[1]}})
75 	if code != 200 || !strings.Contains(page, "Signed in as "+strings.ToLower(email)) {
76 		b.t.Fatalf("verify: %d\n%s", code, page)
77 	}
78 	return page
79 }

85 	b := newBrowser(t, e)
86 
87 	if code, page := b.do("GET", "/account/", nil); code != 200 || !strings.Contains(page, `action="/account/signin"`) {
88 		t.Fatalf("signed-out page: %d\n%s", code, page)
89 	}
90 	page := b.signIn(e, "Reader@Example.com")
91 	if e.mail.sent[0].To != "reader@example.com" {
92 		t.Errorf("sign-in mail went to %q", e.mail.sent[0].To)
93 	}
94 
95 	feed := feedURL.FindString(page)
96 	if feed == "" {
97 		t.Fatalf("no feed URL on the settings page:\n%s", page)
98 	}
99 	// The feed needs no cookie, because a feed reader app has none.
100 	anon := &browser{t: t, client: http.DefaultClient, base: b.base}
101 	code, doc := anon.do("GET", feed, nil)
102 	if code != 200 || wellFormed([]byte(doc)) != nil {
103 		t.Fatalf("feed: %d\n%s", code, doc)
104 	}
105 	for _, want := range []string{"<title>Old article</title>", "old-project.html", "Body of old-article &amp;amp; more."} {
106 		if !strings.Contains(doc, want) {
107 			t.Errorf("feed has no %q:\n%s", want, doc)
108 		}
109 	}
110 
111 	// Mute "projects". The feed then leaves out the project post.
112 	b.do("POST", "/account/settings", url.Values{"email_on": {"1"}, "tag": {"articles"}})
113 	if _, doc := anon.do("GET", feed, nil); strings.Contains(doc, "old-project.html") || !strings.Contains(doc, "old-article.html") {
114 		t.Errorf("muted tag still in feed:\n%s", doc)
115 	}
116 
117 	// A new feed URL revokes the old one.
118 	_, page = b.do("POST", "/account/feed/rotate", nil)
119 	newFeed := feedURL.FindString(page)
120 	if newFeed == "" || newFeed == feed {
121 		t.Fatalf("feed URL did not change: %q", newFeed)
122 	}
123 	if code, _ := anon.do("GET", feed, nil); code != 404 {
124 		t.Errorf("old feed URL gave %d, want 404", code)
125 	}
126 	if code, _ := anon.do("GET", newFeed, nil); code != 200 {
127 		t.Errorf("new feed URL gave %d, want 200", code)
128 	}
129 
130 	// After sign-out, the settings page asks for an address again.
131 	if _, page := b.do("POST", "/account/signout", nil); !strings.Contains(page, `action="/account/signin"`) {
132 		t.Errorf("still signed in after sign-out:\n%s", page)
133 	}
134 }
