31 // routes gives the handler for all paths under /account/. The web server
32 // in front sends only these paths to this service.
33 //
34 // Each change of state uses POST. A GET request from a link in an email
35 // only shows a page with a button, because mail scanners open the links
36 // in email and must not use a sign-in link or unsubscribe a user. The
37 // session cookie is SameSite=Lax, so a form on another site cannot send a
38 // POST request with the cookie.
39 func (a *app) routes() http.Handler {
40 	mux := http.NewServeMux()
41 	mux.HandleFunc("GET /account/{$}", a.home)
42 	mux.HandleFunc("POST /account/signin", a.signin)
43 	mux.HandleFunc("GET /account/verify", a.verifyPage)
44 	mux.HandleFunc("POST /account/verify", a.verify)
45 	mux.HandleFunc("POST /account/settings", a.withUser(a.settings))
46 	mux.HandleFunc("POST /account/feed/rotate", a.withUser(a.rotateFeed))
47 	mux.HandleFunc("POST /account/signout", a.signout)
48 	mux.HandleFunc("POST /account/delete", a.withUser(a.deleteAccount))
49 	mux.HandleFunc("GET /account/unsubscribe", a.unsubscribePage)
50 	mux.HandleFunc("POST /account/unsubscribe", a.unsubscribe)
51 	mux.HandleFunc("GET /account/feed/{token}", a.feed)
52 	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
53 		// The URLs of some pages contain tokens. These headers keep the
54 		// tokens out of caches and out of the Referer header.
55 		w.Header().Set("Cache-Control", "no-store")
56 		w.Header().Set("Referrer-Policy", "no-referrer")
57 		r.Body = http.MaxBytesReader(w, r.Body, 64<<10)
58 		mux.ServeHTTP(w, r)
59 	})
60 }

150 // normalizeEmail returns the address in lower case, or an error if s is
151 // not one plain address such as "name@example.com".
152 func normalizeEmail(s string) (string, error) {
153 	s = strings.TrimSpace(s)
154 	addr, err := mail.ParseAddress(s)
155 	if err != nil || addr.Name != "" || addr.Address != s || len(s) > 254 {
156 		return "", errors.New("not a valid address")
157 	}
158 	return strings.ToLower(s), nil
159 }

182 		})

186 		}

214 func (a *app) settings(w http.ResponseWriter, r *http.Request, u user) {

218 	}

223 	}

226 		checked[t] = true
227 	}
228 	// A tag is muted if the form shows it and the user cleared its box.
