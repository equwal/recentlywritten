154 // TestNotifyNoDuplicatesProperty runs random sequences of new posts, new
155 // users, send failures, and notify runs. At the end no user has a post two
156 // times, and each user has each post that notify first saw at or after
157 // the sign-up.
158 func TestNotifyNoDuplicatesProperty(t *testing.T) {
159 	rapid.Check(t, func(rt *rapid.T) {
160 		e := newTestEnv(t, oldPosts)
161 		if err := e.app.notify(); err != nil { // the first run records the old posts
162 			rt.Fatal(err)
163 		}
164 		index := append([]Post(nil), oldPosts...)
165 		seenAt := map[string]time.Time{}   // slug -> time of the first notify run that saw it
166 		joinedAt := map[string]time.Time{} // email -> sign-up time
167 		run := func() {
168 			for _, p := range index {
169 				if _, ok := seenAt[p.Slug]; !ok {
170 					seenAt[p.Slug] = e.clock.now()
171 				}
172 			}
173 			_ = e.app.notify()
174 		}
175 		for _, p := range oldPosts {
176 			seenAt[p.Slug] = time.Time{}
177 		}
178 
179 		steps := rapid.SliceOfN(rapid.IntRange(0, 3), 1, 30).Draw(rt, "steps")
180 		for i, step := range steps {
181 			e.advance(time.Minute)
182 			switch step {
183 			case 0:
184 				p := Post{Date: "2026-09-02", Slug: fmt.Sprintf("p%d", i), Title: fmt.Sprintf("P%d", i)}
185 				index = append([]Post{p}, index...)
186 				e.writeIndex(rt, index)
187 			case 1:
188 				email := fmt.Sprintf("u%d@example.com", i)
189 				e.signUp(rt, email)
190 				joinedAt[email] = e.clock.now()
191 			case 2:
192 				e.mail.fail = !e.mail.fail
193 			case 3:
194 				run()
195 			}
196 		}
197 		e.advance(time.Minute)
198 		e.mail.fail = false
199 		run()
200 
201 		got := map[string]map[string]int{} // email -> slug -> count
202 		for _, m := range e.mail.sent {
203 			if got[m.To] == nil {
204 				got[m.To] = map[string]int{}
205 			}
206 			for _, line := range strings.Split(m.Body, "\n") {
207 				if s, ok := strings.CutPrefix(line, "https://example.com/"); ok {
208 					got[m.To][strings.TrimSuffix(s, ".html")]++
209 				}
210 			}
211 		}
212 		for email, joined := range joinedAt {
213 			want := 0
214 			for _, p := range index {
215 				if seenAt[p.Slug].Before(joined) {
216 					continue
217 				}
218 				want++
219 				if n := got[email][p.Slug]; n != 1 {
220 					rt.Fatalf("%s got %s %d times, want 1", email, p.Slug, n)
221 				}
222 			}
223 			if len(got[email]) != want {
224 				rt.Fatalf("%s got %v, want %d posts", email, got[email], want)
225 			}
226 		}
227 	})
228 }
