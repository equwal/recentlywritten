15 // captureMailer keeps each message. If fail is true, Send returns an error
16 // and keeps nothing.
17 type captureMailer struct {
18 	mu   sync.Mutex
19 	sent []message
20 	fail bool
21 }
22 
23 func (c *captureMailer) Send(m message) error {
24 	c.mu.Lock()
25 	defer c.mu.Unlock()
26 	if c.fail {
27 		return errors.New("smtp is down")
28 	}
29 	c.sent = append(c.sent, m)
30 	return nil
31 }

52 // newTestEnv makes an app with a :memory: database and a site directory

70 		cfg:  config{BaseURL: "https://example.com", SiteDir: site},
71 		db:   db,
72 		mail: e.mail,
73 		log:  log.New(io.Discard, "", 0),
74 	}
