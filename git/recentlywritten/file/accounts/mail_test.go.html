19 // TestFormatMessageProperty checks that any subject and body give a
20 // message that parses, has exactly the expected headers, and decodes back
21 // to the same subject and body.
22 func TestFormatMessageProperty(t *testing.T) {
23 	rapid.Check(t, func(t *rapid.T) {
24 		valid := rapid.String().Filter(utf8.ValidString)
25 		m := message{
26 			To:          "reader@example.com",
27 			Subject:     valid.Draw(t, "subject"),
28 			Body:        valid.Draw(t, "body"),
29 			Unsubscribe: rapid.SampledFrom([]string{"", "https://example.com/account/unsubscribe?t=ABC"}).Draw(t, "unsub"),
30 		}
31 		data, err := formatMessage(testFrom, m, time.Unix(0, 0))
32 		if err != nil {
33 			t.Fatal(err)
34 		}
35 		msg, err := mail.ReadMessage(bytes.NewReader(data))
36 		if err != nil {
37 			t.Fatalf("message does not parse: %v\n%q", err, data)
38 		}
39 		wantHeaders := 8
40 		if m.Unsubscribe != "" {
41 			wantHeaders += 2
42 		}
43 		if len(msg.Header) != wantHeaders {
44 			t.Fatalf("got headers %v, want %d (a line break in the subject must not add one)", msg.Header, wantHeaders)
45 		}
46 		subject, err := new(mime.WordDecoder).DecodeHeader(msg.Header.Get("Subject"))
47 		if err != nil {
48 			t.Fatal(err)
49 		}
50 		wantSubject := strings.NewReplacer("\r", " ", "\n", " ").Replace(m.Subject)
51 		// net/mail removes spaces at the two ends of a plain header value,
52 		// but not in an encoded word.
53 		if strings.Trim(subject, " \t") != strings.Trim(wantSubject, " \t") {
54 			t.Fatalf("subject: got %q, want %q", subject, wantSubject)
55 		}
56 		body, err := io.ReadAll(quotedprintable.NewReader(msg.Body))
57 		if err != nil {
58 			t.Fatal(err)
59 		}
60 		lf := strings.NewReplacer("\r\n", "\n", "\r", "\n")
61 		if got := lf.Replace(string(body)); got != lf.Replace(m.Body) {
62 			t.Fatalf("body: got %q, want %q", got, m.Body)
63 		}
64 		if got := msg.Header.Get("List-Unsubscribe"); m.Unsubscribe != "" && got != "<"+m.Unsubscribe+">" {
65 			t.Fatalf("List-Unsubscribe = %q", got)
66 		}
67 	})
68 }
