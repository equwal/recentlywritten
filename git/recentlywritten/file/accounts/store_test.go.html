14 // TestTokensProperty checks that tokens are URL-safe, are all different,
15 // and that different tokens have different hashes.
16 func TestTokensProperty(t *testing.T) {
17 	seen := map[string]bool{}
18 	rapid.Check(t, func(t *rapid.T) {
19 		tok := newToken()
20 		if !tokenForm.MatchString(tok) {
21 			t.Fatalf("token %q is not 26 base32 characters", tok)
22 		}
23 		if seen[tok] {
24 			t.Fatalf("token %q came two times", tok)
25 		}
26 		seen[tok] = true
27 		other := rapid.String().Draw(t, "other")
28 		if other != tok && hashToken(other) == hashToken(tok) {
