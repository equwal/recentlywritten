15 // wellFormed reads each token of doc with a strict XML parser.
16 func wellFormed(doc []byte) error {
17 	d := xml.NewDecoder(bytes.NewReader(doc))
18 	for {
19 		_, err := d.Token()
20 		if errors.Is(err, io.EOF) {
21 			return nil
22 		}
23 		if err != nil {
24 			return err
25 		}
26 	}
27 }

29 // anyText draws any string, also strings that are not valid UTF-8 and
30 // strings with characters that XML 1.0 does not permit.
31 func anyText() *rapid.Generator[string] {
32 	return rapid.OneOf(
33 		rapid.String(),
34 		rapid.Map(rapid.SliceOf(rapid.Byte()), func(b []byte) string { return string(b) }),
35 		rapid.SampledFrom([]string{"]]>", "<![CDATA[", "&amp;", "</item>", "\x00", "\r\n", "\uFFFE"}),
36 	)
37 }

39 // xmlText draws strings of characters that XML 1.0 permits.
40 func xmlText() *rapid.Generator[string] {
41 	return rapid.StringOf(rapid.Rune().Filter(func(r rune) bool {
42 		return r == '\t' || r == '\n' || r == '\r' ||
43 			r >= 0x20 && r <= 0xD7FF || r >= 0xE000 && r <= 0xFFFD || r >= 0x10000 && r <= 0x10FFFF
44 	}))
45 }

47 func drawChannel(t *rapid.T, text *rapid.Generator[string]) channel {
48 	ch := channel{Title: text.Draw(t, "title"), Link: text.Draw(t, "link"),
49 		Description: text.Draw(t, "desc"), Language: "en-us"}
50 	for range rapid.IntRange(0, 3).Draw(t, "items") {
51 		ch.Items = append(ch.Items, item{
52 			Title:       text.Draw(t, "item title"),
53 			Link:        text.Draw(t, "item link"),
54 			GUID:        guid{IsPermaLink: "true", Value: text.Draw(t, "guid")},
55 			PubDate:     text.Draw(t, "date"),
56 			Categories:  rapid.SliceOfN(text, 0, 2).Draw(t, "tags"),
57 			Description: text.Draw(t, "body"),
58 		})
59 	}
60 	return ch
61 }
