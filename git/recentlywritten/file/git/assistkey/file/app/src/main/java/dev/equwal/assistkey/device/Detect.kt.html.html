267 <a id="l258" href="#l258">258</a>         val caps = LinkedHashMap&lt;String, State&gt;()
268 <a id="l259" href="#l259">259</a>         caps[KEY_FILTER] = yesNo(Channels.isAccessibilityOn(app))
269 <a id="l260" href="#l260">260</a>         caps[ASSISTANT_ROLE] = yesNo(Channels.isRoleAvailable(app, RoleManager.ROLE_ASSISTANT))
270 <a id="l261" href="#l261">261</a>         caps[CAMERA_ROUTE] = maybe(PowerNative.cameraDoubleTapEnabled(app))

272 <a id="l263" href="#l263">263</a>         caps[SHELL_ACCESS] = yesNo(Shell.ready)
273 <a id="l264" href="#l264">264</a>         caps[EMERGENCY_SOS] = yesNo(PowerNative.canWriteSecure(app) || Shell.ready)
274 <a id="l265" href="#l265">265</a> 
275 <a id="l266" href="#l266">266</a>         val framework = frameworkKeys()
276 <a id="l267" href="#l267">267</a>         if (!Shell.ready) {
277 <a id="l268" href="#l268">268</a>             // Without the shell the light node is usually unreadable, and the
278 <a id="l269" href="#l269">269</a>             // overlays cannot be listed at all. Say so rather than guess.
279 <a id="l270" href="#l270">270</a>             caps[FRONTLIGHT] = if (anyNode()) State.YES else State.UNKNOWN
280 <a id="l271" href="#l271">271</a>             caps[NAV_OVERLAYS] = State.UNKNOWN
281 <a id="l272" href="#l272">272</a>             return done(result(framework, caps))
282 <a id="l273" href="#l273">273</a>         }
283 <a id="l274" href="#l274">274</a>         Shell.run(probeScript()) { r -&gt;
284 <a id="l275" href="#l275">275</a>             val keys = if (r.ok) keysFromGetevent(r.output.substringBefore(MARK)) else emptyList()
285 <a id="l276" href="#l276">276</a>             caps[FRONTLIGHT] = when {
286 <a id="l277" href="#l277">277</a>                 !r.ok -&gt; State.UNKNOWN
287 <a id="l278" href="#l278">278</a>                 section(r.output, &quot;light&quot;).contains(&quot;present&quot;) -&gt; State.YES
288 <a id="l279" href="#l279">279</a>                 else -&gt; State.NO
289 <a id="l280" href="#l280">280</a>             }
290 <a id="l281" href="#l281">281</a>             val navbars = section(r.output, &quot;nav&quot;).trim().toIntOrNull()
291 <a id="l282" href="#l282">282</a>             caps[NAV_OVERLAYS] = when {
292 <a id="l283" href="#l283">283</a>                 !r.ok || navbars == null -&gt; State.UNKNOWN
293 <a id="l284" href="#l284">284</a>                 navbars &gt; 0 -&gt; State.YES
294 <a id="l285" href="#l285">285</a>                 else -&gt; State.NO
295 <a id="l286" href="#l286">286</a>             }
296 <a id="l287" href="#l287">287</a>             done(result(ordered(framework + keys), caps))
297 <a id="l288" href="#l288">288</a>         }

313 <a id="l304" href="#l304">304</a>     /**
314 <a id="l305" href="#l305">305</a>      * What the framework says this device has. [KeyCharacterMap.deviceHasKey]
315 <a id="l306" href="#l306">306</a>      * asks every input device; the second pass asks each device by itself and
316 <a id="l307" href="#l307">307</a>      * leaves out full keyboards, because a typing keyboard is not a button on
317 <a id="l308" href="#l308">308</a>      * this device.
318 <a id="l309" href="#l309">309</a>      */
319 <a id="l310" href="#l310">310</a>     private fun frameworkKeys(): List&lt;HwKey&gt; {
320 <a id="l311" href="#l311">311</a>         val found = LinkedHashSet&lt;HwKey&gt;()
321 <a id="l312" href="#l312">312</a>         HwKey.entries.forEach { key -&gt;
322 <a id="l313" href="#l313">313</a>             if (runCatching { KeyCharacterMap.deviceHasKey(key.code) }.getOrDefault(false)) {
323 <a id="l314" href="#l314">314</a>                 found += key
324 <a id="l315" href="#l315">315</a>             }
325 <a id="l316" href="#l316">316</a>         }
326 <a id="l317" href="#l317">317</a>         val codes = HwKey.entries.map { it.code }.toIntArray()
327 <a id="l318" href="#l318">318</a>         runCatching { InputDevice.getDeviceIds() }.getOrDefault(IntArray(0)).forEach { id -&gt;
328 <a id="l319" href="#l319">319</a>             val device = runCatching { InputDevice.getDevice(id) }.getOrNull() ?: return@forEach
329 <a id="l320" href="#l320">320</a>             if (device.keyboardType == InputDevice.KEYBOARD_TYPE_ALPHABETIC) return@forEach
330 <a id="l321" href="#l321">321</a>             val has = runCatching { device.hasKeys(*codes) }.getOrNull() ?: return@forEach
331 <a id="l322" href="#l322">322</a>             HwKey.entries.forEachIndexed { i, key -&gt; if (has.getOrNull(i) == true) found += key }
332 <a id="l323" href="#l323">323</a>         }
333 <a id="l324" href="#l324">324</a>         return ordered(found)
334 <a id="l325" href="#l325">325</a>     }
