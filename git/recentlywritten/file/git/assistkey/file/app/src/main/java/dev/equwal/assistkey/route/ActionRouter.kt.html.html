157 <a id="l148" href="#l148">148</a>         val w = dm.widthPixels.toFloat()
158 <a id="l149" href="#l149">149</a>         val h = dm.heightPixels.toFloat()
159 <a id="l150" href="#l150">150</a>         val cx = w / 2f
160 <a id="l151" href="#l151">151</a>         val cy = h / 2f
161 <a id="l152" href="#l152">152</a>         val dx = w * 0.35f
162 <a id="l153" href="#l153">153</a>         val dy = h * 0.35f
163 <a id="l154" href="#l154">154</a> 
164 <a id="l155" href="#l155">155</a>         var sx = cx; var sy = cy; var ex = cx; var ey = cy
165 <a id="l156" href="#l156">156</a>         when (dir) {
166 <a id="l157" href="#l157">157</a>             &quot;left&quot; -&gt; { sx = cx + dx; ex = cx - dx }
167 <a id="l158" href="#l158">158</a>             &quot;right&quot; -&gt; { sx = cx - dx; ex = cx + dx }
168 <a id="l159" href="#l159">159</a>             &quot;up&quot; -&gt; { sy = cy + dy; ey = cy - dy }
169 <a id="l160" href="#l160">160</a>             &quot;down&quot; -&gt; { sy = cy - dy; ey = cy + dy }
170 <a id="l161" href="#l161">161</a>             else -&gt; return false
171 <a id="l162" href="#l162">162</a>         }

175 <a id="l166" href="#l166">166</a>         path.lineTo(ex, ey)
