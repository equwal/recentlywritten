231 <a id="l222" href="#l222">222</a> # Use &quot;xargs&quot; to parse quoted args.
232 <a id="l223" href="#l223">223</a> #
233 <a id="l224" href="#l224">224</a> # With -n1 it outputs one arg per line, with the quotes and backslashes removed.
234 <a id="l225" href="#l225">225</a> #
235 <a id="l226" href="#l226">226</a> # In Bash we could simply go:
236 <a id="l227" href="#l227">227</a> #
237 <a id="l228" href="#l228">228</a> #   readarray ARGS &lt; &lt;( xargs -n1 &lt;&lt;&lt;&quot;$var&quot; ) &amp;&amp;
238 <a id="l229" href="#l229">229</a> #   set -- &quot;${ARGS[@]}&quot; &quot;$@&quot;
239 <a id="l230" href="#l230">230</a> #
240 <a id="l231" href="#l231">231</a> # but POSIX shell has neither arrays nor command substitution, so instead we
241 <a id="l232" href="#l232">232</a> # post-process each arg (as a line of input to sed) to backslash-escape any
242 <a id="l233" href="#l233">233</a> # character that might be a shell metacharacter, then use eval to reverse
243 <a id="l234" href="#l234">234</a> # that process (while maintaining the separation between arguments), and wrap
244 <a id="l235" href="#l235">235</a> # the whole thing up as a single &quot;set&quot; statement.
245 <a id="l236" href="#l236">236</a> #
246 <a id="l237" href="#l237">237</a> # This will of course break if any of these variables contains a newline or
247 <a id="l238" href="#l238">238</a> # an unmatched quote.
248 <a id="l239" href="#l239">239</a> #
249 <a id="l240" href="#l240">240</a> 
250 <a id="l241" href="#l241">241</a> eval &quot;set -- $(
