192 <span class="a">+/**</span>
193 <span class="a">+ * Turns a stream of key down/up events into tap-count / hold / chord gestures.</span>
194 <span class="a">+ *</span>
195 <span class="a">+ * Deliberately free of Android types: everything that touches the framework</span>
196 <span class="a">+ * (timers, firing an action, emulating the native key) goes through [Host], so</span>
197 <span class="a">+ * the whole state machine is unit-testable on the JVM.</span>
198 <span class="a">+ *</span>
199 <span class="a">+ * The one rule that matters for feel: a key with no bindings at all is never</span>
200 <span class="a">+ * consumed, and a key whose highest bound tap count is 1 fires on key-up</span>
201 <span class="a">+ * without waiting out the multi-tap window. We only pay the multi-tap latency</span>
202 <span class="a">+ * on keys where the user actually asked for a double/triple tap.</span>
203 <span class="a">+ */</span>
204 <span class="a">+class GestureEngine(</span>
205 <span class="a">+    private val host: Host,</span>
206 <span class="a">+    private val cfg: Config = Config()</span>
207 <span class="a">+) {</span>
208 <span class="a">+</span>

211 <span class="a">+        val multiTapMs: Long = 280,</span>
212 <span class="a">+        /** How long a key must stay down to count as a hold. */</span>
213 <span class="a">+        val holdMs: Long = 450,</span>
214 <span class="a">+        /** Two keys going down within this window form a chord. */</span>
215 <span class="a">+        val chordMs: Long = 140</span>
216 <span class="a">+    )</span>
217 <span class="a">+</span>
218 <span class="a">+    interface Host {</span>

220 <span class="a">+        fun maxTaps(keys: Set&lt;HwKey&gt;): Int</span>
221 <span class="a">+</span>
222 <span class="a">+        /** True when a hold is bound for this key set. */</span>
223 <span class="a">+        fun hasHold(keys: Set&lt;HwKey&gt;): Boolean</span>
224 <span class="a">+</span>
225 <span class="a">+        /** True when this exact gesture has a binding. */</span>
226 <span class="a">+        fun isBound(trigger: Trigger): Boolean</span>
227 <span class="a">+</span>
228 <span class="a">+        /** Any key set that starts with this key and could still become a chord. */</span>

230 <span class="a">+</span>
231 <span class="a">+        fun fire(trigger: Trigger)</span>
232 <span class="a">+</span>
233 <span class="a">+        /** Reproduce what the system would have done, since we ate the event. */</span>

235 <span class="a">+</span>

244 <span class="a">+</span>

247 <span class="a">+    private var tapCount = 0</span>
248 <span class="a">+    private var holdFired = false</span>

252 <span class="a">+    /** @return true to swallow the event. */</span>
253 <span class="a">+    fun onDown(key: HwKey, now: Long, repeatCount: Int): Boolean {</span>
254 <span class="a">+        if (repeatCount &gt; 0) return consuming   // our own timer drives holds</span>
255 <span class="a">+</span>
256 <span class="a">+        down.add(key)</span>
257 <span class="a">+</span>
258 <span class="a">+        val chordUpgrade = consuming &amp;&amp;</span>
259 <span class="a">+            key !in seqKeys &amp;&amp;</span>
260 <span class="a">+            (now - firstDownAt) &lt;= cfg.chordMs &amp;&amp;</span>
261 <span class="a">+            interesting(seqKeys + key)</span>
262 <span class="a">+</span>
263 <span class="a">+        when {</span>
264 <span class="a">+            chordUpgrade -&gt; {</span>
265 <span class="a">+                host.cancel(T_HOLD); host.cancel(T_TAP)</span>
266 <span class="a">+                seqKeys = seqKeys + key</span>
267 <span class="a">+                tapCount = 0</span>
268 <span class="a">+                holdFired = false</span>
269 <span class="a">+            }</span>
270 <span class="a">+            !consuming || key !in seqKeys -&gt; {</span>
271 <span class="a">+                host.cancel(T_HOLD); host.cancel(T_TAP)</span>
272 <span class="a">+                seqKeys = setOf(key)</span>
273 <span class="a">+                tapCount = 0</span>
274 <span class="a">+                holdFired = false</span>
275 <span class="a">+                firstDownAt = now</span>
276 <span class="a">+            }</span>
277 <span class="a">+            else -&gt; {</span>
278 <span class="a">+                // Another tap of a sequence already in progress.</span>
279 <span class="a">+                host.cancel(T_TAP)</span>
280 <span class="a">+            }</span>
281 <span class="a">+        }</span>
282 <span class="a">+</span>
283 <span class="a">+        // Only grab the key if this set, or a chord it could grow into, is bound.</span>
284 <span class="a">+        if (!interesting(seqKeys) &amp;&amp; !couldBecomeChord(seqKeys, now)) {</span>
285 <span class="a">+            reset()</span>
286 <span class="a">+            return false</span>
287 <span class="a">+        }</span>
288 <span class="a">+</span>
289 <span class="a">+        consuming = true</span>
290 <span class="a">+        if (host.hasHold(seqKeys)) {</span>
291 <span class="a">+            host.schedule(T_HOLD, cfg.holdMs) { onHoldElapsed() }</span>
292 <span class="a">+        }</span>
293 <span class="a">+        return true</span>
294 <span class="a">+    }</span>

299 <span class="a">+        if (!consuming) return false</span>

301 <span class="a">+        host.cancel(T_HOLD)</span>
302 <span class="a">+</span>
303 <span class="a">+        if (holdFired) {</span>
304 <span class="a">+            if (down.isEmpty()) reset()</span>

306 <span class="a">+        }</span>
307 <span class="a">+</span>
308 <span class="a">+        // For a chord, wait until every key of the chord is released.</span>
309 <span class="a">+        if (down.any { it in seqKeys }) return true</span>
310 <span class="a">+</span>
311 <span class="a">+        tapCount++</span>

313 <span class="a">+</span>
314 <span class="a">+        if (max == 0) {</span>

317 <span class="a">+            reset()</span>

319 <span class="a">+            return true</span>
320 <span class="a">+        }</span>
321 <span class="a">+</span>
322 <span class="a">+        if (tapCount &gt;= max) {</span>
323 <span class="a">+            commit()</span>
324 <span class="a">+        } else {</span>
325 <span class="a">+            host.schedule(T_TAP, cfg.multiTapMs) { commit() }</span>
326 <span class="a">+        }</span>
327 <span class="a">+        return true</span>
328 <span class="a">+    }</span>
329 <span class="a">+</span>
330 <span class="a">+    fun onCancel() {</span>
331 <span class="a">+        host.cancel(T_HOLD); host.cancel(T_TAP)</span>
332 <span class="a">+        down.clear()</span>
333 <span class="a">+        reset()</span>

336 <span class="a">+    private fun onHoldElapsed() {</span>
337 <span class="a">+        val t = Trigger(seqKeys, GestureType.HOLD)</span>

339 <span class="a">+        tapCount = 0</span>
340 <span class="a">+        if (host.isBound(t)) host.fire(t)</span>

342 <span class="a">+</span>

344 <span class="a">+        host.cancel(T_TAP)</span>
345 <span class="a">+        val keys = seqKeys</span>
346 <span class="a">+        val n = tapCount.coerceIn(1, Trigger.MAX_TAPS)</span>
347 <span class="a">+        reset()</span>
348 <span class="a">+        if (keys.isEmpty()) return</span>
349 <span class="a">+</span>
350 <span class="a">+        val t = Trigger(keys, GestureType.TAP, n)</span>
351 <span class="a">+        if (host.isBound(t)) {</span>
352 <span class="a">+            host.fire(t)</span>
353 <span class="a">+        } else if (n == 1) {</span>
354 <span class="a">+            host.passThrough(keys, longPress = false)</span>
355 <span class="a">+        }</span>
356 <span class="a">+        // A 2+ tap with no binding is intentionally dropped rather than</span>
357 <span class="a">+        // replayed - replaying would fire the native action N times.</span>
358 <span class="a">+    }</span>
359 <span class="a">+</span>

361 <span class="a">+        host.maxTaps(keys) &gt; 0 || host.hasHold(keys)</span>
362 <span class="a">+</span>
363 <span class="a">+    /**</span>
