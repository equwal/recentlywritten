348 <a id="l339" href="#l339">339</a> def levenshtein(a: str, b: str) -&gt; int:
349 <a id="l340" href="#l340">340</a>     prev = list(range(len(b) + 1))
350 <a id="l341" href="#l341">341</a>     for i in range(1, len(a) + 1):
351 <a id="l342" href="#l342">342</a>         cur = [i] + [0] * len(b)
352 <a id="l343" href="#l343">343</a>         for j in range(1, len(b) + 1):
353 <a id="l344" href="#l344">344</a>             cost = 0 if a[i - 1] == b[j - 1] else 1
354 <a id="l345" href="#l345">345</a>             cur[j] = min(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + cost)
355 <a id="l346" href="#l346">346</a>         prev = cur
356 <a id="l347" href="#l347">347</a>     return prev[len(b)]

359 <a id="l350" href="#l350">350</a> def near_duplicates(name: str, ids: list) -&gt; list:
360 <a id="l351" href="#l351">351</a>     &quot;&quot;&quot;Existing ids that name looks like.&quot;&quot;&quot;
361 <a id="l352" href="#l352">352</a>     n = normalize_id(name)
362 <a id="l353" href="#l353">353</a>     out = []
363 <a id="l354" href="#l354">354</a>     for pid in ids:
364 <a id="l355" href="#l355">355</a>         if pid == name:
365 <a id="l356" href="#l356">356</a>             continue
366 <a id="l357" href="#l357">357</a>         m = normalize_id(pid)
367 <a id="l358" href="#l358">358</a>         if n == m or (len(n) &gt;= 4 and len(m) &gt;= 4 and (n in m or m in n)):
368 <a id="l359" href="#l359">359</a>             out.append(pid)
369 <a id="l360" href="#l360">360</a>             continue
370 <a id="l361" href="#l361">361</a>         limit = 1 if min(len(n), len(m)) &lt; 6 else 2
371 <a id="l362" href="#l362">362</a>         if levenshtein(n, m) &lt;= limit:
372 <a id="l363" href="#l363">363</a>             out.append(pid)
373 <a id="l364" href="#l364">364</a>     return out
