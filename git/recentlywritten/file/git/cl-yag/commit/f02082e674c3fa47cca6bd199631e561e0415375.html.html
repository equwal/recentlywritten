160 <span class="a">+  &quot;This function splits a string with separator and returns a list.&quot;</span>
161 <span class="a">+  (let ((text (concatenate &#x27;string text (string separator))))</span>
162 <span class="a">+    (loop for char across text</span>
163 <span class="a">+       counting char into count</span>
164 <span class="a">+       when (char= char separator)</span>
165 <span class="a">+       collect</span>
166 <span class="a">+       ;; we look at the position of the left separator from right to left</span>

170 <span class="a">+                   ;; the string</span>
171 <span class="a">+                   (if left-separator-position (+ 1 left-separator-position) 0)</span>
