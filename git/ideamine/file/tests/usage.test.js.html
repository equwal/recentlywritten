43 test('the price of a row follows the price list', () => {
44   // 1M input, 1M output, 1M cache read, 1M cache write with the 5-minute lifetime, on Opus 5.
45   const row = { model: 'claude-opus-5', input: 1e6, output: 1e6, read: 1e6, write5m: 1e6, write1h: 0 };
46   assert.equal(usage.cost(row), 5 + 25 + 0.5 + 5 * 1.25);
47   // The 1-hour lifetime costs 2 times the input price.
48   assert.equal(usage.cost({ model: 'claude-opus-5', write1h: 1e6 }), 10);
49   // A date at the end of the name names the same model.
