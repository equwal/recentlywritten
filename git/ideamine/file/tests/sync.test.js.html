172 test('two machines with outages end with one archive: no idea lost, none twice, no note twice', async () => {
173   await fc.assert(
174     fc.asyncProperty(fc.array(command, { maxLength: 18 }), async (commands) => {
175       resetServer();
176       const homes = { A: newHome('a'), B: newHome('b') };
177       const up = { A: true, B: true };
178       const added = [];
179       const removed = new Set();
180       let n = 0;
181       const use = (who) => machine(homes[who], up[who] ? server.url : deadUrl);
182       for (const cmd of commands) {
183         if (cmd.c === 'down' || cmd.c === 'up') {
184           up[cmd.who] = cmd.c === 'up';
185           continue;
186         }
187         use(cmd.who);
188         if (cmd.c === 'pull') {
189           await sync.pull().catch((e) => assert.ok(e instanceof sync.SyncError, e));
190           continue;
191         }
192         if (cmd.c === 'add') {
193           const text = `idea ${++n}`;
194           added.push(text);
195           await archive.add([text]);
196           continue;
197         }
198         const ideas = store.load().ideas;
199         if (!ideas.length) continue;
200         const idea = ideas[cmd.pick % ideas.length];
201         if (cmd.c === 'rm') removed.add(idea.text);
202         const run = cmd.c === 'rm' ? archive.remove([idea.id]) : archive.update(idea.id, cmd.c === 'done' ? { status: 'done' } : { note: `note ${++n}` });
203         // The other machine may have deleted the idea already: then the server refuses the change.
204         await run.catch((e) => assert.match(e.message, /^no idea #\d+$/));
205       }
206       for (const who of ['A', 'B', 'A']) {
207         machine(homes[who]);
208         await sync.pull();
209       }
210       const onServer = await serverDb();
211       const texts = onServer.ideas.map((i) => i.text).sort();
212       assert.deepEqual(texts, added.filter((t) => !removed.has(t)).sort());
213       for (const idea of onServer.ideas) {
214         const notes = idea.notes.map((x) => x.text);
215         assert.equal(new Set(notes).size, notes.length, `a note came twice on #${idea.id}`);
216       }
217       for (const who of ['A', 'B']) {
218         machine(homes[who]);
219         assert.deepEqual(store.load().ideas, onServer.ideas);
220         assert.equal(sync.waiting(), 0);
221       }
222     }),
223     { numRuns: 40 },
224   );
225 });
