165 const iso = (ms) => new Date(ms).toISOString();
166 const NOW = 4_000_000;
167 
168 /** An idea with ordered times: created ≤ triaged ≤ started ≤ closed, each stage optional. */
169 const ideaArb = fc
170   .record({
171     id: fc.integer({ min: 1, max: 1000 }),
172     created: fc.integer({ min: 0, max: 1_000_000 }),
173     gaps: fc.tuple(fc.integer({ min: 0, max: 500_000 }), fc.integer({ min: 0, max: 500_000 }), fc.integer({ min: 0, max: 500_000 })),
174     stage: fc.constantFrom('inbox', 'triaged', 'doing', 'done', 'dropped'),
175     triaged: fc.boolean(),
176     hasStart: fc.boolean(),
177     verdict: fc.constantFrom('do', 'maybe', 'skip'),
178   })
179   .map(({ id, created, gaps, stage, triaged, hasStart, verdict }) => {
180     const idea = { id, created: iso(created), status: stage === 'triaged' ? 'triaged' : stage, updated: iso(created) };
181     let t = created;
182     if (stage === 'triaged' || (triaged && stage !== 'inbox')) idea.triage = { at: iso((t += gaps[0])), verdict };
183     if (hasStart && ['doing', 'done', 'dropped'].includes(stage)) idea.started = iso((t += gaps[1]));
184     if (stage === 'done' || stage === 'dropped') idea.closed = iso((t += gaps[2]));
185     idea.updated = iso(t);
186     return idea;
187   });

211       const { times, series } = publish.flow(ideas, { samples, now: NOW });
212       assert.deepEqual(Object.keys(series).sort(), [...store.LANES].sort());
213       for (const lane of store.LANES) assert.equal(series[lane].length, times.length);
214       if (!ideas.length) return assert.equal(times.length, 0);
215       assert.ok(times.length >= 2 && times.length <= samples);

218         const t = Date.parse(times[k]);

220         const total = store.LANES.reduce((sum, lane) => sum + series[lane][k], 0);
221         assert.equal(total, ideas.filter((i) => Date.parse(i.created) <= t).length);
222       }
