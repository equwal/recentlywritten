17 /** A coop hub with the two operator routes that ideamine uses. `answers` are the replies to the next sends. */
18 function startHub() {
19   const h = { sent: [], answers: [], messages: [], auth: [] };
20   const server = http.createServer((req, res) => {
21     const chunks = [];
22     req.on('data', (c) => chunks.push(c));
23     req.on('end', () => {
24       h.auth.push(req.headers.authorization);
25       const reply = (status, json) => res.writeHead(status, { 'content-type': 'application/json' }).end(JSON.stringify(json));
26       const u = new URL(req.url, 'http://hub');
27       if (req.headers.authorization !== `Bearer ${TOKEN}`) return reply(401, { error: 'unauthorized' });
28       if (req.method === 'POST' && u.pathname === '/v1/admin/sessions/workers/messages') {
29         h.sent.push(JSON.parse(Buffer.concat(chunks).toString()));
30         const [status, json] = h.answers.shift() || [404, { error: 'not_found', message: 'no peer in this session can take work now' }];
31         return reply(status, json);
32       }
33       if (req.method === 'GET' && u.pathname === '/v1/admin/sessions/workers/messages') {
34         const after = Number(u.searchParams.get('after'));
35         return reply(200, { messages: h.messages.filter((m) => Number(m.id) > after).slice(0, Number(u.searchParams.get('limit'))) });
36       }
37       reply(404, { error: 'not_found' });
38     });
39   });
40   return new Promise((resolve) =>
41     server.listen(0, '127.0.0.1', () => resolve({ ...h, h, url: `http://127.0.0.1:${server.address().port}`, close: () => new Promise((r) => server.close(r)) })),
42   );
43 }

63 /** An idea that the user dropped in Doing on a board with coop, through the page's own dropCommand. */
64 function dropInDoing(text = 'build the thing') {
65   const [{ idea: added }] = store.addIdeas(text);
66   store.applyTriage([{ id: added.id, verdict: 'do', brief: 'the plan', model: 'sonnet' }]);
67   const page = fs.readFileSync(new URL('../dashboard/index.html', import.meta.url), 'utf8');
68   const dropCommand = new Function('state', `${/function dropCommand[\s\S]*?\n}\n/.exec(page)[0]}; return dropCommand;`)({ data: { live: { coop: true } } });
69   const command = dropCommand({ ...added, lane: 'do' }, 'doing');
70   assert.deepEqual(command, { action: 'update', body: { id: added.id, status: 'doing', agent: 'any' } });
71   const { id, ...patch } = command.body;
72   store.updateIdea(id, patch);
