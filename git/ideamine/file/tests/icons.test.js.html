8 /** The pixels of a PNG that icons.mjs wrote: signature, size, and RGBA rows without a filter. */
9 function decode(png) {
10   assert.deepEqual([...png.subarray(0, 8)], [0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a], 'PNG signature');
11   assert.equal(png.subarray(12, 16).toString('ascii'), 'IHDR');
12   const width = png.readUInt32BE(16);
13   const height = png.readUInt32BE(20);
14   assert.equal(png.subarray(37, 41).toString('ascii'), 'IDAT');
15   const rows = inflateSync(png.subarray(41, 41 + png.readUInt32BE(33)));
16   const at = (x, y) => {
17     const start = y * (width * 4 + 1);
18     assert.equal(rows[start], 0, 'each row says "no filter"');
19     return [...rows.subarray(start + 1 + x * 4, start + 5 + x * 4)];
20   };
21   return { width, height, at };
22 }
