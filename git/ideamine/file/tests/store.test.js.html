75       const script = `for (let i = 0; i < ${each}; i++) require('node:child_process').execFileSync(process.execPath, [${JSON.stringify(BIN)}, 'add', 'proc ${n} idea ' + i]);`;
76       const child = spawn(process.execPath, ['-e', script], { env: process.env, stdio: ['ignore', 'ignore', 'pipe'] });
77       let err = '';
78       child.stderr.on('data', (d) => (err += d));

80     });
81   await Promise.all(Array.from({ length: procs }, (_, n) => run(n)));
