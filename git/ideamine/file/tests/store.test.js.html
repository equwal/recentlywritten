32 test('pickNext takes the best "do" idea, else the best "maybe", and never an idea from another lane', () => {
33   // Every combination of lanes, so that no lane can get into or out of the queue unseen.
34   const LANES = ['do', 'maybe', 'skip', 'inbox', 'doing', 'done', 'dropped'];
35   const VERDICTS = ['do', 'maybe', 'skip'];
36   const make = (lane, id) => ({
37     id,
38     title: lane,
39     status: VERDICTS.includes(lane) ? 'triaged' : lane,
40     triage: lane === 'inbox' ? null : { verdict: VERDICTS.includes(lane) ? lane : 'do', impact: 3, size: 'm', model: 'sonnet' },
41   });
42   for (let mask = 0; mask < 1 << LANES.length; mask++) {
43     const present = LANES.filter((_, bit) => mask & (1 << bit));
44     const db = { ideas: present.map((lane, i) => make(lane, i + 1)) };
45     const want = present.includes('do') ? 'do' : present.includes('maybe') ? 'maybe' : null;
46     assert.equal(store.pickNext(db)?.title ?? null, want, `lanes: ${present.join(', ')}`);
47   }
48 
49   // The verdict counts before the project: a "do" idea elsewhere beats a "maybe" idea here.
50   const db = {
51     ideas: [
52       { id: 1, status: 'triaged', project: '/here', triage: { verdict: 'maybe', impact: 5, size: 'xs', model: 'haiku' } },
53       { id: 2, status: 'triaged', project: '/there', triage: { verdict: 'do', impact: 1, size: 'xl', model: 'opus' } },
54     ],
55   };

126 test('concurrent writers from many processes never lose an idea', async () => {
127   const procs = 8;
128   const each = 6;
129   const run = (n) =>
130     new Promise((resolve, reject) => {
131       // Each process appends its ideas through the CLI, all racing for the same archive.
132       const script = `for (let i = 0; i < ${each}; i++) require('node:child_process').execFileSync(process.execPath, [${JSON.stringify(BIN)}, 'add', 'proc ${n} idea ' + i]);`;
133       const child = spawn(process.execPath, ['-e', script], { env: process.env, stdio: ['ignore', 'ignore', 'pipe'] });
134       let err = '';
135       child.stderr.on('data', (d) => (err += d));
136       child.on('exit', (code) => (code === 0 ? resolve() : reject(new Error(err))));
137     });
138   await Promise.all(Array.from({ length: procs }, (_, n) => run(n)));
139   const ids = store.load().ideas.map((i) => i.id).sort((a, b) => a - b);
140   assert.equal(ids.length, procs * each);
141   assert.deepEqual(ids, Array.from({ length: procs * each }, (_, i) => i + 1));
