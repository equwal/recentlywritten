59 /** A raw request, so a test can send any Host header. */
60 function raw(method, pathname, headers) {
61   return new Promise((resolve, reject) => {
62     const req = http.request(new URL(pathname, url), { method, headers }, (res) => {
63       res.resume();
64       res.on('end', () => resolve(res));
65     });
66     req.on('error', reject);
67     req.end();
68   });

409 /** Put idea `id` in `lane` the way the CLI and Claude do it. `triaged`: it got a verdict first. */
410 function seed(id, lane, triaged) {
411   if (['do', 'maybe', 'skip'].includes(lane)) return store.applyTriage([{ id, verdict: lane }]);
412   if (triaged) store.applyTriage([{ id, verdict: 'do' }]);
413   store.updateIdea(id, { status: lane });
414 }

416 /** A POST from the page at https://mem.equwal.com, the way nginx passes it on. */
417 function httpsPagePost(action, body) {
418   return new Promise((resolve, reject) => {
419     const host = 'mem.equwal.com';
420     const req = http.request(new URL(`api/${action}`, url), { method: 'POST', headers: { host, origin: `https://${host}`, 'content-type': 'application/json' } }, (res) => {
421       const chunks = [];
422       res.on('data', (c) => chunks.push(c));
423       res.on('end', () => resolve({ status: res.statusCode, ...JSON.parse(Buffer.concat(chunks).toString()) }));
424     });
425     req.on('error', reject);
426     req.end(JSON.stringify(body));
427   });
428 }

430 /** Drop idea `id` in `lane` on the board. Resolves to the lane that the idea is in afterwards. */
431 async function drop(id, lane, post = api) {
432   const idea = { ...store.findIdea(store.load(), id) };
433   idea.lane = store.lane(idea);
434   const command = dropCommand(idea, lane);
435   const out = await post(command.action, command.body);
436   assert.equal(out.ok, true, out.error);
437   return store.lane(store.findIdea(store.load(), id));
438 }
