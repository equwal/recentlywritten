17 /** Run the real hook process the way Claude Code does: JSON on stdin, decision on stdout. */
18 function runHook(prompt, cwd = '/work/app') {
19   const input = JSON.stringify({ session_id: 's1', hook_event_name: 'UserPromptSubmit', cwd, prompt });
20   const r = spawnSync(process.execPath, [BIN, 'hook'], { input, encoding: 'utf8', env: process.env });
21   assert.equal(r.status, 0, r.stderr);
22   return r.stdout ? JSON.parse(r.stdout) : null;
23 }
