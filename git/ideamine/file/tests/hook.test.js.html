23 /** Run the real hook process the way Claude Code does: JSON on stdin, decision on stdout. */
24 function runHook(prompt, cwd = '/work/app') {
25   const input = JSON.stringify({ session_id: 's1', hook_event_name: 'UserPromptSubmit', cwd, prompt });
26   const r = spawnSync(process.execPath, [BIN, 'hook'], { input, encoding: 'utf8', env: process.env });
27   assert.equal(r.status, 0, r.stderr);
28   return r.stdout ? JSON.parse(r.stdout) : null;
29 }
