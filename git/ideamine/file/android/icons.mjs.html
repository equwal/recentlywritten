4 import fs from 'node:fs';

9 const RES = path.join(path.dirname(fileURLToPath(import.meta.url)), 'app', 'src', 'main', 'res');

11 // The launcher icon of older Android, and the front layer of the adaptive icon. The front layer is
12 // 108dp wide, and a phone can cut any shape from the middle 72dp, so the lamp stays small there.
13 const DENSITIES = [['mdpi', 1], ['hdpi', 1.5], ['xhdpi', 2], ['xxhdpi', 3], ['xxxhdpi', 4]];
14 
15 for (const [name, scale] of DENSITIES) {
16   const dir = path.join(RES, `mipmap-${name}`);
17   fs.mkdirSync(dir, { recursive: true });
18   fs.writeFileSync(path.join(dir, 'ic_launcher.png'), icon(Math.round(48 * scale)));
19   fs.writeFileSync(path.join(dir, 'ic_foreground.png'), icon(Math.round(108 * scale), { maskable: true }));
20   console.log(`mipmap-${name}: ${Math.round(48 * scale)} and ${Math.round(108 * scale)}`);
21 }
