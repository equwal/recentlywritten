87     /**
88      * The address of the dashboard for this intent: a shared text goes to the composer of the page,
89      * and the "New idea" shortcut opens the composer empty. Null means: only show the board.
90      */
91     private fun urlFor(intent: Intent?): String? {
92         val address = Uri.parse(server).buildUpon()
93         if (intent?.action == NEW_IDEA) return address.appendQueryParameter("new", "1").build().toString()
94         if (intent?.action != Intent.ACTION_SEND) return null
95         val parts = listOfNotNull(
96             intent.getStringExtra(Intent.EXTRA_SUBJECT),
97             intent.getStringExtra(Intent.EXTRA_TEXT),
98         )
99         if (parts.isEmpty()) return null
100         return address.appendQueryParameter("text", parts.joinToString("\n")).build().toString()
101     }

135         /** An address with a scheme and a slash at the end, so that a shared text can go after it. */
136         fun clean(raw: String): String {
137             val trimmed = raw.trim().ifEmpty { DEFAULT_SERVER }
138             val withScheme = if (trimmed.startsWith("http://") || trimmed.startsWith("https://")) trimmed else "http://$trimmed"
139             return if (withScheme.endsWith("/")) withScheme else "$withScheme/"
