ideamine

37 function parseArgs(argv) {
38   const flags = {};
39   const words = [];
40   for (let i = 0; i < argv.length; i++) {
41     const a = argv[i];
42     if (a.startsWith('--')) {
43       const [key, inline] = a.slice(2).split('=', 2);
44       if (inline !== undefined) flags[key] = inline;
45       else if (argv[i + 1] !== undefined && !argv[i + 1].startsWith('--') && ['model', 'limit', 'budget', 'query', 'dir', 'port'].includes(key)) flags[key] = argv[++i];
46       else flags[key] = true;
47     } else words.push(a);
48   }
49   return { flags, words };
50 }

58 function fail(message) {
59   process.stderr.write(`ideamine: ${message}\n`);
60   // A window that the dashboard opened closes when the command ends. Wait for a key, so that the
61   // user can read the error.
62   if (process.env.IDEAMINE_WINDOW && process.stdin.isTTY) {
63     spawnSync(process.env.ComSpec || 'cmd.exe', ['/d', '/c', 'pause'], { stdio: 'inherit' });
64   }
65   process.exit(1);
66 }

104       break;
105     }

107       needId();

109       if (!queued(gone)) for (const idea of gone) console.log(`Removed #${idea.id} ${clip(idea.title, 60)}`);
110       break;
111     }

113     case 'drop':
114     case 'start':
115     case 'doing':
116     case 'reopen': {
117       const status = { drop: 'dropped', start: 'doing' }[command] || command;
118       const note = words.slice(1).join(' ');
119       const idea = await archive.update(needId(), { status, note: note || undefined });
120       if (!queued(idea)) console.log(`#${idea.id} ${clip(idea.title, 60)} → ${store.lane(idea)}`);
121       break;
122     }
123     case 'note': {
124       const note = words.slice(1).join(' ') || fail('usage: ideamine note <id> <text>');
125       const idea = await archive.update(needId(), { note });
126       if (!queued(idea)) console.log(`#${idea.id}: note added`);

128     }
129     case 'model': {
130       const idea = await archive.update(needId(), { model: words[1] || fail('usage: ideamine model <id> <haiku|sonnet|opus|fable>') });
131       if (!queued(idea)) console.log(`#${idea.id} will be built with ${idea.triage.model}`);
132       break;
133     }

135       const idea = store.pickNext(await archive.fresh(), { project: cwd, only: !!flags.here });
136       if (!idea) {
137         console.log('Nothing is ready to build. Triage the inbox first: ideamine sort');

139       }
140       console.log(render.renderIdea(idea));
141       break;
142     }
143     case 'go': {
144       const { goPlan, launchSession } = await import('../src/claude.js');
145       const db = await archive.fresh();
146       const idea = words[0] ? store.findIdea(db, words[0]) : store.pickNext(db, { project: cwd });
147       if (!idea) fail(words[0] ? `no idea #${words[0]}` : 'nothing is ready to build; run: ideamine sort');
148       const { model, dir, prompt } = goPlan(idea, cwd);
149       if (flags.print) {
150         console.log(`directory: ${dir}\nmodel:     ${model}\n\n${prompt}`);
151         break;
152       }
153       queued(await archive.update(idea.id, { status: 'doing' }));
154       console.log(`Opening Claude Code (${model}) in ${dir} for #${idea.id}…`);
155       process.exitCode = launchSession({ model, prompt, cwd: dir });

159       const { headlessTriage } = await import('../src/claude.js');
160       const out = await headlessTriage({
161         model: typeof flags.model === 'string' ? flags.model : undefined,
162         limit: Number(flags.limit) || 20,
163         budget: Number(flags.budget) || 1,
164         dryRun: !!flags['dry-run'],
165       });
166       if (out.message) {
167         console.log(out.message);
168         break;

170       const ok = out.results.filter((r) => !r.error);
171       const spend = out.tokens ? ` (${out.tokens.input} in / ${out.tokens.output} out tokens, ~$${Number(out.cost || 0).toFixed(4)})` : '';
172       console.log(`Triaged ${ok.length}${spend}:`);
173       for (const r of ok) console.log(`  #${r.id} ${r.verdict.padEnd(5)} ${r.verdict === 'skip' ? '' : `${r.model.padEnd(6)} ${r.size.toUpperCase().padEnd(2)} `}${clip(r.title, 60)}`);
174       for (const r of out.results.filter((x) => x.error)) console.log(`  #${r.id} not saved: ${r.error}`);
