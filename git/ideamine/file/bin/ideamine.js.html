ideamine

An idea inbox for Claude Code: /idea saves ideas at zero tokens; Claude triages them and routes each to the cheapest model that can build it.

23 function parseArgs(argv) {
24   const flags = {};
25   const words = [];
26   for (let i = 0; i < argv.length; i++) {
27     const a = argv[i];
28     if (a.startsWith('--')) {
29       const [key, inline] = a.slice(2).split('=', 2);
30       if (inline !== undefined) flags[key] = inline;
31       else if (argv[i + 1] !== undefined && !argv[i + 1].startsWith('--') && ['model', 'limit', 'budget', 'query'].includes(key)) flags[key] = argv[++i];
32       else flags[key] = true;
33     } else words.push(a);
34   }
35   return { flags, words };
36 }

49 async function main() {
50   const [command = 'help', ...rest] = process.argv.slice(2);
51 
52   // The two machine entry points load only what they need.
53   if (command === 'mcp') return (await import('../src/mcp.js')).serve({ prompts: !rest.includes('--plugin') });
54   if (command === 'hook') return (await import('../src/hook.js')).runHook();

56   const store = await import('../src/store.js');
57   const render = await import('../src/render.js');
58   const { splitIdeas, clip } = await import('../src/text.js');
59   const { flags, words } = parseArgs(rest);
60   const cwd = process.cwd();
61   const needId = () => words[0] || fail(`usage: ideamine ${command} <id>`);
62 
63   switch (command) {
64     case 'add': {
65       const text = words.length === 1 && words[0] === '-' ? await readStdin() : words.join(' ');
66       const results = store.addIdeas(splitIdeas(text), { source: 'cli', project: flags['no-project'] ? null : cwd });
67       console.log(render.renderAdded(results, store.load()));
68       break;
69     }
70     case 'ls':
71     case 'list':
72     case 'board': {
73       const lower = words.map((w) => w.toLowerCase());
74       const filter = lower.find((w) => store.FILTERS.includes(w)) || 'open';
75       const here = lower.includes('here') || flags.here;
76       console.log(render.renderBoard(store.load(), { filter, project: here ? cwd : null, query: flags.query || '', cwd }));
77       break;
78     }
79     case 'show': {
80       const idea = store.findIdea(store.load(), needId());
81       if (!idea) fail(`no idea #${words[0]}`);
82       console.log(render.renderIdea(idea));
83       break;
84     }
85     case 'done':
86     case 'drop':
87     case 'start':
88     case 'doing':
89     case 'reopen': {
90       const status = { drop: 'dropped', start: 'doing' }[command] || command;
91       const note = words.slice(1).join(' ');
92       const idea = store.updateIdea(needId(), { status, note: note || undefined });
93       console.log(`#${idea.id} ${clip(idea.title, 60)} → ${store.lane(idea)}`);
94       break;
95     }
96     case 'note': {
97       const note = words.slice(1).join(' ') || fail('usage: ideamine note <id> <text>');
98       const idea = store.updateIdea(needId(), { note });
99       console.log(`#${idea.id}: note added`);
100       break;
101     }
102     case 'model': {
103       const idea = store.updateIdea(needId(), { model: words[1] || fail('usage: ideamine model <id> <haiku|sonnet|opus|fable>') });
104       console.log(`#${idea.id} will be built with ${idea.triage.model}`);
105       break;
106     }
107     case 'next': {
108       const idea = store.pickNext(store.load(), { project: cwd, only: !!flags.here });
109       if (!idea) {
110         console.log('Nothing is ready to build. Triage the inbox first: ideamine triage');
111         break;
112       }
113       console.log(render.renderIdea(idea));
114       break;
115     }
116     case 'go': {
117       const { buildPrompt, launchSession } = await import('../src/claude.js');
118       const db = store.load();
119       const idea = words[0] ? store.findIdea(db, words[0]) : store.pickNext(db, { project: cwd });
120       if (!idea) fail(words[0] ? `no idea #${words[0]}` : 'nothing is ready to build; run: ideamine triage');
121       const model = idea.triage?.model || 'sonnet';
122       const dir = idea.project && fs.existsSync(idea.project) ? idea.project : cwd;
123       const prompt = buildPrompt(idea);
124       if (flags.print) {
125         console.log(`directory: ${dir}\nmodel:     ${model}\n\n${prompt}`);
126         break;
127       }
128       store.updateIdea(idea.id, { status: 'doing' });
129       console.log(`Opening Claude Code (${model}) in ${dir} for #${idea.id}…`);
130       process.exitCode = launchSession({ model, prompt, cwd: dir });
131       break;
132     }
133     case 'triage': {
134       const { headlessTriage } = await import('../src/claude.js');
135       const out = await headlessTriage({
136         model: typeof flags.model === 'string' ? flags.model : undefined,
137         limit: Number(flags.limit) || 20,
138         budget: Number(flags.budget) || 1,
139         dryRun: !!flags['dry-run'],
140       });
141       if (out.message) {
142         console.log(out.message);
143         break;
144       }
145       const ok = out.results.filter((r) => !r.error);
146       const spend = out.tokens ? ` (${out.tokens.input} in / ${out.tokens.output} out tokens, ~$${Number(out.cost || 0).toFixed(4)})` : '';
147       console.log(`Triaged ${ok.length}${spend}:`);
148       for (const r of ok) console.log(`  #${r.id} ${r.verdict.padEnd(5)} ${r.verdict === 'skip' ? '' : `${r.model.padEnd(6)} ${r.size.toUpperCase().padEnd(2)} `}${clip(r.title, 60)}`);
149       for (const r of out.results.filter((x) => x.error)) console.log(`  #${r.id} not saved: ${r.error}`);
150       break;
151     }
152     case 'export': {
153       const md = render.renderMarkdown(store.load());
154       if (words[0]) {
155         fs.writeFileSync(words[0], md);
156         console.log(`Wrote ${words[0]}`);
157       } else process.stdout.write(md);
158       break;
159     }
160     case 'path':
161       console.log(store.dbPath());
162       break;
163     case 'version':
164     case '--version':
165     case '-v':
166       console.log(JSON.parse(fs.readFileSync(new URL('../package.json', import.meta.url), 'utf8')).version);
167       break;
168     case 'help':
169     case '--help':
170     case '-h':
171       console.log(HELP);
172       break;
173     default:
174       fail(`unknown command "${command}". Try: ideamine help`);
175   }
176 }
