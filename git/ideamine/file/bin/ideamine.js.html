39 function parseArgs(argv) {
40   const flags = {};
41   const words = [];
42   for (let i = 0; i < argv.length; i++) {
43     const a = argv[i];
44     if (a.startsWith('--')) {
45       const [key, inline] = a.slice(2).split('=', 2);
46       if (inline !== undefined) flags[key] = inline;
47       else if (argv[i + 1] !== undefined && !argv[i + 1].startsWith('--') && ['model', 'limit', 'budget', 'query', 'dir', 'port'].includes(key)) flags[key] = argv[++i];
48       else flags[key] = true;
49     } else words.push(a);
50   }
51   return { flags, words };
52 }

60 function fail(message) {
61   process.stderr.write(`ideamine: ${message}\n`);
62   // A window that the dashboard opened closes when the command ends. Wait for a key, so that the
63   // user can read the error.
64   if (process.env.IDEAMINE_WINDOW && process.stdin.isTTY) {
65     spawnSync(process.env.ComSpec || 'cmd.exe', ['/d', '/c', 'pause'], { stdio: 'inherit' });
66   }
67   process.exit(1);
68 }
