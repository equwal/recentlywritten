7 /** Claude Code executable: explicit override, else the one running us (desktop app), else PATH. */
8 export function claudeBin() {
9   return process.env.IDEAMINE_CLAUDE_BIN || process.env.CLAUDE_CODE_EXECPATH || 'claude';
10 }

12 function command(args) {
13   const bin = claudeBin();
14   // A .js file is run with node, which keeps a stand-in CLI portable (used by the tests).
15   return /\.[cm]?js$/i.test(bin) ? [process.execPath, [bin, ...args]] : [bin, args];
16 }

18 /**
19  * Environment for a separate, independent Claude Code process: drop the variables that tie a
20  * child to the session that started us (nesting guard, host messaging, the parent's effort).
21  */
22 function childEnv() {
23   const env = { ...process.env };
24   for (const key of Object.keys(env)) {
25     if (
26       key === 'CLAUDECODE' ||
27       key === 'CLAUDE_PID' ||
28       key === 'CLAUDE_EFFORT' ||
29       key === 'AI_AGENT' ||
30       /^CLAUDE_CODE_(ENTRYPOINT|CHILD_SESSION|HOST_SESSION_ID|SESSION_ID|SESSION_ATTENDED|MESSAGING_.*|SDK_.*|EXECPATH)$/.test(key)
31     ) {
32       delete env[key];
33     }
34   }
35   return env;
36 }

38 function parseLooseJson(text) {
39   const s = String(text || '').replace(/^\s*```(?:json)?\s*|\s*```\s*$/g, '');
40   const start = s.indexOf('{');
41   const end = s.lastIndexOf('}');
42   if (start < 0 || end < start) return null;
43   try {
44     return JSON.parse(s.slice(start, end + 1));
45   } catch {
46     return null;
47   }
48 }

50 function run(args, input, timeoutMs) {
51   const [bin, argv] = command(args);
52   return new Promise((resolve, reject) => {
53     const child = spawn(bin, argv, { env: childEnv(), windowsHide: true, stdio: ['pipe', 'pipe', 'pipe'] });
54     let stdout = '';
55     let stderr = '';
56     const timer = setTimeout(() => child.kill(), timeoutMs);
57     child.stdout.on('data', (d) => (stdout += d));
58     child.stderr.on('data', (d) => (stderr += d));
59     child.on('error', (e) => {
60       clearTimeout(timer);
61       reject(e.code === 'ENOENT' ? new Error(`Claude Code CLI not found ("${bin}"). Install it or set IDEAMINE_CLAUDE_BIN.`) : e);
62     });
63     child.on('close', (code) => {
64       clearTimeout(timer);
65       resolve({ code, stdout, stderr });
66     });
67     child.stdin.end(input);
68   });
69 }

71 /**
72  * Triage the inbox with a one-shot `claude -p` call: no tools, no MCP servers, no settings, a
73  * two-line system prompt, and JSON-schema output. Runs on the user's normal Claude Code login and
74  * costs about 400 tokens per idea, whatever model the calling session uses.
75  */
76 export async function headlessTriage({ model = process.env.IDEAMINE_TRIAGE_MODEL || 'sonnet', limit = 20, dryRun = false, budget = 1 } = {}) {
77   const alias = normalizeModel(model) || model;
78   const db = load();
79   const pending = pendingIdeas(db, { limit });
80   if (!pending.length) return { message: 'Nothing to triage: the inbox is empty.' };
81 
82   const prompt = `${triagePrompt(db, pending)}\n\nReturn one verdict for every idea listed above.`;
83   // Skipping settings drops hooks, plugins, and skill listings: ~4k fewer input tokens per call.
84   // Set IDEAMINE_SETTING_SOURCES=user if your login depends on settings.json (apiKeyHelper, env).
85   const args = [
86     '-p',
87     '--model', alias,
88     '--output-format', 'json',
89     '--json-schema', JSON.stringify(BATCH_SCHEMA),
90     '--tools', '',
91     '--strict-mcp-config',
92     '--setting-sources', process.env.IDEAMINE_SETTING_SOURCES ?? '',
93     '--no-session-persistence',
94     '--max-budget-usd', String(budget),
95     '--system-prompt', 'You triage a developer\'s backlog of ideas. Follow the rubric exactly and answer only with the requested JSON.',
96   ];
97   if (alias !== 'haiku') args.push('--effort', 'low');
98   if (dryRun) {
99     const shown = args.map((a) => (/[\s"{]/.test(a) || !a ? JSON.stringify(a) : a)).join(' ');
100     return { message: `${claudeBin()} ${shown}\n\n${prompt}` };
101   }
102 
103   const res = await run(args, prompt, 5 * 60 * 1000);
104   let out;
105   try {
106     out = JSON.parse(res.stdout);
107   } catch {
108     const detail = (res.stderr || res.stdout || '').trim().split(/\r?\n/).slice(-5).join('\n');
109     throw new Error(`claude exited with code ${res.code}: ${detail || 'no output'}`);
110   }
111   if (out.is_error) throw new Error(`claude: ${out.result || out.subtype || 'error'}`);
112   const data = out.structured_output ?? parseLooseJson(out.result);
113   if (!Array.isArray(data?.verdicts)) throw new Error('claude answered without verdicts');
114   const results = applyTriage(data.verdicts, { by: `${alias} (headless)` });
115   const u = out.usage || {};
116   const input = (u.input_tokens || 0) + (u.cache_creation_input_tokens || 0) + (u.cache_read_input_tokens || 0);
117   return { results, model: alias, cost: out.total_cost_usd, tokens: { input, output: u.output_tokens || 0 } };
118 }

120 /** The opening prompt for a fresh session that builds one idea. */
121 export function buildPrompt(idea) {
122   const t = idea.triage;
123   const lines = [`Build idea #${idea.id} from my ideamine archive: ${idea.title}`, ''];
124   if (t?.brief) lines.push(t.brief, '');
125   lines.push('My original note:', idea.text, '');
126   lines.push(
127     `When you finish, record the outcome with the ideamine idea_update tool (id ${idea.id}, status "done", a one-line note), ` +
128       `or run: ideamine done ${idea.id} "<one-line outcome>"`,
129   );
130   return lines.join('\n');
131 }

133 /** Start an interactive Claude Code session on the recommended model. */
134 export function launchSession({ model, prompt, cwd }) {
135   const [bin, argv] = command(['--model', model, prompt]);
136   const res = spawnSync(bin, argv, { cwd, stdio: 'inherit', env: childEnv() });
137   if (res.error) throw res.error;
138   return res.status ?? 0;
