38 export const RUBRIC = `Triage each idea. Be decisive: the user wants a short list worth building, not encouragement.
39 
40 For every idea return:
41 - verdict: "do" (clear value, worth building soon) | "maybe" (unclear value, blocked, or needs a decision from the user) | "skip" (low value, already exists, or costs far more than it returns)
42 - impact: 1-5, how much it would matter to the user or project if it shipped
43 - size: xs (<15 min) | s (<1 h) | m (a few hours) | l (1-2 days) | xl (multi-day, should be split)
44 - model: the CHEAPEST model likely to finish it in one pass. A failed attempt on a weaker model plus a retry costs more than the right model once.
45 ${MODELS.map((m) => `    ${m.alias.padEnd(6)} ${m.price.padEnd(7)} ${m.use}`).join('\n')}
46     (prices in USD per million input/output tokens)
47 - title: imperative, at most 60 characters ("Add ...", "Fix ...")
48 - why: one sentence with the deciding reason
49 - brief: 1-4 sentences an agent can act on without this conversation: goal, scope, and "done when". Leave empty for skip.
50 - dup_of: id of an existing idea this one duplicates (use verdict "skip"), if any`;

82 /** Ideas waiting for a verdict: the inbox, or the given ids (re-triage). */
83 export function pendingIdeas(db, { ids = null, limit = 30 } = {}) {
84   if (Array.isArray(ids) && ids.length) {
85     const wanted = new Set(ids.map(Number));
86     return db.ideas.filter((i) => wanted.has(i.id));
87   }
88   return listIdeas(db, { filter: 'inbox' }).reverse().slice(0, limit); // oldest first
89 }

91 /** The rubric plus the ideas to judge, as one prompt. `projects` (from knownProjects) turns on pairing. */
92 export function triagePrompt(db, pending, projects = []) {
93   const pendingIds = new Set(pending.map((i) => i.id));
94   const existing = db.ideas
95     .filter((i) => !pendingIds.has(i.id) && ['do', 'maybe', 'doing', 'done'].includes(lane(i)))
96     .slice(-60)
97     .map((i) => `#${i.id} ${clip(i.title, 70)} (${lane(i)})`);
98   const lines = [RUBRIC];
99   if (projects.length) lines.push(PAIRING);
100   lines.push('');
101   if (existing.length) lines.push('Existing ideas, for duplicate checks:', ...existing, '');
102   if (projects.length) {
103     lines.push('Projects on this machine:', ...projects.map((p) => `${p.name}: ${p.dir}${p.about ? ` — ${p.about}` : ''}`), '');
104   }
105   lines.push(`Ideas to triage (${pending.length}):`);
106   for (const i of pending) {
107     const ctx = [i.project && `project: ${path.basename(i.project)}`, i.tags?.length && `tags: ${i.tags.join(', ')}`]
108       .filter(Boolean)
109       .join('; ');
110     lines.push(`#${i.id}${ctx ? ` [${ctx}]` : ''}: ${i.text.replace(/\r?\n/g, ' / ')}`);
111   }
112   return lines.join('\n');
113 }
