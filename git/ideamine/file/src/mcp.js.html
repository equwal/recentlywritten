164 function skillBody(name, args) {
165   const file = new URL(`skills/${name}/SKILL.md`, ROOT);
166   const text = fs.readFileSync(fileURLToPath(file), 'utf8').replace(/^---[\s\S]*?\n---\s*\n/, '');

170 function summarizeTriage(results, how = '') {
171   const ok = results.filter((r) => !r.error);
172   const tally = ['do', 'maybe', 'skip'].map((v) => `${ok.filter((r) => r.verdict === v).length} ${v}`).join(' · ');
173   const lines = [`Saved ${ok.length} verdict${ok.length === 1 ? '' : 's'}: ${tally}${how ? ` (${how})` : ''}`];
174   for (const r of ok) {
175     const meta = r.verdict === 'skip' ? '' : ` · ${r.model} · ${r.size.toUpperCase()}`;
176     lines.push(`  #${r.id} ${r.verdict}${meta}  ${clip(r.title, 60)}${r.moved ? `  → ${path.basename(r.moved)}` : ''}`);
177   }
178   for (const r of results.filter((x) => x.error)) lines.push(`  #${r.id} not saved: ${r.error}`);
179   const left = counts(load()).inbox;
180   if (left) lines.push(`${left} still in inbox.`);
181   return lines.join('\n');
182 }

318 async function handleRequest(msg, { prompts }) {
319   const { id, method, params = {} } = msg;
320   switch (method) {
321     case 'initialize': {
322       const asked = params.protocolVersion;
323       const protocolVersion = PROTOCOL_VERSIONS.includes(asked) ? asked : PROTOCOL_VERSIONS[0];
324       const capabilities = { tools: { listChanged: false } };
325       if (prompts) capabilities.prompts = { listChanged: false };
326       return result(id, { protocolVersion, capabilities, serverInfo: { name: 'ideamine', version: VERSION }, instructions: INSTRUCTIONS });
327     }
328     case 'ping':
329       return result(id, {});
330     case 'tools/list':
331       return result(id, { tools: TOOLS });
332     case 'tools/call': {
333       const handler = handlers[params.name];
334       if (!handler) return error(id, -32602, `unknown tool: ${params.name}`);
335       try {
336         const text = await handler(params.arguments || {});
337         return result(id, { content: [{ type: 'text', text }] });
338       } catch (e) {
339         return result(id, { content: [{ type: 'text', text: `Error: ${e.message}` }], isError: true });
340       }
341     }
342     case 'prompts/list':
343       return result(id, { prompts: prompts ? PROMPTS : [] });
344     case 'prompts/get': {
345       const p = prompts && PROMPTS.find((x) => x.name === params.name);
346       if (!p) return error(id, -32602, `unknown prompt: ${params.name}`);
347       const args = Object.values(params.arguments || {}).filter(Boolean).join(' ');
348       return result(id, {
349         description: p.description,
350         messages: [{ role: 'user', content: { type: 'text', text: skillBody(p.name, args) } }],
351       });
352     }
353     case 'resources/list':
354       return result(id, { resources: [] });
355     case 'resources/templates/list':
356       return result(id, { resourceTemplates: [] });
357     case 'logging/setLevel':
358       return result(id, {});
359     default:
360       return error(id, -32601, `method not found: ${method}`);
361   }
362 }

364 async function handleMessage(msg, opts) {
365   if (!msg || typeof msg !== 'object' || msg.jsonrpc !== '2.0') return error(null, -32600, 'invalid request');
366   if (typeof msg.method !== 'string') return null; // a response to a request we never send
367   if (msg.id === undefined) return null; // notification (initialized, cancelled, ...)
368   try {
369     return await handleRequest(msg, opts);
370   } catch (e) {
371     return error(msg.id, -32603, e.message);
372   }
373 }

375 async function handleLine(line, opts) {
376   let msg;
377   try {
378     msg = JSON.parse(line);
379   } catch {
380     return error(null, -32700, 'parse error');
381   }
382   if (!Array.isArray(msg)) return handleMessage(msg, opts);
383   const replies = (await Promise.all(msg.map((m) => handleMessage(m, opts)))).filter(Boolean);
384   return replies.length ? replies : null;
385 }

387 /** Serve MCP on stdin/stdout. `prompts: false` when running as a plugin (skills cover slash commands). */
388 export function serve({ prompts = true } = {}) {
389   const rl = readline.createInterface({ input: process.stdin, crlfDelay: Infinity });
390   const inFlight = new Set();
391   // Requests run concurrently (a headless triage must not stall a ping); replies carry their ids.
392   rl.on('line', (line) => {
393     if (!line.trim()) return;
394     const job = handleLine(line, { prompts }).then((reply) => reply && send(reply));
395     inFlight.add(job);
396     job.finally(() => inFlight.delete(job));
397   });
398   rl.on('close', () => Promise.allSettled([...inFlight]).then(() => process.exit(0)));
