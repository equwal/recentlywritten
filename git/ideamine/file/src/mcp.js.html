126 function skillBody(name, args) {
127   const file = new URL(`skills/${name}/SKILL.md`, ROOT);
128   const text = fs.readFileSync(fileURLToPath(file), 'utf8').replace(/^---[\s\S]*?\n---\s*\n/, '');
129   return text.replaceAll('$ARGUMENTS', args || '').trim();

132 function summarizeTriage(results, how = '') {
133   const ok = results.filter((r) => !r.error);
134   const tally = ['do', 'maybe', 'skip'].map((v) => `${ok.filter((r) => r.verdict === v).length} ${v}`).join(' · ');
135   const lines = [`Saved ${ok.length} verdict${ok.length === 1 ? '' : 's'}: ${tally}${how ? ` (${how})` : ''}`];
136   for (const r of ok) lines.push(`  #${r.id} ${r.verdict}${r.verdict === 'skip' ? '' : ` · ${r.model} · ${r.size.toUpperCase()}`}  ${clip(r.title, 60)}`);
137   for (const r of results.filter((x) => x.error)) lines.push(`  #${r.id} not saved: ${r.error}`);
138   const left = counts(load()).inbox;
139   if (left) lines.push(`${left} still in inbox.`);
140   return lines.join('\n');
141 }

162   async idea_triage({ verdicts, ids, limit, by, headless = false, model }) {
163     if (headless) {
164       const out = await headlessTriage({ model, limit: limit || 20 });
165       return out.message || summarizeTriage(out.results, `${out.model}, ${out.tokens.input} in / ${out.tokens.output} out tokens`);
166     }
167     if (Array.isArray(verdicts) && verdicts.length) {

173     if (!pending.length) return 'Nothing to triage: the inbox is empty.';
174     return (

178     );
179   },

189   idea_next({ id, here = false }) {
190     const db = load();

192     if (id != null) {
193       idea = findIdea(db, id);

228 async function handleRequest(msg, { prompts }) {
229   const { id, method, params = {} } = msg;
230   switch (method) {
231     case 'initialize': {
232       const asked = params.protocolVersion;
233       const protocolVersion = PROTOCOL_VERSIONS.includes(asked) ? asked : PROTOCOL_VERSIONS[0];
234       const capabilities = { tools: { listChanged: false } };
235       if (prompts) capabilities.prompts = { listChanged: false };
236       return result(id, { protocolVersion, capabilities, serverInfo: { name: 'ideamine', version: VERSION }, instructions: INSTRUCTIONS });
237     }
238     case 'ping':
239       return result(id, {});
240     case 'tools/list':
241       return result(id, { tools: TOOLS });
242     case 'tools/call': {
243       const handler = handlers[params.name];
244       if (!handler) return error(id, -32602, `unknown tool: ${params.name}`);
245       try {
246         const text = await handler(params.arguments || {});
247         return result(id, { content: [{ type: 'text', text }] });
248       } catch (e) {
249         return result(id, { content: [{ type: 'text', text: `Error: ${e.message}` }], isError: true });
250       }
251     }
252     case 'prompts/list':
253       return result(id, { prompts: prompts ? PROMPTS.map(({ skill, ...p }) => p) : [] });
254     case 'prompts/get': {
255       const p = prompts && PROMPTS.find((x) => x.name === params.name);
256       if (!p) return error(id, -32602, `unknown prompt: ${params.name}`);
257       const args = Object.values(params.arguments || {}).filter(Boolean).join(' ');
258       return result(id, {
259         description: p.description,
260         messages: [{ role: 'user', content: { type: 'text', text: skillBody(p.skill, args) } }],
261       });
262     }
263     case 'resources/list':
264       return result(id, { resources: [] });
265     case 'resources/templates/list':
266       return result(id, { resourceTemplates: [] });
267     case 'logging/setLevel':
268       return result(id, {});
269     default:
270       return error(id, -32601, `method not found: ${method}`);
271   }
272 }

274 async function handleMessage(msg, opts) {
275   if (!msg || typeof msg !== 'object' || msg.jsonrpc !== '2.0') return error(null, -32600, 'invalid request');
276   if (typeof msg.method !== 'string') return null; // a response to a request we never send
277   if (msg.id === undefined) return null; // notification (initialized, cancelled, ...)
278   try {
279     return await handleRequest(msg, opts);
280   } catch (e) {
281     return error(msg.id, -32603, e.message);
282   }
283 }

285 async function handleLine(line, opts) {
286   let msg;
287   try {
288     msg = JSON.parse(line);
289   } catch {
290     return error(null, -32700, 'parse error');
291   }
292   if (!Array.isArray(msg)) return handleMessage(msg, opts);
293   const replies = (await Promise.all(msg.map((m) => handleMessage(m, opts)))).filter(Boolean);
294   return replies.length ? replies : null;
295 }

297 /** Serve MCP on stdin/stdout. `prompts: false` when running as a plugin (skills cover slash commands). */
298 export function serve({ prompts = true } = {}) {
299   const rl = readline.createInterface({ input: process.stdin, crlfDelay: Infinity });
300   const inFlight = new Set();
301   // Requests run concurrently (a headless triage must not stall a ping); replies carry their ids.
302   rl.on('line', (line) => {
303     if (!line.trim()) return;
304     const job = handleLine(line, { prompts }).then((reply) => reply && send(reply));
305     inFlight.add(job);
306     job.finally(() => inFlight.delete(job));
307   });
308   rl.on('close', () => Promise.allSettled([...inFlight]).then(() => process.exit(0)));
