1 // UserPromptSubmit hook: answers /idea and /ideas locally and blocks the prompt, so the model is
2 // never called. That makes capture free, instant, and possible even when the session is out of
3 // usage. Every other prompt passes through untouched.
4 
5 import { renderAdded, renderBoard, renderIdea } from './render.js';
6 import { addIdeas, FILTERS, findIdea, lane, load, updateIdea } from './store.js';
7 import { clip, splitIdeas } from './text.js';
8 
9 // `/idea ...`, `/ideas ...`, and the plugin-qualified `/ideamine:idea ...` forms.
10 const COMMAND = /^\s*\/(?:ideamine:)?(ideas?)(?=\s|$)([\s\S]*)$/i;
11 const EDIT_VERBS = { done: 'done', finish: 'done', drop: 'dropped', doing: 'doing', start: 'doing', reopen: 'reopen' };
12 
13 const USAGE =
14   'Usage: /idea <your idea>   saves it without calling the model (a bulleted list saves one idea per bullet)\n' +
15   '       /ideas [open|inbox|do|maybe|skip|doing|done|dropped|all|here] · /ideas #12 · /ideas done|drop|start|reopen 12 [note]';
16 
17 function readStdin() {
18   return new Promise((resolve) => {
19     if (process.stdin.isTTY) return resolve('');
20     let data = '';
21     process.stdin.setEncoding('utf8');
22     process.stdin.on('data', (chunk) => (data += chunk));
23     process.stdin.on('end', () => resolve(data));
24     process.stdin.on('error', () => resolve(data));
25   });
26 }

28 /** Handle one prompt. Returns the text to show the user, or null to let the prompt through. */
29 export function handlePrompt(prompt, { cwd = process.cwd(), session = null } = {}) {
30   const m = COMMAND.exec(prompt || '');
31   if (!m) return null;
32   const command = m[1].toLowerCase();
33   const arg = m[2].trim();
34 
35   if (command === 'idea') {
36     if (!arg) return USAGE;
37     const results = addIdeas(splitIdeas(arg), { source: 'hook', project: cwd, session });
38     return renderAdded(results, load());
39   }
40 
41   // /ideas
42   const words = arg.split(/\s+/).filter(Boolean);
43   const db = load();
44 
45   if (!words.length) return renderBoard(db, { cwd, hints: true });
46 
47   if (words.length === 1 && /^#?\d+$/.test(words[0])) {
48     const idea = findIdea(db, words[0]);
49     return idea ? renderIdea(idea) : `No idea ${words[0].startsWith('#') ? words[0] : '#' + words[0]}.`;
50   }
51 
52   const verb = EDIT_VERBS[words[0]?.toLowerCase()];
53   if (verb && words[1] && /^#?\d+$/.test(words[1])) {
54     const note = words.slice(2).join(' ');
55     const idea = updateIdea(words[1], { status: verb, note: note || undefined });
56     return `✓ #${idea.id} ${clip(idea.title, 60)} → ${lane(idea)}${note ? ' (note added)' : ''}`;
57   }
58 
59   const lower = words.map((w) => w.toLowerCase());
60   const filters = lower.filter((w) => FILTERS.includes(w));
61   const here = lower.includes('here');
62   if (filters.length + (here ? 1 : 0) === lower.length && filters.length <= 1) {
63     return renderBoard(db, { filter: filters[0] || 'open', project: here ? cwd : null, cwd, hints: true });
64   }
65   if (lower[0] === 'help') return USAGE;
66 
67   return null; // a question about the ideas: let the model answer it (via the /ideas skill)
68 }

70 export async function runHook() {
71   const raw = await readStdin();
72   let input;
73   try {
74     input = JSON.parse(raw);
75   } catch {
76     return; // not a hook payload; stay out of the way
77   }
78   let message;
79   try {
80     message = handlePrompt(typeof input?.prompt === 'string' ? input.prompt : '', {
81       cwd: input.cwd || process.cwd(),
82       session: input.session_id || null,
83     });
84   } catch (e) {
85     // Never swallow the user's text: let the prompt through so the /idea skill can save it via MCP.
86     process.stderr.write(`ideamine: ${e.message}\n`);
87     return;
88   }
89   if (message == null) return;
90   // Blocking ends the turn before any API request. suppressOriginalPrompt keeps Claude Code from
91   // echoing the command back under our message.
92   process.stdout.write(JSON.stringify({
93     decision: 'block',
94     reason: message,
95     hookSpecificOutput: { hookEventName: 'UserPromptSubmit', suppressOriginalPrompt: true },
96   }));
97 }
