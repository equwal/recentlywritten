21 function stamp(iso) {
22   if (!iso) return '';
23   const d = new Date(iso);
24   const pad = (n) => String(n).padStart(2, '0');
25   return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())} ${pad(d.getHours())}:${pad(d.getMinutes())}`;

28 export function ideaLine(idea, { cwd = null } = {}) {
29   const t = idea.triage;
30   const id = `#${idea.id}`.padEnd(5);
31   const meta = t && lane(idea) !== 'inbox' && t.verdict !== 'skip'
32     ? `${t.model.padEnd(6)} ${t.size.toUpperCase().padEnd(2)} ▲${t.impact}  `
33     : '';
34   const where = idea.project && !samePath(idea.project, cwd) ? `  · ${projectName(idea.project)}` : '';
35   return `  ${id}${meta}${clip(idea.title, 64)}${where}`;
36 }

44 /** The board: ideas grouped by lane. */
45 export function renderBoard(db, { filter = 'open', project = null, query = '', cwd = null, limit = 0, hints = false } = {}) {
46   const ideas = listIdeas(db, { filter, project, query, limit });
47   const out = [summaryLine(db)];
48   if (project || query || filter !== 'open') {
49     const scope = [filter !== 'open' && filter, project && `project ${projectName(project)}`, query && `"${query}"`];
50     out[0] += `  [showing: ${scope.filter(Boolean).join(', ')}]`;
51   }
52   if (!ideas.length) {
53     out.push('', db.ideas.length ? '  (nothing here)' : '  (empty — add one with /idea <text>)');
54   }
55   let current = null;
56   for (const idea of ideas) {
57     const l = lane(idea);
58     if (l !== current) {
59       current = l;
60       out.push('', LANE_TITLES[l] || l.toUpperCase());
61     }
62     out.push(ideaLine(idea, { cwd }));
63   }
64   if (hints) {
65     const c = counts(db);
66     const tips = ['/idea <text> to add', '/ideas #N for details'];
67     if (c.inbox) tips.push('/idea-triage to score the inbox');
68     if (c.do) tips.push('/idea-go to build the top pick');
69     out.push('', tips.join(' · '));
70   }
71   return out.join('\n');
72 }

74 /** One idea in full. */
75 export function renderIdea(idea) {
76   const t = idea.triage;
77   const out = [`#${idea.id} ${idea.title}`];
78   const status = [idea.status];
79   if (t) status.push(t.verdict, t.model, `size ${t.size.toUpperCase()}`, `impact ${t.impact}/5`);
80   out.push(`status   ${status.join(' · ')}`);
81   if (idea.project) out.push(`project  ${idea.project}`);
82   const added = [`${stamp(idea.created)} via ${idea.source}`];
83   if (idea.tags?.length) added.push(`tags: ${idea.tags.join(', ')}`);
84   out.push(`added    ${added.join(' · ')}`);
85   if (idea.dup_of) out.push(`dup of   #${idea.dup_of}`);
86   if (t?.why) out.push(`why      ${t.why}`);
87   if (t?.brief) out.push(`brief    ${t.brief}`);
88   if (t?.by) out.push(`triaged  ${stamp(t.at)} by ${t.by}`);
89   if (idea.text !== idea.title) out.push('', 'text', ...idea.text.split(/\r?\n/).map((l) => `  ${l}`));
90   if (idea.notes?.length) {
91     out.push('', 'notes');
92     for (const n of idea.notes) out.push(`  ${stamp(n.at)}  ${n.text}`);
93   }
94   return out.join('\n');
95 }

97 /** Confirmation after capture, e.g. "💡 Saved #12 · Add dark mode (3 in inbox)". */
98 export function renderAdded(results, db) {
99   const inbox = counts(db).inbox;
100   const lines = [];
101   if (results.length === 1) {
102     const { idea } = results[0];
103     lines.push(`💡 Saved #${idea.id} · ${clip(idea.title, 70)}  (${inbox} in inbox)`);
104   } else {
105     lines.push(`💡 Saved ${results.length} ideas: ${results.map((r) => `#${r.idea.id}`).join(', ')}  (${inbox} in inbox)`);
106   }
107   const seen = new Set(results.map((r) => r.idea.id));
108   for (const r of results) {
109     for (const s of r.similar) {
110       if (seen.has(s.id)) continue;
111       seen.add(s.id);
112       lines.push(`   similar to #${s.id} (${s.status}): ${clip(s.title, 60)}`);
113     }
114   }
115   return lines.join('\n');
116 }

118 /** Markdown export of the whole archive. */
119 export function renderMarkdown(db) {
120   const out = [`# Ideas`, '', `_${summaryLine(db)} · exported ${stamp(new Date().toISOString())}_`];
121   let current = null;
122   for (const idea of listIdeas(db, { filter: 'all' })) {
123     const l = lane(idea);
124     if (l !== current) {
125       current = l;
126       out.push('', `## ${LANE_TITLES[l] || l}`, '');
127     }
128     const t = idea.triage;
129     const meta = t && l !== 'inbox' ? ` — ${t.verdict}, ${t.model}, size ${t.size.toUpperCase()}, impact ${t.impact}/5` : '';
130     out.push(`- **#${idea.id} ${idea.title}**${meta}${idea.project ? ` · \`${projectName(idea.project)}\`` : ''}`);
131     if (t?.brief) out.push(`  - ${t.brief}`);
132     if (idea.text !== idea.title) out.push(`  - > ${idea.text.replace(/\r?\n/g, ' ')}`);
133   }
134   return out.join('\n') + '\n';
