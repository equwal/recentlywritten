57 const configPath = () => path.join(home(), 'config.json');

59 function readFile() {
60   try {
61     return JSON.parse(fs.readFileSync(configPath(), 'utf8'));
62   } catch {
63     return {};
64   }
65 }

67 function setting(key) {
68   const s = SETTINGS[key];
69   if (!s) throw new Error(`unknown setting "${key}" (use: ${Object.keys(SETTINGS).join(', ')})`);
70   return s;
71 }
72 
73 function parse(key, raw) {
74   if (typeof setting(key).value !== 'number') return String(raw).trim();
75   const n = Number(raw);
76   if (String(raw).trim() === '' || !Number.isFinite(n) || n < 0 || n > 1) throw new Error(`${key} must be a number from 0 to 1`);
77   return n;
78 }

80 /** The value and where it comes from: env, file, or default. */
81 function lookup(key) {
82   const s = setting(key);
83   const env = process.env[s.env];
84   if (env !== undefined && env !== '') return { value: parse(key, env), from: s.env };
85   const saved = readFile()[key];
86   if (saved !== undefined) return { value: parse(key, saved), from: 'config.json' };
87   return { value: s.value, from: 'default' };
88 }

90 export function get(key) {
91   return lookup(key).value;
92 }

94 /** Save a setting. An empty value removes it from the file, so the default applies again. */
95 export function set(key, value) {
96   const clean = value == null || String(value).trim() === '' ? undefined : parse(key, value);
97   withLock(() => {
98     const all = readFile();
99     if (clean === undefined) delete all[key];
100     else all[key] = clean;
101     writeAtomic(configPath(), JSON.stringify(all, null, 2) + '\n');
102   });
103   return get(key);
104 }

106 /** Every setting, one line each, with its source. */
107 export function describe() {
108   return Object.keys(SETTINGS)
109     .map((key) => {
110       const { value, from } = lookup(key);
111       return `${key.padEnd(17)} ${String(value === '' ? '(off)' : value).padEnd(28)} ${from} · ${SETTINGS[key].about}`;
112     })
113     .join('\n');
