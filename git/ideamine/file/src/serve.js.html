1 // The dashboard with buttons. `ideamine serve` runs a web server that serves the page of `ideamine
2 // publish`, a live data.json, and an API for the slash commands: add, delete, done, reopen, start,
3 // drop, note, model, triage, build, ask, and the watcher. It has two roles:
4 //
5 // - On a PC, it runs the commands there, where the Claude Code login is. When sync is on, the
6 //   archive that it changes is on the ideamine server, and the Prompts and Memory tabs come from
7 //   there too.
8 // - On a server (sync off), it holds the archive for every machine: the machines sync with /api/db
9 //   and /api/ops, and send their prompts to /api/prompts. With memstate_url, the Memory tab shows
10 //   the memories of a memstated daemon. nginx in front of it names the server in serve_hosts.
11 //
12 // The server listens on 127.0.0.1 only, and it takes commands only from its own page.

56 /**
57  * True when the server may answer a request. The Host header must name this server, which stops a
58  * DNS rebinding page. A POST must send JSON, and its Origin, if any, must be this server. A page of
59  * another site cannot send JSON here without a CORS preflight, and this server allows none.
60  */
61 export function allowed({ method, host, origin, type }, port, hosts = []) {
62   const h = String(host ?? '').toLowerCase();
63   if (h !== `127.0.0.1:${port}` && h !== `localhost:${port}` && !hosts.includes(h)) return false;
64   if (method === 'GET' || method === 'HEAD') return true;
65   if (method !== 'POST' || !/^application\/json\s*(;|$)/i.test(String(type ?? '').trim())) return false;
66   return origin == null || origin === `http://${h}`;
67 }

69 /**
70  * Run `node <args>` in a new console window, so that Claude Code gets a terminal. Windows only.
71  * `start` opens the window. A Windows path cannot hold a quote, so quotes around each argument
72  * keep cmd.exe from reading a & or | in a path. IDEAMINE_WINDOW keeps the window open after an error.
73  */
74 export function openWindow(args, cwd) {
75   if (process.platform !== 'win32') {
76     return Promise.reject(new Error(`the page can open a terminal only on Windows. Run in a terminal: ideamine ${args.slice(1).join(' ')}`));
77   }
78   const line = ['start', '""', ...[process.execPath, ...args].map((a) => `"${a}"`)].join(' ');
79   return new Promise((resolve, reject) => {
80     const child = spawn(process.env.ComSpec || 'cmd.exe', ['/d', '/c', line], {
81       cwd,
82       env: { ...process.env, IDEAMINE_WINDOW: '1' },
83       detached: true,
84       stdio: 'ignore',
85       windowsVerbatimArguments: true,
86     });
87     child.on('error', reject);
88     child.on('spawn', () => {
89       child.unref();
90       resolve();
91     });
92   });
93 }

129 function readBody(req) {
130   return new Promise((resolve, reject) => {
131     const chunks = [];
132     let size = 0;
133     req.on('data', (chunk) => {
134       size += chunk.length;
135       if (size > MAX_BODY) {
136         reject(new Error('the request is too big'));
137         req.destroy();
138       } else chunks.push(chunk);
139     });
140     req.on('end', () => resolve(Buffer.concat(chunks)));
141     req.on('error', reject);
142   });
143 }

234 /**
235  * Apply changes in order. A change that came before (the same oid) is not applied again: its first
236  * result comes back. A machine sends a change again when an answer got lost on the way.
237  */
238 function applyChanges(ops) {
239   if (!Array.isArray(ops)) throw new Error('ops must be a list');
240   const applied = readApplied();
241   const results = ops.map((op) => {
242     if (op?.oid && applied[op.oid]) return applied[op.oid];
243     let result;
244     try {
245       if (!op || !Object.hasOwn(CHANGES, op.op)) throw new Error(`unknown change "${op?.op}"`);
246       result = { ok: true, value: CHANGES[op.op](op) };
247     } catch (e) {
248       result = { ok: false, error: e.message };
249     }
250     if (op?.oid) applied[op.oid] = result;
251     return result;
252   });
253   const kept = Object.entries(applied).slice(-APPLIED_KEPT);
254   store.withLock(() => store.writeAtomic(appliedPath(), JSON.stringify(Object.fromEntries(kept))));
255   return results;
256 }

260 /** The prompt log: one JSON line for each prompt. The copy in memory is kept until the file changes. */
261 function loadPrompts() {
262   const st = fs.statSync(promptsPath(), { throwIfNoEntry: false });
263   if (!st) return { prompts: [], ids: new Set() };
264   if (promptCache?.mtime === st.mtimeMs && promptCache.size === st.size) return promptCache;
265   const prompts = [];
266   for (const line of fs.readFileSync(promptsPath(), 'utf8').split('\n')) {
267     if (!line.trim()) continue;
268     try {
269       prompts.push(JSON.parse(line));
270     } catch {
271       // A line that a crash cut in half.
272     }
273   }
274   promptCache = { mtime: st.mtimeMs, size: st.size, prompts, ids: new Set(prompts.map((p) => p.id)) };
275   return promptCache;
276 }

280 /** Add prompts that the log does not have yet. A prompt without an id, a time, or a text is skipped. */
281 function addPrompts(list) {
282   if (!Array.isArray(list)) throw new Error('prompts must be a list');
283   const { ids } = loadPrompts();
284   const add = [];
285   let skipped = 0;
286   for (const p of list) {
287     const bad = !p || typeof p.id !== 'string' || !p.id || typeof p.prompt !== 'string' || Number.isNaN(Date.parse(p.at));
288     if (bad) skipped++;
289     if (bad || ids.has(p.id)) continue;
290     ids.add(p.id);
291     add.push({ id: p.id.slice(0, 64), at: new Date(p.at).toISOString(), host: field(p.host, 100), session: field(p.session, 100), cwd: field(p.cwd), prompt: p.prompt });
292   }
293   if (add.length) store.withLock(() => fs.appendFileSync(promptsPath(), add.map((p) => `${JSON.stringify(p)}\n`).join('')));
294   return { added: add.length, skipped };
295 }

316 /** Keep the rows of one machine. The new rows take the place of the rows from before. */
317 function saveUsage({ machine, host, rows }) {
318   if (typeof machine !== 'string' || !machine || !Array.isArray(rows)) throw new Error('a machine id and its rows must come together');
319   const clean = rows
320     .filter((r) => r && /^\d{4}-\d{2}-\d{2}$/.test(String(r.day)) && typeof r.project === 'string' && typeof r.model === 'string')
321     .map((r) => ({
322       day: r.day,
323       project: String(r.project).slice(0, 300),
324       model: String(r.model).slice(0, 60),
325       messages: Number(r.messages) || 0,
326       input: Number(r.input) || 0,
327       output: Number(r.output) || 0,
328       read: Number(r.read) || 0,
329       write5m: Number(r.write5m) || 0,
330       write1h: Number(r.write1h) || 0,
331     }));
332   const data = loadUsage();
333   data.machines[machine.slice(0, 64)] = { host: field(host, 100), at: new Date().toISOString(), rows: clean };
334   store.withLock(() => store.writeAtomic(usagePath(), `${JSON.stringify(data)}\n`));
335   return { rows: clean.length, machines: Object.keys(data.machines).length };

354 /** A read from the memstated daemon at memstate_url. */
355 async function memstate(route, body) {
356   const base = config.get('memstate_url');
357   if (!base) throw new Error('this server shows no memories. Set memstate_url, for example: ideamine config memstate_url http://127.0.0.1:8765');
358   const url = `${base.replace(/\/+$/, '')}${route}`;
359   const init = { signal: AbortSignal.timeout(8000) };
360   if (body) Object.assign(init, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(body) });
361   let res;
362   try {
363     res = await fetch(url, init);
364   } catch {
365     throw new Error(`cannot reach memstated at ${base}`);
366   }
367   const json = await res.json().catch(() => null);
368   if (!res.ok) throw new Error(`memstated answered ${res.status}: ${json?.error || 'no reason'}`);
369   return json;
370 }

491 /** The reads and changes of the server role. Resolves to false for a route that it does not have. */
492 async function serverRoute(req, res, route, query) {
493   if (route === 'GET /api/db') return send(res, 200, { ok: true, db: store.load() });
494   if (route === 'POST /api/ops') {
495     const { ops } = await readJson(req);
496     return send(res, 200, { ok: true, results: applyChanges(ops), db: store.load() });
497   }
498   if (route === 'GET /api/usage') {
499     const usage = await import('./usage.js');
500     const sent = usageRows();
501     const rows = sent.length ? sent : await localUsage();
502     // The price comes from the server, so that one price list serves every page.
503     return send(res, 200, { ok: true, rows: rows.map((r) => ({ ...r, cost: usage.cost(r) })) });
504   }
505   if (route === 'POST /api/usage') return send(res, 200, { ok: true, ...saveUsage(await readJson(req)) });
506   if (route === 'POST /api/prompts') {
507     const { prompts } = await readJson(req);
508     return send(res, 200, { ok: true, ...addPrompts(prompts) });
509   }
510   if (route === 'GET /api/prompts') return send(res, 200, { ok: true, prompts: listPrompts(Number(query.get('days')) || 0) });
511   const memory = route.startsWith('GET /api/memory/') && route.slice('GET /api/memory/'.length);
512   if (memory && Object.hasOwn(MEMORY, memory)) return send(res, 200, { ok: true, ...(await MEMORY[memory](query)) });
513   return false;
514 }

516 async function handle(req, res, ctx) {
517   const request = { method: req.method, host: req.headers.host, origin: req.headers.origin, type: req.headers['content-type'] };
518   if (!allowed(request, ctx.port, extraHosts())) return send(res, 403, { ok: false, error: 'forbidden' });
519   const { pathname, search, searchParams } = new URL(req.url, 'http://127.0.0.1');
520   const route = `${req.method === 'HEAD' ? 'GET' : req.method} ${pathname}`;
521   if (route === 'GET /' || route === 'GET /index.html') return sendPage(res);
522   if (req.method === 'GET' && Object.hasOwn(ASSETS, pathname.slice(1))) return sendAsset(res, pathname.slice(1));
523   if (route === 'GET /data.json') return send(res, 200, await liveData());
524   // A ping gets a new connection each time, so it never reaches a server that stops on an old one.
525   if (route === 'GET /api/ping') return send(res, 200, { ok: true, app: 'ideamine' }, { connection: 'close' });
526   if (route === 'POST /v1/embeddings') return proxyEmbeddings(req, res);
527   if (route === 'POST /api/stop') {
528     // After "stopped", no kept-alive connection may answer a request.
529     res.on('finish', () => ctx.server.closeAllConnections?.());
530     send(res, 200, { ok: true, message: 'stopped' }, { connection: 'close' });
531     ctx.server.close();
532     return;
533   }
534   if (sync.enabled()) {
535     if (route === 'GET /api/prompts' || route === 'GET /api/usage' || route.startsWith('GET /api/memory/')) return forward(res, `${pathname.slice(1)}${search}`);
536   } else {
537     try {
538       if ((await serverRoute(req, res, route, searchParams)) !== false) return;
539     } catch (e) {
540       return send(res, 400, { ok: false, error: e.message });
541     }
542   }
543   const action = route.startsWith('POST /api/') ? route.slice('POST /api/'.length) : '';
544   if (!Object.hasOwn(ACTIONS, action)) return send(res, 404, { ok: false, error: 'not found' });
545   try {
546     const message = await ACTIONS[action](await readJson(req), ctx);
547     ctx.log(`${action}: ok`);
548     if (!READ_ONLY.has(action)) {
549       try {
550         ctx.afterChange();
551       } catch {
552         // The upload to the dashboard server must never fail a command.
553       }
554     }
555     return send(res, 200, { ok: true, message });
556   } catch (e) {
557     ctx.log(`${action}: ${e.message}`);
558     return send(res, 400, { ok: false, error: e.message });
559   }
560 }

562 /**
563  * The server, not listening yet. `open` opens the window for a build, `afterChange` runs after a
564  * command changes the archive (by default it uploads the dashboard again), and `log` gets a line
565  * for each command.
566  */
567 export function createServer({ open = openWindow, afterChange = publish.kick, log = () => {} } = {}) {
568   const ctx = { open, afterChange, log, port: null };
569   ctx.server = http.createServer((req, res) => {
570     handle(req, res, ctx).catch((e) => {
571       if (!res.headersSent) send(res, 500, { ok: false, error: e.message });
572     });
573   });
574   ctx.server.on('listening', () => (ctx.port = ctx.server.address().port));
575   return ctx.server;
576 }
577 
578 /** Listen on 127.0.0.1. Resolves to { server, url }. */
579 export function start({ port: p = port(), ...options } = {}) {
580   const server = createServer(options);
581   return new Promise((resolve, reject) => {
582     server.once('error', reject);
583     server.listen(p, '127.0.0.1', () => resolve({ server, url: address(server.address().port) }));
584   });
585 }

587 /** True when an ideamine server answers at `url`. */
588 async function ping(url) {
589   try {
590     const res = await fetch(new URL('api/ping', url), { signal: AbortSignal.timeout(1000) });
591     return res.ok && (await res.json()).app === 'ideamine';
592   } catch {
593     return false;
594   }
595 }

597 /** Start `ideamine serve` in the background. Its output goes to serve.log, and so does a failed start. */
598 function startProcess() {
599   fs.mkdirSync(store.home(), { recursive: true });
600   const log = fs.openSync(logPath(), 'a');
601   try {
602     const child = spawn(process.execPath, [BIN, 'serve'], { cwd: store.home(), detached: true, stdio: ['ignore', log, log], windowsHide: true });
603     child.on('error', (e) => fs.appendFileSync(logPath(), `${logLine(`cannot start: ${e.message}`)}\n`));
604     child.unref();
605   } finally {
606     fs.closeSync(log);
607   }
608 }

610 /** /ideas-web: start the server when it does not run. Resolves to the text for the user. */
611 export async function ensureRunning({ startServer = startProcess, waitMs = 5000 } = {}) {
612   const url = address();
613   if (await ping(url)) return `ideamine web: ${url}`;
614   startServer();
615   for (const deadline = Date.now() + waitMs; Date.now() < deadline; ) {
616     await sleep(150);
617     if (await ping(url)) return `ideamine web: ${url} (started)`;
618   }
619   return `ideamine web did not start at ${url}. The reason is in ${logPath()}.`;
620 }
621 
622 /** /ideas-web off */
623 export async function stopRunning() {
624   const url = address();
625   if (!(await ping(url))) return 'ideamine web: not running.';
626   try {
627     await fetch(new URL('api/stop', url), { method: 'POST', headers: { 'content-type': 'application/json' }, body: '{}' });
628   } catch {
629     // The server can close the connection before its answer arrives. The ping below tells the result.
630   }
631   return (await ping(url)) ? `ideamine web: still running at ${url}` : 'ideamine web: stopped.';
632 }
