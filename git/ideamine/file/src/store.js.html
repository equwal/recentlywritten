2 // Writers take a directory lock and replace the file atomically, so concurrent sessions,

38 /** Read-only snapshot. Safe without the lock because writes are atomic renames. */
39 export function load() {
40   const file = dbPath();
41   let raw;
42   try {
43     raw = fs.readFileSync(file, 'utf8').replace(/^﻿/, '');
44   } catch (e) {
45     if (e.code === 'ENOENT') return emptyDb();
46     throw e;
47   }
48   if (!raw.trim()) return emptyDb();
49   let db;
50   try {
51     db = JSON.parse(raw);
52   } catch (e) {
53     throw new Error(`${file} is not valid JSON (${e.message}). Fix it, or restore ${file}.bak`);
54   }
55   if (!db || !Array.isArray(db.ideas)) throw new Error(`${file} has no "ideas" array`);
56   const maxId = db.ideas.reduce((m, i) => Math.max(m, Number(i.id) || 0), 0);
57   db.next_id = Math.max(Number(db.next_id) || 1, maxId + 1);
58   return db;
59 }

61 function acquireLock() {
62   fs.mkdirSync(home(), { recursive: true });
63   const lockDir = path.join(home(), '.lock');
64   const deadline = Date.now() + LOCK_WAIT_MS;
65   for (let attempt = 0; ; attempt++) {
66     try {
67       fs.mkdirSync(lockDir);
68       return () => fs.rmSync(lockDir, { recursive: true, force: true });
69     } catch (e) {
70       if (e.code !== 'EEXIST') throw e;
71     }
72     try {
73       // A writer holds the lock for milliseconds; an old lock means a crashed process.
74       if (Date.now() - fs.statSync(lockDir).mtimeMs > LOCK_STALE_MS) {
75         fs.rmSync(lockDir, { recursive: true, force: true });
76         continue;
77       }
78     } catch {
79       // The lock vanished between mkdir and stat; the next attempt will likely get it.
80     }
81     if (Date.now() > deadline) {
82       throw new Error(`idea store is busy (lock: ${lockDir}). Delete that folder if no ideamine process is running.`);
83     }
84     sleepSync(Math.min(10 + attempt * 5, 100));
85   }
86 }

88 function renameWithRetry(from, to) {
89   // Windows refuses to replace a file that another program (editor, antivirus, sync) has open.
90   for (let attempt = 0; ; attempt++) {
91     try {
92       fs.renameSync(from, to);
93       return;
94     } catch (e) {
95       if (attempt >= 40 || !['EPERM', 'EBUSY', 'EACCES'].includes(e.code)) {
96         fs.rmSync(from, { force: true });
97         throw e;
98       }
99       sleepSync(25);
100     }
101   }
102 }

104 function save(db) {
105   const file = dbPath();
106   const tmp = `${file}.${process.pid}.${Date.now()}.tmp`;
107   fs.writeFileSync(tmp, JSON.stringify(db, null, 2) + '\n');
108   try {
109     fs.copyFileSync(file, `${file}.bak`);
110   } catch (e) {
111     if (e.code !== 'ENOENT') throw e;
112   }
113   renameWithRetry(tmp, file);
114 }

116 /** Lock, read, apply `fn(db)`, write. Nothing is written if `fn` throws. */
117 export function mutate(fn) {
118   const release = acquireLock();
119   try {
120     const db = load();
121     const result = fn(db);
122     save(db);
123     return result;
124   } finally {
125     release();
126   }
127 }

132 export function samePath(a, b) {
133   if (!a || !b) return false;
134   const norm = (p) => {
135     const r = path.resolve(p).replace(/[\\/]+$/, '');
136     return process.platform === 'win32' ? r.toLowerCase() : r;
137   };
138   return norm(a) === norm(b);

165 export function matchesFilter(idea, filter) {
166   const l = lane(idea);
167   switch (filter) {
168     case 'all':
169       return true;
170     case 'open':
171       return ['doing', 'do', 'maybe', 'inbox'].includes(l);
172     default:
173       return l === filter;
174   }

177 export function listIdeas(db, { filter = 'open', project = null, query = '', limit = 0 } = {}) {
178   if (!FILTERS.includes(filter)) throw new Error(`unknown filter "${filter}" (use: ${FILTERS.join(', ')})`);
179   const words = String(query || '').toLowerCase().split(/\s+/).filter(Boolean);
180   let ideas = db.ideas.filter((i) => matchesFilter(i, filter));
181   if (project) ideas = ideas.filter((i) => samePath(i.project, project));
182   if (words.length) {
183     ideas = ideas.filter((i) => {
184       const hay = `${i.title}\n${i.text}\n${(i.tags || []).join(' ')}`.toLowerCase();
185       return words.every((w) => hay.includes(w));
186     });
187   }
188   ideas.sort((a, b) => {
189     const la = LANE_ORDER.indexOf(lane(a));
190     const lb = LANE_ORDER.indexOf(lane(b));
191     if (la !== lb) return la - lb;
192     if (['done', 'dropped', 'inbox'].includes(lane(a))) return b.id - a.id; // newest first
193     return priority(b) - priority(a) || a.id - b.id;
194   });
195   return limit > 0 ? ideas.slice(0, limit) : ideas;
196 }

198 export function counts(db) {
199   const c = { inbox: 0, do: 0, maybe: 0, skip: 0, doing: 0, done: 0, dropped: 0 };
200   for (const i of db.ideas) c[lane(i)] = (c[lane(i)] || 0) + 1;
201   c.open = c.inbox + c.do + c.maybe + c.doing;

205 /** Best idea to build next: verdict "do", not started, current project first, then value per effort. */
206 export function pickNext(db, { project = null, only = false } = {}) {
207   let ready = db.ideas.filter((i) => lane(i) === 'do');
208   if (only) ready = ready.filter((i) => samePath(i.project, project));
209   ready.sort((a, b) => {
210     const pa = project && samePath(a.project, project) ? 1 : 0;
211     const pb = project && samePath(b.project, project) ? 1 : 0;
212     return pb - pa || priority(b) - priority(a) || a.id - b.id;
213   });
214   return ready[0] || null;
215 }

217 function similarTo(db, text, excludeId) {
218   const words = wordSet(text);
219   if (words.size < 2) return [];
220   return db.ideas
221     .filter((i) => i.id !== excludeId && i.status !== 'dropped' && isSimilar(words, wordSet(`${i.title} ${i.text}`)))
222     .slice(-2)
223     .map((i) => ({ id: i.id, title: i.title, status: lane(i) }));
224 }

229 export function addIdeas(texts, { source = 'mcp', project = null, session = null, tags = [] } = {}) {
230   const items = (Array.isArray(texts) ? texts : [texts]).map((t) => String(t ?? '').trim()).filter(Boolean);
231   if (!items.length) throw new Error('idea text is empty');
232   return mutate((db) =>
233     items.map((text) => {
234       const similar = similarTo(db, text, null);
235       const stamp = now();
236       const idea = {
237         id: db.next_id++,
238         title: deriveTitle(text),
239         text,
240         status: 'inbox',
241         tags: normalizeTags([...extractTags(text), ...normalizeTags(tags)]),
242         project: project ? path.resolve(project) : null,
243         source,
244         created: stamp,
245         updated: stamp,
246         triage: null,
247         notes: [],
248       };
249       if (session) idea.session = String(session);
250       db.ideas.push(idea);
251       return { idea, similar };
252     }),
253   );
254 }

261 function cleanVerdict(v) {
262   const verdict = String(v.verdict || '').toLowerCase();
263   if (!VERDICTS.includes(verdict)) throw new Error(`verdict must be one of ${VERDICTS.join('/')}`);
264   const impact = Math.min(5, Math.max(1, Math.round(Number(v.impact) || 3)));
265   const size = SIZES.includes(String(v.size || '').toLowerCase()) ? String(v.size).toLowerCase() : 'm';
266   const model = normalizeModel(v.model) || 'sonnet';
267   return {
268     verdict,
269     impact,
270     size,
271     model,
272     why: clip(v.why || '', 300),
273     brief: verdict === 'skip' ? '' : String(v.brief || '').trim().slice(0, 2000),
274   };
275 }

277 /** Save triage verdicts. Bad items are reported individually instead of failing the batch. */
278 export function applyTriage(verdicts, { by = null } = {}) {
279   if (!Array.isArray(verdicts) || !verdicts.length) throw new Error('no verdicts given');
280   return mutate((db) =>
281     verdicts.map((v) => {
282       const idea = findIdea(db, v?.id);
283       if (!idea) return { id: v?.id, error: 'no such idea' };
284       let t;
285       try {
286         t = cleanVerdict(v);
287       } catch (e) {
288         return { id: idea.id, error: e.message };
289       }
290       idea.triage = { ...t, at: now(), ...(by ? { by } : {}) };
291       if (v.title) idea.title = clip(v.title, 90);
292       if (Array.isArray(v.tags)) idea.tags = normalizeTags([...(idea.tags || []), ...v.tags]);
293       const dup = Number(v.dup_of);
294       if (dup && dup !== idea.id && findIdea(db, dup)) idea.dup_of = dup;
295       if (idea.status === 'inbox') idea.status = 'triaged';
296       idea.updated = now();
297       return { id: idea.id, verdict: t.verdict, model: t.model, size: t.size, title: idea.title };
298     }),
299   );
300 }

304 export function updateIdea(id, patch = {}) {
305   return mutate((db) => {
306     const idea = findIdea(db, id);
307     if (!idea) throw new Error(`no idea #${id}`);
308     if (patch.status) {
309       let status = String(patch.status).toLowerCase();
310       status = STATUS_ALIASES[status] || status;
311       if (status === 'reopen' || status === 'open') status = idea.triage ? 'triaged' : 'inbox';
312       if (!STATUSES.includes(status)) throw new Error(`status must be one of ${STATUSES.join(', ')}, or reopen`);
313       idea.status = status;
314       if (status === 'done' || status === 'dropped') idea.closed = now();
315       else delete idea.closed;
316     }
317     if (patch.text) idea.text = String(patch.text).trim();
318     if (patch.title) idea.title = clip(patch.title, 90);
319     else if (patch.text) idea.title = deriveTitle(idea.text);
320     if (Array.isArray(patch.tags)) idea.tags = normalizeTags(patch.tags);
321     if (patch.project !== undefined) idea.project = patch.project ? path.resolve(patch.project) : null;
322     if (patch.model) {
323       const model = normalizeModel(patch.model);
324       if (!model) throw new Error(`model must be one of ${MODEL_ALIASES.join(', ')}`);
325       idea.triage = { ...(idea.triage || { verdict: 'do', impact: 3, size: 'm', why: '', brief: '' }), model };
326       if (idea.status === 'inbox') idea.status = 'triaged';
327     }
328     if (patch.note) {
329       idea.notes ||= [];
330       idea.notes.push({ at: now(), text: String(patch.note).trim() });
331     }
332     idea.updated = now();
333     return idea;
334   });
