22 ;; This is an implementation of an elisp-based FORTH. This means that elisp
23 ;; is used to write the function definitions. In fact, no FORTH function
24 ;; definitions are provided at all to the Forthe macro from the library.
25 ;; The main benefit comes from reducing the verbosity of elisp in a small
26 ;; Domain Specific Language the programmer defines.

30 (defun forthe--inner (lang list)
31   "Internal function for the `forthe' macro.
32 LANG are the bindings for the forth.
33 LIST is full of the Forthe stack (RPN)."
34   (let (s)
35     (dolist (i list)
36       (if (assoc i lang)
37           (let ((fn (cadr (assoc i lang))))
38             (if s
39                 (progn (apply fn s) (setf s nil))
40               (funcall fn)))
41         (push i s)))))
42 
43 (defmacro forthe (lang list)
44   "Define and execute a FORTH language (called Forthe).
45 LANG are the let-bindings for the FORTH's operators.
46 LIST is the forth's stack."
47   `(forthe--inner ,lang ',list))
