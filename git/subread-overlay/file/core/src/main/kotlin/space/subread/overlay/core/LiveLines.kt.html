3 /**
4  * The lines that a caption app sends while it makes captions from live audio.
5  *
6  * A caption app sends a line many times while the speech goes on: a partial line, which grows.
7  * When the sentence is finished, it sends the line once more as a final line. The panel shows the
8  * newest line. Only a final line becomes the line before: a partial line goes away when the next
9  * line comes, because the final line of the same speech takes its place.
10  */
11 class LiveLines {
12 
13     /** The newest line. Empty before the first line. */
14     var now: String = ""
15         private set
16 
17     /** True when [now] is a partial line: the speech that it comes from goes on. */
18     var partial: Boolean = false
19         private set
20 
21     /** The last final line before [now]; null when there is none. */
22     var before: String? = null
23         private set
24 
25     /** A new line from the caption app. */
26     fun line(text: String, partial: Boolean) {
27         if (!this.partial && now.isNotEmpty()) before = now
28         now = text
29         this.partial = partial
30     }
31 
32     /** Forgets each line: the caption app stopped. */
33     fun clear() {
34         now = ""
35         partial = false
36         before = null
37     }
