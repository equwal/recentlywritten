3 /**
4  * Finds the line to show for a position in the media.
5  *
6  * The line stays on the screen until the next line starts, also in the silence after its end
7  * time. A reader who looks a word up needs the line to stay; an empty overlay helps nobody.
8  */
9 class CueIndex(cues: List<Cue>) {
10 
11     val cues: List<Cue> = cues.sortedBy { it.startMs }
12     private val starts = LongArray(this.cues.size) { this.cues[it].startMs }
13 
14     val size: Int get() = cues.size
15 
16     /** The index of the last cue that starts at or before [positionMs]; -1 before the first cue. */
17     fun indexAt(positionMs: Long): Int {
18         var low = 0
19         var high = starts.lastIndex
20         var found = -1
21         while (low <= high) {
22             val mid = (low + high) ushr 1
23             if (starts[mid] <= positionMs) {
24                 found = mid
25                 low = mid + 1
26             } else {
27                 high = mid - 1
28             }
29         }
30         return found
31     }
32 
33     /** The position at which another line comes on after [positionMs]; null after the last cue. */
34     fun nextChangeAfter(positionMs: Long): Long? {
35         val next = indexAt(positionMs) + 1
36         return if (next < starts.size) starts[next] else null
37     }
