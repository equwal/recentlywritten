3 /**
4  * What a media player said about its position, and when it said it.
5  *
6  * A player does not report each millisecond. It reports a position, the time of the report on
7  * the clock of the device, and the speed. The position now is computed from these three.
8  */
9 data class PlayClock(
10     val positionMs: Long,
11     /** The time of the report, on the same clock as the `nowMs` parameters. */
12     val reportedAtMs: Long,
13     val speed: Float,
14     val playing: Boolean,
15 ) {
16     /** The position in the media at [nowMs]. */
17     fun positionAt(nowMs: Long): Long =
18         if (!playing) positionMs else positionMs + ((nowMs - reportedAtMs) * speed.toDouble()).toLong()
19 
20     /**
21      * How long to wait, on the clock of the device, until the media is at [targetMs].
22      * Null when the media does not move forward, or is already there.
23      */
24     fun waitUntil(targetMs: Long, nowMs: Long): Long? {
25         if (!playing || speed <= 0f) return null
26         val left = targetMs - positionAt(nowMs)
27         if (left <= 0) return null
28         return Math.ceil(left / speed.toDouble()).toLong()
29     }
