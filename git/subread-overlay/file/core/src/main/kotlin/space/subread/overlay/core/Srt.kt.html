6 /**
7  * Reads SubRip (`.srt`) text.
8  *
9  * The reader is lenient: a block without a valid time line is skipped and the rest of the file
10  * is kept. Subtitle files from the wild have stray numbers, comments and broken blocks.
11  */
12 object Srt {
13 
14     private val TIMES = Regex(
15         """(\d{1,3}):(\d{2}):(\d{2})[.,](\d{1,3})\s*-->\s*(\d{1,3}):(\d{2}):(\d{2})[.,](\d{1,3})""",
16     )
17     private val TAG = Regex("""<[^>]+>|\{\\[^}]*\}""")
18 
19     /**
20      * The cues of [content], sorted by start time.
21      *
22      * Accepts a byte order mark, CRLF, CR or LF line ends, and `.` or `,` before the milliseconds.
23      * Removes format tags (`<i>`, `{\an8}`): the overlay shows plain text.
24      */
25     fun parse(content: String): List<Cue> =
26         content.removePrefix("\uFEFF")
27             .replace("\r\n", "\n").replace('\r', '\n')
28             .split(Regex("\n\\s*\n"))
29             .mapNotNull(::block)
30             .sortedBy { it.startMs }
31 
32     private fun block(block: String): Cue? {
33         val lines = block.lines().map { it.trim() }.filter { it.isNotEmpty() }
34         val at = lines.indexOfFirst { TIMES.containsMatchIn(it) }
35         if (at == -1) return null
36         val g = TIMES.find(lines[at])!!.groupValues
37         val start = millis(g[1], g[2], g[3], g[4])
38         val end = millis(g[5], g[6], g[7], g[8])
39         val text = lines.drop(at + 1).joinToString("\n") { it.replace(TAG, "") }.trim()
40         if (end <= start || text.isEmpty()) return null
41         return Cue(start, end, text)
42     }
43 
44     private fun millis(h: String, m: String, s: String, fraction: String): Long =
45         h.toLong() * 3_600_000 + m.toLong() * 60_000 + s.toLong() * 1_000 +
46             fraction.padEnd(3, '0').take(3).toLong()
