20     private fun stamp(ms: Long) =
21         "%02d:%02d:%02d,%03d".format(ms / 3_600_000, ms / 60_000 % 60, ms / 1_000 % 60, ms % 1_000)
22 
23     private fun write(cues: List<Cue>) = cues.withIndex().joinToString("\n\n") { (i, c) ->
24         "${i + 1}\n${stamp(c.startMs)} --> ${stamp(c.endMs)}\n${c.text}"
25     }
26 
27     /** Cues that do not share a start time, so that the sorted order is one order only. */
28     private val cues = Arb.list(
29         Arb.int(1..20_000).map { it.toLong() },
30         0..40,
31     ).map { gaps ->
32         var at = 0L
33         gaps.mapIndexed { i, gap ->
34             at += gap
35             Cue(at, at + 1 + gap, "line $i\nsecond row")
36         }
37     }

40     fun whatIsWrittenIsReadBack(): Unit = runBlocking {
41         checkAll(cues) { list ->
42             assertEquals(list, Srt.parse(write(list)))
43             assertEquals(list, Srt.parse("\uFEFF" + write(list).replace("\n", "\r\n")))
44         }
45     }

120     @Test
121     fun afterTheWaitTheMediaIsAtTheTarget(): Unit = runBlocking {
122         checkAll(
123             Arb.long(0L..36_000_000L),
124             Arb.long(0L..10_000_000L),
125             Arb.float(0.25f..4f),
126             Arb.long(1L..600_000L),
127             Arb.boolean(),
128         ) { position, now, speed, ahead, reportedLate ->
129             val clock = PlayClock(position, if (reportedLate) now else 0, speed, playing = true)
130             val target = clock.positionAt(now) + ahead
131             val wait = clock.waitUntil(target, now)!!
132             // Not early, and late by less than one step of the device clock at this speed.
133             assertTrue("early", clock.positionAt(now + wait) >= target - 1)
134             assertTrue("late", clock.positionAt(now + wait) <= target + speed.toLong() + 1)
135         }
136     }
