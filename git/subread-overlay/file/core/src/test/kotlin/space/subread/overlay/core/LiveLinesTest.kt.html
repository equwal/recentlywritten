21     @Test
22     fun theLineBeforeIsTheLastFinalLineBeforeTheNewestLine(): Unit = runBlocking {
23         checkAll(feeds) { feed ->

27             assertEquals(last?.first.orEmpty(), lines.now)
28             assertEquals(last?.second == true, lines.partial)

30             assertEquals(finalsBefore.lastOrNull()?.first, lines.before)
31         }
