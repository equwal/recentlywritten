83     private val lookedUp = CopyOnWriteArrayList<String>()

86     private val events = object : OverlayView.Events {
87         override fun onDrag(dx: Float, dy: Float) = Unit
88         override fun onDragEnd() = Unit
89         override fun onClose() = Unit
90         override fun onTouchWord() { touches++ }
91         override fun onLookUp(word: String) { lookedUp += word }
92         override fun onShare(word: String) { shared += word }
93         override fun onTogglePlay() = Unit
94         override fun onShiftLines(steps: Int) = Unit
95         override fun onNudge(ms: Long) = Unit

98     /** The x of the middle of the character at [offset], in the panel. */
99     private fun OverlayView.middleOf(offset: Int): Float {
100         val layout = text.layout
101         return text.totalPaddingLeft + (layout.getPrimaryHorizontal(offset) + layout.getPrimaryHorizontal(offset + 1)) / 2
102     }
103 
104     /** A y just over the baseline of the row of the character at [offset], in the panel. */
105     private fun OverlayView.rowOf(offset: Int): Float =
106         text.totalPaddingTop + text.layout.getLineBaseline(text.layout.getLineForOffset(offset)) - 5f
107 
108     /** Gives a touch event at [x], [y] of the text view to the text view. */
109     private fun OverlayView.touch(action: Int, x: Float, y: Float) {
110         val now = SystemClock.uptimeMillis()
111         val event = android.view.MotionEvent.obtain(now, now, action, x, y, 0)
112         text.dispatchTouchEvent(event)
113         event.recycle()
114     }

245     private fun waitFor(line: String?) {
246         val until = SystemClock.elapsedRealtime() + 5_000
247         while (lines.lastOrNull() != line && SystemClock.elapsedRealtime() < until) Thread.sleep(20)
248         assertEquals(lines.toString(), line, lines.lastOrNull())

251     @Test
252     fun theLineFollowsThePlayerThroughPauseSeekSpeedAndOffset() {
253         val session = MediaSession(context, "test player")
254         lateinit var follower: Follower
255         try {
256             session.setPlaybackState(state(PlaybackState.STATE_PAUSED, 1_500))
257             session.isActive = true
258             onMain {
259                 follower = Follower(main) { at, _, _ -> lines += follower.index.cues.getOrNull(at)?.text }
260                 follower.index = index
261                 follower.follow(listOf(MediaController(context, session.sessionToken)))
262             }
263             waitFor("one")
264 
265             // A seek. The player is paused: the line changes at once and then stays.
266             session.setPlaybackState(state(PlaybackState.STATE_PAUSED, 500))
267             waitFor(null)
268 
269             // Play at double speed from 59.6 s: "two" is 0.2 s away, "three" 0.2 s after it.
270             val started = SystemClock.elapsedRealtime()
271             session.setPlaybackState(state(PlaybackState.STATE_PLAYING, 59_600, speed = 2f))
272             waitFor("one")
273             waitFor("two")
274             waitFor("three")
275             val took = SystemClock.elapsedRealtime() - started
276             assertTrue("the follower waited $took ms; at double speed 0.8 s of media is 0.4 s", took in 350..1_500)
277 
278             // The subtitles are 60 s late for this media: the user shifts them.
279             session.setPlaybackState(state(PlaybackState.STATE_PAUSED, 100))
280             waitFor(null)
281             onMain { follower.offsetMs = 60_000 }
282             waitFor("two")
283 
284             // "line ▶" makes the next line the line of now.
285             onMain { follower.shiftLines(1) }
286             waitFor("three")
287             onMain { assertEquals(60_400L - 100L, follower.offsetMs) }
288         } finally {
289             onMain { follower.stop() }
290             session.release()
291         }
