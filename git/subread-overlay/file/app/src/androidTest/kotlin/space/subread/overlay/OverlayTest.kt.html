39     @Test
40     fun aWordOfJapaneseThatHasNoSpaces() {
41         val text = "女のいない男たちは、東京で暮らしている。"
42         // The dictionary of ICU decides the cut. The test asks for what each good cut has:
43         // a word is shorter than the sentence, stops at the comma, and has the character under the finger.
44         for (at in listOf(0, text.indexOf("男"), text.indexOf("東京"), text.indexOf("暮"))) {
45             val range = Words.at(text, at)!!
46             assertTrue("$range has $at", at in range)
47             assertTrue("$range is a word, not the sentence", range.last - range.first < 6)
48             assertTrue("no comma in a word", '、' !in text.substring(range.first, range.last + 1))
49         }
50         assertEquals("東京", word(text, text.indexOf("東京")))
51         assertNull(word(text, text.indexOf('、')))
52     }

81     /** The x of the middle of the character at [offset], in the panel. */
82     private fun OverlayView.middleOf(offset: Int): Float {
83         val layout = text.layout
84         return text.totalPaddingLeft + (layout.getPrimaryHorizontal(offset) + layout.getPrimaryHorizontal(offset + 1)) / 2
85     }
86 
87     /** A y just over the baseline of the row of the character at [offset], in the panel. */
88     private fun OverlayView.rowOf(offset: Int): Float =
89         text.totalPaddingTop + text.layout.getLineBaseline(text.layout.getLineForOffset(offset)) - 5f

194     @Test
195     fun theLineFollowsThePlayerThroughPauseSeekSpeedAndOffset() {
196         val session = MediaSession(context, "test player")
197         lateinit var follower: Follower
198         try {
199             session.setPlaybackState(state(PlaybackState.STATE_PAUSED, 1_500))
200             session.isActive = true
201             onMain {
202                 follower = Follower(main) { at, _, _ -> lines += follower.index.cues.getOrNull(at)?.text }
203                 follower.index = index
204                 follower.follow(listOf(MediaController(context, session.sessionToken)))
205             }
206             waitFor("one")
207 
208             // A seek. The player is paused: the line changes at once and then stays.
209             session.setPlaybackState(state(PlaybackState.STATE_PAUSED, 500))
210             waitFor(null)
211 
212             // Play at double speed from 59.6 s: "two" is 0.2 s away, "three" 0.2 s after it.
213             val started = SystemClock.elapsedRealtime()
214             session.setPlaybackState(state(PlaybackState.STATE_PLAYING, 59_600, speed = 2f))
215             waitFor("one")
216             waitFor("two")
217             waitFor("three")
218             val took = SystemClock.elapsedRealtime() - started
219             assertTrue("the follower waited $took ms; at double speed 0.8 s of media is 0.4 s", took in 350..1_500)
220 
221             // The subtitles are 60 s late for this media: the user shifts them.
222             session.setPlaybackState(state(PlaybackState.STATE_PAUSED, 100))
223             waitFor(null)
224             onMain { follower.offsetMs = 60_000 }
225             waitFor("two")
226 
227             // "line ▶" makes the next line the line of now.
228             onMain { follower.shiftLines(1) }
229             waitFor("three")
230             onMain { assertEquals(60_400L - 100L, follower.offsetMs) }
231         } finally {
232             onMain { follower.stop() }
233             session.release()
