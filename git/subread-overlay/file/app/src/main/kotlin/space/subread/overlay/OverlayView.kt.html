23 /**
24  * The panel over the media player: the subtitle line, and a few buttons.
25  *
26  * A tap on the line selects the word under the finger. A drag makes the selection longer, word
27  * by word. The selection goes to a dictionary with "Look up". The system text selection is not
28  * used: it needs a window with the input focus, and then the player below gets no keys.
29  *
30  * Black on white and nothing that moves, so that the panel is usable on an e-ink screen.
31  */

129     fun showLine(value: String, before: String? = null, after: String? = null) {
130         val whole = listOfNotNull(before, value, after).joinToString("\n")
131         if (whole == line) return
132         line = whole
133         around = buildList {
134             if (before != null) add(0 until before.length)
135             if (after != null) add(whole.length - after.length until whole.length)

151     /** Selects the word at the point [x], [y] of the text view. For a tap, and for tests. */
152     fun selectAt(x: Float, y: Float, extend: Boolean = false) {
153         val offset = offsetAt(x, y) ?: return select(if (extend) selection else null)
154         val from = anchor
155         if (extend && from != null) {
156             select(Words.between(line, from, offset))
157         } else {
158             anchor = Words.at(line, offset)
159             select(anchor)
160         }
161     }

163     private fun select(range: IntRange?) {
164         selection = range
165         if (range == null) anchor = null
166         val marked = SpannableString(line)
167         for (part in around) {
168             marked.setSpan(ForegroundColorSpan(Color.DKGRAY), part.first, part.last + 1, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
169             marked.setSpan(RelativeSizeSpan(0.85f), part.first, part.last + 1, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
170         }
171         if (range != null) {
172             marked.setSpan(BackgroundColorSpan(Color.BLACK), range.first, range.last + 1, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
173             marked.setSpan(ForegroundColorSpan(Color.WHITE), range.first, range.last + 1, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
174         }
175         text.text = marked
176         lookUpRow.visibility = if (range == null) View.GONE else View.VISIBLE
177         if (range == null) events.onSelectionCleared()
178     }

180     /** The index of the character under the point; null when the point is not on a character. */
181     private fun offsetAt(x: Float, y: Float): Int? {
182         val layout = text.layout ?: return null
183         if (line.isEmpty()) return null
184         val inX = x - text.totalPaddingLeft
185         val inY = y - text.totalPaddingTop
186         if (inY < 0 || inY > layout.height) return null
187         val row = layout.getLineForVertical(inY.toInt())
188         if (inX < layout.getLineLeft(row) || inX > layout.getLineRight(row)) return null
189         // The offset of a point is a place between two characters. The character under the
190         // finger is the one before that place when the finger is on its right half.
191         val between = layout.getOffsetForHorizontal(row, inX)
192         val under = if (between > layout.getLineStart(row) && layout.getPrimaryHorizontal(between) > inX) between - 1 else between
193         return under.coerceIn(0, line.length - 1)
194     }

211     @SuppressLint("ClickableViewAccessibility") // The buttons of the panel do the same for a screen reader.
212     private fun selectWith(view: TextView) {
213         view.setOnTouchListener { _, event ->
214             when (event.actionMasked) {
215                 MotionEvent.ACTION_DOWN -> selectAt(event.x, event.y)
216                 MotionEvent.ACTION_MOVE -> selectAt(event.x, event.y, extend = true)
217             }
218             true
219         }
220     }

222     @SuppressLint("ClickableViewAccessibility") // A drag has no equal for a screen reader; the panel works where it is.
223     private fun dragWith(handle: View) {
224         var lastX = 0f
225         var lastY = 0f
226         handle.setOnTouchListener { _, event ->
227             when (event.actionMasked) {
228                 MotionEvent.ACTION_DOWN -> {
229                     lastX = event.rawX
230                     lastY = event.rawY
231                 }
232                 MotionEvent.ACTION_MOVE -> {
233                     events.onDrag(event.rawX - lastX, event.rawY - lastY)
234                     lastX = event.rawX
235                     lastY = event.rawY
236                 }
237                 MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> events.onDragEnd()
238             }
239             true
240         }
241     }
