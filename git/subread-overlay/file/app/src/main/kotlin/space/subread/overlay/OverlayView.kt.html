23 /**
24  * The panel over the media player: the subtitle line, and a few buttons.
25  *
26  * A tap on the line selects the word under the finger and pauses the player at once. A drag
27  * makes the selection longer, word by word. When the finger lifts, the selection goes to the
28  * dictionary. When the dictionary closes, the selection goes away. The system text selection is not used: it needs a window with the input focus,
29  * and then the player below gets no keys.
30  *
31  * With SubRead Anki installed, the selection row has an "Anki" button: one tap makes a card
32  * from the word and the line.
33  *
34  * Black on white and nothing that moves, so that the panel is usable on an e-ink screen.
35  */
36 @SuppressLint("ViewConstructor", "SetTextI18n")
37 class OverlayView(context: Context, private val events: Events, anki: Boolean = false) : LinearLayout(context) {

138     fun showLine(value: String, before: String? = null, after: String? = null) {
139         val whole = listOfNotNull(before, value, after).joinToString("\n")
140         if (whole == line) return
141         line = whole
142         around = buildList {
143             if (before != null) add(0 until before.length)
144             if (after != null) add(whole.length - after.length until whole.length)
145         }
146         text.setTypeface(Typeface.DEFAULT, Typeface.NORMAL)
147         select(null)
148     }

160     /** Selects the word at the point [x], [y] of the text view. For a tap, and for tests. */
161     fun selectAt(x: Float, y: Float, extend: Boolean = false) {
162         val offset = offsetAt(x, y) ?: return select(if (extend) selection else null)
163         val from = anchor
164         if (extend && from != null) {
165             select(Words.between(line, from, offset))
166         } else {
167             anchor = Words.at(line, offset)
168             select(anchor)
169         }
170     }

172     private fun select(range: IntRange?) {
173         selection = range
174         if (range == null) anchor = null
175         val marked = SpannableString(line)
176         for (part in around) {
177             marked.setSpan(ForegroundColorSpan(Color.DKGRAY), part.first, part.last + 1, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
178             marked.setSpan(RelativeSizeSpan(0.85f), part.first, part.last + 1, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
179         }
180         if (range != null) {
181             marked.setSpan(BackgroundColorSpan(Color.BLACK), range.first, range.last + 1, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
182             marked.setSpan(ForegroundColorSpan(Color.WHITE), range.first, range.last + 1, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
183         }
184         text.text = marked
185         lookUpRow.visibility = if (range == null) View.GONE else View.VISIBLE
186         if (range == null) events.onSelectionCleared()
187     }

189     /** The index of the character under the point; null when the point is not on a character. */
190     private fun offsetAt(x: Float, y: Float): Int? {
191         val layout = text.layout ?: return null
192         if (line.isEmpty()) return null
193         val inX = x - text.totalPaddingLeft
194         val inY = y - text.totalPaddingTop
195         if (inY < 0 || inY > layout.height) return null
196         val row = layout.getLineForVertical(inY.toInt())
197         if (inX < layout.getLineLeft(row) || inX > layout.getLineRight(row)) return null
198         // The offset of a point is a place between two characters. The character under the
199         // finger is the one before that place when the finger is on its right half.
200         val between = layout.getOffsetForHorizontal(row, inX)
201         val under = if (between > layout.getLineStart(row) && layout.getPrimaryHorizontal(between) > inX) between - 1 else between
202         return under.coerceIn(0, line.length - 1)
203     }

224     @SuppressLint("ClickableViewAccessibility") // The buttons of the panel do the same for a screen reader.
225     private fun selectWith(view: TextView) {
226         view.setOnTouchListener { _, event ->
227             when (event.actionMasked) {
228                 MotionEvent.ACTION_DOWN -> {
229                     selectAt(event.x, event.y)
230                     if (selection != null) events.onTouchWord()
231                 }
232                 MotionEvent.ACTION_MOVE -> selectAt(event.x, event.y, extend = true)
233                 MotionEvent.ACTION_UP -> lookUp()
234             }
235             true
236         }
237     }

239     @SuppressLint("ClickableViewAccessibility") // A drag has no equal for a screen reader; the panel works where it is.
240     private fun dragWith(handle: View) {
241         var lastX = 0f
242         var lastY = 0f
243         handle.setOnTouchListener { _, event ->
244             when (event.actionMasked) {
245                 MotionEvent.ACTION_DOWN -> {
246                     lastX = event.rawX
247                     lastY = event.rawY
248                 }
249                 MotionEvent.ACTION_MOVE -> {
250                     events.onDrag(event.rawX - lastX, event.rawY - lastY)
251                     lastX = event.rawX
252                     lastY = event.rawY
253                 }
254                 MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> events.onDragEnd()
255             }
256             true
257         }
258     }
