162     /** The apps that take a selected text, with the name that each one gives the entry. */
163     private fun dictionaries(): List<Pair<String, ComponentName>> =
164         packageManager.queryIntentActivities(Lookup.probe(), 0).map {
165             val app = it.activityInfo.applicationInfo.loadLabel(packageManager)
166             val entry = it.loadLabel(packageManager)
167             val name = if (entry.toString() == app.toString()) "$app" else "$app: $entry"
168             name to ComponentName(it.activityInfo.packageName, it.activityInfo.name)
169         }.sortedBy { it.first }
170 
171     private fun chooseDictionary() {
172         val found = dictionaries()
173         if (found.isEmpty()) {
174             AlertDialog.Builder(this).setMessage(R.string.dictionary_none).setPositiveButton(android.R.string.ok, null).show()
175             return
176         }
177         val names = listOf(getString(R.string.dictionary_ask)) + found.map { it.first }
178         AlertDialog.Builder(this).setItems(names.toTypedArray()) { _, which ->
179             store.dictionary = found.getOrNull(which - 1)?.second
180             draw()
181         }.show()
182     }

184     @Deprecated("The platform Activity has no other result API, and this app has no AndroidX activity.")
185     override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
186         val uri = data?.data
187         if (requestCode != PICK_SUBTITLES || resultCode != RESULT_OK || uri == null) return
188         // The listener reads the file later, also after a restart of the device.
189         runCatching { contentResolver.takePersistableUriPermission(uri, Intent.FLAG_GRANT_READ_URI_PERMISSION) }
190         val lines = runCatching {
191             contentResolver.openInputStream(uri)!!.use { Srt.parse(it.readBytes().decodeToString()) }.size
192         }.getOrDefault(0)
193         store.subtitles = uri
194         store.subtitlesName = resources.getQuantityString(R.plurals.step_file_cues, lines, displayName(uri), lines)
195         MediaListener.instance?.reload()
196         draw()
197     }
