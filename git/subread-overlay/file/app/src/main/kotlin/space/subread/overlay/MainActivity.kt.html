196     /** The apps that take a selected text, with the name that each one gives the entry. */
197     private fun dictionaries(): List<Pair<String, ComponentName>> =
198         packageManager.queryIntentActivities(Lookup.probe(), 0).map {
199             val app = it.activityInfo.applicationInfo.loadLabel(packageManager)
200             val entry = it.loadLabel(packageManager)
201             val name = if (entry.toString() == app.toString()) "$app" else "$app: $entry"
202             name to ComponentName(it.activityInfo.packageName, it.activityInfo.name)
203         }.sortedBy { it.first }
204 
205     private fun chooseDictionary() {
206         val found = dictionaries()
207         if (found.isEmpty()) {
208             AlertDialog.Builder(this).setMessage(R.string.dictionary_none).setPositiveButton(android.R.string.ok, null).show()
209             return
210         }
211         val names = listOf(getString(R.string.dictionary_ask)) + found.map { it.first }
212         AlertDialog.Builder(this).setItems(names.toTypedArray()) { _, which ->
213             store.dictionary = found.getOrNull(which - 1)?.second
214             draw()
215         }.show()
216     }

218     @Deprecated("The platform Activity has no other result API, and this app has no AndroidX activity.")
219     override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
220         val uri = data?.data
221         if (requestCode != PICK_SUBTITLES || resultCode != RESULT_OK || uri == null) return
222         // The listener reads the file later, also after a restart of the device.
223         runCatching { contentResolver.takePersistableUriPermission(uri, Intent.FLAG_GRANT_READ_URI_PERMISSION) }
224         val lines = runCatching {
225             contentResolver.openInputStream(uri)!!.use { Srt.parse(it.readBytes().decodeToString()) }.size
226         }.getOrDefault(0)
227         store.subtitles = uri
228         store.subtitlesName = resources.getQuantityString(R.plurals.step_file_cues, lines, displayName(uri), lines)
229         MediaListener.instance?.reload()
230         draw()
231     }
