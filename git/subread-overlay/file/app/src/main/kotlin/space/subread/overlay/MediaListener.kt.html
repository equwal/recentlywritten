19 /**
20  * Holds the panel, and follows the media sessions of the device.
21  *
22  * Android gives the media sessions of other apps only to a notification listener. This service
23  * reads no notification: it has no `onNotificationPosted`. The system keeps a listener running,
24  * so the panel needs no foreground service and no notification of its own.
25  *
26  * A caption app can send lines through [PlayerProvider]. While it does, the panel shows its
27  * newest line and not the subtitle file.
28  */
29 class MediaListener : NotificationListenerService(), OverlayView.Events {

34     private var panel: OverlayView? = null

37     /** True while a caption app sends lines. The follower then does not draw on the panel. */
38     private var liveOn = false
39 
40     /** True when a live line came while a word was selected: the panel shows it after the selection. */
41     private var livePending = false
42     private val liveTimeout = Runnable { endLive() }
43     private val params = WindowManager.LayoutParams(
44         WindowManager.LayoutParams.MATCH_PARENT,
45         WindowManager.LayoutParams.WRAP_CONTENT,
46         WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
47         // The player below keeps the keys and each touch that is not on the panel.
48         WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or WindowManager.LayoutParams.FLAG_NOT_TOUCH_MODAL,

52     private val sessions by lazy { getSystemService(MediaSessionManager::class.java) }
53     private val me by lazy { ComponentName(this, MediaListener::class.java) }
54     private val onSessions = MediaSessionManager.OnActiveSessionsChangedListener {
55         // A hidden panel follows nothing: no work while the user does not read along.
56         if (panel != null) follower.follow(it.orEmpty())
57     }

59     override fun onListenerConnected() {
60         store = Store(this)
61         follower = Follower(handler) { at, hasPlayer, playing -> show(at, hasPlayer, playing) }
62         sessions.addOnActiveSessionsChangedListener(onSessions, me, handler)
63         instance = this
64         if (store.shown) showPanel()
65     }
66 
67     override fun onListenerDisconnected() {
68         instance = null
69         sessions.removeOnActiveSessionsChangedListener(onSessions)
70         follower.stop()
71         handler.removeCallbacks(liveTimeout)
72         removePanel()
73     }

75     /** Puts the panel on the screen. False when the user did not allow the panel yet. */
76     fun showPanel(): Boolean {
77         if (!Settings.canDrawOverlays(this)) return false
78         store.shown = true
79         if (panel == null) {
80             params.x = store.x
81             params.y = store.y
82             panel = OverlayView(this, this).also { getSystemService(WindowManager::class.java).addView(it, params) }
83         }
84         reload()
85         return true
86     }

88     fun hidePanel() {
89         store.shown = false
90         follower.stop()
91         removePanel()
92     }

94     /** Reads the settings and the subtitle file again. */
95     fun reload() {
96         val view = panel ?: return
97         view.setTextSize(store.textSizeSp)
98         view.setTransparency(store.transparencyPercent)
99         view.showOffset(store.offsetMs)
100         follower.offsetMs = store.offsetMs
101         val file = store.subtitles
102         if (file == null) {
103             follower.index = CueIndex(emptyList())
104             return
105         }
106         thread(name = "subtitles") {
107             val cues = runCatching {
108                 contentResolver.openInputStream(file)!!.use { Srt.parse(it.readBytes().decodeToString()) }
109             }.getOrDefault(emptyList())
110             handler.post {
111                 follower.index = CueIndex(cues)
112                 follower.follow(sessions.getActiveSessions(me))
113             }
114         }
115     }

117     private fun removePanel() {
118         panel?.let { getSystemService(WindowManager::class.java).removeView(it) }
119         panel = null
120     }

122     /**
123      * A line from a caption app, from any thread. Returns [PlayerProvider.LIVE_OK] when the panel
124      * shows it, else the reason: the panel needs the overlay permission, and the user must have
125      * it on the screen. A panel that the user closed does not come back for a line.
126      */
127     fun liveLine(text: String, partial: Boolean): String {
128         if (!Settings.canDrawOverlays(this)) return PlayerProvider.ERROR_NO_OVERLAY
129         if (!store.shown) return PlayerProvider.ERROR_PANEL_HIDDEN
130         handler.post {
131             if (!liveOn) {
132                 liveOn = true
133                 live.clear()
134             }
135             live.line(text, partial)
136             // A caption app that dies without `end` must not leave its last line for ever.
137             handler.removeCallbacks(liveTimeout)
138             handler.postDelayed(liveTimeout, LIVE_TIMEOUT_MS)
139             if (panel == null) showPanel() else showLive()
140         }
141         return PlayerProvider.LIVE_OK
142     }
143 
144     /** The caption app stopped: the panel goes back to the subtitle file. From any thread. */
145     fun endLive(): String {
146         handler.post {
147             if (!liveOn) return@post
148             liveOn = false
149             livePending = false
150             live.clear()
151             handler.removeCallbacks(liveTimeout)
152             if (panel != null) reload()
153         }
154         return PlayerProvider.LIVE_OK
155     }
156 
157     /**
158      * Shows the newest live line. While a word is selected, the line waits: a line that changes
159      * under the finger would take the selection away before the lookup.
160      */
161     private fun showLive() {
162         val view = panel ?: return
163         if (view.selection != null) {
164             livePending = true
165             return
166         }
167         livePending = false
168         val now = if (live.partial) live.now + " …" else live.now
169         if (store.linesAround) view.showLine(now, live.before, null) else view.showLine(now)
170     }
171 
172     private fun show(at: Int, hasPlayer: Boolean, playing: Boolean) {
173         val view = panel ?: return
174         if (liveOn) {
175             showLive()
176             return
177         }
178         view.showPlaying(playing)
179         val cues = follower.index.cues
180         val line = cues.getOrNull(at)?.text
181         when {
182             store.subtitles == null -> view.showStatus(getString(R.string.status_no_file))
183             cues.isEmpty() -> view.showStatus(getString(R.string.status_empty_file))
184             !hasPlayer -> view.showStatus(getString(R.string.status_no_player))
185             line == null -> view.showStatus(getString(R.string.status_before_first_line))
186             store.linesAround -> view.showLine(line, cues.getOrNull(at - 1)?.text, cues.getOrNull(at + 1)?.text)
187             else -> view.showLine(line)
188         }
189     }
