+  (let ((destination (gethash number *links*)))
+    (if (not destination)

+        (format t "gopher://~a~a/~a~a~%"

+                (let ((port (location-port destination)))
+                  (if (= 70 port)
+                      ""
+                      (format nil ":~a" port)))
+                (location-type destination)
