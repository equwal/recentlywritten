43 #-win32
44 (eval-when (:compile-toplevel :load-toplevel :execute)
45   (pushnew (with-standard-io-syntax
46              (read-from-string
47               (format nil ":~(~A~)" (posix:uname-sysname (posix:uname)))))
48            *features*))

56 (let ((cpu (cond
57              ((or (member :pc386 *features*)
58                   (member (machine-type) '("x86" "x86_64")
59                           :test #'string-equal))
60               (if (member :word-size=64 *features*)
61                   '#:x86-64
62                   '#:x86))
63              ((string= (machine-type) "POWER MACINTOSH")
64               '#:ppc)
65              ((or (member (machine-type) '("SPARC" "SPARC64")
66                           :test #'string-equal))
67               (if (member :word-size=64 *features*)
68                   '#:sparc64
69                   '#:sparc)))))
70   (when cpu
71     (pushnew (intern (symbol-name cpu) '#:keyword)
72              *features*)))
