37            (let ((ptr (fli:alloca :type :byte :nelems 2)))
38              (setf (fli:dereference ptr :type '(:unsigned :short)) #xfeff)
39              (ecase (fli:dereference ptr :type '(:unsigned :byte))
40                (#xfe (intern "BIG-ENDIAN" :keyword))
41                (#xff (intern "LITTLE-ENDIAN" :keyword)))))
