57 (define-constant +ebcdic-encode-table+
58   (loop with rt = (make-array 256 :element-type 'ub8 :initial-element 0)
59         for code across +ebcdic-decode-table+ for i from 0 do
60         (assert (= 0 (aref rt code)))
61         (setf (aref rt code) i)
62         finally (return rt))
63   :test #'equalp)

65 (define-unibyte-encoder :ebcdic-us (code)
66   (if (>= code 256)
67       (handle-error)
68       (aref +ebcdic-encode-table+ code)))

70 (define-unibyte-decoder :ebcdic-us (octet)
71   (aref +ebcdic-decode-table+ octet))
