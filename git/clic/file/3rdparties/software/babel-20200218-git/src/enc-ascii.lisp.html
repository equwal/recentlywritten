29 (define-character-encoding :ascii
30     "A 7-bit, fixed-width character encoding in which all
31 character codes map to their Unicode equivalents."
32   :aliases '(:us-ascii)
33   :literal-char-code-limit 128)

35 (define-unibyte-encoder :ascii (code)
36   (if (>= code 128)
37       (handle-error)
38       code))
39 
40 (define-unibyte-decoder :ascii (octet)
41   (if (>= octet 128)
42       (handle-error)
