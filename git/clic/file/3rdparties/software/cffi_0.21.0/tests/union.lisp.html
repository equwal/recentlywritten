30 (defcunion uint32-bytes
31   (int-value :unsigned-int)
32   (bytes :unsigned-char :count 4))
33 
34 (defctype uint32-bytes (:union uint32-bytes))
35 
36 (defun int-to-bytes (n)
37   "Convert N to a list of bytes using a union."
38   (with-foreign-object (obj 'uint32-bytes)
39     (setf (foreign-slot-value obj 'uint32-bytes 'int-value) n)
40     (loop for i from 0 below 4
41           collect (mem-aref
42                    (foreign-slot-value obj 'uint32-bytes 'bytes)
43                    :unsigned-char i))))
