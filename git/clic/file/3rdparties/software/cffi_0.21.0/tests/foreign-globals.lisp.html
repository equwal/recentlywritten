117 ;; The *.set.* tests restore the old values so that the *.ref.*
118 ;; don't fail when re-run.
119 (defmacro with-old-value-restored ((place) &body body)
120   (let ((old (gensym)))
121     `(let ((,old ,place))
122        (prog1
123            (progn ,@body)
124          (setq ,place ,old)))))
