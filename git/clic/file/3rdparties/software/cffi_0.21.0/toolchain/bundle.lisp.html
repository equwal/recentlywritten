78   ;; Dependencies of a gather-op are the actions of the dependent operation
79   ;; for all the (sorted) required components for loading the system.
80   ;; Monolithic operations typically use lib-op as the dependent operation,
81   ;; and all system-level dependencies as required components.
82   ;; Non-monolithic operations typically use compile-op as the dependent operation,
83   ;; and all transitive sub-components as required components (excluding other systems).
84   (defmethod component-depends-on ((o gather-operation) (s system))
85     (let* ((mono (operation-monolithic-p o))
86            (go (make-operation (or (gather-operation o) 'compile-op)))
87            (bundle-p (typep go 'bundle-op))
88            ;; In a non-mono operation, don't recurse to other systems.
89            ;; In a mono operation gathering bundles, don't recurse inside systems.
90            (component-type (if mono (if bundle-p 'system t) '(not system)))
91            ;; In the end, only keep system bundles or non-system bundles, depending.
92            (keep-component (if bundle-p 'system '(not system)))
93            (deps
94             ;; Required-components only looks at the dependencies of an action, excluding the action
95             ;; itself, so it may be safely used by an action recursing on its dependencies (which
96             ;; may or may not be an overdesigned API, since in practice we never use it that way).
97             ;; Therefore, if we use :goal-operation 'load-op :keep-operation 'load-op, which looks
98             ;; cleaner, we will miss the load-op on the requested system itself, which doesn't
99             ;; matter for a regular system, but matters, a lot, for a package-inferred-system.
100             ;; Using load-op as the goal operation and basic-compile-op as the keep-operation works
101             ;; for our needs of gathering all the files we want to include in a bundle.
102             ;; Note that we use basic-compile-op rather than compile-op so it will still work on
103             ;; systems when *load-system-operation* is load-bundle-op.
104             (required-components
105              s :other-systems mono :component-type component-type :keep-component keep-component
106              :goal-operation 'load-op :keep-operation 'basic-compile-op)))
107       `((,go ,@deps) ,@(call-next-method))))

212   ;; From the ASDF-internal bundle-type identifier, get a filesystem-usable pathname type.
213   (defun bundle-pathname-type (bundle-type)
214     (etypecase bundle-type
215       ((or null string) ;; pass through nil or string literal
216        bundle-type)
217       ((eql :no-output-file) ;; marker for a bundle-type that has NO output file
218        (error "No output file, therefore no pathname type"))
219       ((eql :fasl) ;; the type of a fasl
220        #-(or clasp ecl mkcl) (compile-file-type) ; on image-based platforms, used as input and output
221        #+(or clasp ecl mkcl) "fasb") ; on C-linking platforms, only used as output for system bundles
222       ((member :image)
223        #+allegro "dxl"
224        #+(and clisp os-windows) "exe"
225        #-(or allegro (and clisp os-windows)) "image")
226       ;; NB: on CLASP and ECL these implementations, we better agree with
227       ;; (compile-file-type :type bundle-type))
228       ((eql :object) ;; the type of a linkable object file
229        (os-cond ((os-unix-p) "o")
230                 ((os-windows-p) (if (featurep '(:or :mingw32 :mingw64)) "o" "obj"))))
231       ((member :lib :static-library) ;; the type of a linkable library
232        (os-cond ((os-unix-p) "a")
233                 ((os-windows-p) (if (featurep '(:or :mingw32 :mingw64)) "a" "lib"))))
234       ((member :dll :shared-library) ;; the type of a shared library
235        (os-cond ((os-macosx-p) "dylib") ((os-unix-p) "so") ((os-windows-p) "dll")))
236       ((eql :program) ;; the type of an executable program
237        (os-cond ((os-unix-p) nil) ((os-windows-p) "exe")))))

241     (let ((bundle-type (bundle-type o)))
242       (unless (or (eq bundle-type :no-output-file) ;; NIL already means something regarding type.
243                   (and (null (input-files o c)) (not (member bundle-type '(:image :program)))))

426   (defmethod perform ((o deliver-asd-op) (s system))
427     (let* ((inputs (input-files o s))
428            (fasl (first inputs))
429            (library (second inputs))
430            (asd (first (output-files o s)))
431            (name (if (and fasl asd) (pathname-name asd) (return-from perform)))
432            (version (component-version s))
433            (dependencies
434              (if (operation-monolithic-p o)
435                  ;; We want only dependencies, and we use basic-load-op rather than load-op so that
436                  ;; this will keep working on systems when *load-system-operation* is load-bundle-op
437                  (remove-if-not 'builtin-system-p
438                                 (required-components s :component-type 'system
439                                                        :keep-operation 'basic-load-op))
440                  (while-collecting (x) ;; resolve the sideway-dependencies of s
441                    (map-direct-dependencies
442                     t 'load-op s
443                     #'(lambda (o c)
444                         (when (and (typep o 'load-op) (typep c 'system))
445                           (x c)))))))
446            (depends-on (mapcar 'coerce-name dependencies)))
447       (when (pathname-equal asd (system-source-file s))
448         (cerror "overwrite the asd file"
449                 "~/asdf-action:format-action/ is going to overwrite the system definition file ~S ~
450 which is probably not what you want; you probably need to tweak your output translations."
451                 (cons o s) asd))
452       (with-open-file (s asd :direction :output :if-exists :supersede
453                              :if-does-not-exist :create)
454         (format s ";;; Prebuilt~:[~; monolithic~] ASDF definition for system ~A~%"
455                 (operation-monolithic-p o) name)
456         (format s ";;; Built for ~A ~A on a ~A/~A ~A~%"
457                 (lisp-implementation-type)
458                 (lisp-implementation-version)
459                 (software-type)
460                 (machine-type)
461                 (software-version))
462         (let ((*package* (find-package :asdf-user)))
463           (pprint `(defsystem ,name
464                      :class prebuilt-system
465                      :version ,version
466                      :depends-on ,depends-on
467                      :components ((:compiled-file ,(pathname-name fasl)))
468                      ,@(when library `(:lib ,(file-namestring library))))
469                   s)
470           (terpri s)))))
