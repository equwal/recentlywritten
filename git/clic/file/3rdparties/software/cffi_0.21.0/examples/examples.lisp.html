37 ;; A simple libc function.

41 ;; This definition uses the STRING type translator to automatically
42 ;; convert Lisp strings to foreign strings and vice versa.
43 (defcfun "getenv" :string
44   (name :string))
45 
46 ;; Calling a varargs function.
47 (defun sprintf-test ()
48   "Test calling a varargs function."
49   (with-foreign-pointer-as-string ((buf buf-size) 255)
50     (foreign-funcall
51      "snprintf" :pointer buf :int buf-size
52      :string "%d %f #x%x!" :int 666
53      :double (coerce pi 'double-float)
54      :unsigned-int #xcafebabe
55      :void)))
56 
57 ;; Defining an emerated type.
58 (defcenum test-enum
59   (:invalid 0)
60   (:positive 1)
61   (:negative -1))
62 
63 ;; Use the absolute value function to test keyword/enum translation.
64 (defcfun ("abs" c-abs) test-enum
65   (n test-enum))

67 (defun cffi-version ()
68   (asdf:component-version (asdf:find-system 'cffi)))

70 (defun run-examples ()
71   (format t "~&;;; CFFI version ~A on ~A ~A:~%"
72           (cffi-version) (lisp-implementation-type)
73           (lisp-implementation-version))
74   (format t "~&;; shell:             ~A~%" (getenv "SHELL"))
75   (format t "~&;; sprintf test:      ~A~%" (sprintf-test))
76   (format t "~&;; (c-abs :positive): ~A~%" (c-abs :positive))
77   (format t "~&;; (c-abs :negative): ~A~%" (c-abs :negative))
78   (force-output))
