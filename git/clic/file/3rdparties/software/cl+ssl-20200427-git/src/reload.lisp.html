24 ;; Windows builds have been naming librypto and libssl DLLs in several different ways:
25 ;;
26 ;; - libeay32.dll, libssl32.dll
27 ;; - libeay32.dll, ssleay32.dll
28 ;;
29 ;; Note, the above names were used both for 32 and 64 -bit versions.
30 ;;
31 ;; - libcrypto-1_1-x64.dll, libssl-1_1-x64.dll
32 ;;
33 ;; The above are used for 64-bit only.
34 ;;
35 ;; - libcrypto-1_1.dll, libssl-1_1.dll
