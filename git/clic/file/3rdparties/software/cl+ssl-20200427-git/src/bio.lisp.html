92 (cffi:defcallback lisp-read :int ((bio :pointer) (buf :pointer) (n :int))
93   bio buf n
94   (let ((i 0))
95     (handler-case
96   (unless (or (cffi:null-pointer-p buf) (null n))
97     (clear-retry-flags bio)
98     (when (or *blockp* (listen *socket*))
99             (setf (cffi:mem-ref buf :unsigned-char i) (read-byte *socket*))
100             (incf i))
101     (loop
102         while (and (< i n)
103                          (or (null *partial-read-p*) (listen *socket*)))
104         do
105     (setf (cffi:mem-ref buf :unsigned-char i) (read-byte *socket*))
106     (incf i))
107     #+(or)
108     (when (zerop i) (set-retry-read bio)))
109       (end-of-file ()))
110     i))
