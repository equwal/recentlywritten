12 (defun random-bytes (count)
13   "Generates COUNT cryptographically strong pseudo-random bytes. Returns
14 the bytes as a SIMPLE-ARRAY with ELEMENT-TYPE '(UNSIGNED-BYTE 8). Signals
15 an ERROR in case of problems, for example when the OpenSSL random number
16 generator has not been seeded with enough randomness to ensure an
17 unpredictable byte sequence."
18   (let* ((result (make-array count :element-type '(unsigned-byte 8)))
19          (buf (make-buffer count))
20          (ret (with-pointer-to-vector-data (ptr buf)
21                 (rand-bytes ptr count))))
22     (when (/= 1 ret)
23       (error "RANDOM-BYTES failed: error reported by the OpenSSL RAND_bytes function. ~A."
24              (format-ssl-error-queue nil (read-ssl-error-queue))))
25     (s/b-replace result buf)))
