17 (defun sanitize-as-path (str)
18   ;; Substitute dots by hypens if our main process is CCL, it 
19   ;; prepends the > symbol before dots;
20   ;; for example: 1.1.0.36.mswinmt.1201-284e340 => 1>.1>.0>.36>.mswinmt.1201-284e340
21   ;; When we pass such a pathname to other lisps, they can't handle it.
22   (substitute #\- #\. str))
