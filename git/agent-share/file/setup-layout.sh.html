1 #!/bin/sh
2 # Create the agent folder layout in a mounted 9P store.
3 # Usage: AGENTFS=/mnt/agentfs AGENTS="claude codex copilot antigravity" ./setup-layout.sh
4 set -eu
5 : "${AGENTFS:?set AGENTFS to the mount point}"
6 AGENTS=${AGENTS:-"claude codex copilot antigravity"}
7 mkdir -p "$AGENTFS/shared" "$AGENTFS/handoff"
8 for a in $AGENTS; do mkdir -p "$AGENTFS/scratch/$a" "$AGENTFS/inbox/$a"; done
9 cp "$(dirname "$0")/RULES.md" "$AGENTFS/RULES.md"
10 ls -R "$AGENTFS"
