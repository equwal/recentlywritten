Decode subtitle files in UTF-8, UTF-16 or Shift_JIS

The reader read each subtitle file as UTF-8. A Shift_JIS copy of
sample-jpn.srt showed its cues as mojibake, and no cue matched the
book.

decodeSubtitles reads the bytes of the file. A UTF-8 BOM or valid UTF-8
gives UTF-8. A UTF-16 BOM gives UTF-16. Other bytes give Shift_JIS. The
text has no BOM.

+/**
+ * Decodes the bytes of a subtitle file to text, and removes the BOM.
+ *
+ * - A UTF-8 BOM, or bytes that are valid UTF-8: UTF-8.
+ * - A UTF-16 BOM: UTF-16, little endian or big endian as the BOM tells.
+ * - Other bytes: Shift_JIS, the usual encoding of old Japanese files.
+ *   Japanese text in Shift_JIS is almost never valid UTF-8.
+ */
+export function decodeSubtitles(bytes: ArrayBuffer): string {
+  const b = new Uint8Array(bytes);
+  // TextDecoder removes the BOM of its own encoding.
+  if (b[0] === 0xef && b[1] === 0xbb && b[2] === 0xbf) return new TextDecoder('utf-8').decode(b);
+  if (b[0] === 0xff && b[1] === 0xfe) return new TextDecoder('utf-16le').decode(b);
+  if (b[0] === 0xfe && b[1] === 0xff) return new TextDecoder('utf-16be').decode(b);
+  try {
+    return new TextDecoder('utf-8', { fatal: true }).decode(b);
+  } catch {
+    return new TextDecoder('shift_jis').decode(b);
+  }
+}

+function utf8(text: string): ArrayBuffer {
+  const bytes = new TextEncoder().encode(text);
+  return bytes.buffer.slice(bytes.byteOffset, bytes.byteOffset + bytes.byteLength);
+}

+function utf16(text: string, littleEndian: boolean): ArrayBuffer {
+  const view = new DataView(new ArrayBuffer(text.length * 2));
+  for (let i = 0; i < text.length; i++) view.setUint16(2 * i, text.charCodeAt(i), littleEndian);
+  return view.buffer;
+}
