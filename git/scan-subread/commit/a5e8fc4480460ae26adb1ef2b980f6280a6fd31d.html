Handle Back, the app state, the screen and player problems on Android

- Back destroyed the activity, and the reader lost the book. Back now
  closes the page jump, else the drawer, else puts the reader in the
  background (App.minimizeApp). It never finishes the activity
  (backStep). On the web the browser handles Back.
- In the background the clock stops, so no poll of the overlay runs,
  and the page goes to the meta data at once. Back in front, the clock
  reads the player at once, and the reader checks for a SubRead result
  and for the apps of the suite.
- The screen stays on while a book is open, the player plays and the
  follow is not off (keepScreenOn). The plugin gets a call only when
  the value changes.
- With SubRead Overlay installed but no notification access, play and
  "audio to page" stayed on, and the strip showed
  "PlayerError: no_notification_access". playerCommandsOff now turns
  both off for no_overlay, no_notification_access and, on the web,
  no_player. The strip gives an "Open SubRead Overlay" button for no
  notification access, and for an installed overlay that does not
  answer. Without the overlay, the link to its releases stays.

+/**
+ * The step of the Back button: close the page jump, close the drawer, or

+ */
+export type BackStep = 'close-jump' | 'close-drawer' | 'minimize';
+
+/**
+ * The step of the Back button. The reader never finishes its activity:
+ * Android then destroys the WebView, and the reader loses its state.
+ */
+export function backStep(open: { jump: boolean; drawer: boolean }): BackStep {
+  if (open.jump) return 'close-jump';
+  if (open.drawer) return 'close-drawer';
+  return 'minimize';
+}

+export function keepScreenOn(s: {
+  active: boolean;
+  book: boolean;
+  playing: boolean;
+  mode: FollowMode;
+}): boolean {
+  return s.active && s.book && s.playing && s.mode !== 'off';
+}
