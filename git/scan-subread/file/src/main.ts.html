338 /**
339  * Opens `file` as the book. The book that is open stays until the new
340  * file loads: a file that is not a PDF, or is damaged, changes nothing.
341  *
342  * `restore` comes from the restore of the last book: the book opens at the
343  * page of its meta data. An open without it is a file that the user
344  * picked, and its meta data comes from IndexedDB after the open. The page
345  * shows and the reading starts at once, and the meta data applies when it
346  * comes: an upgrade of the database can hold IndexedDB for many seconds.
347  * Gives true when the new book is open.
348  */
349 async function openBook(file: File, restore?: Restore): Promise<boolean> {
350   const meta = restore?.meta;
351   const attempt = beginOpen(opens, restore === undefined);
352   /** Ends the attempt that failed. The strip shows `text` only when failureShows allows it. */
353   const fail = (text: string): false => {
354     endOpen(opens, attempt, false);
355     claimSubtitles();
356     if (failureShows(opens, attempt, restore?.since)) say(text);
357     return false;
358   };
359   if (file.size === 0) return fail(EMPTY_FILE_TEXT);
360   let bytes: ArrayBuffer;
361   try {
362     bytes = await file.arrayBuffer();
363   } catch (err) {
364     // The copy of the last book is gone, for example after the storage was cleared.
365     return fail(restore ? LAST_BOOK_GONE_TEXT : `Cannot read ${file.name}: ${String(err)}`);
366   }
367   let pdf: PdfDoc;
368   try {
369     // pdf.js loads with the first PDF, so the start card shows sooner.
370     const { loadPdf } = await import('./pdf');
371     pdf = await loadPdf(bytes);
372   } catch (err) {
373     return fail(pdfOpenText(err));
374   }
375   // A file that the user picked later is open already.
376   if (!endOpen(opens, attempt, true)) {
377     void pdf.destroy();
378     claimSubtitles();
379     return false;
380   }
381   const key = bookKey(file);
382   // The page of the book before goes to its meta data now.
383   flushPageSave();
384   const old = state.pdf;
385   state.pdf = pdf;
386   state.file = file;
387   state.book = key;
388   state.pages = [];
389   state.tokens = [];
390   state.spans = [];
391   state.cueParts = [];
392   state.activeCue = -1;
393   state.markedCue = -1;
394   state.held = false;
395   state.changed = { page: false, forceOcr: false };
396   // The subtitles of another book go at once. Subtitles of no book, loaded
397   // while no book was open or while this book loaded, are for this book.
398   if (state.subtitles && state.subtitlesBook !== null && state.subtitlesBook !== key) {
399     clearSubtitles();
400   }
401   claimSubtitles();
402   const subtitlesSeq = state.subtitlesSeq;
403   // The SubRead status of the book before goes, unless a job runs.
404   if (!keepMakeStatus(key, state.making, state.makeWait?.book ?? null)) {
405     ui.makeStatus.textContent = '';
406   }
407   // A SubRead job that waits for the pages of another book does not start.
408   if (state.makeWait && state.makeWait.book !== key) state.makeWait = null;
409   setForceOcr(meta?.forceOcr ?? false);
410   ui.forceOcr.disabled = false;
411   ui.clearBook.disabled = false;
412   // One SubRead job at a time.
413   ui.makeSubs.disabled = state.making !== null;
414   ui.empty.hidden = true;
415   ui.page.hidden = false;
416   if (state.clock) showPlayerStatus(state.clock);
417   say(pdfLoadedText(pdf.numPages));
418   // showPage sets the current page at once, so the reading starts there.
419   // This page is the saved page, or page 1 until the meta data comes: no
420   // save, so page 1 does not go over the saved page of the book.
421   void showPage(validPage(meta?.page, pdf.numPages) ?? 0, false);
422   // readBook stops the reading of the old book before its document closes.
423   void readBook();
424   void old?.destroy();
425   showSrtRow();
426   if (meta) {
427     applyMeta(meta, attempt, subtitlesSeq);
428   } else {
429     void putLastBook(file);
430     void getMeta(key).then((m) => applyMeta(m, attempt, subtitlesSeq));
431   }
432   return true;
433 }

442 /**
443  * Applies the meta data of the book that open attempt `attempt` opened.
444  * `subtitlesSeq` is the count of subtitle loads at the open: subtitles
445  * that the user loaded after the open stay. A page turn and a change of
446  * "Force OCR" after the open stay too (lateMeta).
447  */
448 function applyMeta(meta: BookMeta, attempt: number, subtitlesSeq: number): void {
449   const key = state.book;
450   const pdf = state.pdf;
451   if (attempt !== opens.shown || key === null || !pdf) return;
452   const late = lateMeta(meta, state.changed);
453   const page = validPage(late.page, pdf.numPages);
454   // The saved page needs no new save.
455   if (page !== undefined && page !== state.currentPage) void showPage(page, false);
456   if (late.forceOcr !== undefined && late.forceOcr !== state.forceOcr) {
457     setForceOcr(late.forceOcr);
458     void readBook();
459   }
460   if (state.subtitlesSeq === subtitlesSeq) {
461     const loaded = state.subtitles ? state.subtitlesBook : undefined;
462     switch (subtitlesOnOpen(loaded, key, meta.subtitles !== undefined)) {
463       case 'keep':
464         claimSubtitles();
465         break;
466       case 'load':
467         if (meta.subtitles) loadSubtitles(meta.subtitles, { book: key, save: false });
468         break;
469       case 'clear':
470         clearSubtitles();
471         break;
472     }
473   }
474   // After the saved subtitles, so that a SubRead result asks first when it
475   // replaces a file of the user. A result that came while another book was
476   // open gets the same offer as the result for the open book.
477   const kept = meta.subread;
478   if (kept) {
479     void putMeta(key, { subread: undefined });
480     if (offerSubreadSubtitles(kept.srt, kept)) ui.makeStatus.textContent = subreadResultText(kept);
481   }
482   void checkPendingSubtitles();
483 }

520 /**
521  * Opens the last book at its page, with its subtitles. Android stops the
522  * reader while the user is in the dictionary or in the player app, and a
523  * reload forgets the file that the user opened.
524  *
525  * A book that the user opens while the restore waits for the database
526  * wins, also while it still loads (userOpened).
527  */
528 async function restoreLastBook(): Promise<void> {
529   const since = opens.byUser;
530   const file = await getLastBook();
531   if (!file || userOpened(opens, since)) {
532     showStart();
533     return;
534   }
535   const meta = await getMeta(bookKey(file));
536   if (userOpened(opens, since)) {
537     showStart();
538     return;
539   }
540   if (await openBook(file, { meta, since })) return;
541   // The copy cannot be read, or it is no PDF now: forget it. After an open
542   // by the user, the copy can be of the user's book, so it stays.
543   if (!userOpened(opens, since)) await clearLastBook(bookKey(file));
544   showStart();
545 }

569 /**
570  * The tokens of one page: from the cache, else the text layer, else OCR.
571  * `force` skips the text layer. Rejects with OcrStartError when the page
572  * needs OCR and OCR cannot start or its start takes too long, and with
573  * OcrTimeoutError when the OCR of the page takes too long.
574  */
575 async function readPage(
576   pdf: PdfDoc,
577   file: File,
578   page: number,
579   lang: string,
580   force: boolean,
581   ocr: OcrJob<HTMLCanvasElement>,
582 ): Promise<PageEntry> {
583   if (!force) {
584     const cached = await getPage(pageKey(file, page, lang, 'text'));
585     if (cached) return cached;
586   }
587   const cachedOcr = await getPage(pageKey(file, page, lang, 'ocr'));
588   if (cachedOcr) return cachedOcr;
589   if (!force) {
590     const text = await pdf.pageText(page, OCR_WIDTH);
591     if (text.tokens) {
592       const entry: PageEntry = { ...text, tokens: text.tokens, source: 'text' };
593       await putPage(pageKey(file, page, lang, 'text'), entry);
594       return entry;
595     }
596   }
597   // OCR starts before the render: when it cannot start, no page is rendered for nothing.
598   await ocr.start();
599   const canvas = await pdf.renderPage(page, OCR_WIDTH);
600   const tokens = await ocr.recognize(canvas, page, pageTimeLimit(canvas.width * canvas.height));
601   const entry: PageEntry = { tokens, width: canvas.width, height: canvas.height, source: 'ocr' };
602   await putPage(pageKey(file, page, lang, 'ocr'), entry);
603   return entry;
604 }

612 /**

617  * A page that cannot be read does not stop the other pages. At the end the
618  * strip tells how many pages could not be read, with Retry. `retry` reads
619  * only the pages that are not read.
620  */
621 async function readBook(retry = false): Promise<void> {
622   const { pdf, file } = state;
623   if (!pdf || !file) return;
624   const seq = ++state.readSeq;
625   readingRuns = true;
626   // The OCR of the reading before stops at once. It does not finish its
627   // page: two quick changes of a setting ran two OCR jobs at the same time.
628   readingOcr?.stop();
629   if (!retry) {
630     state.pages = [];
631     state.tokens = [];
632     // The spans index the old tokens, so they go too.
633     state.spans = [];
634     state.cueParts = [];
635     state.markedCue = -1;
636   }
637   // A SubRead job that waits shows the reading again, for example after Retry.
638   startMakeWhenRead();
639   const total = pdf.numPages;
640   const pending = new Set(
641     Array.from({ length: total }, (_, i) => i).filter((i) => !state.pages[i]),
642   );
643   const counts = { text: 0, ocr: 0 };
644   for (const entry of state.pages) if (entry) counts[entry.source]++;
645   const read = (): number => counts.text + counts.ocr;
646   let failed = 0;
647   let startFailed = false;
648   const lang = ui.lang.value;
649   const force = state.forceOcr;
650   /** The page that is read now. */
651   let page = -1;
652   const onOcrProgress = (p: number): void => {
653     if (seq === state.readSeq) {
654       setReading(readingText(read(), total, `page ${page + 1}: OCR ${Math.round(p * 100)}%`));
655     }
656   };
657   // The OCR worker, and tesseract.js itself, load on the first page that
658   // needs them. After a page that fails or takes too long, the job stops
659   // the worker, and the next page starts a new one. A start that takes too
660   // long gives each page the start error, and the strip shows Retry.
661   const ocr = createOcrJob<HTMLCanvasElement>((signal) =>
662     import('./ocr').then(
663       ({ createOcr }) => createOcr(lang, onOcrProgress, signal),
664       (err: unknown) => {
665         throw new OcrStartError(err);
666       },
667     ),
668   );
669   readingOcr = ocr;
670   try {
671     while (pending.size > 0) {
672       page = nextPage(pending, state.currentPage);
673       pending.delete(page);
674       setReading(readingText(read(), total, `page ${page + 1}`));
675       let entry: PageEntry;
676       try {
677         entry = await readPage(pdf, file, page, lang, force, ocr);
678       } catch (err) {
679         if (seq !== state.readSeq) return; // Another book or another setting took over.
680         failed++;
681         // Each page that needs OCR gives the same start error. Tell it once.
682         if (err instanceof OcrStartError) {
683           if (!startFailed) console.warn(err);
684           startFailed = true;
685         } else {
686           console.warn(`Page ${page + 1} could not be read.`, err);
687         }
688         continue;
689       }
690       if (seq !== state.readSeq) return; // Another book or another setting took over.
691       counts[entry.source]++;
692       state.pages[page] = entry;
693       // Put the tokens of the page at their place in page order. The spans
694       // keep their tokens until the next alignment: the mark and the Anki
695       // card use them.
696       const at = state.pages.slice(0, page).reduce((n, p) => n + (p?.tokens.length ?? 0), 0);
697       state.tokens.splice(at, 0, ...entry.tokens);
698       state.spans = shiftSpans(state.spans, at, entry.tokens.length);
699       const how = entry.source === 'text' ? 'text layer' : 'OCR';
700       setReading(readingText(read(), total, `page ${page + 1}: ${how}`));
701       if (page === state.currentPage) view.mark();
702       scheduleAlign();
703       startMakeWhenRead();
704     }
705     const matched = align();
706     say(
707       pagesReadText(counts) +
708         (matched === null ? '' : ` ${matchedText(matched, state.cues.length, failed === 0)}`),
709     );
710   } catch (err) {
711     if (seq === state.readSeq) say(`Cannot read the pages: ${String(err)}`);
712   } finally {
713     ocr.stop();
714     if (seq === state.readSeq) {
715       readingRuns = false;
716       setReading(readingEndMessage(failed, startFailed));
717       // A SubRead job that waits now waits for pages that could not be read.
718       startMakeWhenRead();
719     }
720   }
721 }

902  * Else tells how far the reading is, or after the reading, how many pages

919 /** Asks SubRead for the subtitles of the book `book`, and loads them after the checks. */
920 async function makeSubtitles(audio: { uri: string; name: string }, book: string): Promise<void> {
921   const name = resultName(book);
922   state.making = book;
923   ui.makeSubs.disabled = true;
924   ui.makeStatus.textContent = `SubRead makes the subtitles for ${audio.name}...`;
925   let result: SubtitlesResult;
926   try {
927     result = await SubRead.makeSubtitles({
928       audio: audio.uri,
929       bookText: bookText(state.tokens),
930       language: ui.subLang.value,
931       resultName: name,
932     });
933   } catch (err) {
934     result = { error: String(err) };
935   } finally {
936     state.making = null;
937     ui.makeSubs.disabled = state.book === null;
938   }
939   if (result.error === 'not_installed') {
940     fill(ui.makeStatus, SUBREAD_NOT_INSTALLED);
941     return;
942   }
943   // SubRead 0.10 and later also write the .srt into the result file. Read
944   // it now, so that the file goes, and use it when the answer has no .srt.
945   const copy = await SubRead.pendingSubtitles({ resultName: name }).catch(() => ({}) as PendingSrt);
946   const srt = result.srt ?? copy.srt;
947   if (srt === undefined) {
948     ui.makeStatus.textContent = subreadErrorText(result.error ?? 'no file');
949     return;
950   }
951   // Another book opened during the job: the result waits for its book,
952   // with what SubRead told about it, for the checks of the offer.
953   if (state.book !== book) {
954     const { cues, matchRate, language } = result;
955     keepResult(book, { srt, cues, matchRate, language });
956     return;
957   }
958   ui.makeStatus.textContent = subreadResultText(result);
959   offerSubreadSubtitles(srt, result);
960 }

1132 /**
1133  * The follow for the last state of the clock: mark the cue of now, and
1134  * turn to the page of the audio unless the follow is held. In the silence
1135  * between two cues, the cue before stays the cue of now. In a cue on two
1136  * pages, the page follows the time.
1137  */
1138 function runFollow(event: FollowEvent): void {
1139   const s = state.clock;
1140   const t = s?.positionMs == null ? null : s.positionMs / 1000;
1141   const cue = t === null ? -1 : lastCueAt(state.cues, t);
1142   if (cue !== state.activeCue) {
1143     markCueInList(cue);
1144     state.activeCue = cue;
1145   }
1146   const mode = followMode();
1147   const page =
1148     t === null || cue < 0 ? null : audioPage(state.cueParts, cue, cueProgress(state.cues[cue]!, t));
1149   const out = follow({
1150     mode,
1151     currentPage: state.currentPage,
1152     cue,
1153     matched: (state.cueParts[cue]?.length ?? 0) > 0,
1154     page,
1155     held: state.held,
1156     event,
1157   });
1158   state.held = out.held;
1159   // The Follow button shows only when it has a page to go to.
1160   const paused = out.held && mode !== 'off' && page !== null;
1161   ui.followNow.hidden = !paused;
1162   ui.followNote.hidden = !paused;
1163   const marked = out.highlightCue ?? -1;
1164   const markMoved = marked !== state.markedCue;
1165   state.markedCue = marked;
1166   if (out.turnToPage !== null) void showPage(out.turnToPage);
1167   else if (markMoved) view.mark();
1168 }

1353 /**
1354  * The token under a point of the screen, in client pixels: an index of
1355  * state.tokens, or -1 when no token is near. Null when the point is not on
1356  * the page, or the page is not read yet: then the strip tells how many
1357  * pages are read.
1358  */
1359 function tokenAtPoint(clientX: number, clientY: number): number | null {
1360   const canvas = ui.page.querySelector('canvas');
1361   const pdf = state.pdf;
1362   if (!canvas || !pdf || ui.page.hidden) return null;
1363   const rect = canvas.getBoundingClientRect();
1364   const inside =
1365     clientX >= rect.left && clientX <= rect.right && clientY >= rect.top && clientY <= rect.bottom;
1366   if (!inside) return null;
1367   const size = state.pages[state.currentPage];
1368   if (!size) {
1369     say(notReadText(state.pages.filter((p) => p).length, pdf.numPages));
1370     return null;
1371   }
1372   const x = ((clientX - rect.left) / rect.width) * size.width;
1373   const y = ((clientY - rect.top) / rect.height) * size.height;
1374   const zoom = window.visualViewport?.scale ?? 1;
1375   return tokenAt(state.tokens, state.currentPage, x, y, tapTolerance(size.width, rect.width, zoom));
1376 }

1450 /**
1451  * Runs `open`, an app over the reader: the dictionary or SubRead Anki.
1452  * With "Pause on lookup", the player pauses first, and plays again when
1453  * `open` gives true: the app closed. When the pause fails, `open` still
1454  * runs.
1455  */
1456 async function withPause(open: () => Promise<boolean>): Promise<void> {
1457   const pause = ui.pauseLookup.checked && state.clock?.playing === true && !state.lookupPaused;
1458   let paused = false;
1459   if (pause) {
1460     state.lookupPaused = true;
1461     try {
1462       await clock.pause();
1463       paused = true;
1464     } catch (err) {
1465       sayError(err);
1466     }
1467   }
1468   const resume = await open();
1469   try {
1470     if (paused && resume) await clock.play();
1471   } catch (err) {
1472     sayError(err);
1473   } finally {
1474     if (pause) state.lookupPaused = false;
1475   }
1476 }
