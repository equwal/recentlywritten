25 /**
26  * The full texts of the parts of the strip that show, for the panel that a
27  * tap on the strip opens: the title of each part, else its text. A part
28  * with no text gives nothing. On the phone the one-line strip cut a long
29  * message, and a phone cannot hover to read the title.
30  */
31 export function stripTexts(parts: readonly StripPart[]): string[] {
32   return parts
33     .filter((part) => part.shown)
34     .map((part) => part.title || part.text.trim())
35     .filter((text) => text !== '');
36 }

63 /** A position as mm:ss, or h:mm:ss from one hour. */
64 export function clockText(ms: number): string {
65   const s = Math.floor(Math.max(0, ms) / 1000);
66   const pad = (n: number): string => String(n).padStart(2, '0');
67   const h = Math.floor(s / 3600);
68   return `${h > 0 ? `${h}:` : ''}${pad(Math.floor((s % 3600) / 60))}:${pad(s % 60)}`;
69 }

73 /**
74  * What is wrong with the player, and what to do, for a reason of
75  * `PlayerError` or `ClockState.error`. Null for another reason.
76  * `overlayInstalled` is true when SubRead Overlay is on the device: then a
77  * button opens it, so that the user can give it notification access.
78  */
79 export function playerProblem(
80   reason: string,
81   android: boolean,
82   overlayInstalled = false,
83 ): Message | null {
84   switch (reason) {
85     case 'no_player':
86       return {
87         text: android
88           ? 'No audiobook is playing. Start it in your player app.'
89           : 'Load the audio file first.',
90       };
91     case 'no_overlay':
92       return overlayInstalled
93         ? { text: 'SubRead Overlay does not answer.', action: OPEN_OVERLAY }
94         : {
95             text: 'SubRead Overlay is not installed.',
96             link: { href: OVERLAY_RELEASES, text: 'Get it' },
97           };
98     case 'no_notification_access':
99       return { text: 'Allow notification access in SubRead Overlay.', action: OPEN_OVERLAY };
100     default:
101       return null;
102   }
103 }

138 /** The player part of the strip: the time and the play state, or the problem. */
139 export function playerMessage(
140   s: ClockState | null,
141   android: boolean,
142   overlayInstalled = false,
143 ): Message {
144   const name = android ? 'Player' : 'Audio';
145   if (!s) return { text: `${name}: not read yet.` };
146   if (s.error !== null) {
147     return (
148       playerProblem(s.error, android, overlayInstalled) ?? {
149         text: `${name}: cannot read it (${s.error}).`,
150       }
151     );
152   }
153   if (s.positionMs === null) return { text: `${name}: no position.` };
154   return { text: `${name} ${clockText(s.positionMs)}, ${s.playing ? 'playing' : 'paused'}` };
155 }

225 /** What SubRead made: the cues, the language and the share of the lines found in the book. */
226 export function subreadResultText(r: {
227   cues?: number;
228   language?: string | null;
229   matchRate?: number;
230 }): string {
231   const cues = r.cues !== undefined && r.cues >= 0 ? String(r.cues) : '?';
232   const rate = r.matchRate !== undefined && r.matchRate >= 0 ? r.matchRate : null;
233   return (
234     `${cues} cues, language ${r.language ?? '?'}` +
235     (rate === null ? '.' : `, ${Math.round(rate * 100)}% of the lines found in the book.`) +
236     (rate !== null && rate < 0.8
237       ? ' Under 80% usually means another edition or the wrong language.'
238       : '')
239   );
240 }

303 /**
304  * The count of matched cues. When all pages are read and less than
305  * MISMATCH_SHARE of the cues is matched, the hint follows.
306  */
307 export function matchedText(matched: number, cues: number, allRead: boolean): string {
308   if (!allRead) return `${matched}/${cues} cues matched to the pages read so far.`;
309   const text = `${matched}/${cues} cues matched.`;
310   return cues > 0 && matched / cues < MISMATCH_SHARE ? `${text} ${MISMATCH_TEXT}` : text;
