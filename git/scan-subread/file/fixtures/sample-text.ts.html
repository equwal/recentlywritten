33 /** One cue per line. */
34 function lineCues(pages: string[][]): string[] {
35   return pages.flat();
36 }
37 
38 /**
39  * One cue per line, except that the last line of each page and the first
40  * line of the next page form one cue. That cue crosses the page boundary.
41  */
42 function crossingCues(pages: string[][]): string[] {
43   const cues: string[] = [];
44   pages.forEach((lines, p) => {
45     const first = p > 0 ? 1 : 0;
46     const last = p < pages.length - 1 ? lines.length - 1 : lines.length;
47     cues.push(...lines.slice(first, last));
48     if (p < pages.length - 1) cues.push(`${lines[lines.length - 1]} ${pages[p + 1]![0]}`);
49   });
50   return cues;
51 }

113 /** Vertical pixel band that line `i` of a page occupies. */
114 export function lineBand(i: number): { top: number; bottom: number } {
115   const top = PAGE.firstBaseline + i * PAGE.lineHeight - PAGE.lineHeight * 0.75;
116   return { top, bottom: top + PAGE.lineHeight };

119 function srtTime(seconds: number): string {
120   const ms = Math.round(seconds * 1000);
121   const h = Math.floor(ms / 3600000);
122   const m = Math.floor((ms % 3600000) / 60000);
123   const s = Math.floor((ms % 60000) / 1000);
124   const frac = ms % 1000;
125   const pad = (n: number, w: number) => String(n).padStart(w, '0');
126   return `${pad(h, 2)}:${pad(m, 2)}:${pad(s, 2)},${pad(frac, 3)}`;
127 }

129 /** SRT text with one cue per entry of `cues`, CUE_SECONDS each, starting at 0. */
130 export function toSrt(fixture: Fixture): string {
131   return fixture.cues
132     .map((line, i) => {
133       const start = i * CUE_SECONDS;
134       return `${i + 1}\n${srtTime(start)} --> ${srtTime(start + CUE_SECONDS)}\n${line}\n`;
135     })
136     .join('\n');
137 }
