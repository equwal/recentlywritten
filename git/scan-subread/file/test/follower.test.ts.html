6 /** The page of each cue. Null for an unmatched cue. */
7 const pages: (number | null)[] = [0, 0, null, 1, 1, null, null, 2];
8 const parts: CueParts[] = pages.map((p) => (p === null ? [] : [{ page: p, share: 1 }]));
9 
10 function input(over: Partial<FollowInput>): FollowInput {
11   return {
12     mode: 'highlight',
13     currentPage: 0,
14     cue: 0,
15     matched: true,
16     page: 0,
17     held: false,
18     event: 'tick',
19     ...over,
20   };
21 }

23 /** The input of the follow at the start of cue `cue` of `pages`. */
24 function at(cue: number, over: Partial<FollowInput> = {}): FollowInput {
25   return input({
26     cue,
27     matched: pages[cue] !== null && pages[cue] !== undefined,
28     page: audioPage(parts, cue, 0),
29     ...over,
30   });
31 }
