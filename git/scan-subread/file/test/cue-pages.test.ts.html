15 /** Tokens of one text line, one per character; a space gets no token. */
16 function lineTokens(text: string, page: number): OcrToken[] {
17   return [...text]
18     .filter((ch) => ch !== ' ')
19     .map((ch, i) => ({ text: ch, page, line: page, bbox: { x0: i, y0: 0, x1: i + 1, y1: 1 } }));
20 }

44   it('gives shares that add up to 1, one part per page in page order', () => {
45     const token = fc.record({
46       text: fc.constantFrom('a', 'B', '猫', ',', '。', ' '),
47       page: fc.nat({ max: 3 }),
48     });
49     const input = fc
50       .array(token, { minLength: 1, maxLength: 60 })
51       .map((ts) => ts.sort((x, y) => x.page - y.page))
52       .chain((ts) =>
53         fc.tuple(fc.constant(ts), fc.nat({ max: ts.length - 1 }), fc.nat({ max: ts.length - 1 })),
54       );
55     fc.assert(
56       fc.property(input, ([ts, a, b]) => {
57         const tokens: OcrToken[] = ts.map((t, i) => ({
58           ...t,
59           line: t.page,
60           bbox: { x0: i, y0: 0, x1: i + 1, y1: 1 },
61         }));
62         const span = { start: Math.min(a, b), end: Math.max(a, b) + 1, matched: true };
63         const [parts] = cueParts(tokens, [span]);
64         // A slow reference: count the normalized characters of each page.
65         const counts = new Map<number, number>();
66         for (const t of tokens.slice(span.start, span.end)) {
67           const n = normalizeText(t.text).length;
68           if (n > 0) counts.set(t.page, (counts.get(t.page) ?? 0) + n);
69         }
70         const total = [...counts.values()].reduce((x, y) => x + y, 0);
71         expect(parts!.map((p) => p.page)).toEqual([...counts.keys()]);
72         parts!.forEach((p) => expect(p.share).toBeCloseTo(counts.get(p.page)! / total, 12));
73         if (parts!.length > 0) {
74           expect(parts!.reduce((x, p) => x + p.share, 0)).toBeCloseTo(1, 12);
75         }
76       }),
77     );
78   });
