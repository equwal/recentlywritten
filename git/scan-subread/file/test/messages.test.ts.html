345   it('adds the hint when all pages are read and less than 30% of the cues match', () => {
346     // sample-jpn.srt on the English PDF.
347     expect(matchedText(0, 3, true)).toBe(`0/3 cues matched. ${MISMATCH_TEXT}`);
348     expect(matchedText(4, 15, true)).toBe(`4/15 cues matched. ${MISMATCH_TEXT}`);
349     expect(matchedText(5, 15, true)).toBe('5/15 cues matched.');
350     expect(matchedText(15, 15, true)).toBe('15/15 cues matched.');
351   });
352 
353   it('gives the hint exactly for all pages read and less than 30% matched', () => {
354     expect(MISMATCH_SHARE).toBe(0.3);
355     // 3 of 10 is 30%: no hint. In floating point 0.3 * 10 is more than 3.
356     expect(matchedText(3, 10, true)).toBe('3/10 cues matched.');
357     const counts = fc
358       .tuple(fc.nat({ max: 500 }), fc.nat({ max: 500 }))
359       .map(([a, b]) => ({ matched: Math.min(a, b), cues: Math.max(a, b) }));
360     fc.assert(
361       fc.property(counts, fc.boolean(), ({ matched, cues }, allRead) => {
362         const hint = matchedText(matched, cues, allRead).endsWith(MISMATCH_TEXT);
363         // Whole numbers: 10 * matched < 3 * cues is exact.
364         expect(hint).toBe(allRead && 10 * matched < 3 * cues);
365       }),
366     );
367   });
