8 /** Tokens for one page, one per character: `lines` is a list of lines, each a list of words. */
9 function pageTokens(page: number, lines: readonly (readonly string[])[]): OcrToken[] {
10   const out: OcrToken[] = [];
11   lines.forEach((words, li) => {
12     words.forEach((word, wi) => {
13       for (const ch of word) {
14         out.push({
15           text: ch,
16           page,
17           line: page * 1000 + li,
18           word: page * 1000 + li * 100 + wi,
19           bbox: box,
20         });
21       }
22     });
23   });
24   return out;
25 }

27 /** The index of token number `n` (from 0) with the text `ch`. */
28 function find(tokens: readonly OcrToken[], ch: string, n = 0): number {
29   let seen = 0;
30   for (let i = 0; i < tokens.length; i++) {
31     if (tokens[i]!.text !== ch) continue;
32     if (seen === n) return i;
33     seen++;
34   }
35   throw new Error(`No token ${ch} number ${n}.`);
36 }
