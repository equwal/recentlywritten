18 /**
19  * Tokens of text lines, one token per character, 10 px per character and
20  * 20 px per line. A space separates two words and gets no token.
21  */
22 function textTokens(lines: string[], page = 0): OcrToken[] {
23   const out: OcrToken[] = [];
24   let word = 0;
25   lines.forEach((text, row) => {
26     [...text].forEach((ch, col) => {
27       if (ch === ' ') {
28         word++;
29         return;
30       }
31       const x0 = col * 10;
32       const y0 = row * 20;
33       out.push({ text: ch, page, line: row, word, bbox: { x0, y0, x1: x0 + 10, y1: y0 + 20 } });
34     });
35     word++;
36   });
37   return out;
38 }

40 /** A matched span of the tokens `from` to `to`, `to` not included. */
41 function spanOf(from: number, to: number): TokenSpan {
42   return { start: from, end: to, matched: true };
43 }
