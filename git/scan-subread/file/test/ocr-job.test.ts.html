13 /** The reasons of the unhandled rejections while `run` runs, and 20 ms after it. */
14 async function unhandledDuring(run: () => void): Promise<unknown[]> {
15   const reasons: unknown[] = [];
16   const record = (reason: unknown): void => {
17     reasons.push(reason);
18   };
19   process.on('unhandledRejection', record);
20   try {
21     run();
22     await new Promise((resolve) => setTimeout(resolve, 20));
23   } finally {
24     process.off('unhandledRejection', record);
25   }
26   return reasons;
27 }
