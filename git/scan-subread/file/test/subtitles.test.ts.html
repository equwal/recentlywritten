60 function utf8(text: string): ArrayBuffer {
61   const bytes = new TextEncoder().encode(text);
62   return bytes.buffer.slice(bytes.byteOffset, bytes.byteOffset + bytes.byteLength);
63 }

65 function utf16(text: string, littleEndian: boolean): ArrayBuffer {
66   const view = new DataView(new ArrayBuffer(text.length * 2));
67   for (let i = 0; i < text.length; i++) view.setUint16(2 * i, text.charCodeAt(i), littleEndian);
68   return view.buffer;
69 }
