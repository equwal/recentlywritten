98 /** One step of a script of opens: start the next attempt, or end one that loads. */
99 interface Step {
100   begin: boolean;
101   /** The attempt that starts or ends. */
102   attempt: number;
103   /** For a start: the user picked the file. */
104   byUser: boolean;
105   /** For an end: the file loaded as a PDF. */
106   loaded: boolean;
107 }
108 
109 /**
110  * Scripts of up to 6 attempts. Each attempt starts in number order and ends
111  * after its start, in any order, with or without a loaded PDF.
112  */
113 const scripts: fc.Arbitrary<Step[]> = fc
114   .record({
115     count: fc.integer({ min: 1, max: 6 }),
116     byUser: fc.array(fc.boolean(), { minLength: 6, maxLength: 6 }),
117     loaded: fc.array(fc.boolean(), { minLength: 6, maxLength: 6 }),
118     picks: fc.array(fc.nat(), { minLength: 12, maxLength: 12 }),
119   })
120   .map(({ count, byUser, loaded, picks }) => {
121     const steps: Step[] = [];
122     const loading: number[] = [];
123     let begun = 0;
124     const end = (attempt: number): Step => ({
125       begin: false,
126       attempt,
127       byUser: false,
128       loaded: loaded[attempt - 1]!,
129     });
130     for (const pick of picks) {
131       if (begun < count && (loading.length === 0 || pick % 2 === 0)) {
132         loading.push(++begun);
133         steps.push({ begin: true, attempt: begun, byUser: byUser[begun - 1]!, loaded: false });
134       } else if (loading.length > 0) {
135         steps.push(end(loading.splice(pick % loading.length, 1)[0]!));
136       }
137     }
138     // The attempts that still load end in their order.
139     for (const attempt of loading) steps.push(end(attempt));
140     return steps;
141   });
142 
143 /** Runs one step of a script. */
144 function run(o: Opens, step: Step): void {
145   if (step.begin) beginOpen(o, step.byUser);
146   else endOpen(o, step.attempt, step.loaded);
147 }

149 describe('open attempts', () => {
150   it('shows the book of the last attempt that loaded, whatever the order of the loads', () => {
151     fc.assert(
152       fc.property(scripts, (steps) => {
153         const o = createOpens();
154         let lastLoaded = 0;
155         for (const step of steps) {
156           const before = o.shown;
157           if (step.begin) {
158             expect(beginOpen(o, step.byUser)).toBe(step.attempt);
159           } else {
160             const shows = endOpen(o, step.attempt, step.loaded);
161             // An attempt shows when it loaded and no later attempt shows.
162             expect(shows).toBe(step.loaded && step.attempt > before);
163             if (step.loaded) lastLoaded = Math.max(lastLoaded, step.attempt);
164           }
165           // The book that shows never goes back to an older attempt.
166           expect(o.shown).toBeGreaterThanOrEqual(before);
167           // An open loads that can replace the book that shows.
168           expect(opening(o)).toBe([...o.loading].some((a) => a > o.shown));
169         }
170         expect(o.shown).toBe(lastLoaded);
171         expect(opening(o)).toBe(false);
172       }),
173     );
174   });
175 });
