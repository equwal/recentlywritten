5 /** Reads a binary PPM file: the header fields and the RGB bytes after the header. */
6 function decodePpm(ppm: Uint8Array): { width: number; height: number; max: number; rgb: number[] } {
7   // The header is ASCII: "P6", then width, height and maximum value, each after white space,
8   // then one white space character.
9   const head = new TextDecoder('latin1').decode(ppm.subarray(0, 64));
10   const match = /^P6\s+(\d+)\s+(\d+)\s+(\d+)\s/.exec(head);
11   if (!match) throw new Error(`No PPM header: ${JSON.stringify(head)}`);
12   return {
13     width: Number(match[1]),
14     height: Number(match[2]),
15     max: Number(match[3]),
16     rgb: Array.from(ppm.subarray(match[0].length)),
17   };
18 }

20 /** The RGB bytes of RGBA pixels. */
21 function rgbOf(data: Uint8ClampedArray): number[] {
22   return Array.from(data).filter((_, i) => i % 4 !== 3);
