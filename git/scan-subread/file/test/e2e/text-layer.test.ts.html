27 async function textTokens(file: string): Promise<{ tokens: OcrToken[]; height: number }> {
28   const data = new Uint8Array(await readFile(file));
29   const task = getDocument({ data });
30   const doc = await task.promise;
31   const tokens: OcrToken[] = [];
32   let height = 0;
33   for (let p = 0; p < doc.numPages; p++) {
34     const page = await doc.getPage(p + 1);
35     const base = page.getViewport({ scale: 1 });
36     const viewport = page.getViewport({ scale: WIDTH / base.width });
37     height = viewport.height;
38     const content = await page.getTextContent();
39     const items = textItems(content.items);
40     const toPixel = (x: number, y: number) =>
41       viewport.convertToViewportPoint(x, y) as [number, number];
42     tokens.push(...textItemsToTokens(items, p, toPixel));
43   }
44   await task.destroy();
45   return { tokens, height };
46 }

53 describe.each(withText)('text layer of $name-text.pdf', (fixture) => {
54   it('carries every character in order, line by line', async () => {
55     const { tokens, height } = await textTokens(
56       path.resolve('fixtures', `${fixture.name}-text.pdf`),
57     );
58     const expected = fixture.pages.map((lines) => lines.join('\n')).join('\n\n') + '\n';
59     expect(bookText(tokens)).toBe(expected);
60     expect(tokens.map((t) => t.text).join('')).toBe(expected.replace(/\s/g, ''));
61     // Each page: every box inside the page, lines from top to bottom.
62     let lastLine = -1;
63     let lastY = -Infinity;
64     for (const t of tokens) {
65       expect(t.bbox.x0).toBeGreaterThanOrEqual(0);
66       expect(t.bbox.x1).toBeLessThanOrEqual(WIDTH);
67       expect(t.bbox.y0).toBeGreaterThanOrEqual(0);
68       expect(t.bbox.y1).toBeLessThanOrEqual(height);
69       if (t.line !== lastLine) {
70         if (t.line % 100000 !== 0) expect(t.bbox.y0).toBeGreaterThan(lastY);
71         lastLine = t.line;
72         lastY = t.bbox.y0;
73       }
74     }
75   });
76 
77   it('aligns every cue to the text layer', async () => {
78     const { tokens } = await textTokens(path.resolve('fixtures', `${fixture.name}-text.pdf`));
79     const srt = await readFile(path.resolve('fixtures', `${fixture.name}.srt`), 'utf8');
80     const spans = alignCuesToTokens(
81       parseSubtitles(srt).map((c) => c.text),
82       tokens,
83     );
84     expect(spans).toHaveLength(fixture.cues.length);
85     spans.forEach((span, i) => {
86       expect(span.matched, `cue ${i} matched`).toBe(true);
87       const text = tokens
88         .slice(span.start, span.end)
89         .map((t) => t.text)
90         .join('');
91       // The alignment drops punctuation, so a span can leave out a comma at its edge.
92       expect(normalizeText(text)).toBe(normalizeText(fixture.cues[i]!));
93     });
94   });
95 });
