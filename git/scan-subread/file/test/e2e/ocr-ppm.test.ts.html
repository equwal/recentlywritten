20 /** A PPM file of the pixels of a PNG file: RGBA pixels, as getImageData gives them in the app. */
21 async function ppmOf(png: Buffer): Promise<Buffer> {
22   const { data, info } = await sharp(png).ensureAlpha().raw().toBuffer({ resolveWithObject: true });
23   expect(info.channels).toBe(4);
24   const ppm = imageToPpm({
25     width: info.width,
26     height: info.height,
27     data: new Uint8ClampedArray(data.buffer, data.byteOffset, data.byteLength),
28   });
29   return Buffer.from(ppm.buffer, ppm.byteOffset, ppm.byteLength);
30 }

32 /** How far apart two readings of one page are. */
33 function difference(a: OcrToken[], b: OcrToken[]): string {
34   const text = (tokens: OcrToken[]) => tokens.map((t) => t.text).join('');
35   let box = 0;
36   let unequal = 0;
37   for (let i = 0; i < Math.min(a.length, b.length); i++) {
38     const p = a[i]!;
39     const q = b[i]!;
40     const d = Math.max(
41       Math.abs(p.bbox.x0 - q.bbox.x0),
42       Math.abs(p.bbox.y0 - q.bbox.y0),
43       Math.abs(p.bbox.x1 - q.bbox.x1),
44       Math.abs(p.bbox.y1 - q.bbox.y1),
45     );
46     box = Math.max(box, d);
47     if (d > 0 || p.text !== q.text || p.line !== q.line || p.word !== q.word) unequal++;
48   }
49   return (
50     `${a.length} and ${b.length} symbols, text ${text(a) === text(b) ? 'equal' : 'not equal'}, ` +
51     `${unequal} unequal symbols, largest box difference ${box} px`
52   );
53 }

63       const ppm = await ppmOf(png);
64       // tesseract.js writes the cache file only when the directory exists.
65       await mkdir(CACHE, { recursive: true });
66       const worker = await createWorker(lang, 1, { cachePath: CACHE });
67       try {
68         const read = async (image: Buffer): Promise<OcrToken[]> => {
69           const start = performance.now();
70           const { data } = await worker.recognize(image, {}, { blocks: true, text: false });
71           console.log(
72             `[${file}] ${image.length} bytes read in ${Math.round(performance.now() - start)} ms`,
73           );
74           return pageToTokens(data, 0);
75         };
76         const fromPng = await read(png);
77         const fromPpm = await read(ppm);
78         console.log(`[${file}] PNG and PPM: ${difference(fromPng, fromPpm)}`);
79         expect(fromPng.length).toBeGreaterThan(0);
80         expect(fromPpm).toEqual(fromPng);
81       } finally {
