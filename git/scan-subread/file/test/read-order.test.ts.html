5 /** The whole numbers from `from` up to, but not including, `to`. */
6 function range(from: number, to: number): number[] {
7   return Array.from({ length: Math.max(0, to - from) }, (_, i) => from + i);
8 }

10 /** All pages of a book of `count` pages, in the order of nextPage, while the current page stays. */
11 function readAll(count: number, current: number): number[] {
12   const pending = new Set(range(0, count));
13   const order: number[] = [];
14   while (pending.size > 0) {
15     const page = nextPage(pending, current);
16     pending.delete(page);
17     order.push(page);
18   }
19   return order;
20 }
