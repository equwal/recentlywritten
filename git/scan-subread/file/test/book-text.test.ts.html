8 /** The non-blank lines of a book text, trimmed. */
9 function paragraphs(text: string): string[] {
10   return text
11     .split('\n')
12     .map((l) => l.trim())
13     .filter((l) => l.length > 0);
14 }
15 
16 /** Tokens for one page: `lines` is a list of lines, each a list of words. */
17 function pageTokens(page: number, lines: readonly (readonly string[])[]): OcrToken[] {
18   const out: OcrToken[] = [];
19   lines.forEach((words, li) => {
20     words.forEach((word, wi) => {
21       for (const ch of word) {
22         out.push({
23           text: ch,
24           page,
25           line: page * 1000 + li,
26           word: page * 1000 + li * 100 + wi,
27           bbox: box,
28         });
29       }
30     });
31   });
32   return out;
33 }

35 describe('bookText', () => {
