199 // The bounds leave room for a busy machine: the aligner takes about 1 s and
200 // 30 ms here, and took 4.3 s once under heavy load. The old global diff took
201 // 13 s for the full book, and the same again for each page, so the test still
202 // catches a return to it.
203 describe('createAligner speed', () => {
204   it('aligns a book of 300 000 characters in under 8 s, and one more page in under 500 ms', () => {
205     const LENGTH = 300_000;
206     const PAGE = 600;
207     // 5% noise: 3 codes in 60 are a drop, a substitution and an insertion.
208     const codes = [...'k'.repeat(57), 'd', 's', 'i'];
209     const [sample] = fc.sample(
210       fc.tuple(
211         fc.string({ unit: char, minLength: LENGTH, maxLength: LENGTH }),
212         fc.string({ unit: fc.constantFrom(...codes), minLength: LENGTH, maxLength: LENGTH }),
213       ),
214       { seed: 1, numRuns: 1 },
215     );
216     const [text, noise] = sample!;
217     const cues = text.match(/.{1,40}/g)!;
218     const pages: Token[][] = [];
219     for (let i = 0; i < LENGTH; i++) {
220       const page = Math.floor(i / PAGE);
221       const tokens = (pages[page] ??= []);
222       const ch = text[i]!;
223       const other = text[(i + 1) % LENGTH]!;
224       if (noise[i] === 's') tokens.push({ text: other, page });
225       else if (noise[i] === 'i') tokens.push({ text: other, page }, { text: ch, page });
226       else if (noise[i] !== 'd') tokens.push({ text: ch, page });
227     }
228 
229     let t = performance.now();
230     const aligner = createAligner(cues);
231     const spans = aligner.spans(pages);
232     const all = performance.now() - t;
233     // Read one page again: a new token array, as after a change of the OCR language.
234     pages[250] = [...pages[250]!];
235     t = performance.now();
236     aligner.spans(pages);
237     const one = performance.now() - t;
238 
239     expect(spans.filter((s) => s.matched).length).toBeGreaterThanOrEqual(0.99 * cues.length);
240     expect(all).toBeLessThan(8000);
241     expect(one).toBeLessThan(500);
242   }, 30_000);
243 });
