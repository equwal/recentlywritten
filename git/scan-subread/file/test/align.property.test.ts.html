43 /** The spans of the matched cues rise and stay inside [0, total]. */
44 function expectRising(spans: readonly TokenSpan[], total: number, disjoint: boolean): void {
45   let last: TokenSpan | null = null;
46   for (const span of spans) {
47     if (!span.matched) {
48       expect(span).toEqual({ start: 0, end: 0, matched: false });
49       continue;
50     }
51     expect(span.start).toBeGreaterThanOrEqual(0);
52     expect(span.end).toBeGreaterThan(span.start);
53     expect(span.end).toBeLessThanOrEqual(total);
54     if (last) {
55       expect(span.start).toBeGreaterThanOrEqual(disjoint ? last.end : last.start);
56       expect(span.end).toBeGreaterThanOrEqual(last.end);
57     }
58     last = span;
59   }
60 }

73 describe('risingChain', () => {
74   /** The length of a longest strictly rising chain, by the slow O(n²) method. */
75   function slowLength(values: readonly number[]): number {
76     const best = values.map(() => 1);
77     for (let i = 0; i < values.length; i++) {
78       for (let j = 0; j < i; j++) {
79         if (values[j]! < values[i]!) best[i] = Math.max(best[i]!, best[j]! + 1);
80       }
81     }
82     return Math.max(0, ...best);
83   }

88         const chain = risingChain(values);
89         for (let k = 1; k < chain.length; k++) {

91           expect(values[chain[k]!]!).toBeGreaterThan(values[chain[k - 1]!]!);
92         }
93         expect(chain.length).toBe(slowLength(values));
94       }),

254   /** A span of `n` tokens: matched and inside [0, n], or unmatched. */
255   function span(n: number): fc.Arbitrary<TokenSpan> {
256     if (n === 0) return fc.constant(unmatched);
257     const matched = fc
258       .nat({ max: n - 1 })
259       .chain((start) =>
260         fc.integer({ min: start + 1, max: n }).map((end) => ({ start, end, matched: true })),
261       );
262     return fc.oneof(fc.constant(unmatched), matched);
263   }

373 interface Edit {
374   pos: number;
375   kind: 'sub' | 'del' | 'ins';
376   ch: string;
377 }
378 
379 function applyEdits(text: string, edits: readonly Edit[]): string {
380   const chars = [...text];
381   // Apply from the end so earlier positions stay valid.
382   for (const e of [...edits].sort((x, y) => y.pos - x.pos)) {
383     if (e.kind === 'sub') chars[e.pos] = e.ch;
384     else if (e.kind === 'del') chars.splice(e.pos, 1);
385     else chars.splice(e.pos, 0, e.ch);
386   }
387   return chars.join('');
