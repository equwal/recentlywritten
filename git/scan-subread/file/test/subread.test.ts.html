4 /** A fake Screen Wake Lock API that keeps each lock it gives. */
5 function fakeWakeLock() {
6   const locks: { released: boolean; release: () => Promise<void> }[] = [];
7   const request = vi.fn(async (type: string) => {
8     expect(type).toBe('screen');
9     const lock = {
10       released: false,
11       release: async () => {
12         lock.released = true;
13       },
14     };
15     locks.push(lock);
16     return lock;
17   });
18   return { wakeLock: { request }, request, locks };
19 }
