14 /** Tokens for one page: `lines` laid out left to right, 10 px per character. */
15 function tokensOf(lines: string[], page = 0): OcrToken[] {
16   const tokens: OcrToken[] = [];
17   lines.forEach((line, row) => {
18     [...line].forEach((ch, col) => {
19       tokens.push({
20         text: ch,
21         page,
22         line: page * 100 + row,
23         bbox: { x0: col * 10, y0: row * 20, x1: col * 10 + 10, y1: row * 20 + 20 },
24       });
25     });
26   });
27   return tokens;
28 }

83 /** Tokens of one line of words, with a word id per word. Spaces are not tokens. */
84 function wordTokens(words: string[], page = 0, line = 0): OcrToken[] {
85   const tokens: OcrToken[] = [];
86   let col = 0;
87   words.forEach((w, wi) => {
88     for (const ch of w) {
89       tokens.push({
90         text: ch,
91         page,
92         line,
93         word: wi,
94         bbox: { x0: col * 10, y0: 0, x1: col * 10 + 10, y1: 20 },
95       });
96       col++;
97     }
98     col++;
99   });
100   return tokens;
101 }

103 /** Tokens of a page of text lines, a word id per word. A space separates two words. */
104 function pageTokens(lines: string[], page = 0): OcrToken[] {
105   return lines.flatMap((line, row) => wordTokens(line.split(' '), page, page * 100 + row));
106 }

108 /** Index of the token of the `n`-th `ch` (from 0) on line `line`. */
109 function tapOn(tokens: readonly OcrToken[], line: number, ch: string, n = 0): number {
110   let seen = 0;
111   for (let i = 0; i < tokens.length; i++) {
112     const t = tokens[i]!;
113     if (t.line === line && t.text === ch && seen++ === n) return i;
114   }
115   throw new Error(`No ${ch} number ${n} on line ${line}`);
116 }
