6 /**
7  * Tells when an app that the plugin started for a result is closed.
8  *
9  * Android gives the activity result when the other app closes. But an app
10  * that opens in a new task (launchMode singleTask or singleInstance, or
11  * FLAG_ACTIVITY_NEW_TASK) gives RESULT_CANCELED at once, while it stays on
12  * the screen. The reader must not start the player again at that time.
13  *
14  * The other app covers the reader, so the reader pauses. A result that
15  * comes after such a pause is final: the watch runs its finish at once. For
16  * an immediate result, the watch keeps the finish and runs it at the next
17  * resume that follows a pause: then the user is back. When no pause comes
18  * in {@link #OPEN_MS}, the other app did not open, and the watch runs the
19  * finish.
20  *
21  * Android also pauses a resumed activity to give it a result, and resumes
22  * it after. That pause and the result come in one message of the main
23  * thread. The watch records each pause one message later, so that such a
24  * pause is not a pause after the launch.
25  *
26  * All methods run on the main thread, except {@link #launched()}, which the
27  * plugin calls on its own thread.
28  */
29 final class ReturnWatch {
30 
31     /** Runs code later on the main thread. A test gives its own clock. */
32     interface Later {
33         void run(Runnable code, long delayMs);
34     }
35 
36     /** The time that the other app has to open after an immediate result. */
37     static final long OPEN_MS = 1500;
38 
39     private final Later later;
40 
41     /** True when the activity paused after the last launch. */
42     private boolean pausedSinceLaunch;
43 
44     /** The finish of a result that came at once. Null when no result waits. */
45     private Runnable waiting;
46 
47     /** True when the activity paused after the result that waits. */
48     private boolean pausedSinceResult;
49 
50     /** A watch that runs its later code on the main thread. */
51     ReturnWatch() {
52         this(new Handler(Looper.getMainLooper())::postDelayed);
53     }
54 
55     ReturnWatch(Later later) {
56         this.later = later;
57     }
58 
59     /** The plugin calls this before it starts an activity for a result. */
60     synchronized void launched() {
61         // A new launch ends the wait for the result before it.
62         finishWaiting();
63         pausedSinceLaunch = false;
64     }
65 
66     /** The plugin calls this from handleOnPause. */
67     synchronized void paused() {
68         if (waiting != null) pausedSinceResult = true;
69         later.run(this::markPaused, 0);
70     }
71 
72     private synchronized void markPaused() {
73         pausedSinceLaunch = true;
74     }
75 
76     /** The plugin calls this from handleOnResume. */
77     synchronized void resumed() {
78         if (pausedSinceResult) finishWaiting();
79     }
80 
81     /**
82      * The plugin calls this from its activity callback, with the code that
83      * ends the call. The watch runs `finish` now when the other app is
84      * closed, else when the user is back.
85      */
86     synchronized void result(Runnable finish) {
87         finishWaiting();
88         if (pausedSinceLaunch) {
89             finish.run();
90             return;
91         }
92         waiting = finish;
93         pausedSinceResult = false;
94         later.run(() -> notOpened(finish), OPEN_MS);
95     }
96 
97     /** No pause came after the immediate result: the other app did not open. */
98     private synchronized void notOpened(Runnable finish) {
99         if (waiting == finish && !pausedSinceResult) finishWaiting();
100     }
101 
102     private void finishWaiting() {
103         Runnable finish = waiting;
104         waiting = null;
105         pausedSinceResult = false;
106         if (finish != null) finish.run();
107     }
