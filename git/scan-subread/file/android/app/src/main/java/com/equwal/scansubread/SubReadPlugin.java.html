149     /** The state line of the first overlay that answers, or `error=no_overlay`. */
150     private String queryState() {
151         ContentResolver resolver = getContext().getContentResolver();
152         for (String a : candidates()) {
153             // query returns null when no app has the provider. It throws when
154             // the provider is not visible to this app.
155             try (Cursor c = resolver.query(stateUri(a), null, null, null, null)) {
156                 if (c != null && c.moveToFirst()) {
157                     authority = a;
158                     String line = c.getString(0);
159                     return line == null ? NO_OVERLAY : line;
160                 }
161             } catch (RuntimeException e) {
162                 // Try the next authority.
163             }
164         }
165         return NO_OVERLAY;
166     }

253     @PluginMethod
254     public void lookup(PluginCall call) {
255         String text = call.getString("text");
256         if (text == null) {
257             call.reject("text is required.");
258             return;
259         }
260         Intent send = probe()
261             .putExtra(Intent.EXTRA_PROCESS_TEXT, text)
262             .putExtra(Intent.EXTRA_PROCESS_TEXT_READONLY, true);
263         String chosen = prefs().getString(PREF_DICTIONARY, "");
264         ComponentName component = chosen.isEmpty() ? null : ComponentName.unflattenFromString(chosen);
265         if (component != null) {
266             returns.launched();
267             if (startForResult(call, new Intent(send).setComponent(component), "lookupResult") == null) return;
268             // The chosen app is gone, or it does not let this app start it. Ask.
269             prefs().edit().remove(PREF_DICTIONARY).apply();
270         }
271         returns.launched();
272         RuntimeException failed = startForResult(call, Intent.createChooser(send, null), "lookupResult");
273         if (failed != null) call.reject("Cannot open the dictionary: " + failed.getMessage(), failed);
274     }

389      * Tells which apps of the suite are installed, for a checklist. The
390      * packages are in the queries of the manifest, else Android hides them.

392     @PluginMethod
393     public void suite(PluginCall call) {
394         // The version name of SubRead, or null when SubRead is not installed.
395         PackageInfo subread = packageInfo(SUBREAD_PACKAGE);
396         Object version = JSObject.NULL;
397         if (subread != null) version = subread.versionName == null ? "" : subread.versionName;
398         int dictionaries = 0;
399         PackageManager pm = getContext().getPackageManager();
400         for (ResolveInfo info : pm.queryIntentActivities(probe(), PackageManager.MATCH_DEFAULT_ONLY)) {
401             // This app is not a dictionary.
402             if (info.activityInfo != null && !info.activityInfo.packageName.equals(getContext().getPackageName())) {
403                 dictionaries++;
404             }
405         }
406         JSObject ret = new JSObject();
407         ret.put("overlay", packageInfo(OVERLAY_PACKAGE) != null);
408         ret.put("overlayDebug", packageInfo(OVERLAY_DEBUG_PACKAGE) != null);
409         ret.put("subread", version);
410         ret.put("anki", firstInstalled(ANKI_PACKAGES, name -> packageInfo(name) != null) != null);
411         ret.put("dictionaries", dictionaries);
412         call.resolve(ret);
413     }

461     /**
462      * A safe file name: the name keeps A-Z, a-z, 0-9, ".", "_" and "-", and
463      * each other character becomes "_". So the name cannot point to another
464      * folder. Null when no file can have the name.
465      */
466     static String safeFileName(String name) {
467         String safe = name.replaceAll("[^A-Za-z0-9._-]", "_");
468         if (safe.isEmpty() || safe.equals(".") || safe.equals("..") || safe.length() > 255) return null;
469         return safe;
470     }

490     /**
491      * Asks SubRead for the .srt of the audio and the book text. The optional
492      * `resultName` names a result file, see pendingSubtitles.
493      */
494     @PluginMethod
495     public void makeSubtitles(PluginCall call) {
496         String audio = call.getString("audio");
497         String bookText = call.getString("bookText");
498         String language = call.getString("language", "auto");
499         String resultName = call.getString("resultName");
500         if (audio == null || bookText == null) {
501             call.reject("audio and bookText are required.");
502             return;
503         }
504         File result = resultName == null ? null : resultFile(resultName);
505         if (resultName != null && result == null) {
506             call.reject("resultName is not a usable file name.");
507             return;
508         }
509         Uri bookUri;
510         try {
511             bookUri = shareFile("book.txt", bookText);
512         } catch (IOException e) {
513             call.reject("Cannot write the book text: " + e.getMessage(), e);
514             return;
515         }
516         Uri resultUri = null;
517         if (result != null) {
518             try {
519                 resultUri = clearResult(result);
520             } catch (IOException e) {
521                 call.reject("Cannot make the result file: " + e.getMessage(), e);
522                 return;
523             }
524         }
525         // While SubRead runs, Capacitor keeps the options of this call two
526         // times in the saved state of the activity. That state has a limit of
527         // about 1 MB. With a large book text, the reader crashes when SubRead
528         // covers it. SubRead reads the book text from the file, so remove it.
529         call.getData().remove("bookText");
530         Uri audioUri = Uri.parse(audio);
531         Intent ask = new Intent(ACTION_ALIGN)
532             .setPackage(SUBREAD_PACKAGE)
533             .putExtra(EXTRA_AUDIO, audioUri)
534             .putExtra(EXTRA_BOOK, bookUri)
535             .putExtra(EXTRA_LANGUAGE, language);
536         // SubRead must be able to read both files.
537         ClipData clip = ClipData.newRawUri("audio", audioUri);
538         clip.addItem(new ClipData.Item(bookUri));
539         if (resultUri != null) {
540             // SubRead 0.10.0 and later write the .srt into this file before
541             // they answer, so the .srt is safe when Android stops the reader
542             // during the job. Older versions ignore the extra.
543             ask.putExtra(EXTRA_RESULT, resultUri);
544             clip.addItem(new ClipData.Item(resultUri));
545         }
546         ask.setClipData(clip);
547         ask.addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION);
548         if (ask.resolveActivity(getContext().getPackageManager()) == null) {
549             call.resolve(errorObject("not_installed"));
550             return;
551         }
552         if (resultUri != null) {
553             // SubRead must be able to write the result file. The intent flag
554             // FLAG_GRANT_WRITE_URI_PERMISSION asks for write access to each
555             // file of the clip, the audio file too. This app can only read the
556             // audio file, so startActivity would throw SecurityException. So
557             // only the result file gets write access.
558             getContext().grantUriPermission(SUBREAD_PACKAGE, resultUri, Intent.FLAG_GRANT_WRITE_URI_PERMISSION);
559         }
560         RuntimeException failed = startForResult(call, ask, "subtitlesResult");
561         if (failed != null) {
562             // SubRead did not start, so it needs no write access.
563             if (resultUri != null) getContext().revokeUriPermission(resultUri, Intent.FLAG_GRANT_WRITE_URI_PERMISSION);
564             call.resolve(errorObject("cannot_start: " + failed.getMessage()));
565         }
566     }

568     /**
569      * Gives the .srt that SubRead wrote into the result file `resultName`,
570      * and removes the file. SubRead can finish while Android has stopped the
571      * reader. Then the activity result is lost, but the file is there.
572      * Resolves no srt when the file is not there or is empty.
573      */
574     @PluginMethod
575     public void pendingSubtitles(PluginCall call) {
576         String resultName = call.getString("resultName");
577         if (resultName == null) {
578             call.reject("resultName is required.");
579             return;
580         }
581         File file = resultFile(resultName);
582         if (file == null) {
583             call.reject("resultName is not a usable file name.");
584             return;
585         }
586         JSObject ret = new JSObject();
587         if (file.isFile() && file.length() > 0) {
588             try (InputStream in = new FileInputStream(file)) {
589                 ret.put("srt", readAll(in));
590             } catch (IOException e) {
591                 call.reject("Cannot read the result file: " + e.getMessage(), e);
592                 return;
593             }
594             // The job is done, so SubRead needs no more write access.
595             Uri uri = FileProvider.getUriForFile(getContext(), getContext().getPackageName() + ".fileprovider", file);
596             getContext().revokeUriPermission(uri, Intent.FLAG_GRANT_WRITE_URI_PERMISSION);
597             // When the delete fails, the next call gives the same .srt again.
598             file.delete();
599         }
600         call.resolve(ret);
601     }

603     private static String readAll(InputStream in) throws IOException {
604         ByteArrayOutputStream out = new ByteArrayOutputStream();
605         byte[] buf = new byte[64 * 1024];
606         int n;
607         while ((n = in.read(buf)) > 0) out.write(buf, 0, n);
608         return new String(out.toByteArray(), StandardCharsets.UTF_8);
609     }

611     @ActivityCallback
612     private void subtitlesResult(PluginCall call, ActivityResult result) {
613         if (call == null) return;
614         Intent data = result.getData();
615         Uri srtUri = data == null ? null : data.getData();
616         if (result.getResultCode() != Activity.RESULT_OK || srtUri == null) {
617             String error = data == null ? null : data.getStringExtra(EXTRA_ERROR);
618             call.resolve(errorObject(error == null ? "cancelled" : error));
619             return;
620         }
621         // The grant ends with the calling activity: copy the file now, off the UI thread.
622         new Thread(
623             () -> {
624                 try (InputStream in = getContext().getContentResolver().openInputStream(srtUri)) {
625                     if (in == null) {
626                         call.resolve(errorObject("cannot_read"));
627                         return;
628                     }
629                     JSObject ret = new JSObject();
630                     ret.put("srt", readAll(in));
631                     ret.put("cues", data.getIntExtra(EXTRA_CUES, -1));
632                     ret.put("matchRate", data.getDoubleExtra(EXTRA_MATCH_RATE, -1));
633                     String language = data.getStringExtra(EXTRA_LANGUAGE);
634                     ret.put("language", language == null ? JSObject.NULL : language);
635                     call.resolve(ret);
636                 } catch (IOException | RuntimeException e) {
637                     call.resolve(errorObject("cannot_read: " + e.getMessage()));
638                 }
639             },
640             "SubReadSrt"
641         )
642             .start();
643     }

655     @ActivityCallback
656     private void audioResult(PluginCall call, ActivityResult result) {
657         if (call == null) return;
658         Intent data = result.getData();
659         Uri uri = data == null ? null : data.getData();
660         if (result.getResultCode() != Activity.RESULT_OK || uri == null) {
661             call.reject("No audio file was picked.");
662             return;
663         }
664         ContentResolver resolver = getContext().getContentResolver();
665         try {
666             // Keep the permission, so the Uri can be given to SubRead later.
667             resolver.takePersistableUriPermission(uri, Intent.FLAG_GRANT_READ_URI_PERMISSION);
668         } catch (SecurityException e) {
669             // Some providers give no persistable grant. The Uri still works now.
670         }
671         String name = uri.getLastPathSegment();
672         try (Cursor c = resolver.query(uri, new String[] { OpenableColumns.DISPLAY_NAME }, null, null, null)) {
673             if (c != null && c.moveToFirst() && !c.isNull(0)) name = c.getString(0);
674         } catch (RuntimeException e) {
675             // Keep the path segment.
676         }
677         JSObject ret = new JSObject();
678         ret.put("uri", uri.toString());
679         ret.put("name", name == null ? "" : name);
680         call.resolve(ret);
681     }

683     /** Opens the share sheet with a text file, for example the finished .srt. */
684     @PluginMethod
685     public void shareText(PluginCall call) {
686         String name = call.getString("name");
687         String text = call.getString("text");
688         if (name == null || text == null) {
689             call.reject("name and text are required.");
690             return;
691         }
692         try {
693             Uri uri = shareFile(name, text);
694             Intent send = new Intent(Intent.ACTION_SEND)
695                 .setType("text/plain")
696                 .putExtra(Intent.EXTRA_STREAM, uri)
697                 .addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION);
698             // The chooser reads the file for its preview. createChooser gives
699             // the chooser the clip of this intent, with the read grant. Without
700             // a clip the chooser has no grant, and its preview gets a
701             // SecurityException.
702             send.setClipData(ClipData.newRawUri(name, uri));
703             getActivity().startActivity(Intent.createChooser(send, name));
704             call.resolve();
705         } catch (IOException | RuntimeException e) {
706             call.reject("Cannot share: " + e.getMessage(), e);
707         }
708     }
