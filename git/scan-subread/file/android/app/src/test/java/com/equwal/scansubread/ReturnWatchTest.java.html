9 /**
10  * The order of the pauses, the resumes and the result, as Android gives
11  * them. Each block of calls with no `main.advance` between them is one
12  * message of the main thread.
13  */
14 public class ReturnWatchTest {
15 
16     /** A main thread for the test: posted code runs when the test moves the clock. */
17     private static final class FakeMain implements ReturnWatch.Later {
18 
19         private static final class Task {
20 
21             final long at;
22             final Runnable code;
23 
24             Task(long at, Runnable code) {
25                 this.at = at;
26                 this.code = code;
27             }
28         }
29 
30         private final List<Task> tasks = new ArrayList<>();
31         private long now;
32 
33         @Override
34         public void run(Runnable code, long delayMs) {
35             tasks.add(new Task(now + delayMs, code));
36         }
37 
38         /** Moves the clock and runs the code that is due, the earliest first. */
39         void advance(long ms) {
40             now += ms;
41             while (true) {
42                 Task next = null;
43                 for (Task t : tasks) if (t.at <= now && (next == null || t.at < next.at)) next = t;
44                 if (next == null) return;
45                 tasks.remove(next);
46                 next.code.run();
47             }
48         }
49     }
50 
51     private final FakeMain main = new FakeMain();
52     private final ReturnWatch watch = new ReturnWatch(main);
53     private int finished;
54     private final Runnable finish = () -> finished++;
55 
56     @Test
57     public void aResultAfterThePauseForTheOtherAppIsFinal() {
58         watch.launched();
59         // The dictionary covers the reader.
