Docker: run the server behind Caddy with docker compose

compose.yaml builds the image from this source and puts Caddy in front.
Caddy gets the TLS certificate for SBM_DOMAIN by itself. sbm-sync uses
the network of the caddy container. So Caddy connects on 127.0.0.1, and
the sign-in limits see the address of each client.

The image holds only the static binary, on distroless, as user nonroot.
The data is in the volume sbm-data.

The README tells how to run the server with and without Docker, and how
to connect the devices to it.

This computer has no Docker, so no image was built. Two checks passed:
the build command works on the files that the Dockerfile copies, and
compose.yaml passes the Compose schema.

+# Image of sbm-sync: the static binary on distroless, as user nonroot.
+# compose.yaml builds it and puts Caddy in front for TLS.
+FROM golang:1.26-alpine AS build
+WORKDIR /src
+COPY go.mod go.sum ./
+RUN go mod download
+COPY *.go pages.html ./
+RUN CGO_ENABLED=0 go build -trimpath -ldflags='-s -w' -o /sbm-sync . && mkdir /data
+
+FROM gcr.io/distroless/static-debian13:nonroot
+COPY --from=build /sbm-sync /usr/local/bin/sbm-sync
+# A new volume on /data gets the owner of this directory: nonroot.
+COPY --from=build --chown=65532:65532 /data /data
+ENV SBM_ADDR=:8750 SBM_DATA=/data
+EXPOSE 8750
+ENTRYPOINT ["/usr/local/bin/sbm-sync"]

 ## Run your own server
 
+This repository has all the source code of the server, under the
+AGPL-3.0. You need a machine with a domain name that points to it.
+
+### With Docker
+
+Ports 80 and 443 must be open. Caddy gets the TLS certificate by itself.
+
+    git clone https://github.com/equwal/sbm-sync
+    cd sbm-sync
+    cp .env.example .env    # then set SBM_DOMAIN in .env
+    docker compose up -d
+
+The data is in the Docker volume `sbm-data`. To back it up:
+
+    docker compose cp sbm-sync:/data ./backup
+
+To update, run `git pull`, then `docker compose up -d --build`.
+
+### Without Docker
+
     go build
     SBM_URL=https://sbm.example.org ./sbm-sync

+# sbm-sync behind Caddy. Caddy gets and renews the TLS certificate for
+# SBM_DOMAIN by itself. Copy .env.example to .env, set SBM_DOMAIN in it,
+# then run: docker compose up -d
+services:
+  caddy:
+    image: caddy:2
+    restart: unless-stopped
+    ports:
+      - "80:80"
+      - "443:443"
+      - "443:443/udp"
+    environment:
+      SBM_DOMAIN: ${SBM_DOMAIN:?set SBM_DOMAIN in .env}
+    volumes:
+      - ./contrib/Caddyfile:/etc/caddy/Caddyfile:ro
+      - caddy-data:/data
+      - caddy-config:/config
+
+  sbm-sync:
+    build: .
+    restart: unless-stopped
+    # sbm-sync uses the network of caddy. Caddy connects to it on
+    # 127.0.0.1, and sbm-sync trusts X-Real-IP only from 127.0.0.1. So the
+    # sign-in limits count the address of each client, not of Caddy.
+    network_mode: service:caddy
+    env_file: .env
+    environment:
+      SBM_URL: https://${SBM_DOMAIN}
+      SBM_ADDR: 127.0.0.1:8750
+    volumes:
+      - sbm-data:/data
+
+volumes:
+  caddy-data:
+  caddy-config:
+  sbm-data:
