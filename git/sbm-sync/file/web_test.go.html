36 // upload sends a file with the import form, as a browser does.
37 func (e *harness) upload(c *http.Client, content string) (int, string, string) {
38 	var body bytes.Buffer
39 	m := multipart.NewWriter(&body)
40 	f, _ := m.CreateFormFile("file", "bookmarks.html")
41 	io.WriteString(f, content)
42 	m.Close()
43 	req, _ := http.NewRequest("POST", e.web.URL+"/bookmarks/import", &body)
44 	req.Header.Set("Content-Type", m.FormDataContentType())
45 	req.Header.Set("Origin", e.web.URL)
46 	resp, err := c.Do(req)
47 	if err != nil {
48 		e.t.Fatal(err)
49 	}
50 	defer resp.Body.Close()
51 	b, _ := io.ReadAll(resp.Body)
52 	return resp.StatusCode, resp.Header.Get("Location"), string(b)

75 // editForm gives the fields v and n of the edit link of the bookmark with the
76 // title on the page.
77 func editForm(t *testing.T, page, title string) url.Values {
78 	t.Helper()
79 	for _, li := range strings.Split(page, "<li>") {
80 		if strings.Contains(li, ">"+title+"</a>") {
81 			if m := editLink.FindStringSubmatch(li); m != nil {
82 				return url.Values{"v": {m[1]}, "n": {m[2]}}
83 			}
84 		}
85 	}
86 	t.Fatalf("no edit link for %q on the page:\n%s", title, page)
87 	return nil
88 }

90 var rowTitle = regexp.MustCompile(`<li>(?:<a href="[^"]*"[^>]*>)?([^<]*)`)

92 // titles gives the titles of the bookmarks on the page, in their order.
93 func titles(page string) []string {
94 	var out []string
95 	for _, m := range rowTitle.FindAllStringSubmatch(page, -1) {
96 		out = append(out, m[1])
97 	}
98 	return out

494 func TestLiveSearch(t *testing.T) {
495 	e := newEnv(t, false)
496 	c, _, _ := e.account("https://a.org\tA\t\ngemini://b.org\tB\t\n")
497 	resp, err := c.Get(e.web.URL + "/bookmarks")
498 	if err != nil {
499 		t.Fatal(err)
500 	}
501 	b, _ := io.ReadAll(resp.Body)
502 	resp.Body.Close()
503 	// Scripts come only from this server, and fetch only from it.
504 	csp := resp.Header.Get("Content-Security-Policy")
505 	if !strings.Contains(csp, "script-src 'self';") || !strings.Contains(csp, "connect-src 'self';") || strings.Contains(csp, "unsafe-eval") {
506 		t.Errorf("Content-Security-Policy: %q", csp)
507 	}
508 	// The script finds the form, the results, and the links that Enter opens.
509 	// A bookmark without a web address has no such link.
510 	for _, want := range []string{`<form id="search" `, `<div id="results">`, `<a href="https://a.org" class="open">A</a>`,
511 		`<li>B<br>`, `<script src="/search.js" defer></script>`} {
512 		if !strings.Contains(string(b), want) {
513 			t.Errorf("the page has no %q", want)
514 		}
515 	}
516 	resp, err = e.browser().Get(e.web.URL + "/search.js")
517 	if err != nil {
518 		t.Fatal(err)
519 	}
520 	b, _ = io.ReadAll(resp.Body)
521 	resp.Body.Close()
522 	if resp.StatusCode != http.StatusOK || resp.Header.Get("Content-Type") != "text/javascript; charset=utf-8" || string(b) != searchJS {
523 		t.Errorf("GET /search.js: %s %q, %d bytes", resp.Status, resp.Header.Get("Content-Type"), len(b))
524 	}
525 }
