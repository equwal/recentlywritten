468 // A URL that feedURL takes has only printable ASCII without single quotes,
469 // so that it holds between single quotes in the sfeedrc. Each such http or
470 // https URL with a host is taken.
471 func TestFeedURLIsShellSafe(t *testing.T) {
472 	rapid.Check(t, func(t *rapid.T) {
473 		raw := feedURLs.Draw(t, "url")
474 		u := strings.TrimSpace(raw)
475 		safe := u != "" && len(u) <= maxFeedURL
476 		for i := 0; i < len(u); i++ {
477 			if c := u[i]; c <= ' ' || c > '~' || c == '\'' {
478 				safe = false
479 			}
480 		}
481 		p, err := url.Parse(u)
482 		want := safe && err == nil && p.Host != "" && (p.Scheme == "http" || p.Scheme == "https")
483 		got, err := feedURL(raw)
484 		if (err == nil) != want {
485 			t.Fatalf("feedURL(%q) = %q, %v; want ok %v", raw, got, err, want)
486 		}
487 		if err != nil {
488 			return
489 		}
490 		if got != u {
491 			t.Fatalf("feedURL(%q) = %q, want %q", raw, got, u)
492 		}
493 		rc := sfeedrc("/data/items", []Feed{{URL: got, Name: "x", File: "x"}})
494 		line := "\tfeed 'x' '" + got + "'\n"
495 		if !strings.Contains(rc, line) || strings.Count(line, "'") != 4 {
496 			t.Fatalf("sfeedrc for %q:\n%s", got, rc)
497 		}
498 	})
499 }
