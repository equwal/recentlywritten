102 // and letters of more than one byte, so that searches often match.

105 // matches is a slow reference for the search: case aside, the letters of
106 // each word of the query occur in the text in their order.
107 func matches(query, text string) bool {
108 	hay := []rune(strings.ToLower(text))
109 	for _, word := range strings.FieldsFunc(strings.ToLower(query), func(r rune) bool { return r == ' ' || r == '\t' }) {
110 		w, i := []rune(word), 0
111 		for _, r := range hay {
112 			if i < len(w) && r == w[i] {
113 				i++
114 			}
115 		}
116 		if i < len(w) {
117 			return false
118 		}
119 	}
120 	return true
121 }

202 func TestMergeRowsAddsEachURLOnce(t *testing.T) {
203 	rapid.Check(t, func(t *rapid.T) {
204 		text, rows := file(t, "file"), lines(file(t, "rows"))
205 		add, skipped := mergeRows(text, rows)
206 		seen := map[string]bool{}
207 		for _, l := range lines(text) {
208 			seen[key(l)] = true
209 		}
210 		for _, a := range add {
211 			if k := key(a); k == "" || seen[k] {
212 				t.Fatalf("added %q, but its URL is known", a)
213 			}
214 			seen[key(a)] = true
215 		}
216 		empty := 0
217 		for _, r := range rows {
218 			if key(r) == "" {
219 				empty++
220 			}
221 		}
222 		if len(add)+skipped+empty != len(rows) {
223 			t.Fatalf("%d added, %d skipped and %d empty of %d rows", len(add), skipped, empty, len(rows))
224 		}
