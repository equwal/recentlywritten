1 // sbm-sync keeps the sbm bookmark file of a user the same on each device.
2 //
3 // A device sends its copy of the file and the name of the version that it
4 // got last time. The server merges the changes of the device into its own
5 // copy, keeps the result and sends it back. Accounts are an email address
6 // and a password. Billing through Stripe is off unless STRIPE_SECRET_KEY is
7 // set.

414 // webhook takes the events of Stripe that change a subscription.
415 func (s *server) webhook(w http.ResponseWriter, r *http.Request) {
416 	if s.stripe == nil {
417 		http.NotFound(w, r)
418 		return
419 	}
420 	body, err := io.ReadAll(http.MaxBytesReader(w, r.Body, 1<<20))
421 	if err != nil {
422 		http.Error(w, "cannot read the event", http.StatusBadRequest)
423 		return
424 	}
425 	if err := verify(body, r.Header.Get("Stripe-Signature"), s.stripe.Webhook, time.Now()); err != nil {
426 		http.Error(w, err.Error(), http.StatusBadRequest)
427 		return
428 	}
429 	var ev struct {
430 		Type    string `json:"type"`
431 		Created int64  `json:"created"`
432 		Data    struct {
433 			Object struct {
434 				Customer  string            `json:"customer"`
435 				Reference string            `json:"client_reference_id"`
436 				Status    string            `json:"status"`
437 				Metadata  map[string]string `json:"metadata"`
438 			} `json:"object"`
439 		} `json:"data"`
440 	}
441 	if err := json.Unmarshal(body, &ev); err != nil {
442 		http.Error(w, "cannot read the event", http.StatusBadRequest)
443 		return
444 	}
445 	o := ev.Data.Object
446 	switch ev.Type {
447 	case "checkout.session.completed":
448 		if a := s.store.byID(o.Reference); a != nil && o.Customer != "" {
449 			err = s.store.setCustomer(a, o.Customer)
450 		}
451 	case "customer.subscription.created", "customer.subscription.updated", "customer.subscription.deleted":
452 		a := s.store.byCustomer(o.Customer)
453 		if a == nil {
454 			if a = s.store.byID(o.Metadata["account"]); a != nil {
455 				err = s.store.setCustomer(a, o.Customer)
456 			}
457 		}
458 		if a != nil && err == nil {
459 			status := o.Status
460 			if ev.Type == "customer.subscription.deleted" {
461 				status = "canceled"
462 			}
463 			err = s.store.setStatus(a, status, ev.Created)
464 			log.Printf("account %s: subscription %s", a.ID, status)
465 		}
466 	}
467 	if err != nil {
468 		s.fail(w, err) // Stripe sends the event again later
469 		return
470 	}
471 	w.WriteHeader(http.StatusOK)
472 }

514 // sync takes the bookmark file of a device as the body, and the version that
515 // the device got last time as ?base=. It gives the merged file, with the
516 // name of its version in the Sbm-Version header. The device writes the file
517 // and sends that name as base next time.
518 func (s *server) sync(w http.ResponseWriter, r *http.Request) {
519 	u := s.apiUser(r)
520 	if u == nil {
521 		http.Error(w, "not signed in: sign in again", http.StatusUnauthorized)
522 		return
523 	}
524 	if !s.active(s.store.view(u)) {
525 		http.Error(w, "sbm Sync is paused for this account: see "+s.site+"/account", http.StatusPaymentRequired)
526 		return
527 	}
528 	base := r.URL.Query().Get("base")
529 	if base != "" && !hashName.MatchString(base) {
530 		http.Error(w, "base is not a version name", http.StatusBadRequest)
531 		return
532 	}
533 	body, err := io.ReadAll(http.MaxBytesReader(w, r.Body, maxFile))
534 	var tooBig *http.MaxBytesError
535 	if errors.As(err, &tooBig) {
536 		http.Error(w, "the bookmark file is too large", http.StatusRequestEntityTooLarge)
537 		return
538 	} else if err != nil {
539 		http.Error(w, "cannot read the bookmark file", http.StatusBadRequest)
540 		return
541 	}
542 	text, version, err := s.store.sync(u, base, string(body))
543 	if errors.Is(err, errGone) {
544 		http.Error(w, "not signed in: sign in again", http.StatusUnauthorized)
545 		return
546 	} else if err != nil {
547 		s.fail(w, err)
548 		return
549 	}
550 	w.Header().Set("Content-Type", "text/plain; charset=utf-8")
551 	w.Header().Set("Sbm-Version", version)
552 	w.Header().Set("Access-Control-Expose-Headers", "Sbm-Version")
553 	io.WriteString(w, text)
554 }

561 // clientIP gives the address of the client. Behind a proxy on the same
562 // machine, the proxy gives it in X-Real-IP.
563 func clientIP(r *http.Request) string {
564 	host, _, err := net.SplitHostPort(r.RemoteAddr)
565 	if err != nil {
566 		host = r.RemoteAddr
567 	}
568 	if ip := net.ParseIP(host); ip != nil && ip.IsLoopback() {
569 		if real := r.Header.Get("X-Real-IP"); real != "" {
570 			return real
571 		}
572 	}
573 	return host
574 }

576 // limiter counts attempts for each key in a sliding window.
577 type limiter struct {
578 	mu   sync.Mutex
579 	hits map[string][]time.Time
580 }
581 
582 func (l *limiter) allow(key string, n int, per time.Duration) bool {
583 	l.mu.Lock()
584 	defer l.mu.Unlock()
585 	now := time.Now()
586 	if l.hits == nil || len(l.hits) > 10000 {
587 		l.hits = map[string][]time.Time{}
588 	}
589 	recent := l.hits[key][:0]
590 	for _, t := range l.hits[key] {
591 		if now.Sub(t) < per {
592 			recent = append(recent, t)
593 		}
594 	}
595 	if len(recent) >= n {
596 		l.hits[key] = recent
597 		return false
598 	}
599 	l.hits[key] = append(recent, now)
600 	return true
601 }
