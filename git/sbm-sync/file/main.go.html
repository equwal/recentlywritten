1 // sbm-sync keeps the sbm bookmark file of a user the same on each device.
2 //
3 // A device sends its copy of the file and the name of the version that it
4 // got last time. The server merges the changes of the device into its own
5 // copy, keeps the result and sends it back. Accounts are an email address
6 // and a password. Billing through Stripe is off unless STRIPE_SECRET_KEY is
7 // set. Accounts confirm their email address only when SBM_SMTP_HOST is set.

58 func main() {
59 	site := strings.TrimRight(env("SBM_URL", "http://localhost:8750"), "/")
60 	u, err := url.Parse(site)
61 	if err != nil || u.Host == "" {
62 		log.Fatalf("SBM_URL: not an address: %q", site)
63 	}
64 	days, err := strconv.Atoi(env("SBM_TRIAL_DAYS", "30"))
65 	if err != nil {
66 		log.Fatalf("SBM_TRIAL_DAYS: %v", err)
67 	}
68 	var since time.Time
69 	if v := os.Getenv("SBM_BILLING_START"); v != "" {
70 		if since, err = time.Parse("2006-01-02", v); err != nil {
71 			log.Fatalf("SBM_BILLING_START: %v", err)
72 		}
73 	}
74 	store, err := openStore(env("SBM_DATA", "data"))
75 	if err != nil {
76 		log.Fatal(err)
77 	}
78 	if store.defaultFeeds, err = defaultFeeds(os.Getenv("SBM_DEFAULT_FEEDS")); err != nil {
79 		log.Fatal(err)
80 	}
81 	s := &server{
82 		store:      store,
83 		site:       site,
84 		origin:     u.Scheme + "://" + u.Host,
85 		secure:     u.Scheme == "https",
86 		trial:      time.Duration(days) * 24 * time.Hour,
87 		since:      since,
88 		contact:    os.Getenv("SBM_CONTACT"),
89 		teams:      link("SBM_TEAMS_URL"),
90 		teamsLarge: link("SBM_TEAMS_LARGE_URL"),
91 		support:    link("SBM_SUPPORT_URL"),
92 		manage:     link("SBM_MANAGE_URL"),
93 		prices:     map[string]string{"month": os.Getenv("STRIPE_PRICE_MONTH"), "year": os.Getenv("STRIPE_PRICE_YEAR")},
94 		labels:     map[string]string{"month": "monthly", "year": "yearly"},
95 	}
96 	if key := os.Getenv("STRIPE_SECRET_KEY"); key != "" {
97 		for _, v := range []string{"STRIPE_WEBHOOK_SECRET", "STRIPE_PRICE_MONTH", "STRIPE_PRICE_YEAR"} {
98 			if os.Getenv(v) == "" {
99 				log.Fatalf("STRIPE_SECRET_KEY is set, so %s must be set too", v)
100 			}
101 		}
102 		s.stripe = &Stripe{Key: key, Webhook: os.Getenv("STRIPE_WEBHOOK_SECRET"),
103 			API: "https://api.stripe.com", Client: &http.Client{Timeout: 20 * time.Second}}
104 		for plan, price := range s.prices {
105 			if l, err := s.stripe.label(price); err != nil {
106 				log.Printf("price of the %s plan: %v", plan, err)
107 			} else {
108 				s.labels[plan] = l
109 			}
110 		}
111 	}
112 	if host := os.Getenv("SBM_SMTP_HOST"); host != "" {
113 		from, err := mail.ParseAddress(env("SBM_MAIL_FROM", os.Getenv("SBM_SMTP_USER")))
114 		if err != nil {
115 			log.Fatalf("SBM_MAIL_FROM: %v", err)
116 		}
117 		port := env("SBM_SMTP_PORT", "587")
118 		m := &Mailer{Addr: net.JoinHostPort(host, port), TLS: port == "465",
119 			User: os.Getenv("SBM_SMTP_USER"), Password: os.Getenv("SBM_SMTP_PASSWORD"),
120 			From: from, Hello: u.Hostname()}
121 		s.mail = m.send
122 	}
123 	if len(os.Args) > 1 {
124 		if os.Args[1] != "feed" {
125 			log.Fatalf("unknown command %q: the one command is feed", os.Args[1])
126 		}
127 		s.feeds = execTools()
128 		s.runFeeds()
129 		return
130 	}
131 	addr := env("SBM_ADDR", "127.0.0.1:8750")
132 	log.Printf("sbm-sync on %s for %s, billing %v, email check %v", addr, site, s.stripe != nil, s.mail != nil)
133 	hs := &http.Server{
134 		Addr:              addr,
135 		Handler:           s.routes(),
136 		ReadHeaderTimeout: 10 * time.Second,
137 		ReadTimeout:       60 * time.Second,
138 		WriteTimeout:      60 * time.Second,
139 		IdleTimeout:       120 * time.Second,
140 		MaxHeaderBytes:    16 << 10,
141 	}
142 	log.Fatal(hs.ListenAndServe())
143 }

323 		// The account page tells how to send the email again.
324 		if err := s.sendCheck(a.Email, code); err != nil {
325 			log.Printf("email check for %s: %v", a.ID, err)

346 // resend sends the email check again, with a new code.
347 func (s *server) resend(w http.ResponseWriter, r *http.Request) {
348 	u := s.user(r)
349 	if !s.sameSite(w, r) {
350 		return
351 	}
352 	if u == nil || s.verified(s.store.view(u)) {
353 		http.Redirect(w, r, "/account", http.StatusSeeOther)
354 		return
355 	}

358 		s.showAccount(w, http.StatusTooManyRequests, a, "Too many emails in one hour. Try again later.")
359 		return

362 	if err != nil {
363 		s.fail(w, err)
364 		return
365 	}
366 	if err := s.sendCheck(a.Email, code); err != nil {
367 		log.Printf("email check for %s: %v", a.ID, err)
368 		s.showAccount(w, http.StatusBadGateway, a, "The email did not go out. Try again later.")
369 		return
370 	}
371 	http.Redirect(w, r, "/account", http.StatusSeeOther)

484 // state tells in one sentence if sync works for the account, and why.
485 func (s *server) state(a Account) string {
486 	left := s.trialLeft(a)
487 	switch {
488 	case s.stripe == nil:
489 		return "Sync is on."
490 	case a.Status == "past_due":
491 		return "Your last payment failed. Update your card in Manage billing."
492 	case paid(a.Status):
493 		return "Your subscription is active."
494 	case left > 0:
495 		days := int(left.Hours()/24) + 1
496 		if days == 1 {
497 			return "Free trial: 1 day left."
498 		}
499 		return "Free trial: " + strconv.Itoa(days) + " days left."
500 	default:
501 		return "Your free trial has ended. Sync is paused until you subscribe."
502 	}

541 	}

543 	if err != nil {
544 		log.Printf("portal for %s: %v", u.ID, err)
545 		s.showAccount(w, http.StatusBadGateway, a, "Stripe did not answer. Try again later.")
546 		return
547 	}
548 	http.Redirect(w, r, to, http.StatusSeeOther)

551 func (s *server) remove(w http.ResponseWriter, r *http.Request) {
552 	u := s.user(r)
553 	if !s.sameSite(w, r) {
554 		return
555 	}
556 	if u == nil {
557 		http.Redirect(w, r, "/login", http.StatusSeeOther)
558 		return
559 	}
560 	a := s.store.view(u)
561 	if _, err := s.store.check(a.Email, r.PostFormValue("password")); err != nil {
562 		s.showAccount(w, http.StatusUnauthorized, a, "Wrong password. The account is not deleted.")
563 		return
564 	}
565 	// End the subscription first: a deleted account must not pay.
566 	if a.Customer != "" && s.stripe != nil {
567 		if err := s.stripe.deleteCustomer(a.Customer); err != nil {
568 			log.Printf("delete customer of %s: %v", a.ID, err)
569 			s.showAccount(w, http.StatusBadGateway, a, "Stripe did not answer, so the account is not deleted. Try again later.")
570 			return
571 		}
572 	}
573 	if err := s.store.remove(u); err != nil {
574 		s.fail(w, err)
575 		return
576 	}
577 	log.Printf("deleted account %s", a.ID)
578 	s.setCookie(w, "", -1)
579 	http.Redirect(w, r, "/", http.StatusSeeOther)
580 }

589 // webhook takes the events of Stripe that change a subscription.
590 func (s *server) webhook(w http.ResponseWriter, r *http.Request) {
591 	if s.stripe == nil {
592 		http.NotFound(w, r)
593 		return
594 	}
595 	body, err := io.ReadAll(http.MaxBytesReader(w, r.Body, 1<<20))
596 	if err != nil {
597 		http.Error(w, "cannot read the event", http.StatusBadRequest)
598 		return
599 	}
600 	if err := verify(body, r.Header.Get("Stripe-Signature"), s.stripe.Webhook, time.Now()); err != nil {
601 		http.Error(w, err.Error(), http.StatusBadRequest)
602 		return
603 	}
604 	var ev struct {
605 		Type    string `json:"type"`
606 		Created int64  `json:"created"`
607 		Data    struct {
608 			Object struct {
609 				Customer  string            `json:"customer"`
610 				Reference string            `json:"client_reference_id"`
611 				Status    string            `json:"status"`
612 				Metadata  map[string]string `json:"metadata"`
613 			} `json:"object"`
614 		} `json:"data"`
615 	}
616 	if err := json.Unmarshal(body, &ev); err != nil {
617 		http.Error(w, "cannot read the event", http.StatusBadRequest)
618 		return
619 	}
620 	o := ev.Data.Object
621 	switch ev.Type {
622 	case "checkout.session.completed":
623 		if a := s.store.byID(o.Reference); a != nil && o.Customer != "" {
624 			err = s.store.setCustomer(a, o.Customer)
625 		}
626 	case "customer.subscription.created", "customer.subscription.updated", "customer.subscription.deleted":
627 		a := s.store.byCustomer(o.Customer)
628 		if a == nil {
629 			if a = s.store.byID(o.Metadata["account"]); a != nil {
630 				err = s.store.setCustomer(a, o.Customer)
631 			}
632 		}
633 		if a != nil && err == nil {
634 			status := o.Status
635 			if ev.Type == "customer.subscription.deleted" {
636 				status = "canceled"
637 			}
638 			err = s.store.setStatus(a, status, ev.Created)
639 			log.Printf("account %s: subscription %s", a.ID, status)
640 		}
641 	}
642 	if err != nil {
643 		s.fail(w, err) // Stripe sends the event again later
644 		return
645 	}
646 	w.WriteHeader(http.StatusOK)
647 }

689 // sync takes the bookmark file of a device as the body, and the version that
690 // the device got last time as ?base=. It gives the merged file, with the
691 // name of its version in the Sbm-Version header. The device writes the file
692 // and sends that name as base next time.
693 func (s *server) sync(w http.ResponseWriter, r *http.Request) {
694 	u := s.apiOwner(w, r)
695 	if u == nil {
696 		return
697 	}
698 	base := r.URL.Query().Get("base")
699 	if base != "" && !hashName.MatchString(base) {
700 		http.Error(w, "base is not a version name", http.StatusBadRequest)
701 		return
702 	}
703 	body, err := io.ReadAll(http.MaxBytesReader(w, r.Body, maxFile))
704 	var tooBig *http.MaxBytesError
705 	if errors.As(err, &tooBig) {
706 		http.Error(w, "the bookmark file is too large", http.StatusRequestEntityTooLarge)
707 		return
708 	} else if err != nil {
709 		http.Error(w, "cannot read the bookmark file", http.StatusBadRequest)
710 		return
711 	}
712 	text, version, err := s.store.sync(u, base, string(body))
713 	if errors.Is(err, errGone) {
714 		http.Error(w, "not signed in: sign in again", http.StatusUnauthorized)
715 		return
716 	} else if err != nil {
717 		s.fail(w, err)
718 		return
719 	}
720 	w.Header().Set("Content-Type", "text/plain; charset=utf-8")
721 	w.Header().Set("Sbm-Version", version)
722 	w.Header().Set("Access-Control-Expose-Headers", "Sbm-Version")
723 	io.WriteString(w, text)
724 }

836 // clientIP gives the address of the client. Behind a proxy on the same
837 // machine, the proxy gives it in X-Real-IP.
838 func clientIP(r *http.Request) string {
839 	host, _, err := net.SplitHostPort(r.RemoteAddr)
840 	if err != nil {
841 		host = r.RemoteAddr
842 	}
843 	if ip := net.ParseIP(host); ip != nil && ip.IsLoopback() {
844 		if real := r.Header.Get("X-Real-IP"); real != "" {
845 			return real
846 		}
847 	}
848 	return host
849 }

851 // limiter counts attempts for each key in a sliding window.
852 type limiter struct {
853 	mu   sync.Mutex
854 	hits map[string][]time.Time
855 }
856 
857 func (l *limiter) allow(key string, n int, per time.Duration) bool {
858 	l.mu.Lock()
859 	defer l.mu.Unlock()
860 	now := time.Now()
861 	if l.hits == nil || len(l.hits) > 10000 {
862 		l.hits = map[string][]time.Time{}
863 	}
864 	recent := l.hits[key][:0]
865 	for _, t := range l.hits[key] {
866 		if now.Sub(t) < per {
867 			recent = append(recent, t)
868 		}
869 	}
870 	if len(recent) >= n {
871 		l.hits[key] = recent
872 		return false
873 	}
874 	l.hits[key] = append(recent, now)
875 	return true
876 }
