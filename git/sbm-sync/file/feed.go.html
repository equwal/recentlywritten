3 // Feeds. An account can follow RSS and Atom feeds. The list of an account
4 // is one plain file, feeds.txt, with one feed on each line:
5 //
6 //	URL<tab>name
7 //
8 // The command `sbm-sync feed` (feedjob.go) fetches the feeds with
9 // sfeed_update into one sfeed(5) file for each feed, looks through the
10 // bookmarks of the account for feeds with sfeed_web when the account asked
11 // for it, and sends a daily digest of the new items by email. The page /feed
12 // shows the items, the feeds and the settings. GET /api/feed gives the items
13 // in the sfeed format, for sfeed_plain and the other sfeed tools.
14 //
15 // The files of an account are under feeds/<account ID>/:
16 //
17 //	feeds.txt    the feeds, written by the web pages
18 //	sfeedrc      the sfeed_update settings, made from feeds.txt by the job
19 //	items/<file> the items of one feed, written by sfeed_update
20 //	explore      the feeds found on the bookmarked pages, written by the job
21 //	mailed       the keys of the items that went out in a digest
22 //	digest       the time of the last digest

93 // sharedFeeds gives the feeds of a share link: the url and name fields, in
94 // pairs. The values that are not feed addresses come back in bad.
95 func sharedFeeds(q url.Values) (feeds []Feed, bad []string) {
96 	names := q["name"]
97 	for i, u := range q["url"] {
98 		if i >= maxShare {
99 			break
100 		}
101 		addr, err := feedURL(u)
102 		if err != nil {
103 			bad = append(bad, u)
104 			continue
105 		}
106 		name := ""
107 		if i < len(names) {
108 			name = names[i]

110 		feeds = append(feeds, Feed{URL: addr, Name: feedName(name, addr)})
111 	}
112 	return feeds, bad

115 // feedURL checks the URL of a feed. It takes only http and https addresses
116 // of printable ASCII without quotes, so that the URL can go into the
117 // sfeedrc, a shell script, between single quotes.
118 func feedURL(raw string) (string, error) {
119 	u := strings.TrimSpace(raw)
120 	if u == "" {
121 		return "", errNoURL
122 	}
123 	if len(u) > maxFeedURL {
124 		return "", errFeedLong
125 	}
126 	for i := 0; i < len(u); i++ {
127 		if c := u[i]; c <= ' ' || c > '~' || c == '\'' {
128 			return "", errFeedURL
129 		}
130 	}
131 	p, err := url.Parse(u)
132 	if err != nil || p.Host == "" || (p.Scheme != "http" && p.Scheme != "https") {
133 		return "", errFeedURL
134 	}
135 	return u, nil
136 }

164 // parseFeeds gives the feeds of feeds.txt, in the order of the file. Each
165 // feed gets a file name that no other feed of the file has.
166 func parseFeeds(text string) []Feed {
167 	var out []Feed
168 	used := map[string]bool{}
169 	for _, l := range lines(text) {
170 		u, name, _ := strings.Cut(l, "\t")
171 		if u == "" {
172 			continue
173 		}
174 		f := Feed{URL: u, Name: name, File: fileName(name)}
175 		for i := 2; used[f.File]; i++ {
176 			f.File = fileName(name) + "-" + strconv.Itoa(i)
177 		}
178 		used[f.File] = true
179 		out = append(out, f)
180 	}
181 	return out
182 }

196 // unescape undoes the escapes of the sfeed format: \t, \n and \\.
197 func unescape(s string) string {
198 	if !strings.Contains(s, `\`) {
199 		return s
200 	}
201 	var b strings.Builder
202 	for i := 0; i < len(s); i++ {
203 		if s[i] == '\\' && i+1 < len(s) {
204 			switch s[i+1] {
205 			case 't':
206 				b.WriteByte('\t')
207 				i++
208 				continue
209 			case 'n':
210 				b.WriteByte('\n')
211 				i++
212 				continue
213 			case '\\':
214 				b.WriteByte('\\')
215 				i++
216 				continue
217 			}
218 		}
219 		b.WriteByte(s[i])
220 	}
221 	return b.String()
222 }

224 // parseItems gives the items of the sfeed file of a feed, in the order of
225 // the file.
226 func parseItems(text string, f Feed) []Item {
227 	var out []Item
228 	for _, l := range lines(text) {
229 		c := strings.Split(l, "\t")
230 		if len(c) < 3 {
231 			continue
232 		}
233 		for len(c) < 9 {
234 			c = append(c, "")
235 		}
236 		it := Item{Title: unescape(c[1]), Author: unescape(c[6]), Feed: f.Name, File: f.File, Raw: l}
237 		if t, err := strconv.ParseInt(c[0], 10, 64); err == nil {
238 			it.Time = time.Unix(t, 0).UTC()
239 		}
240 		link := unescape(c[2])
241 		if link != "" {
242 			it.Link = Bookmark{URL: link}.Link()
243 		}
244 		id := c[5]
245 		if id == "" {
246 			id = c[2]
247 		}
248 		if id == "" {
249 			id = c[1]
250 		}
251 		it.Key = f.File + "\t" + id
252 		if it.Title == "" {
253 			it.Title = link
254 		}
255 		if it.Title == "" {
256 			it.Title = "(no title)"
257 		}
258 		out = append(out, it)
259 	}
260 	return out
261 }

362 // suggestions gives the feeds of the explore file that the account does not
363 // follow, the number of bookmarked pages that the job looked at, and the
364 // number of bookmarks with a web address.
365 func (s *server) suggestions(a *Account, feeds []Feed) ([]suggestion, int, int, error) {
366 	text, _, err := s.store.read(a, "")
367 	if err != nil {
368 		return nil, 0, 0, err
369 	}
370 	names := map[string]string{}
371 	total := 0
372 	for _, b := range parse(text) {
373 		if l := b.Link(); l != "" {
374 			names[l] = b.Desc
375 			total++
376 		}
377 	}
378 	b, err := os.ReadFile(filepath.Join(s.store.feedDir(a.ID), "explore"))
379 	if err != nil && !errors.Is(err, fs.ErrNotExist) {
380 		return nil, 0, 0, err
381 	}
382 	known := map[string]bool{}
383 	for _, f := range feeds {
384 		known[norm(f.URL)] = true
385 	}
386 	pages := map[string]bool{}
387 	var out []suggestion
388 	for _, l := range lines(string(b)) {
389 		f := strings.Split(l, "\t")
390 		if len(f) < 2 {
391 			continue
392 		}
393 		if _, ok := names[f[0]]; ok {
394 			pages[f[0]] = true // a page that is a bookmark still
395 		}
396 		if f[1] == "-" || known[norm(f[1])] {
397 			continue
398 		}
399 		known[norm(f[1])] = true
400 		out = append(out, suggestion{URL: f[1], Page: f[0], Name: feedName(names[f[0]], f[1])})
401 	}
402 	return out, len(pages), total, nil
403 }

433 // feedData gives the feed page of the account: the items from start, or
434 // only the items of the feed with the file name filter.
435 func (s *server) feedData(a *Account, filter string, start int) (feedPage, error) {
436 	p := feedPage{page: s.page("Your feed"), Filter: filter, Start: start, HasMail: s.mail != nil,
437 		Settings: s.store.view(a).Feed}
438 	text, err := s.store.readFeeds(a)
439 	if err != nil {
440 		return p, err
441 	}
442 	p.Feeds = parseFeeds(text)
443 	p.ShareAll = shareLink(p.Feeds)
444 	items := s.feedItems(a, p.Feeds)
445 	if filter != "" {
446 		items = slices.DeleteFunc(items, func(it Item) bool { return it.File != filter })
447 		for _, f := range p.Feeds {
448 			if f.File == filter {
449 				p.FilterName = f.Name
450 			}
451 		}
452 	}
453 	p.Total = len(items)
454 	if start > len(items) {
455 		start = len(items)
456 	}
457 	p.Items = items[start:min(start+itemsOnPage, len(items))]
458 	if start+itemsOnPage < len(items) {
459 		p.Next = start + itemsOnPage
460 	}
461 	if p.Settings.Explore {
462 		p.Suggestions, p.Checked, p.Bookmarks, err = s.suggestions(a, p.Feeds)
463 	}
464 	return p, err
465 }

567 // followFeeds adds the feeds of the form to feeds.txt, as the button of the
568 // follow page sends them. Feeds that the account follows already are
569 // skipped.
570 func (s *server) followFeeds(w http.ResponseWriter, r *http.Request) {
571 	if !s.sameSite(w, r) {
572 		return
573 	}
574 	u := s.owner(w, r)
575 	if u == nil {
576 		return
577 	}
578 	r.Body = http.MaxBytesReader(w, r.Body, maxForm)
579 	if err := r.ParseForm(); err != nil {
580 		http.Error(w, "cannot read the form", http.StatusBadRequest)
581 		return
582 	}
583 	feeds, _ := sharedFeeds(r.PostForm)
584 	added, known := 0, 0
585 	err := s.store.changeFeeds(u, func(text string) (string, error) {
586 		ls := lines(text)
587 		have := map[string]bool{}
588 		for _, f := range parseFeeds(text) {
589 			have[norm(f.URL)] = true
590 		}
591 		for _, f := range feeds {
592 			switch {
593 			case have[norm(f.URL)]:
594 				known++
595 			case len(have) >= maxFeeds:
596 				return "", errFeedMany
597 			default:
598 				have[norm(f.URL)] = true
599 				ls = append(ls, f.URL+"\t"+f.Name)
600 				added++
601 			}
602 		}
603 		return join(ls), nil
604 	})
605 	if errors.Is(err, errFeedMany) {
606 		p, perr := s.feedData(u, "", 0)
607 		if perr != nil {
608 			s.fail(w, perr)
609 			return
610 		}
611 		p.Error = sentence(err)
612 		s.render(w, http.StatusBadRequest, "feed", p)
613 		return
614 	} else if err != nil {
615 		s.fail(w, err)
616 		return
617 	}
618 	http.Redirect(w, r, fmt.Sprintf("/feed?done=followed&added=%d&known=%d", added, known), http.StatusSeeOther)
619 }

631 	raw, name := r.PostFormValue("url"), r.PostFormValue("name")
632 	addr, err := feedURL(raw)
633 	if err == nil {
634 		err = s.store.changeFeeds(u, func(text string) (string, error) {
635 			feeds := parseFeeds(text)
636 			if len(feeds) >= maxFeeds {
637 				return "", errFeedMany
638 			}
639 			for _, f := range feeds {
640 				if norm(f.URL) == norm(addr) {
641 					return "", errFeedKnown
642 				}
643 			}
644 			return join(append(lines(text), addr+"\t"+feedName(name, addr))), nil
645 		})
646 	}
647 	status := http.StatusBadRequest
648 	switch {
649 	case err == nil:
650 		http.Redirect(w, r, "/feed?done=added", http.StatusSeeOther)
651 		return
652 	case errors.Is(err, errFeedKnown):
653 		status = http.StatusConflict
654 	case errors.Is(err, errNoURL), errors.Is(err, errFeedURL), errors.Is(err, errFeedLong), errors.Is(err, errFeedMany):
655 	default:
656 		s.fail(w, err)
657 		return
658 	}
659 	p, perr := s.feedData(u, "", 0)
660 	if perr != nil {
661 		s.fail(w, perr)
662 		return
663 	}
664 	p.Error, p.Open, p.Form.URL, p.Form.Name = sentence(err), "add", raw, name
665 	s.render(w, status, "feed", p)
666 }

668 // unfollowFeed removes the feed with the URL of the form from feeds.txt.
669 func (s *server) unfollowFeed(w http.ResponseWriter, r *http.Request) {
670 	if !s.sameSite(w, r) {
671 		return
672 	}
673 	u := s.owner(w, r)
674 	if u == nil {
675 		return
676 	}
677 	r.Body = http.MaxBytesReader(w, r.Body, maxForm)
678 	k := norm(strings.TrimSpace(r.PostFormValue("url")))
679 	err := s.store.changeFeeds(u, func(text string) (string, error) {
680 		var out []string
681 		for _, l := range lines(text) {
682 			if k == "" || key(l) != k {
683 				out = append(out, l)
684 			}
685 		}
686 		return join(out), nil
687 	})
688 	if err != nil {
689 		s.fail(w, err)
690 		return
691 	}
692 	http.Redirect(w, r, "/feed?done=deleted", http.StatusSeeOther)
693 }

723 // apiOwner gives the account of the token when it can use its bookmarks
724 // and feeds: as for sync, its email is confirmed and sync is not paused.
725 // Else apiOwner answers the request and gives nil.
726 func (s *server) apiOwner(w http.ResponseWriter, r *http.Request) *Account {
727 	u := s.apiUser(r)
728 	if u == nil {
729 		http.Error(w, "not signed in: sign in again", http.StatusUnauthorized)
730 		return nil
731 	}
732 	if !s.verified(s.store.view(u)) {
733 		http.Error(w, "confirm your email address first: open the link in the email from sbm Sync, or see "+s.site+"/account", http.StatusForbidden)
734 		return nil
735 	}
736 	if !s.active(s.store.view(u)) {
737 		http.Error(w, "sbm Sync is paused for this account: see "+s.site+"/account", http.StatusPaymentRequired)
738 		return nil
739 	}
740 	return u
741 }

743 // apiFeed gives the items of all feeds of the account in the sfeed format,
744 // newest first, for the sfeed tools:
745 //
746 //	curl -H "Authorization: Bearer $token" https://sbmsync.com/api/feed | sfeed_plain
747 func (s *server) apiFeed(w http.ResponseWriter, r *http.Request) {
748 	u := s.apiOwner(w, r)
749 	if u == nil {
750 		return
751 	}

754 		s.fail(w, err)

756 	}
757 	w.Header().Set("Content-Type", "text/plain; charset=utf-8")
758 	for _, it := range s.feedItems(u, parseFeeds(text)) {
759 		io.WriteString(w, it.Raw+"\n")

761 }
