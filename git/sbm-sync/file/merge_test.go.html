11 // file draws a bookmark file from few URLs and descriptions, so that files
12 // share lines, and lines share bookmarks, as often as real files do.
13 func file(t *rapid.T, label string) string {
14 	pool := []string{
15 		"https://suckless.org/\tsuckless\ttools",
16 		"https://suckless.org\tsuckless software\ttools c",
17 		"http://www.suckless.org\tsuckless",
18 		"https://example.org/a\tA\t",
19 		"https://example.org/a\tA, edited\tx",
20 		"https://example.org/b\tB\t",
21 		"https://jisho.org\tJisho: 日本語の辞書\tja",
22 		"https://jisho.org\tJisho\t",
23 		"# comment",
24 		"",
25 	}
26 	ls := rapid.SliceOfN(rapid.SampledFrom(pool), 0, 8).Draw(t, label)
27 	return join(ls)
28 }

30 // unique draws a file without two equal lines.
31 func unique(t *rapid.T, label string) string {
32 	ls := lines(file(t, label))
33 	var out []string
34 	for _, l := range ls {
35 		if !slices.Contains(out, l) {
36 			out = append(out, l)
37 		}
38 	}
39 	return join(out)
40 }

142 		}

158 func TestMergeExamples(t *testing.T) {
159 	cases := []struct{ name, base, local, remote, want string }{
160 		{"first device", "", "a\t1\n", "", "a\t1\n"},
161 		{"second device: union", "", "b\t2\n", "a\t1\n", "a\t1\nb\t2\n"},
162 		{"local add goes to the end", "a\t1\n", "a\t1\nc\t3\n", "a\t1\nb\t2\n", "a\t1\nb\t2\nc\t3\n"},
163 		{"local delete", "a\t1\nb\t2\n", "b\t2\n", "a\t1\nb\t2\nc\t3\n", "b\t2\nc\t3\n"},
164 		{"local edit keeps its place", "a\t1\nb\t2\n", "a\tone\nb\t2\n", "a\t1\nb\t2\n", "a\tone\nb\t2\n"},
165 		{"both edit: the device wins", "a\t1\n", "a\tlocal\n", "a\tremote\n", "a\tlocal\n"},
166 		{"local delete, remote edit: the edit stays", "a\t1\n", "", "a\tremote\n", "a\tremote\n"},
167 		{"CRLF", "", "a\t1\r\nb\t2\r\n", "", "a\t1\nb\t2\n"},
168 		{"no final newline", "", "a\t1", "", "a\t1\n"},
169 		{"a file of one empty line", "", "\n", "", "\n"},
170 		{"an empty line from the server", "a\n", "", "a\n\n", "\n"},
171 		{"same page, other form", "", "https://www.x.org/\tx\n", "http://x.org\tx\n", "https://www.x.org/\tx\n"},
172 	}
173 	for _, c := range cases {
174 		if got := merge(c.base, c.local, c.remote); got != c.want {
175 			t.Errorf("%s: got %q, want %q", c.name, got, c.want)
176 		}
177 	}
178 }
