1 # The feed job of sbm-sync: it fetches the feeds of every account with
2 # sfeed_update, looks through bookmarks for feeds, and sends the daily
3 # digests. sbm-feed.timer starts it every 30 minutes. Install:
4 #   apt-get install sfeed
5 #   install -m 644 contrib/sbm-feed.service contrib/sbm-feed.timer /etc/systemd/system/
6 #   systemctl daemon-reload && systemctl enable --now sbm-feed.timer
