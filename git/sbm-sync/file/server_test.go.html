29 func (f *fakeStripe) ServeHTTP(w http.ResponseWriter, r *http.Request) {
30 	r.ParseForm()
31 	f.mu.Lock()
32 	f.calls = append(f.calls, r.Method+" "+r.URL.Path+" "+r.PostForm.Encode())
33 	f.mu.Unlock()
34 	switch {
35 	case r.URL.Path == "/v1/checkout/sessions":
36 		io.WriteString(w, `{"url": "https://checkout.stripe.test/s1"}`)
37 	case r.URL.Path == "/v1/billing_portal/sessions":
38 		io.WriteString(w, `{"url": "https://billing.stripe.test/p1"}`)
39 	case strings.HasPrefix(r.URL.Path, "/v1/customers/"):
40 		io.WriteString(w, `{"deleted": true}`)
41 	default:
42 		w.WriteHeader(http.StatusNotFound)
43 		io.WriteString(w, `{"error": {"message": "no such thing"}}`)
44 	}
45 }

402 func sign(body []byte, secret string, t int64) string {
403 	mac := hmac.New(sha256.New, []byte(secret))
404 	fmt.Fprintf(mac, "%d.%s", t, body)
405 	return fmt.Sprintf("t=%d,v1=%s", t, hex.EncodeToString(mac.Sum(nil)))
406 }

408 func TestVerify(t *testing.T) {
409 	body := []byte(`{"type":"x"}`)
410 	now := time.Unix(1_800_000_000, 0)
411 	good := sign(body, "whsec_a", now.Unix())
412 	if err := verify(body, good, "whsec_a", now); err != nil {
413 		t.Errorf("good signature: %v", err)
414 	}
415 	if err := verify(body, "t=1,v1=00,"+strings.TrimPrefix(good, "t=1800000000,"), "whsec_a", now); err == nil {
416 		t.Error("a signature for another time passes")
417 	}
418 	if err := verify(body, "v1=00,"+good, "whsec_a", now); err != nil {
419 		t.Errorf("one good signature among more: %v", err)
420 	}
421 	for name, c := range map[string]struct {
422 		body   string
423 		header string
424 		secret string
425 		now    time.Time
426 	}{
427 		"changed body":  {`{"type":"y"}`, good, "whsec_a", now},
428 		"other secret":  {string(body), good, "whsec_b", now},
429 		"old":           {string(body), good, "whsec_a", now.Add(6 * time.Minute)},
430 		"no header":     {string(body), "", "whsec_a", now},
431 		"no secret set": {string(body), good, "", now},
432 	} {
433 		if err := verify([]byte(c.body), c.header, c.secret, c.now); err == nil {
434 			t.Errorf("%s: passes", name)
435 		}
436 	}
437 	e := newEnv(t, true)
438 	req, _ := http.NewRequest("POST", e.web.URL+"/stripe", strings.NewReader(`{}`))

587 // api sends a request with the token of a device, as the add-on does, and
588 // gives the status and the body of the answer.
589 func (e *harness) api(method, token, path string, form url.Values) (int, string) {
590 	req, _ := http.NewRequest(method, e.web.URL+path, strings.NewReader(form.Encode()))
591 	req.Header.Set("Authorization", "Bearer "+token)
592 	req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
593 	resp, err := http.DefaultClient.Do(req)
594 	if err != nil {
595 		e.t.Fatal(err)
596 	}
597 	defer resp.Body.Close()
598 	b, _ := io.ReadAll(resp.Body)
599 	return resp.StatusCode, string(b)
600 }
