91 // parseLine gives the bookmark of a line, and false for a comment or an
92 // empty line. A line without a tab has the old format "URL description |
93 // tags". Until bm-migrate converts it, its first word is the URL, as in the
94 // add-on.
95 func parseLine(line string) (Bookmark, bool) {
96 	if strings.TrimSpace(line) == "" || strings.HasPrefix(line, "#") {
97 		return Bookmark{}, false
98 	}
99 	fields := strings.Split(line, "\t")
100 	if len(fields) == 1 {
101 		words := strings.Fields(line)
102 		return Bookmark{URL: words[0], Desc: strings.Join(words[1:], " ")}, true
103 	}
104 	b := Bookmark{URL: strings.TrimSpace(fields[0]), Desc: fields[1]}
105 	if len(fields) > 2 {
106 		b.Tags = strings.FieldsFunc(fields[2], func(r rune) bool { return r == ' ' })
107 	}
108 	return b, b.URL != ""
109 }

126 // line gives the line of the bookmark, as bm and the add-on write it: the URL
127 // without white space, the description without tabs and line breaks, and the
128 // tags with one space between them.
129 func (e entry) line() (string, error) {
130 	u := strings.Join(strings.Fields(e.URL), "")
131 	switch {
132 	case u == "":
133 		return "", errNoURL
134 	case strings.HasPrefix(u, "#"):
135 		return "", errHashURL
136 	}
137 	desc := strings.TrimSpace(strings.NewReplacer("\t", " ", "\r", " ", "\n", " ").Replace(e.Desc))
138 	return u + "\t" + desc + "\t" + strings.Join(strings.Fields(e.Tags), " "), nil

143 // score gives the score of query in text, higher for a better match, or -1
144 // when a word of the query does not occur in text. This is the search of the
145 // add-on (lib/fuzzy.js), as in fzf: each word of the query must occur in the
146 // text, with its letters in order but not always together. Case does not
147 // matter.
148 func score(query, text string) int {
149 	hay := strings.ToLower(text)
150 	total := 0
151 	for _, word := range strings.FieldsFunc(strings.ToLower(query), func(r rune) bool { return r == ' ' || r == '\t' }) {
152 		s := scoreWord(word, hay)
153 		if s < 0 {
154 			return -1
155 		}
156 		total += s
157 	}
158 	return total
159 }
160 
161 // scoreWord gives the score of one word in hay. The whole word in one place
162 // scores highest, most at the start of a word.
163 func scoreWord(word, hay string) int {
164 	if at := strings.Index(hay, word); at >= 0 {
165 		s := 100 + utf8.RuneCountInString(word)
166 		if startsWord(hay, at) {
167 			s += 20
168 		}
169 		return s
170 	}
171 	s, end := 0, -1 // end: the byte after the last letter found
172 	for _, c := range word {
173 		from := max(end, 0)
174 		i := strings.IndexRune(hay[from:], c)
175 		if i < 0 {
176 			return -1
177 		}
178 		i += from
179 		s++
180 		if i == end {
181 			s += 5
182 		}
183 		if startsWord(hay, i) {
184 			s += 3
185 		}
186 		_, n := utf8.DecodeRuneInString(hay[i:])
187 		end = i + n
188 	}
189 	return s
190 }

192 // startsWord tells if the letter at byte i of hay starts a word: no letter
193 // or digit is before it.
194 func startsWord(hay string, i int) bool {
195 	if i == 0 {
196 		return true
197 	}
198 	r, _ := utf8.DecodeLastRuneInString(hay[:i])
199 	return !unicode.IsLetter(r) && !unicode.IsNumber(r)
200 }

202 // search gives the bookmarks that match query, best first. Bookmarks with the
203 // same score keep their order.
204 func search(bs []Bookmark, query string) []Bookmark {
205 	type hit struct {
206 		b     Bookmark
207 		score int
208 	}
209 	var hits []hit
210 	for _, b := range bs {
211 		if s := score(query, b.Desc+" "+strings.Join(b.Tags, " ")+" "+b.URL); s >= 0 {
212 			hits = append(hits, hit{b, s})
213 		}
214 	}
215 	slices.SortStableFunc(hits, func(x, y hit) int { return y.score - x.score })
216 	out := make([]Bookmark, len(hits))
217 	for i, h := range hits {
218 		out[i] = h.b
219 	}
220 	return out
221 }

266 // fromHTML gives the bookmark lines of the Netscape HTML that browsers
267 // export, as bm-import does: <H3> opens a folder, </DL> closes it, and the
268 // folders of a bookmark become its tags. Only http, https, ftp and file URLs
269 // stay.
270 func fromHTML(text string) []string {
271 	var folders, out []string
272 	for _, line := range lines(text) {
273 		upper := upperASCII(line)
274 		if at := strings.Index(upper, "<H3"); at >= 0 {
275 			name := line[at:]
276 			name = name[strings.IndexByte(name, '>')+1:]
277 			if end := strings.Index(upperASCII(name), "</H3"); end >= 0 {
278 				name = name[:end]
279 			}
280 			folders = append(folders, folderTag(htmlText(name)))
281 		}
282 		if at := strings.Index(upper, `HREF="`); at >= 0 {
283 			u, rest, ok := strings.Cut(line[at+len(`HREF="`):], `"`)
284 			name := rest[strings.IndexByte(rest, '>')+1:]
285 			if end := strings.Index(upperASCII(name), "</A"); end >= 0 {
286 				name = name[:end]
287 			}
288 			var tags []string
289 			for _, f := range folders {
290 				if f != "" && !slices.Contains(tags, f) {
291 					tags = append(tags, f)
292 				}
293 			}
294 			if u = htmlText(u); ok && imported.MatchString(u) {
295 				out = append(out, u+"\t"+htmlText(name)+"\t"+strings.Join(tags, " "))
296 			}
297 		}
298 		if strings.Contains(upper, "</DL>") && len(folders) > 0 {
299 			folders = folders[:len(folders)-1]
300 		}
301 	}
302 	return out
303 }

305 // upperASCII gives s with the letters a to z in upper case. Each byte keeps
306 // its index, so an index in the result is an index in s.
307 func upperASCII(s string) string {
308 	b := []byte(s)
309 	for i, c := range b {
310 		if 'a' <= c && c <= 'z' {
311 			b[i] = c - 'a' + 'A'
312 		}
313 	}
314 	return string(b)
315 }

329 // mergeRows gives the rows whose URL is not in text and not in a row before
330 // them, as bm --merge does, with the number of rows that it skipped for that
331 // reason. Empty rows and comments go.
332 func mergeRows(text string, rows []string) ([]string, int) {
333 	seen := map[string]bool{}
334 	for _, l := range lines(text) {
335 		seen[key(l)] = true
336 	}
337 	var add []string
338 	skipped := 0
339 	for _, row := range rows {
340 		k := key(row)
341 		switch {
342 		case k == "":
343 		case seen[k]:
344 			skipped++
345 		default:
346 			seen[k] = true
347 			f := append(strings.SplitN(row, "\t", 4), "", "")
348 			add = append(add, f[0]+"\t"+f[1]+"\t"+f[2])
349 		}
350 	}
351 	return add, skipped
352 }

403 // listing gives the list page for the bookmarks in text that match the query
404 // and the tag, in the order.
405 func (s *server) listing(text, version, query, tag, order string) bookmarksPage {
406 	p := bookmarksPage{page: s.page("Your bookmarks"), Query: query, Tag: tag, Version: version}
407 	all := parse(text)
408 	count := map[string]int{}
409 	var rows []Bookmark
410 	for _, b := range all {
411 		for _, t := range b.Tags {
412 			count[t]++
413 		}
414 		if tag == "" || slices.Contains(b.Tags, tag) {
415 			rows = append(rows, b)
416 		}
417 	}
418 	for name, n := range count {
419 		p.Tags = append(p.Tags, tagCount{name, n})
420 	}
421 	slices.SortFunc(p.Tags, func(x, y tagCount) int { return strings.Compare(x.Name, y.Name) })
422 	if orders[order] == nil {
423 		order = "recent"
424 	}
425 	p.Sort = order
426 	slices.SortStableFunc(rows, orders[order])
427 	if strings.TrimSpace(query) != "" {
428 		if rows = search(rows, query); len(rows) == 0 {
429 			p.Target = target(query)
430 		}
431 	}
432 	p.Total, p.Found = len(all), len(rows)
433 	p.Rows = rows[:min(len(rows), maxRows)]
434 	return p
435 }

473 // addBookmark adds a bookmark at the end of the file, where bm adds them. As
474 // bm does, it refuses a URL that is a bookmark already.
475 func (s *server) addBookmark(w http.ResponseWriter, r *http.Request) {
476 	if !s.sameSite(w, r) {
477 		return
478 	}
479 	u := s.owner(w, r)
480 	if u == nil {
481 		return
482 	}
483 	r.Body = http.MaxBytesReader(w, r.Body, maxForm)
484 	e := entry{URL: r.PostFormValue("url"), Desc: r.PostFormValue("desc"), Tags: r.PostFormValue("tags")}
485 	line, err := e.line()
486 	if err != nil {

488 		return
489 	}
490 	text, version, err := s.store.read(u, "")
491 	if err != nil {
492 		s.fail(w, err)
493 		return
494 	}
495 	for _, b := range parse(text) {
496 		if norm(b.URL) == key(line) {
497 			s.refuse(w, u, http.StatusConflict, "add", "This URL is a bookmark already: "+b.Title()+".", e)
498 			return
499 		}
500 	}
501 	if _, _, err := s.store.sync(u, version, join(append(lines(text), line))); err != nil {
502 		s.fail(w, err)
503 		return
504 	}
505 	http.Redirect(w, r, "/bookmarks?done=added", http.StatusSeeOther)

508 // lineAt gives line n of version v of the file of the account, and n as a
509 // number. It gives errChanged when the version is gone, or when the line is
510 // not a bookmark.
511 func (s *server) lineAt(u *Account, v, n string) (string, int, error) {
512 	i, err := strconv.Atoi(n)
513 	if err != nil || !hashName.MatchString(v) {
514 		return "", 0, errChanged
515 	}
516 	text, _, err := s.store.read(u, v)
517 	if errors.Is(err, fs.ErrNotExist) {
518 		return "", 0, errChanged
519 	} else if err != nil {
520 		return "", 0, err
521 	}
522 	ls := lines(text)
523 	if i < 0 || i >= len(ls) {
524 		return "", 0, errChanged
525 	}
526 	if _, ok := parseLine(ls[i]); !ok {
527 		return "", 0, errChanged
528 	}
529 	return ls[i], i, nil
530 }

552 // saveBookmark changes or deletes the bookmark of the edit page in the
553 // current file. When the current file does not have the line of the page,
554 // another device changed it, and nothing changes.
555 func (s *server) saveBookmark(w http.ResponseWriter, r *http.Request) {
556 	if !s.sameSite(w, r) {
557 		return
558 	}
559 	u := s.owner(w, r)
560 	if u == nil {
561 		return
562 	}
563 	r.Body = http.MaxBytesReader(w, r.Body, maxForm)
564 	v := r.PostFormValue("v")
565 	old, n, err := s.lineAt(u, v, r.PostFormValue("n"))
566 	if errors.Is(err, errChanged) {
567 		http.Redirect(w, r, "/bookmarks?done=changed", http.StatusSeeOther)
568 		return
569 	} else if err != nil {
570 		s.fail(w, err)
571 		return
572 	}
573 	e := entry{URL: r.PostFormValue("url"), Desc: r.PostFormValue("desc"), Tags: r.PostFormValue("tags")}
574 	problem := func(status int, text string) {
575 		p := bookmarksPage{page: s.page("Edit a bookmark"), Version: v, N: n, Form: e}
576 		p.Error = text
577 		s.render(w, status, "edit", p)
578 	}
579 	line, msg := "", "deleted"
580 	if r.PostFormValue("do") != "delete" {
581 		if line, err = e.line(); err != nil {
582 			problem(http.StatusBadRequest, sentence(err))
583 			return
584 		}
585 		msg = "saved"
586 	}
587 	text, version, err := s.store.read(u, "")
588 	if err != nil {
589 		s.fail(w, err)
590 		return
591 	}
592 	if b, _ := parseLine(old); line != "" && key(line) != norm(b.URL) {
593 		for _, other := range parse(text) {
594 			if norm(other.URL) == key(line) {
595 				problem(http.StatusConflict, "Another bookmark has this URL: "+other.Title()+".")
596 				return
597 			}
598 		}
599 	}
600 	var out []string
601 	found := false
602 	for _, l := range lines(text) {
603 		if l != old {
604 			out = append(out, l)
605 		} else if !found {
606 			found = true
607 			if line != "" {
608 				out = append(out, line)
609 			}
610 		}
611 	}
612 	if !found {
613 		http.Redirect(w, r, "/bookmarks?done=changed", http.StatusSeeOther)
614 		return
615 	}
616 	if _, _, err := s.store.sync(u, version, join(out)); err != nil {
617 		s.fail(w, err)
618 		return
619 	}
620 	http.Redirect(w, r, "/bookmarks?done="+msg, http.StatusSeeOther)
621 }

638 // saveFile takes the whole file of the file page. As for the file of a
639 // device, the server merges the changes since the version that the page
640 // showed.
641 func (s *server) saveFile(w http.ResponseWriter, r *http.Request) {
642 	if !s.sameSite(w, r) {
643 		return
644 	}
645 	u := s.owner(w, r)
646 	if u == nil {
647 		return
648 	}
649 	// A form sends a tab as %09, so the form can be three times as large as
650 	// the file.
651 	r.Body = http.MaxBytesReader(w, r.Body, 3*maxFile+maxForm)
652 	if err := r.ParseForm(); err != nil {
653 		var tooBig *http.MaxBytesError
654 		if errors.As(err, &tooBig) {
655 			http.Error(w, "the bookmark file is too large", http.StatusRequestEntityTooLarge)
656 		} else {
657 			http.Error(w, "cannot read the form", http.StatusBadRequest)
658 		}
659 		return
660 	}
661 	base, text := r.PostFormValue("base"), join(lines(r.PostFormValue("text")))
662 	if base != "" && !hashName.MatchString(base) {
663 		http.Error(w, "base is not a version name", http.StatusBadRequest)
664 		return
665 	}
666 	if len(text) > maxFile {
667 		p := bookmarksPage{page: s.page("Edit the file"), Version: base, Text: text}
668 		p.Error = "The file is larger than 4 MB. Nothing changed."
669 		s.render(w, http.StatusRequestEntityTooLarge, "file", p)
670 		return
671 	}
672 	merged, _, err := s.store.sync(u, base, text)
673 	if err != nil {
674 		s.fail(w, err)
675 		return
676 	}
677 	msg := "file"
678 	if merged != text {
679 		msg = "merged"
680 	}
681 	http.Redirect(w, r, "/bookmarks?done="+msg, http.StatusSeeOther)
682 }

684 // importFile adds the bookmarks of a file, as bm-import and bm --merge do. A
685 // URL that is a bookmark already is skipped.
686 func (s *server) importFile(w http.ResponseWriter, r *http.Request) {
687 	if !s.sameSite(w, r) {
688 		return
689 	}
690 	u := s.owner(w, r)
691 	if u == nil {
692 		return
693 	}
694 	r.Body = http.MaxBytesReader(w, r.Body, maxImport)
695 	f, _, err := r.FormFile("file")
696 	if err != nil {
697 		s.refuse(w, u, http.StatusBadRequest, "import", "Choose a file of 32 MB or less.", entry{})
698 		return
699 	}
700 	defer f.Close()
701 	b, err := io.ReadAll(f)
702 	if err != nil {
703 		s.fail(w, err)
704 		return
705 	}
706 	text, version, err := s.store.read(u, "")
707 	if err != nil {
708 		s.fail(w, err)
709 		return
710 	}
711 	add, skipped := mergeRows(text, importRows(string(b)))
712 	merged := join(append(lines(text), add...))
713 	if len(merged) > maxFile {
714 		s.refuse(w, u, http.StatusRequestEntityTooLarge, "import",
715 			"With these bookmarks, the file would be larger than 4 MB. Nothing changed.", entry{})
716 		return
717 	}
718 	if len(add) > 0 {
719 		if _, _, err := s.store.sync(u, version, merged); err != nil {
720 			s.fail(w, err)
721 			return
722 		}
723 	}
724 	http.Redirect(w, r, fmt.Sprintf("/bookmarks?done=imported&added=%d&known=%d", len(add), skipped), http.StatusSeeOther)
725 }

727 // download gives the current file. It works also when sync is paused,
728 // because the bookmarks belong to their owner.
729 func (s *server) download(w http.ResponseWriter, r *http.Request) {
730 	u := s.user(r)
731 	if u == nil {
732 		http.Redirect(w, r, "/login", http.StatusSeeOther)
733 		return
734 	}
735 	text, _, err := s.store.read(u, "")
736 	if err != nil {
737 		s.fail(w, err)
738 		return
739 	}
740 	w.Header().Set("Content-Type", "text/plain; charset=utf-8")
741 	w.Header().Set("Content-Disposition", `attachment; filename="bookmarks.txt"`)
742 	io.WriteString(w, text)
743 }
