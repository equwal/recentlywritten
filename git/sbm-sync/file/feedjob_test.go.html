15 // fakeTools are feed tools that need no sfeed: "sfeed_update" writes the
16 // items of a table, and "sfeed_web" answers from a table.
17 type fakeTools struct {
18 	items   map[string]string   // feed URL: the sfeed lines of the feed
19 	pages   map[string][]string // page URL: the "URL<tab>type" lines that sfeed_web finds
20 	updates []string            // the sfeedrc of each update
21 	crawled []string            // the pages given to discover
22 	plain   int                 // calls of plain
23 	now     time.Time
24 }

26 func (f *fakeTools) tools() feedTools {
27 	return feedTools{
28 		update: func(rc string) error {
29 			f.updates = append(f.updates, rc)
30 			b, err := os.ReadFile(rc)
31 			if err != nil {
32 				return err
33 			}
34 			dir := ""
35 			for _, l := range lines(string(b)) {
36 				if v, ok := strings.CutPrefix(l, "sfeedpath='"); ok {
37 					dir = strings.TrimSuffix(v, "'")
38 				}
39 				if v, ok := strings.CutPrefix(l, "\tfeed '"); ok {
40 					file, u, _ := strings.Cut(strings.TrimSuffix(v, "'"), "' '")
41 					if err := os.WriteFile(filepath.Join(dir, file), []byte(f.items[u]), 0o600); err != nil {
42 						return err
43 					}
44 				}
45 			}
46 			return nil
47 		},
48 		plain: func(files []string) (string, error) {
49 			f.plain++
50 			var b strings.Builder
51 			for _, path := range files {
52 				text, err := os.ReadFile(path)
53 				if err != nil {
54 					return "", err
55 				}
56 				b.WriteString(filepath.Base(path) + ":\n")
57 				for _, l := range lines(string(text)) {
58 					c := strings.Split(l, "\t")
59 					b.WriteString("  " + c[1] + "  " + c[2] + "\n")
60 				}
61 			}
62 			return b.String(), nil
63 		},
64 		discover: func(page string) ([]string, error) {
65 			f.crawled = append(f.crawled, page)
66 			return f.pages[page], nil
67 		},
68 		lookup: func(host string) ([]net.IP, error) {
69 			switch host {
70 			case "internal.test":
71 				return []net.IP{net.ParseIP("10.66.0.4")}, nil
72 			case "nowhere.test":
73 				return nil, errors.New("no such host")
74 			}
75 			return []net.IP{net.ParseIP("93.184.216.34")}, nil
76 		},
77 		now: func() time.Time { return f.now },
78 	}
79 }
