1 #!/bin/sh
2 # Build the kernel image for the STM32F446 from source: SBCL runs the Thumb-2 assembler
3 # (asm.lisp) and the metacompiler (kernel.lisp, kernel.fs) and writes kernel-f446.bin.
4 # No third-party Forth. The image runs on QEMU netduinoplus2, the closest QEMU board
5 # to the F446 (console: USART2 = QEMU's second -serial); a real board takes it with
6 #   st-flash write kernel-f446.bin 0x8000000
7 set -e
8 HERE=$(cd "$(dirname "$0")" && pwd)
9 nice "${SBCL:-sbcl}" --script "$HERE/kernel.lisp" "$HERE/kernel-f446.bin"
