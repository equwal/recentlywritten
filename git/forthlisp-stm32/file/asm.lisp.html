17 (defun fixup (size fn)
18   "Reserve SIZE bytes; segment-bytes later calls (FN position) to emit them with labels known.
19 sb-assem's emit-back-patch resolved only the first of several patches in a row here (SBCL
20 2.6.9), so label references go through this list instead."
21   (push (list (sb-assem::segment-current-posn *seg*) size fn) *fixups*)
22   (dotimes (i size) (sb-assem:emit-byte *seg* 0)))
23 (defun segment-bytes ()
24   (sb-assem:finalize-segment *seg*)
25   (let ((v (copy-seq (coerce (sb-assem:segment-contents-as-vector *seg*) '(simple-array (unsigned-byte 8) (*)))))
26         (main *seg*))
27     (unwind-protect
28          (dolist (f *fixups* v)
29            (destructuring-bind (posn size fn) f
30              (setf *seg* (sb-assem:make-segment))
31              (funcall fn posn)
32              (sb-assem:finalize-segment *seg*)
33              (let ((bytes (sb-assem:segment-contents-as-vector *seg*)))
34                (assert (= (length bytes) size) () "fixup at ~d wrote ~d bytes, not ~d" posn (length bytes) size)
35                (replace v bytes :start1 posn))))
36       (setf *seg* main))))
