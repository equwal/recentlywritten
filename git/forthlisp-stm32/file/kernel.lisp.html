2 ;;;;
3 ;;;; Indirect-threaded. r4 = IP, r5 = data stack pointer (full descending, holds the second
4 ;;;; item), r6 = TOS, r7 = W (current xt), sp = return stack. NEXT: ldr r7,[r4],#4;
5 ;;;; ldr r0,[r7]; bx r0. Header: link, flags (80 immediate, 40 hidden), length, name,
6 ;;;; pad to 4, code field, body. Primitives are `defcode` templates (a name plus a Thumb-2
7 ;;;; body, NEXT appended); the high-level kernel is metacompiled from kernel.fs.
8 ;;;; Usage: sbcl --script kernel.lisp OUT.bin

258 (defun tokenize (text)
259   "Whitespace tokens; \\ and ( ) comments dropped; the text after .\" s\" .( becomes (:string ...)."
260   (let ((toks '()) (i 0) (n (length text)))
261     (flet ((ws (c) (member c '(#\Space #\Tab #\Newline #\Return)))
262            (upto (ch) (let ((e (or (position ch text :start i) n))) (prog1 (subseq text i e) (setf i (min n (1+ e)))))))
263       (loop
264         (loop while (and (< i n) (ws (char text i))) do (incf i))
265         (when (>= i n) (return))
266         (let* ((e (or (position-if #'ws text :start i) n)) (tok (subseq text i e)))
267           (setf i e)
268           (cond ((equal (first toks) ":") (push tok toks))     ; a name, even if it looks special
269                 ((string= tok "\\") (upto #\Newline))
270                 ((string= tok "(") (upto #\)))
271                 ((member tok '(".\"" "s\"") :test #'string=)
272                  (incf i) (push tok toks) (push (list :string (upto #\")) toks))
273                 (t (push tok toks)))))
274       (nreverse toks))))

286 (defun compile-def (name body)
287   "BODY: tokens between the name and ;. Control structures resolve to labels here."
288   (let ((cs '()))
289     (loop while body do
290       (let ((tok (pop body)))
291         (cond
292           ((consp tok) (error "stray string in ~a" name))
293           ((string= tok "if") (cell "0branch") (let ((l (label))) (abs32 l) (push l cs)))
294           ((string= tok "else") (cell "branch") (let ((l (label))) (abs32 l) (mark (pop cs)) (push l cs)))
295           ((string= tok "then") (mark (pop cs)))
296           ((string= tok "begin") (push (mark (label)) cs))
297           ((string= tok "until") (cell "0branch") (abs32 (pop cs)))
298           ((string= tok "again") (cell "branch") (abs32 (pop cs)))
299           ((string= tok "while") (cell "0branch") (let ((l (label)) (dest (pop cs))) (abs32 l) (push l cs) (push dest cs)))
300           ((string= tok "repeat") (cell "branch") (abs32 (pop cs)) (mark (pop cs)))
301           ((member tok '("do" "?do") :test #'string=)
302            (cell (if (string= tok "do") "(do)" "(?do)"))
303            (let ((leave (label))) (abs32 leave) (push leave cs) (push (mark (label)) cs)))
304           ((member tok '("loop" "+loop") :test #'string=)
305            (cell (if (string= tok "loop") "(loop)" "(+loop)")) (abs32 (pop cs)) (mark (pop cs)))
306           ((string= tok "recurse") (cell name))
307           ((string= tok "[']") (lit (xt (pop body))))
308           ((string= tok "[char]") (lit (char-code (char (pop body) 0))))
309           ((member tok '(".\"" "s\"") :test #'string=)
310            (let ((s (second (pop body))))
311              (cell "(s\")") (byte8 (length s)) (loop for c across s do (byte8 (char-code c))) (align4)
312              (when (string= tok ".\"") (cell "type"))))
313           ((parse-num tok) (lit (parse-num tok)))
314           (t (cell tok)))))
315     (when cs (error "unbalanced control structure in ~a" name))
316     (cell "exit")))

318 (defun metacompile (path)
319   (let ((toks (with-open-file (s path) (let ((str (make-string (file-length s)))) (tokenize (subseq str 0 (read-sequence str s)))))))
320     (loop while toks do
321       (let ((tok (pop toks)))
322         (unless (equal tok ":") (error "kernel.fs: expected : at ~s" tok))
323         (let* ((name (pop toks)) (end (position ";" toks :test #'equal)) (body (subseq toks 0 end))
324                (imm (equal (nth (1+ end) toks) "immediate")))
325           (setf toks (nthcdr (+ end (if imm 2 1)) toks))
326           (header name (if imm #x80 0))
327           (abs32 *docol* 1)
328           (compile-def name body))))))
