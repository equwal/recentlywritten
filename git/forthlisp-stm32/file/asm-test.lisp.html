63 (defun run-lisp ()
64   (new-segment)
65   (let ((labels (make-hash-table :test 'equal)) (ranges '()))
66     (flet ((lab (n) (or (gethash n labels) (setf (gethash n labels) (label)))))
67       (dolist (c *cases*)
68         (let ((f (car c)) (start (here)))
69           (if (eq (car f) :label)
70               (mark (lab (second f)))
71               (apply (car f) (mapcar (lambda (a) (if (stringp a) (lab a) a)) (cdr f))))
72           (push (list start (here) (cdr c)) ranges))))
73     (values (segment-bytes) (nreverse ranges))))

99 (multiple-value-bind (mine ranges) (run-lisp)
100   (let ((gas (run-gas)) (fail 0) (n 0))
101     (dolist (r ranges)
102       (destructuring-bind (a z text) r
103         (unless (= a z)
104           (incf n)
105           (let ((m (subseq mine a z)) (g (subseq gas a (min z (length gas)))))
106             (unless (equalp m g)
107               (incf fail)
108               (format t "FAIL ~a: mine ~{~2,'0x~^ ~} gas ~{~2,'0x~^ ~}~%"
109                       (substitute #\Space #\Newline text) (coerce m 'list) (coerce g 'list)))))))
110     (incf n)
