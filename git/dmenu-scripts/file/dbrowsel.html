1 #!/bin/bash

3 # Dmenu for selecting which browser to open.
4 # from github.com/AlexChaplinBraz/dmenu-scripts/blob/master/dbrowsel
5 # (with edits)
6 
7 THEBROWSER="1. \$BROWSER $BROWSER + clipboard"
8 BROWSER2='2. firefox + clipboard'
9 XSELECT='3. clipboard'
10 
11 BROWSERS="$THEBROWSER
12 $BROWSER2
13 $XSELECT"
14 
15 [ -z "$1" ] && URL="$(xclip -o)"
16 [ -z "$1" ] || URL="$1"
17 
18 SHORTURL="$(echo "$URL" | sed 's!https?:?/?/?!!' | sed 's!/.*!!')"
19 
20 CHOSEN=$(printf %s\\n "$BROWSERS" | dmenu -i -l 5 -p "Select browser for ($SHORTURL...): ")
21 
22 [ -z "$CHOSEN" ] && exit 1
23 
24 case $CHOSEN in
25     "$THEBROWSER") command -v "$BROWSER"    && echo "$URL" | xclip -i -sel clip && $BROWSER "$URL";;
26 	  "$TOR")        command -v "tor-browser" && echo "$URL" | xclip -i -sel clip && tor-browser "$URL";; # tor-browser in PATH?
27     "$BROWSER2")  command -v "firefox"     && firefox "$URL";;
28     "$XSELECT")    command -v "xclip"       && echo "$URL" | xclip -i -sel clip ;; # TODO: maybe support xsel
29 	*) eval "$CHOSEN" "$URL";;
30 esac
