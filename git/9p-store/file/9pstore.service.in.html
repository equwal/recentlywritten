9pstore.service.in (524 bytes)

1 # 9P file store served by diod. install.sh fills in the @...@ values.
2 # No 9P auth: the private-address bind and the firewall are the gate.
3 # Every client user maps to @SQUASH_USER@.
4 [Unit]
5 Description=9P file store (diod)
6 After=network-online.target
7 Wants=network-online.target
8 
9 [Service]
10 ExecStart=/usr/sbin/diod -f -n -S -U @SQUASH_USER@ -e @EXPORT_DIR@ -l @LISTEN_ADDR@:@PORT@ -l 127.0.0.1:@PORT@ --nwthreads 2 -L stderr
11 Restart=on-failure
12 RestartSec=5
13 MemoryMax=64M
14 MemoryHigh=48M
15 
16 [Install]
17 WantedBy=multi-user.target
