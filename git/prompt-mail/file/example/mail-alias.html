3 # Helper program for neomutt!
4 # configure ~/.config/prompt-mail.conf to contain just the path to the
5 # aliases filie and optional comments.
6 
7 
8 CONFIG="$(sed '/^#.*$/d' ~/.config/prompt-mail.conf | sed -E 's/^alias // ; s/#.*$//')"
9 
10 add_email () {
11     out=$(sed -ne 's/^From:[ \t]*//p' /dev/stdin)
12     # Ensure adding aliases is idempotent with respect to mail address.
13     grep -q -F -- $(echo "$out" | sed 's/^.*<// ; s/>.*$>//') "$CONFIG"
14     active=$?
15     [ $active != 0 ] && y-or-n "Add $out? " && echo "alias $(dmenu -p 'Alias shorthand? ') $out" >> "$CONFIG"
16 }
17 
18 [ "$1" = "--add-email" ] || [ "$1" = "-a" ] &&  add_email </dev/stdin
19 [ "$1" = "--query" ] || [ "$1" = "-q" ] && grep "$2" "$CONFIG"
