1 #!/bin/sh
2 # Edit this if you want a different prompter.
3 PROMPT="dmenu -l 15 -i -p"
4 REMCOMMENT=
5 
6 verify_email () {
7     IN=$(cat /dev/stdin)
8     echo "$IN" | grep -q '@' && echo "$IN" && return 0
9     echo "$IN Not an email." >/dev/stderr ; return 1
10 }
11 
12 mail_addrs () {
13     sed 's/[\\t ]*#.*//' "$XDG_CONFIG_HOME"/prompt-mail.conf \
14         | xargs sed -E "s/^alias[\\t ]+[a-zA-Z0-9]+[ \\t]+// ; s/[ \\t]+$//" \
15         | sed -E "s:~:$HOME:g" \
16         | sort \
17         | uniq
18 }
19 
20 mail_addrs \
21     | $PROMPT 'mail address:' \
22     | sed  's/ *#.*//' \
23     | verify_email
