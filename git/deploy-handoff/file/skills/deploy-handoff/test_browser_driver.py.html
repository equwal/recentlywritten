49 def front_matter(text: str) -> dict[str, str]:
50     """Return the fields of the front matter of text that have their value on one line."""
51     lines = text.splitlines()
52     end = lines.index("---", 1) if lines[:1] == ["---"] else 0
53     fields: dict[str, str] = {}
54     for line in lines[1:end]:
55         key, colon, value = line.partition(":")
56         if colon and key.isidentifier():
57             fields[key] = value.strip()
58     return fields

61 def human_steps(text: str) -> list[str]:
62     """Return the items of the list after HUMAN_STEPS_LINE in text."""
63     lines = text.splitlines()
64     items: list[str] = []
65     for line in lines[lines.index(HUMAN_STEPS_LINE) + 1 :]:
66         if line.startswith("- "):
67             items.append(line.removeprefix("- "))
68         elif items or line.strip():
69             break
70     return items
