1 """Open a web page and wait while the user does a step that only a human may do.
2 
3 A tool calls this script when it reaches a step that an AI agent must not do:
4 the final deploy, publish, or billing click, a sign-in, an OAuth consent, or
5 the creation of a GitHub pull request. The script opens the page in the
6 default browser and shows a small dialog with the steps. It never clicks for
7 the user. It prints the answer of the user as one JSON object on stdout.
8 
9 Exit codes: 0 done, 1 error, 2 bad arguments, 3 not done, 4 timeout.
10 """

55 # A plain DNS name in lowercase ASCII. The script refuses all other host forms,
56 # so the browser and this script always read the same host from a URL.
57 HOST_RE = re.compile(r"[a-z0-9-]+(?:\.[a-z0-9-]+)+")
58 # Printable ASCII without a space. Callers percent-encode all other characters.
59 URL_CHARS_RE = re.compile(r"[!-~]+")
60 REPO_RE = re.compile(r"[A-Za-z0-9-]+/[A-Za-z0-9._-]+")
61 GITHUB_REMOTE_RE = re.compile(
62     r"(?:https://(?:[^@/]+@)?github\.com/|git@github\.com:|ssh://git@github\.com/)"
63     r"(?P<repo>[A-Za-z0-9-]+/[A-Za-z0-9._-]+?)(?:\.git)?/?"

82 def match_domain(host: str, domains: Iterable[str]) -> str | None:
83     """Return the most specific domain that is host or a parent domain of host."""
84     matches = [domain for domain in domains if host == domain or host.endswith("." + domain)]
85     return max(matches, key=len, default=None)

88 def check_url(url: str, domains: Iterable[str]) -> str:
89     """Return the domain that allows url. Raise HandoffError if the script must not open url."""
90     if "\\" in url or not URL_CHARS_RE.fullmatch(url):
91         raise HandoffError(
92             f"The URL must be printable ASCII with no spaces or backslashes: {url!r}"
93         )
94     try:
95         parts = urlsplit(url)
96     except ValueError as exc:
97         raise HandoffError(f"The URL is not valid: {url}") from exc
98     host = parts.hostname or ""
99     # The netloc must be the host only: no user name, no password, and no port.
100     if parts.scheme != "https" or parts.netloc.lower() != host or not HOST_RE.fullmatch(host):
101         raise HandoffError(f"The URL must use https and a plain host name: {url}")
102     domain = match_domain(host, domains)
103     if domain is None:
104         raise HandoffError(
105             f"{host} is not an allowed host. Ask the user to add it to allowed_hosts in "
106             f"{USER_CONFIG}. Do not add it yourself."
107         )
108     return domain

111 def check_text(title: str, steps: Iterable[str]) -> None:
112     """Raise HandoffError if the title or a step is not short Simplified Technical English."""
113     named = [("The title", title), *((f"Step {n}", step) for n, step in enumerate(steps, 1))]
114     for name, text in named:
115         words = len(text.split())
116         if not 0 < words <= MAX_STEP_WORDS or ";" in text:
117             raise HandoffError(
118                 f"{name} is not short Simplified Technical English ({words} words). "
119                 f"Give one action in 1 to {MAX_STEP_WORDS} words, with no semicolons."
120             )

123 def read_hosts(path: Path) -> list[str]:
124     """Return the allowed_hosts list of the TOML file at path."""
125     try:
126         data = tomllib.loads(path.read_text(encoding="utf-8"))
127     except (OSError, tomllib.TOMLDecodeError) as exc:
128         raise HandoffError(f"Cannot read {path}: {exc}") from exc
129     hosts = data.get("allowed_hosts", [])
130     if not isinstance(hosts, list) or not all(
131         isinstance(host, str) and HOST_RE.fullmatch(host) for host in hosts
132     ):
133         raise HandoffError(f"{path}: allowed_hosts must be a list of lowercase host names.")
134     return hosts

137 def allowed_hosts() -> list[str]:
138     """Return the hosts in the shipped file, and in the user file if it exists."""
139     paths = [SHIPPED_CONFIG, USER_CONFIG] if USER_CONFIG.is_file() else [SHIPPED_CONFIG]
140     return [host for path in paths for host in read_hosts(path)]

149 def github_repo(remote_url: str) -> str:
150     """Return "owner/name" for the URL of a github.com remote."""
151     match = GITHUB_REMOTE_RE.fullmatch(remote_url.strip())
152     if match is None:
153         raise HandoffError(f"{remote_url} is not a github.com remote. Give --repo OWNER/NAME.")
154     return match["repo"]

157 def compare_url(repo: str, base: str | None, head: str, title: str, body: str) -> str:
158     """Return the GitHub page that shows a new pull request form with title and body filled in."""
159     # Without a base, GitHub compares head with the default branch.
160     refs = quote(head) if base is None else f"{quote(base)}...{quote(head)}"
161     query = urlencode({"expand": "1", "title": title, "body": body}, quote_via=quote)
162     url = f"https://github.com/{repo}/compare/{refs}?{query}"
163     if len(url) > MAX_PR_URL_LENGTH:
164         raise HandoffError(
165             f"The pull request URL has {len(url)} characters. The limit is {MAX_PR_URL_LENGTH}. "
166             "Make the body shorter."
167         )
168     return url

171 def git(repo_dir: Path, *args: str) -> str:
172     """Run git in repo_dir and return its output. Raise HandoffError if git fails."""
173     command = ["git", "-C", str(repo_dir), *args]
174     try:
175         result = subprocess.run(
176             command, capture_output=True, encoding="utf-8", errors="replace", timeout=60
177         )
178     except (OSError, subprocess.TimeoutExpired) as exc:
179         raise HandoffError(f"{' '.join(command)} failed: {exc}") from exc
180     if result.returncode != 0:
181         raise HandoffError(f"{' '.join(command)} failed: {result.stderr.strip()}")
182     return result.stdout.strip()

185 def current_branch(repo_dir: Path) -> str:
186     """Return the branch that HEAD is on."""
187     branch = git(repo_dir, "branch", "--show-current")
188     if not branch:
189         raise HandoffError("HEAD is not on a branch. Give --head BRANCH.")
190     return branch

193 def check_pushed(repo_dir: Path, remote: str, branch: str) -> None:
194     """Raise HandoffError if remote does not have the commit of the local branch."""
195     ref = f"refs/heads/{branch}"
196     local = git(repo_dir, "rev-parse", "--verify", ref)
197     lines = git(repo_dir, "ls-remote", remote, ref).splitlines()
198     remote_commits = [line.split()[0] for line in lines if line.split()[1:] == [ref]]
199     if remote_commits != [local]:
200         raise HandoffError(
201             f"{remote} does not have the local commit of {branch}. Push {branch} first."
202         )

205 def find_open_pr(repo: str, head: str) -> str | None:
206     """Return the URL of the open pull request from head, or None if gh cannot find one."""
207     command = ["gh", "pr", "list", "--repo", repo, "--head", head, "--state", "open"]
208     command += ["--json", "url", "--jq", ".[0].url // empty"]
209     try:
210         result = subprocess.run(command, capture_output=True, encoding="utf-8", timeout=60)
211     except (OSError, subprocess.TimeoutExpired):
212         return None
213     url = result.stdout.strip()
214     return url if result.returncode == 0 and url else None

217 def pr_request(args: argparse.Namespace) -> Request:
218     """Make the request that opens the GitHub form for a new pull request."""
219     repo_dir: Path = args.repo_dir
220     repo: str = args.repo or github_repo(git(repo_dir, "remote", "get-url", args.remote))
221     if not REPO_RE.fullmatch(repo):
222         raise HandoffError(f"{repo!r} is not a repository name of the form OWNER/NAME.")
223     head: str = args.head or current_branch(repo_dir)
224     check_pushed(repo_dir, args.remote, head)
225     try:
226         body: str = args.body_file.read_text(encoding="utf-8") if args.body_file else args.body
227     except OSError as exc:
228         raise HandoffError(f"Cannot read {args.body_file}: {exc}") from exc
229     url = compare_url(repo, args.base, head, args.title, body)
230     return Request(f"Create pull request: {args.title}", url, PR_STEPS, pr=(repo, head))

233 class Dialog:
234     """A small window on top of the browser. It shows the steps and records the answer."""
235 
236     def __init__(self, request: Request, timeout_minutes: float, reopen: Callable[[], None]):
237         self.status = "not_done"
238         self.note_text = ""
239         if sys.platform == "win32":
240             # Draw sharp text on high-DPI screens, as IDLE does. The call fails if the
241             # process has already set the DPI awareness. That is not a problem.
242             with contextlib.suppress(OSError):
243                 ctypes.OleDLL("shcore").SetProcessDpiAwareness(1)
244         self.root = tk.Tk()
245         scale = self.root.winfo_fpixels("1i") / 96
246         wrap = round(420 * scale)
247         self.root.title(f"Claude handoff: {request.title}")
248         self.root.attributes("-topmost", True)
249         self.root.resizable(False, False)
250         # Put the window in the bottom-left corner, above the taskbar. Deploy pages
251         # usually put the final button on the right side.
252         self.root.geometry(f"+{round(24 * scale)}-{round(80 * scale)}")
253         self.root.protocol("WM_DELETE_WINDOW", lambda: self.finish("not_done"))
254 
255         parts = urlsplit(request.url)
256         shown = f"https://{parts.netloc}{parts.path}"
257         if len(shown) > 80:
258             shown = shown[:77] + "..."
259         steps = "\n".join(f"{number}. {step}" for number, step in enumerate(request.steps, 1))
260         # Tk deletes a font when its Python object goes away, so the dialog keeps it.
261         self.bold = tkfont.nametofont("TkDefaultFont").copy()
262         self.bold.configure(weight="bold")
263         frame = ttk.Frame(self.root, padding=round(12 * scale))
264         frame.grid()
265         # The user must see first what to do, and then where to do it.
266         ttk.Label(frame, text=request.title, font=self.bold, wraplength=wrap).grid(sticky="w")
267         ttk.Label(frame, text=HEADING, font=self.bold).grid(sticky="w", pady=(8, 0))
268         ttk.Label(frame, text=steps, wraplength=wrap, justify="left").grid(sticky="w")
269         site = ttk.Label(frame, text=f"Site: {parts.hostname}", font=self.bold)
270         site.grid(sticky="w", pady=(8, 0))
271         link = ttk.Label(frame, text=shown, foreground="blue", cursor="hand2", wraplength=wrap)
272         link.grid(sticky="w")
273         link.bind("<Button-1>", lambda _event: reopen())
274         for text, color in ((FINISH, ""), (WARNING, "#b00020")):
275             label = ttk.Label(frame, text=text, foreground=color, wraplength=wrap, justify="left")
276             label.grid(sticky="w", pady=(8, 0))
277         ttk.Label(frame, text="Note for Claude (optional):").grid(sticky="w", pady=(8, 0))
278         self.note = ttk.Entry(frame)
279         self.note.grid(sticky="ew")
280         buttons = ttk.Frame(frame)
281         buttons.grid(sticky="e", pady=(12, 0))
282         self.done_button = ttk.Button(buttons, text="Done", command=lambda: self.finish("done"))
283         self.done_button.grid(row=0, column=0, padx=(0, 8))
284         self.not_done_button = ttk.Button(
285             buttons, text="Not done", command=lambda: self.finish("not_done")
286         )
287         self.not_done_button.grid(row=0, column=1)
288         self.timer = self.root.after(round(timeout_minutes * 60_000), self.finish, "timeout")
289 
290     def finish(self, status: str) -> None:
291         """Record the answer and close the window."""
292         self.root.after_cancel(self.timer)
293         self.status = status
294         self.note_text = self.note.get().strip()
295         self.root.destroy()
296 
297     def run(self) -> tuple[str, str]:
298         """Show the window until the user answers or the time ends."""
299         self.root.mainloop()
300         return self.status, self.note_text

303 def parse_args(argv: list[str] | None) -> argparse.Namespace:
304     """Read the command line."""
305     common = argparse.ArgumentParser(add_help=False)
306     common.add_argument(
307         "--timeout",
308         type=float,
309         default=30,
310         metavar="MINUTES",
311         help="the time to wait for the user (default: 30)",
312     )
313     parser = argparse.ArgumentParser(
314         description="Open a web page and wait while the user does the final step."
315     )
316     commands = parser.add_subparsers(dest="command", required=True)
317     page = commands.add_parser("open", parents=[common], help="open a page and show the steps")
318     page.add_argument("--url", required=True, help="the https URL of the page")
319     page.add_argument("--title", required=True, help="the action, for example: Publish 1.2")
320     page.add_argument(
321         "--step", action="append", required=True, help="one step for the user (give it again)"
322     )
323     page.add_argument(
324         "--no-open", action="store_true", help="show only the dialog: the page is already open"
325     )
326     pr = commands.add_parser("pr", parents=[common], help="open the GitHub pull request form")
327     pr.add_argument("--title", required=True, help="the title of the pull request")
328     body = pr.add_mutually_exclusive_group()
329     body.add_argument("--body", default="", help="the description of the pull request")
330     body.add_argument("--body-file", type=Path, help="a file with the description")
331     pr.add_argument("--base", help="the branch to merge into (default: the default branch)")
332     pr.add_argument("--head", help="the branch to merge (default: the current branch)")
333     pr.add_argument("--repo", help="OWNER/NAME on GitHub (default: from the remote URL)")
334     pr.add_argument("--remote", default="origin", help="the remote with the branch")
335     pr.add_argument("--repo-dir", type=Path, default=Path(), help="the local repository")
336     return parser.parse_args(argv)

339 def report(result: dict[str, str | None]) -> int:
340     """Print result as JSON. Return the exit code for its status."""
341     print(json.dumps(result))
342     return EXIT_CODES[str(result["status"])]

345 def main(argv: list[str] | None = None) -> int:
346     """Do the handoff. Return the exit code."""
347     args = parse_args(argv)
348     try:
349         hosts = allowed_hosts()
350         if args.command == "pr":
351             request = pr_request(args)
352         else:
353             check_text(args.title, args.step)
354             request = Request(args.title, args.url, tuple(args.step))
355         check_url(request.url, hosts)
356         # With --no-open, the caller drove a browser to the page already.
357         if args.command == "pr" or not args.no_open:
358             open_page(request.url)
359     except HandoffError as exc:
360         return report({"status": "error", "error": str(exc)})
361     status, note = Dialog(request, args.timeout, lambda: open_page(request.url)).run()
362     result: dict[str, str | None] = {"status": status, "note": note}
363     if request.pr is not None and status == "done":
364         result["pr_url"] = find_open_pr(*request.pr)
365     return report(result)
