62     """Make a branch name that git accepts."""
63     parts = draw(st.lists(st.text(REF_CHAR, min_size=1, max_size=6), min_size=1, max_size=3))

72     """Make a host near an allowed domain: the domain, a subdomain, or a look-alike."""
73     domains = draw(st.lists(DOMAIN, min_size=1, max_size=4))

81 def reference_match(host: str, domains: list[str]) -> str | None:
82     """Do the work of match_domain slowly: compare the labels from the right."""
83     labels = host.split(".")
84     found = [d for d in domains if labels[-len(d.split(".")) :] == d.split(".")]
85     return max(found, key=len, default=None)

103 def make_repo(root: Path) -> Path:
104     """Make a repository with one commit on main, and a bare remote with the name origin."""
105     remote, work = root / "remote.git", root / "work"
106     run_git("init", "--bare", str(remote))
107     run_git("init", "-b", "main", str(work))
108     run_git("-C", str(work), "commit", "--allow-empty", "-m", "first")
109     run_git("-C", str(work), "remote", "add", "origin", str(remote))
110     return work

277     def test_round_trip(
278         self, repo: str, base: str | None, head: str, title: str, body: str
