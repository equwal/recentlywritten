593     def test_terminal_step_names_the_terminal_and_shows_no_page(self) -> None:
594         # On 2026-10-04 a tool asked "Install ...? [Y/n]" in a terminal tab. The step had no
595         # page, so the dialog could not show it: the script wanted a URL on an allowed host.
596         request = Request(
597             "Install Herdr on basedmatrix",
598             None,
599             ('Press "Y".',),
600             place='Terminal tab "herdr: add basedmatrix"',
601         )
602         dialog = handoff.Dialog(request, 1, lambda: None)
603         texts = label_texts(dialog.root)
604         # Let the window run before it closes: on macOS, Tk stops the process (SIGTRAP) when
605         # a second window that never ran is closed in one test process.
606         dialog.root.after(50, dialog.done_button.invoke)
607         self.assertEqual(dialog.run(), ("done", ""))
608         expected = [
609             "Install Herdr on basedmatrix",
610             "Do these steps in the terminal:",
611             '1. Press "Y".',
612             'Where: Terminal tab "herdr: add basedmatrix"',
613         ]
614         self.assertEqual(texts[:4], expected)
615         self.assertIn(handoff.TERMINAL_WARNING, texts)
616         # No site, no address, and no text about the address bar of a browser.
617         self.assertFalse([text for text in texts if "Site:" in text or "https://" in text])
618         self.assertNotIn(handoff.WARNING, texts)

624     def setUp(self) -> None:
625         which = mock.patch("shutil.which", return_value="/usr/bin/bemenu")
626         which.start()
627         self.addCleanup(which.stop)
628         display = mock.patch.dict("os.environ", {"WAYLAND_DISPLAY": "wayland-1"})

630         self.addCleanup(display.stop)
631         self.request = Request("Publish 1.2", "https://github.com/o", ("Click it.",))

633     def answer(self, *results: subprocess.CompletedProcess[str] | Exception) -> mock.MagicMock:
634         run = mock.MagicMock(side_effect=list(results))
635         patcher = mock.patch("subprocess.run", run)
636         patcher.start()
637         self.addCleanup(patcher.stop)
638         return run
639 
640     def pick(self, choice: str, code: int = 0) -> subprocess.CompletedProcess[str]:
641         return subprocess.CompletedProcess(["bemenu"], code, stdout=choice + "\n")
642 
643     def run_menu(self, timeout_minutes: float = 1) -> tuple[str, str]:
644         return handoff.MenuDialog(self.request, timeout_minutes, lambda: None).run()
645 
646     def test_done(self) -> None:

648         self.assertEqual(self.run_menu(), ("done", ""))
649         command = run.call_args.args[0]
650         self.assertEqual(command[0], "bemenu")
651         self.assertIn("Publish 1.2", command)
652         self.assertIn("1. Click it.", run.call_args.kwargs["input"].splitlines())
653         self.assertIn("Site: github.com", run.call_args.kwargs["input"].splitlines())

661         self.assertFalse([line for line in lines if line.startswith("Site:")])

718     def run_main(self, *argv: str) -> tuple[int, dict[str, object]]:
719         output = StringIO()
