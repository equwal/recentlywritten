12 // genRow makes a file line: a URL, a description, tags and sometimes a feed,
13 // with tabs, spaces and line breaks in awkward places.
14 func genRow(t *rapid.T) string {
15 	host := rapid.SampledFrom([]string{"a.com", "www.a.com", "b.org", "c.net"}).Draw(t, "host")
16 	path := rapid.SampledFrom([]string{"", "/", "/x", "/x/", "/y"}).Draw(t, "path")
17 	sch := rapid.SampledFrom([]string{"https://", "http://", ""}).Draw(t, "scheme")
18 	desc := rapid.StringMatching(`[a-z \t\n]{0,8}`).Draw(t, "desc")
19 	tags := rapid.StringMatching(`[a-z ]{0,8}`).Draw(t, "tags")
20 	feed := rapid.SampledFrom([]string{"", "", "https://a.com/feed", "b.org/rss", " c.net/atom.xml "}).Draw(t, "feed")
21 	return line(sch+host+path, desc, tags, feed)
22 }

105 			}
106 		}
107 		if feeds != filled {
108 			t.Fatalf("merge gave %d feeds, want %d", feeds, filled)
109 		}
110 		for _, s := range lines(out) {
111 			if l, _ := parseLine(s); l.Feed != want[key(s)] {
