1 # nginx snippet for sbm-webpublish. deploy.sh installs it as
2 # /etc/nginx/snippets/sbm-webpublish.conf. Include it in the HTTPS server
3 # block of the site:
4 #
5 #     include snippets/sbm-webpublish.conf;
6 #
7 # nginx continues to serve the static site. It sends only /links/ to
8 # sbm-webpublish. Serve the site over HTTPS: the admin page uses basic auth.
9 # "^~" stops the search for a regular expression location, so that a
10 # location such as "~ \.xml$" of the site does not take /links/atom.xml.
11 location ^~ /links/ {
12     proxy_pass http://127.0.0.1:8082;
13     proxy_set_header Host $host;
14     proxy_set_header X-Forwarded-Proto $scheme;
15     client_max_body_size 10m;
16 }
