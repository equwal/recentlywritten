213 // genPublicRow makes a file line with the characters that XML and HTML must
214 // escape, and with links that the outputs must leave out.
215 func genPublicRow(t *rapid.T) string {
216 	u := rapid.SampledFrom([]string{"https://a.com/", "http://b.org/?x=1&y=<2>", "c.net/", "javascript:alert(1)//", "ftp://d.com/"}).Draw(t, "url")
217 	path := rapid.StringMatching(`[a-z]{0,3}`).Draw(t, "path")
218 	desc := rapid.StringMatching(`[a-z<>&"' \]\t]{0,10}`).Draw(t, "desc")
219 	tags := rapid.StringMatching(`[a-z<>&" ]{0,8}`).Draw(t, "tags")
220 	feed := rapid.SampledFrom([]string{"", "https://a.com/feed", "http://b.org/rss?x=<1>&y", "c.net/atom.xml", "javascript:alert(2)"}).Draw(t, "feed")
221 	return line(u+path, desc, tags, feed)
222 }

224 // TestOutputsRoundTrip: links.txt parses back to the published links, and the
225 // feed gives each published link, newest first, with its time and tags.
226 func TestOutputsRoundTrip(t *testing.T) {
227 	s, h := newServer(t)
228 	rapid.Check(t, func(t *rapid.T) {
229 		text := ""
230 		for range rapid.IntRange(0, 3).Draw(t, "batches") {
231 			rows := rapid.SliceOf(rapid.Custom(genPublicRow)).Draw(t, "rows")
232 			when := time.Unix(rapid.Int64Range(0, 4102444800).Draw(t, "when"), 0)
233 			text, _, _, _ = addDated(text, rows, when)
234 		}
235 		write(t, s, text)
236 		want := published(text)
237 
238 		got := parse(get(h, "/links/links.txt").Body.String())
239 		if len(got) != len(want) {
240 			t.Fatalf("links.txt has %d links, want %d", len(got), len(want))
241 		}
242 		for i := range want {
243 			if got[i].URL != want[i].URL || got[i].Desc != want[i].Desc || !slices.Equal(got[i].Tags, want[i].Tags) || got[i].Feed != want[i].Feed {
244 				t.Fatalf("links.txt link %d is %+v, want %+v", i, got[i], want[i])
245 			}
246 		}
247 
248 		f := feedOf(t, get(h, "/links/atom.xml"))
249 		if len(f.Entries) != len(want) {
250 			t.Fatalf("feed has %d entries, want %d", len(f.Entries), len(want))
251 		}
252 		for i, e := range f.Entries {
253 			l := want[len(want)-1-i]
254 			var tags []string
255 			for _, c := range e.Categories {
256 				tags = append(tags, c.Term)
257 			}
258 			if e.Title != l.Title() || e.Link.Href != l.URL || e.ID != l.URL || e.Updated != atomTime(l.Added) || !slices.Equal(tags, l.Tags) {
259 				t.Fatalf("entry %d is %+v, want %+v", i, e, l)
260 			}
261 		}
262 	})
263 }
