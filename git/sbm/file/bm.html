19 # Two URLs are the same bookmark when they differ only in scheme, a leading
20 # www., trailing slashes or the case of the host.
21 AWK_NORM='
22 function norm(u,    host, rest, i) {
23     sub(/^[A-Za-z][A-Za-z0-9+.-]*:\/\//, "", u)
24     i = index(u, "/")
25     if (i) { host = substr(u, 1, i - 1); rest = substr(u, i) }
26     else   { host = u; rest = "" }
27     host = tolower(host)
28     sub(/^www\./, "", host)
29     sub(/\/+$/, "", rest)
30     return host rest
31 }'

64 # dmenu whenever there is a display to draw it on. fzf stands in on a bare

66 pickmenu () {
67     [ -n "$SBM_MENU" ] && return

94 # fzfpick <fzf options>: the chosen line, or else what was typed, which is

103 pick () {
104     case $SBM_MENU in

111 # pickmulti <prompt>: like pick, for any number of lines.
112 pickmulti () {
113     choices=$(cat)
114     case $SBM_MENU in
115         fzf)
116             while :; do
117                 out=$(printf '%s\n' "$choices" | fzfcmd --multi --print-query \
118                       --prompt="$1 " --header='TAB marks, ENTER confirms. No match: ENTER creates the typed tag, ESC finishes.')
119                 chosen=$(printf '%s\n' "$out" | sed 1d)
120                 if [ -n "$chosen" ]; then
121                     printf '%s\n' "$chosen"
122                     break
123                 fi
124                 typed=$(printf '%s\n' "$out" | sed -n 1p)
125                 [ -n "$typed" ] || break
126                 printf '%s\n' "$typed"
127             done
128             ;;
129         dmenu)
130             while :; do
131                 one=$(printf 'done | (choose when done)\n%s\n' "$choices" \
132                       | dmenu -i -l "$DMENULINES" -p "$1")
133                 # An empty answer means the menu was closed.
134                 [ -z "$one" ] || [ "${one%% *}" = 'done' ] && break
135                 printf '%s\n' "$one"
136             done
137             ;;
138         *)  printf '%s\n' "$choices" | custommenu multi "$1" ;;
139     esac

193 # Bookmarks restricted to $tagfilter and ordered by $sortorder, each followed
194 # by a fourth field: how often it was used.
195 rows () {
196     SBM_TAG=$tagfilter SBM_ORDER=$sortorder SBM_USAGE=$USAGE awk -F'\t' '
197         BEGIN {
198             while ((getline line < ENVIRON["SBM_USAGE"]) > 0) {
199                 split(line, f, "\t")
200                 used[f[1]] = f[2]
201             }
202             order = ENVIRON["SBM_ORDER"]
203         }
204         NF == 0 || /^#/ { next }
205         ENVIRON["SBM_TAG"] != "" {
206             hit = 0
207             n = split($3, tags, " ")
208             for (i = 1; i <= n; i++)
209                 if (tags[i] == ENVIRON["SBM_TAG"]) hit = 1
210             if (!hit) next
211         }
212         {
213             if (order == "used")        key = sprintf("%09d", 999999999 - used[$1])
214             else if (order == "recent") key = sprintf("%09d", 999999999 - NR)
215             else if (order == "desc")   key = tolower($2)
216             else if (order == "tag")    key = tolower($3)
217             else                        key = ""
218             printf "%s\t%s\t%s\t%s\t%s\t%d\n", key, tolower($1), $1, $2, $3, used[$1]
219         }' "$BOOKMARKS" | LC_ALL=C sort -t"$TAB" -k1,1 -k2,2 | cut -f3-
220 }

235 # For fzf a fifth field is added, and it is all that is listed. The first
236 # four feed the preview, which follows the cursor.
237 PREVIEW="printf '%s\n%s\ntags: %s\nused: %s\n' {1} {2} {3} {4}"
238 fzfrows () {
239     awk -F'\t' 'NF {
240         desc = $2
241         if (length(desc) > 44) desc = substr(desc, 1, 42) ".."
242         printf "%s\t%-44s  %-18s  %s\n", $0, desc, $3, $1
243     }'
244 }

273 # find_dup <url>: print the bookmark that is the same page as <url>, if any.
274 find_dup () {
275     SBM_URL=$1 awk -F'\t' "$AWK_NORM"'

280 # tmpfile <file>: create an empty temporary file and print its name. Beside
281 # <file> if that directory can be written to, else in $TMPDIR. noclobber makes
282 # the shell create it with O_EXCL, so a file or symlink planted under that
283 # name in a shared /tmp is refused rather than followed: mktemp, in POSIX.
284 tmpfile () {
285     for candidate in "$1.tmp.$$" "${TMPDIR:-/tmp}/sbm.$$.${1##*/}"; do
286         if (umask 077; set -C; : > "$candidate") 2>/dev/null; then
287             printf '%s\n' "$candidate"
288             return 0
289         fi
290     done
291     return 1
292 }

294 # rewrite <file>: replace <file> with stdin. Through cat, not mv, so that a
295 # symlinked file stays a symlink and keeps its mode.
296 rewrite () {
297     tmp=$(tmpfile "$1") || die "cannot create a temporary file for $1"
298     cat > "$tmp" && cat "$tmp" > "$1"
299     rc=$?
300     rm -f "$tmp"
301     return $rc
302 }

304 # bump <url>: count a use, for the "used" order. The counts live beside the
305 # bookmark file, which stays a plain list: url, count, time of last use.
306 # srand() returns the previous seed, and the first seed is the time of day:
307 # that is "date +%s" for a POSIX that has no %s.
308 bump () {
309     [ -e "$USAGE" ] || : > "$USAGE" || return 0
310     SBM_URL=$1 awk -F'\t' '
311         BEGIN { OFS = "\t"; srand(); now = srand() }
312         $1 == ENVIRON["SBM_URL"] { $2 = $2 + 1; $3 = now; seen = 1 }
313         { print }
314         END { if (!seen) print ENVIRON["SBM_URL"], 1, now }
315     ' "$USAGE" | rewrite "$USAGE"
316 }

362 # urlencode <text>: for a query string. awk has no ord(), hence the table.
363 urlencode () {
364     printf '%s' "$1" | LC_ALL=C awk '
365         BEGIN { for (i = 1; i < 256; i++) ord[sprintf("%c", i)] = i }
366         {
367             for (i = 1; i <= length($0); i++) {
368                 c = substr($0, i, 1)
369                 if (c ~ /[A-Za-z0-9._~-]/) out = out c
370                 else if (c == " ")         out = out "+"
371                 else                       out = out sprintf("%%%02X", ord[c])
372             }
373         }
374         END { printf "%s", out }'
375 }

377 # go <text>: open text that is not a bookmark. One word with "://" or a dot
378 # in it is an address. Anything else is a web search: by the engine whose
379 # keyword is the first word ($ENGINES holds "keyword | url with %s" lines),
380 # or else by $SEARCH.
381 go () {
382     case $1 in
383         *' '*) ;;
384         *://*) opener "$1"; return ;;
385         *.*)   opener "https://$1"; return ;;
386     esac
387     engine=$SEARCH
388     query=$1
389     if [ "${1%% *}" != "$1" ] && [ -r "$ENGINES" ]; then
390         found=$(SBM_KEY=${1%% *} awk -F' *[|] *' \
391                 '$1 == ENVIRON["SBM_KEY"] { print $2; exit }' "$ENGINES")
392         if [ -n "$found" ]; then
393             engine=$found
394             query=${1#* }
395         fi
396     fi
397     opener "$(printf '%s\n' "$engine" | sed "s|%s|$(urlencode "$query")|")"
398 }

406 # Exit status 3 means "already bookmarked".
407 add () {
408     url=$1
409     if [ -z "$url" ]; then
410         url=$(ask 'url' "$(clip_in | grep '://' | head -n 1)")
411     fi
412     url=$(printf '%s' "$url" | tr -d '[:space:]')
413     [ -n "$url" ] || exit 1
414 
415     dup=$(find_dup "$url")
416     if [ -n "$dup" ]; then
417         printf 'bm: already bookmarked:\n%s\n' "$dup" >&2
418         exit 3
419     fi
420 
421     # bm-title, where installed, fetches the page's title.
422     title=
423     if have bm-title; then title=$(bm-title "$url"); fi
424     desc=$(ask 'description' "$title" | tr '\t' ' ')
425 
426     [ -e "$USERTAGS" ] || : > "$USERTAGS"
427     names=$(pickmulti 'tags:' < "$USERTAGS" | sed 's/[ |].*//' | sed '/^$/d' \
428             | sort -u)
429     printf '%s\n' "$names" | learn_tags
430     tags=$(printf '%s\n' "$names" | paste -sd ' ' -)
431 
432     printf '%s\t%s\t%s\n' "$url" "$desc" "$tags" >> "$BOOKMARKS"
433     changed "bm: add $url"
434 }

436 # merge: bookmark lines on stdin (a bare URL per line will do); the ones not
437 # known yet are appended. This is how the output of bm-import gets in.
438 merge () {
439     new=$(tmpfile "$BOOKMARKS") || die 'cannot create a temporary file'
440     SBM_FILE=$BOOKMARKS SBM_NEW=$new awk -F'\t' "$AWK_NORM"'
441         BEGIN {
442             while ((getline line < ENVIRON["SBM_FILE"]) > 0) {
443                 split(line, f, "\t")
444                 seen[norm(f[1])] = 1
445             }
446         }
447         NF == 0 || /^#/ { next }
448         seen[norm($1)]++ { skipped++; next }
449         { printf "%s\t%s\t%s\n", $1, $2, $3 > ENVIRON["SBM_NEW"]; added++ }
450         END { printf "added %d, skipped %d duplicates\n", added, skipped }'
451     cut -f3 "$new" | tr ' ' '\n' | sort -u | learn_tags
452     cat "$new" >> "$BOOKMARKS"
453     rm -f "$new"
454     changed 'bm: merge'
455 }

486 # browse_fzf: fzf can bind keys, so one list does everything. Opening and
487 # copying leave it, the other keys come back to it.
488 browse_fzf () {
489     while :; do
490         header="ENTER open  ^Y copy  ^A add  ^D delete  ^E edit  ^T tag [${tagfilter:-all}]  ^S sort [$sortorder]"
491         out=$(rows | fzfrows | withpreview fzfcmd --print-query \
492               --expect=ctrl-y,ctrl-a,ctrl-d,ctrl-e,ctrl-t,ctrl-s \
493               --header="$header" --prompt='bookmark: ')
494         [ -n "$out" ] || exit 1
495         typed=$(printf '%s\n' "$out" | sed -n 1p)
496         key=$(printf '%s\n' "$out" | sed -n 2p)
497         url=$(printf '%s\n' "$out" | sed -n 3p | cut -f1)
498 
499         case $key in
500             ctrl-a) add_inside; continue ;;
501             ctrl-t) picktag; continue ;;
502             ctrl-s) nextorder; continue ;;
503         esac
504         if [ -z "$url" ]; then
505             [ -n "$key" ] && continue
506             [ -n "$typed" ] || exit 1
507             go "$typed"
508             exit 0
509         fi
510         case $key in
511             ctrl-y) copy_url "$url"; exit 0 ;;
512             ctrl-e) edit_url "$url" ;;
513             ctrl-d)
514                 case $(ask "delete $url? (y/N)") in
515                     y|Y|yes) delete_url "$url" ;;
516                 esac
517                 ;;
518             *)      open_url "$url"; exit 0 ;;
519         esac
520     done
521 }

523 # browse_menu: dmenu can neither bind keys nor preview, so a second menu does
524 # both: it shows the bookmark in full and asks what to do with it. Closing it
525 # goes back to the list; closing the list leaves.
526 browse_menu () {
527     while :; do
528         chosen=$( { printf '[add]\n'; rows | cut -f1-3; } | pick 'bookmark:')
529         [ -n "$chosen" ] || exit 1
530         if [ "$chosen" = '[add]' ]; then
531             add_inside
532             continue
533         fi
534         url=$(printf '%s\n' "$chosen" | cut -f1)
535         if ! is_bookmark "$url"; then
536             go "$chosen"
537             exit 0
538         fi
539         action=$( { printf 'open\ncopy\nedit\ndelete\n'
540                     details "$url" | sed 's/^/  /'; } | pick 'do:')
541         case $action in
542             open)   open_url "$url"; exit 0 ;;
543             copy)   copy_url "$url"; exit 0 ;;
544             edit)   edit_url "$url" ;;
545             delete) delete_url "$url" ;;
546         esac
547     done
548 }

560 # needarg <flag> <count of remaining arguments> <value>
561 needarg () {
562     if [ "$2" -lt 2 ] || [ -z "$3" ]; then
563         printf 'bm: %s needs an argument\n' "$1" >&2
564         exit 2
565     fi
