21 # Two URLs are the same bookmark when they differ only in scheme, a leading
22 # www., trailing slashes or the case of the host.
23 AWK_NORM='
24 function norm(u,    host, rest, i) {
25     sub(/^[A-Za-z][A-Za-z0-9+.-]*:\/\//, "", u)
26     i = index(u, "/")
27     if (i) { host = substr(u, 1, i - 1); rest = substr(u, i) }
28     else   { host = u; rest = "" }
29     host = tolower(host)
30     sub(/^www\./, "", host)
31     sub(/\/+$/, "", rest)
32     return host rest
33 }'

113 # pickmulti <prompt>: like pick, for any number of lines.
114 pickmulti () {
115     choices=$(cat)
116     case $SBM_MENU in
117         fzf)
118             while :; do
119                 out=$(printf '%s\n' "$choices" | fzfcmd --multi --print-query \
120                       --prompt="$1 " --header='TAB marks, ENTER confirms. No match: ENTER creates the typed tag, ESC finishes.')
121                 chosen=$(printf '%s\n' "$out" | sed 1d)
122                 if [ -n "$chosen" ]; then
123                     printf '%s\n' "$chosen"
124                     break
125                 fi
126                 typed=$(printf '%s\n' "$out" | sed -n 1p)
127                 [ -n "$typed" ] || break
128                 printf '%s\n' "$typed"
129             done
130             ;;
131         dmenu)
132             while :; do
133                 one=$(printf 'done | (choose when done)\n%s\n' "$choices" \
134                       | dmenu -i -l "$DMENULINES" -p "$1")
135                 # An empty answer means the menu was closed.
136                 [ -z "$one" ] || [ "${one%% *}" = 'done' ] && break
137                 printf '%s\n' "$one"
138             done
139             ;;
140         *)  printf '%s\n' "$choices" | custommenu multi "$1" ;;
141     esac

195 # Bookmarks restricted to $tagfilter and ordered by $sortorder, each followed
196 # by a fourth field: how often it was used.
197 rows () {
198     SBM_TAG=$tagfilter SBM_ORDER=$sortorder SBM_USAGE=$USAGE awk -F'\t' '
199         BEGIN {
200             while ((getline line < ENVIRON["SBM_USAGE"]) > 0) {
201                 split(line, f, "\t")
202                 used[f[1]] = f[2]
203             }
204             order = ENVIRON["SBM_ORDER"]
205         }
206         NF == 0 || /^#/ { next }
207         ENVIRON["SBM_TAG"] != "" {
208             hit = 0
209             n = split($3, tags, " ")
210             for (i = 1; i <= n; i++)
211                 if (tags[i] == ENVIRON["SBM_TAG"]) hit = 1
212             if (!hit) next
213         }
214         {
215             if (order == "used")        key = sprintf("%09d", 999999999 - used[$1])
216             else if (order == "recent") key = sprintf("%09d", 999999999 - NR)
217             else if (order == "desc")   key = tolower($2)
218             else if (order == "tag")    key = tolower($3)
219             else                        key = ""
220             printf "%s\t%s\t%s\t%s\t%s\t%d\n", key, tolower($1), $1, $2, $3, used[$1]
221         }' "$BOOKMARKS" | LC_ALL=C sort -t"$TAB" -k1,1 -k2,2 | cut -f3-
222 }

237 # For fzf a fifth field is added, and it is all that is listed. The first
238 # four feed the preview, which follows the cursor.
239 PREVIEW="printf '%s\n%s\ntags: %s\nused: %s\n' {1} {2} {3} {4}"
240 fzfrows () {
241     awk -F'\t' 'NF {
242         desc = $2
243         if (length(desc) > 44) desc = substr(desc, 1, 42) ".."
244         printf "%s\t%-44s  %-18s  %s\n", $0, desc, $3, $1
245     }'

282 # tmpfile <file>: create an empty temporary file and print its name. Beside
283 # <file> if that directory can be written to, else in $TMPDIR. noclobber makes
284 # the shell create it with O_EXCL, so a file or symlink planted under that
285 # name in a shared /tmp is refused rather than followed: mktemp, in POSIX.
286 tmpfile () {
287     for candidate in "$1.tmp.$$" "${TMPDIR:-/tmp}/sbm.$$.${1##*/}"; do
288         if (umask 077; set -C; : > "$candidate") 2>/dev/null; then
289             printf '%s\n' "$candidate"
290             return 0
291         fi
292     done
293     return 1
294 }

296 # rewrite <file>: replace <file> with stdin. Through cat, not mv, so that a
297 # symlinked file stays a symlink and keeps its mode.
298 rewrite () {
299     tmp=$(tmpfile "$1") || die "cannot create a temporary file for $1"
300     cat > "$tmp" && cat "$tmp" > "$1"
301     rc=$?
302     rm -f "$tmp"
303     return $rc
304 }

306 # lock: take the write lock of the bookmark file, a directory beside it. bm
307 # holds it while it changes the file, and bm-sync while it reads or writes
308 # the file, so that neither writes over a change of the other. A lock whose
309 # process is gone is taken over. After $SBM_LOCK_WAIT seconds (default 10)
310 # bm stops.
311 lock () {
312     waited=0
313     until mkdir "$WLOCK" 2>/dev/null; do
314         pid=$(cat "$WLOCK/pid" 2>/dev/null)
315         if [ -n "$pid" ] && ! kill -0 "$pid" 2>/dev/null; then
316             [ "$(cat "$WLOCK/pid" 2>/dev/null)" != "$pid" ] || rm -rf "$WLOCK"
317             continue
318         fi
319         [ "$waited" -lt "${SBM_LOCK_WAIT:-10}" ] \
320             || die "$BOOKMARKS is in use (bm -e or bm-sync). If neither runs, remove $WLOCK"
321         sleep 1
322         waited=$((waited + 1))
323     done
324     echo $$ > "$WLOCK/pid"
325     locked=1
326     # Also in the subshell of add_inside, which does not keep these traps.
327     trap unlock EXIT
328     trap 'exit 1' HUP INT TERM
329 }
330 
331 unlock () {
332     [ -z "$locked" ] || rm -rf "$WLOCK"
333     locked=
334 }

336 # bump <url>: count a use, for the "used" order. The counts live beside the
337 # bookmark file, which stays a plain list: url, count, time of last use.
338 # srand() returns the previous seed, and the first seed is the time of day:
339 # that is "date +%s" for a POSIX that has no %s.
340 bump () {
341     [ -e "$USAGE" ] || : > "$USAGE" || return 0
342     SBM_URL=$1 awk -F'\t' '
343         BEGIN { OFS = "\t"; srand(); now = srand() }
344         $1 == ENVIRON["SBM_URL"] { $2 = $2 + 1; $3 = now; seen = 1 }
345         { print }
346         END { if (!seen) print ENVIRON["SBM_URL"], 1, now }
347     ' "$USAGE" | rewrite "$USAGE"
348 }

350 # learn_tags: tag names on stdin; the ones $USERTAGS lacks are appended to it.

353     while IFS= read -r name; do

403 # urlencode <text>: for a query string. awk has no ord(), hence the table.
404 urlencode () {
405     printf '%s' "$1" | LC_ALL=C awk '
406         BEGIN { for (i = 1; i < 256; i++) ord[sprintf("%c", i)] = i }
407         {
408             for (i = 1; i <= length($0); i++) {
409                 c = substr($0, i, 1)
410                 if (c ~ /[A-Za-z0-9._~-]/) out = out c
411                 else if (c == " ")         out = out "+"
412                 else                       out = out sprintf("%%%02X", ord[c])
413             }
414         }
415         END { printf "%s", out }'
416 }

418 # go <text>: open text that is not a bookmark. One word with "://" or a dot
419 # in it is an address. Anything else is a web search: by the engine whose
420 # keyword is the first word ($ENGINES holds "keyword | url with %s" lines),
421 # or else by $SEARCH.
422 go () {
423     case $1 in
424         *' '*) ;;
425         *://*) opener "$1"; return ;;
426         *.*)   opener "https://$1"; return ;;
427     esac
428     engine=$SEARCH
429     query=$1
430     if [ "${1%% *}" != "$1" ] && [ -r "$ENGINES" ]; then
431         found=$(SBM_KEY=${1%% *} awk -F' *[|] *' \
432                 '$1 == ENVIRON["SBM_KEY"] { print $2; exit }' "$ENGINES")
433         if [ -n "$found" ]; then
434             engine=$found
435             query=${1#* }
436         fi
437     fi
438     opener "$(printf '%s\n' "$engine" | sed "s|%s|$(urlencode "$query")|")"
439 }

447 # Exit status 3 means "already bookmarked".
448 add () {
449     url=$1
450     if [ -z "$url" ]; then
451         url=$(ask 'url' "$(clip_in | grep '://' | head -n 1)")
452     fi
453     url=$(printf '%s' "$url" | tr -d '[:space:]')
454     [ -n "$url" ] || exit 1
455 
456     dup=$(find_dup "$url")
457     if [ -n "$dup" ]; then
458         printf 'bm: already bookmarked:\n%s\n' "$dup" >&2
459         exit 3
460     fi
461 
462     # bm-title, where installed, fetches the page's title.
463     title=
464     if have bm-title; then title=$(bm-title "$url"); fi
465     desc=$(ask 'description' "$title" | tr '\t' ' ')
466 
467     [ -e "$USERTAGS" ] || : > "$USERTAGS"
468     names=$(pickmulti 'tags:' < "$USERTAGS" | sed 's/[ |].*//' | sed '/^$/d' \
469             | sort -u)
470     printf '%s\n' "$names" | learn_tags
471     tags=$(printf '%s\n' "$names" | paste -sd ' ' -)
472 
473     lock
474     printf '%s\t%s\t%s\n' "$url" "$desc" "$tags" >> "$BOOKMARKS"
475     unlock
476     changed "bm: add $url"

479 # merge: bookmark lines on stdin (a bare URL per line will do); the ones not
480 # known yet are appended. This is how the output of bm-import gets in.
481 merge () {
482     new=$(tmpfile "$BOOKMARKS") || die 'cannot create a temporary file'
483     lock
484     SBM_FILE=$BOOKMARKS SBM_NEW=$new awk -F'\t' "$AWK_NORM"'
485         BEGIN {
486             while ((getline line < ENVIRON["SBM_FILE"]) > 0) {
487                 split(line, f, "\t")
488                 seen[norm(f[1])] = 1
489             }
490         }
491         NF == 0 || /^#/ { next }
492         seen[norm($1)]++ { skipped++; next }
493         { printf "%s\t%s\t%s\n", $1, $2, $3 > ENVIRON["SBM_NEW"]; added++ }
494         END { printf "added %d, skipped %d duplicates\n", added, skipped }'
495     cut -f3 "$new" | tr ' ' '\n' | sort -u | learn_tags
496     cat "$new" >> "$BOOKMARKS"
497     unlock
498     rm -f "$new"
499     changed 'bm: merge'
500 }

521 # browse_fzf: fzf can bind keys, so one list does everything. Opening and
522 # copying leave it, the other keys come back to it.
523 browse_fzf () {
524     while :; do
525         header="ENTER open  ^Y copy  ^A add  ^D delete  ^E edit  ^S sort [$sortorder]"
526         out=$(rows | fzfrows | withpreview fzfcmd --print-query \
527               --expect=ctrl-y,ctrl-a,ctrl-d,ctrl-e,ctrl-s \
528               --header="$header" --prompt='bookmark: ')
529         [ -n "$out" ] || exit 1
530         typed=$(printf '%s\n' "$out" | sed -n 1p)
531         key=$(printf '%s\n' "$out" | sed -n 2p)
532         url=$(printf '%s\n' "$out" | sed -n 3p | cut -f1)
533 
534         case $key in
535             ctrl-a) add_inside; continue ;;
536             ctrl-s) nextorder; continue ;;
537         esac
538         if [ -z "$url" ]; then
539             [ -n "$key" ] && continue
540             [ -n "$typed" ] || exit 1
541             go "$typed"
542             exit 0
543         fi
544         case $key in
545             ctrl-y) copy_url "$url"; exit 0 ;;
546             ctrl-e) edit_url "$url" ;;
547             ctrl-d)
548                 case $(ask "delete $url? (y/N)") in
549                     y|Y|yes) delete_url "$url" ;;
550                 esac
551                 ;;
552             *)      open_url "$url"; exit 0 ;;
553         esac
554     done
555 }

557 # browse_menu: dmenu can neither bind keys nor preview, so a second menu does
558 # both: it shows the bookmark in full and asks what to do with it. Closing it
559 # goes back to the list; closing the list leaves.
560 browse_menu () {
561     while :; do
562         chosen=$( { printf '[add]\n'; rows | cut -f1-3; } | pick 'bookmark:')
563         [ -n "$chosen" ] || exit 1
564         if [ "$chosen" = '[add]' ]; then
565             add_inside
566             continue
567         fi
568         url=$(printf '%s\n' "$chosen" | cut -f1)
569         if ! is_bookmark "$url"; then
570             go "$chosen"
571             exit 0
572         fi
573         action=$( { printf 'open\ncopy\nedit\ndelete\n'
574                     details "$url" | sed 's/^/  /'; } | pick 'do:')
575         case $action in
576             open)   open_url "$url"; exit 0 ;;
577             copy)   copy_url "$url"; exit 0 ;;
578             edit)   edit_url "$url" ;;
579             delete) delete_url "$url" ;;
580         esac
581     done
582 }
