3 # Scripted stand-in for fzf, installed as "fzf" on the test PATH. Consumes one
4 # line of $SBM_TEST_ANSWERS per call, like test/fakemenu. What it prints
5 # depends on the options it was given, as with the real thing:
6 #
7 #   --expect             answer key:text   the typed text, the key ("enter"
8 #                                          is the empty key), then the first
9 #                                          row containing text, if any
10 #   --multi              answer text:words the typed text, then the choices
11 #                                          that start with each word
12 #   --print-query alone  answer text       the typed text, then the first row
13 #                                          containing it, if any
14 #   none of these        answer text       the first row containing text
15 #
16 # An empty answer is ESC: no output, exit 130.
17

26     esac
27 done
