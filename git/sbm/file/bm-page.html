3 # bm-page: print a web page as plain text: its title, a blank line, then its
4 # text. When it is installed, the fzf list of bm shows the page of the
5 # bookmark under the cursor in the preview, live. SBM_FETCH=0 keeps it off
6 # the network.
7 #
8 #     bm-page https://suckless.org/ | less
9 #
10 # Only http and https pages are fetched. curl stops after 5 seconds, and dd
11 # passes at most 512k of the page. Scripts, styles and comments are left
12 # out. Control characters are removed, so that a page cannot send escape
13 # sequences to the terminal.
14 
15 if [ $# -ne 1 ]; then
16     printf 'usage: bm-page <url>\n' >&2
17     exit 2
18 fi
19 [ "${SBM_FETCH:-1}" != 0 ] || exit 0
20 case $(printf '%s' "$1" | tr '[:upper:]' '[:lower:]') in
21     http://*|https://*) ;;
22     *) exit 0 ;;
23 esac

25 # The first awk reads the page as one record (RS is a control character),
26 # then splits it at each "<". Each part starts with a tag and ends with the
27 # text after it, so one pass through the parts reads the page in linear
28 # time. The second awk tidies the white space.
29 curl -sL --max-time 5 -A 'Mozilla/5.0' -- "$1" 2>/dev/null \

31     function utf8(c) {
32         if (c == 160) return " "
33         if (c < 128) return sprintf("%c", c)
34         if (c < 2048) return sprintf("%c%c", 192 + int(c / 64), 128 + c % 64)
35         if (c < 65536)
36             return sprintf("%c%c%c", 224 + int(c / 4096), 128 + int(c / 64) % 64, 128 + c % 64)
37         if (c < 1114112)
38             return sprintf("%c%c%c%c", 240 + int(c / 262144), 128 + int(c / 4096) % 64,
39                            128 + int(c / 64) % 64, 128 + c % 64)
40         return ""
41     }
42     function entity(name,    n, i) {
43         if (name ~ /^#[0-9]+$/) return utf8(substr(name, 2) + 0)
44         if (name ~ /^#[xX][0-9A-Fa-f]+$/) {
45             n = 0
46             for (i = 3; i <= length(name); i++)
47                 n = n * 16 + index("0123456789abcdef", tolower(substr(name, i, 1))) - 1
48             return utf8(n)
49         }
50         if (name in named) return named[name]
51         return "&" name ";"
52     }
53     function decode(s,    out, i, j, name) {
54         out = ""
55         while ((i = index(s, "&")) > 0) {
56             out = out substr(s, 1, i - 1)
57             s = substr(s, i + 1)
58             j = index(s, ";")
59             name = substr(s, 1, j - 1)
60             if (j > 1 && j <= 12 && name ~ /^(#[0-9]+|#[xX][0-9A-Fa-f]+|[A-Za-z][A-Za-z0-9]*)$/) {
61                 out = out entity(name)
62                 s = substr(s, j + 1)
63             } else {
64                 out = out "&"
65             }
66         }
67         return out s
68     }
69     # text: the text of the page, or of its title. As in a browser, a line
70     # break in the text is a space. The line breaks come from the tags: brk
71     # tells how many are waiting for the next text.
72     function text(s) {
73         if (intitle) {
74             title = title " " s
75             return
76         }
77         s = decode(s)
78         gsub(/[ \t\r\n]+/, " ", s)

81             return
82         }

85             brk = 0
86             sub(/^ /, "", s)
87         }
88         printf "%s", s
