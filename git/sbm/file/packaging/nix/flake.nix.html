18       packages = forAllSystems (pkgs: rec {
19         sbm = pkgs.callPackage ./default.nix { };
20         default = sbm;
21       });
22 
23       # Build the working tree instead of the tagged release:

26       # What "make check" wants, plus the runtime programs. dmenu is left out

28       devShells = forAllSystems (pkgs: {
