1 #!/bin/sh
2 # Exercise the installed programs. --list and --merge are the two actions
3 # that never open a menu, so this needs no display and no menu program.
4 set -e
5 
6 work=${AUTOPKGTEST_TMP:?AUTOPKGTEST_TMP is not set}
7 
8 BOOKMARKS="$work/bookmarks"
9 USERTAGS="$work/usertags"
10 export BOOKMARKS USERTAGS
11 SBM_GIT=0
12 SBM_FETCH=0
13 export SBM_GIT SBM_FETCH
14 
15 printf '%s\t%s\t%s\n' 'https://equwal.com' "Spenser Truex's website" 'users' \
16 	| bm --merge
17 
18 bm --list | grep -q 'https://equwal.com'
19 
20 # The same URL again, with a trailing slash, must be refused as a duplicate.
21 printf '%s\t%s\t%s\n' 'https://equwal.com/' 'duplicate' 'users' \
22 	| bm --merge | grep -q 'skipped 1'
23 
24 test "$(bm --list | wc -l)" = 1
25 
26 bm-html | grep -q '<li>'
27 
28 echo 'smoke: ok'
