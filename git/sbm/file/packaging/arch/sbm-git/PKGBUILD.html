31 pkgver() {
32   cd "$_pkgname"
33   # There are no tags yet, so git describe has nothing to work from. Fall
34   # back to the version in config.mk plus the commit count. Once v0.3 is
35   # tagged the first branch takes over on its own.
36   # The exit status of a pipeline is sed's, which always succeeds: ask git
37   # first, and only then format.
38   if git describe --long --tags >/dev/null 2>&1; then
39     git describe --long --tags | sed 's/^v//; s/\([^-]*-g\)/r/; s/-/./g'
40   else
41     printf '%s.r%s.g%s'       "$(sed -n 's/^VERSION=//p' config.mk)"       "$(git rev-list --count HEAD)"       "$(git rev-parse --short HEAD)"
42   fi
43 }
