61 # Find a POSIX sh. Git for Windows ships one; the package depends on git for
62 # exactly that reason. Look in several places rather than one hardcoded path,
63 # because Git installs per machine, per user, 32 bit, and through Chocolatey.
64 # ---------------------------------------------------------------------------
65 
66 function Get-PosixSh {
67   $candidates = New-Object 'System.Collections.Generic.List[string]'
68 
69   $gitCmd = Get-Command 'git.exe' -ErrorAction SilentlyContinue
70   if ($gitCmd) {
71     # ...\Git\cmd\git.exe and ...\Git\bin\git.exe both give ...\Git here.
72     $gitRoot = Split-Path -Parent (Split-Path -Parent $gitCmd.Source)
73     $candidates.Add((Join-Path $gitRoot 'bin\sh.exe'))
74     $candidates.Add((Join-Path $gitRoot 'usr\bin\sh.exe'))
75   }
76 
77   $keys = @(
78     'HKLM:\SOFTWARE\GitForWindows',
79     'HKLM:\SOFTWARE\WOW6432Node\GitForWindows',
80     'HKCU:\SOFTWARE\GitForWindows'
81   )
82   foreach ($key in $keys) {
83     try {
84       $installPath = (Get-ItemProperty -Path $key -Name 'InstallPath' -ErrorAction Stop).InstallPath
85       if ($installPath) {
86         $candidates.Add((Join-Path $installPath 'bin\sh.exe'))
87         $candidates.Add((Join-Path $installPath 'usr\bin\sh.exe'))
88       }
89     } catch {
90       # No such key on this machine.
91     }
92   }
93 
94   $roots = @(
95     $env:ProgramFiles,
96     ${env:ProgramFiles(x86)},
97     $env:ProgramW6432,
98     $(if ($env:LOCALAPPDATA) { Join-Path $env:LOCALAPPDATA 'Programs' }),
99     $(if ($env:ChocolateyInstall) { Join-Path $env:ChocolateyInstall 'lib\git.install\tools' })
100   )
101   foreach ($root in $roots) {
102     if ($root) {
103       $candidates.Add((Join-Path $root 'Git\bin\sh.exe'))
104       $candidates.Add((Join-Path $root 'Git\usr\bin\sh.exe'))
105     }
106   }
107 
108   $shCmd = Get-Command 'sh.exe' -ErrorAction SilentlyContinue
109   if ($shCmd) { $candidates.Add($shCmd.Source) }
110 
111   foreach ($candidate in $candidates) {
112     if ($candidate -and (Test-Path -LiteralPath $candidate -PathType Leaf)) { return $candidate }
113   }
114   return $null
115 }

117 $sh = Get-PosixSh
118 if (-not $sh) {
119   throw @'
120 sbm: no POSIX sh found.
121 
122 sbm is POSIX shell and needs one to run. Git for Windows ships sh.exe and is
123 a dependency of this package:

125     choco install git
126 
127 If Git is installed somewhere unusual, sbm still works: put that sh.exe on
128 PATH and reinstall this package.
