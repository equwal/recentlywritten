1 ; sbm.nsi: the Windows installer of sbm. contrib/windows/build-installer.sh
2 ; runs makensis on this file.
3 ;
4 ; The installer needs Cygwin with mintty. It puts bm and the bm-* tools in
5 ; /usr/local/bin, fzf.exe in /usr/local/libexec/sbm, and the settings in
6 ; /etc/profile.d/sbm.sh. It makes a desktop shortcut and a Start menu entry
7 ; that open bm in a mintty window. The key of the desktop shortcut is
8 ; Ctrl+Alt+Shift+F24. Almost no keyboard has an F24 key, so keyboard firmware
9 ; can use this key combination without conflicts. Only one shortcut has the
10 ; key: when two shortcuts have the same key, Explorer can lose the key.
11 ;
12 ; A Startup entry runs bm-watch in a hidden mintty at each login. bm-watch
13 ; brings the bookmarks of the Chromium browsers into bm, and then the new
14 ; bookmarks as you add them. The installer also starts bm-watch at once.
15 ;
16 ; The uninstaller stops bm-watch and removes these files. It keeps the
17 ; bookmarks of the user.

59 Function .onInit
60   ; Cygwin setup is a 64-bit program. It writes to the 64-bit registry.
61   SetRegView 64
62   ReadRegStr $Cygwin HKLM "Software\Cygwin\setup" "rootdir"
63   ${If} $Cygwin == ""
64     ReadRegStr $Cygwin HKCU "Software\Cygwin\setup" "rootdir"
65   ${EndIf}
66   ${IfNot} ${FileExists} "$Cygwin\bin\mintty.exe"
67     MessageBox MB_ICONSTOP "sbm needs Cygwin with mintty.$\r$\nInstall Cygwin from https://cygwin.com, then start this installer again."
68     Abort
69   ${EndIf}
70 FunctionEnd
