1 # sbm.sh: settings for bm under Cygwin. The installer puts this file in
2 # /etc/profile.d, and each login shell reads it. The shell reads
3 # ~/.bash_profile and ~/.profile later, so settings there win.
4 
5 # Open URLs with the default browser of Windows.
6 export SBM_OPEN="${SBM_OPEN:-cygstart}"
7 # Use the Windows clipboard. sed removes the CR at the end of each line.
8 export SBM_COPY="${SBM_COPY:-cp /dev/stdin /dev/clipboard}"
9 export SBM_PASTE="${SBM_PASTE:-sed s/\r\$// /dev/clipboard}"
10 # fzf.exe reads non-ASCII input only in full screen.
11 export SBM_FZF_OPTS="${SBM_FZF_OPTS:---no-height}"
