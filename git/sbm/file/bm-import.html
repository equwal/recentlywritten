3 # bm-import: turn a browser's bookmarks into bm's format, on stdout.
4 #
5 #     bm-import bookmarks.html | bm --merge
6 #
7 # Reads the Netscape HTML that every browser exports, or the JSON file that
8 # Chromium style browsers keep (this needs jq). Folder names become tags:
9 # "Dev Tools" is dev-tools. Only http, https, ftp and file URLs are kept.
10 
11 usage () {
12     cat <<'EOF2'
13 usage: bm-import <file> | vivaldi | chrome | brave | edge | chromium
14 EOF2
15 }
16 
17 die () {
18     printf 'bm-import: %s\n' "$*" >&2
19     exit 1
20 }

23     usage >&2

32     vivaldi)  src=vivaldi ;;

46 # Chromium JSON. The trash (Vivaldi has one) is left out.
47 from_json () {
48     command -v jq >/dev/null 2>&1 || die 'a Chromium bookmark file needs jq'
49     jq -r '
50         def slug: ascii_downcase | gsub("[^a-z0-9]+"; "-")
51                   | sub("^-"; "") | sub("-$"; "");
52         def emit(path):
53             if .type == "url" then
54                 [.url, (.name | gsub("[\t\r\n]+"; " ")),
55                  (path | unique | join(" "))] | join("\t")
56             elif (.children? | type) == "array" then
57                 (path + ([.name | slug] | map(select(. != "")))) as $sub
58                 | .children[] | emit($sub)
59             else empty end;
60         .roots | to_entries[]
61         | select(.key != "trash" and (.value | type) == "object")
62         | .value.children[]? | emit([])' "$1"
63 }
64 
65 # Netscape HTML. <H3> opens a folder, </DL> closes it.
66 from_html () {
67     awk '
68     function decode(s) {
69         gsub(/&lt;/, "<", s); gsub(/&gt;/, ">", s); gsub(/&quot;/, "\"", s)
70         gsub(/&#39;|&#x27;|&apos;/, "\047", s); gsub(/&nbsp;/, " ", s)
71         gsub(/&amp;/, "\\&", s)
72         gsub(/[ \t\r]+/, " ", s); sub(/^ /, "", s); sub(/ $/, "", s)
73         return s
74     }
75     function slug(s) {
76         s = tolower(s)
77         gsub(/[^a-z0-9]+/, "-", s); sub(/^-/, "", s); sub(/-$/, "", s)
78         return s
79     }
80     {
81         upper = toupper($0)
82         if ((at = index(upper, "<H3")) > 0) {
83             name = substr($0, at)
84             name = substr(name, index(name, ">") + 1)
85             end = index(toupper(name), "</H3")
86             if (end) name = substr(name, 1, end - 1)
87             folder[++depth] = slug(decode(name))
88         }
89         if ((at = index(upper, "HREF=\"")) > 0) {
90             url = substr($0, at + 6)
91             rest = substr(url, index(url, "\"") + 1)
92             url = decode(substr(url, 1, index(url, "\"") - 1))
93             name = substr(rest, index(rest, ">") + 1)
94             end = index(toupper(name), "</A")
95             if (end) name = substr(name, 1, end - 1)
96             tags = ""
97             for (i = 1; i <= depth; i++)
98                 if (folder[i] != "" && index(" " tags " ", " " folder[i] " ") == 0)
99                     tags = tags (tags == "" ? "" : " ") folder[i]
100             printf "%s\t%s\t%s\n", url, decode(name), tags
101         }
102         if (index(upper, "</DL>") > 0 && depth > 0) depth--
103     }' "$1"
104 }

106 case $(LC_ALL=C tr -d ' \t\r\n' < "$src" | dd bs=1 count=1 2>/dev/null) in
107     '{') from_json "$src" ;;
108     *)   from_html "$src" ;;
109 esac | awk -F'\t' '$1 ~ /^(https?|ftp|file):\/\//'
