30 def inline(s):
31     out = []
32     for i, part in enumerate(re.split(r"(`[^`]*`)", s)):
33         if i % 2:
34             out.append("<code>%s</code>" % esc(part[1:-1]))
35         else:
36             t = esc(part)
37             t = re.sub(r"\*\*(.+?)\*\*", r"<b>\1</b>", t)
38             t = re.sub(r"\[([^\]]+)\]\(([^)]+)\)", r'<a href="\2">\1</a>', t)
39             out.append(t)
40     return "".join(out)

43 def md(text, toc, prefix, base=2):
44     """Small Markdown subset: headings, paragraphs, lists, tables, code."""
45     out, lines, i = [], text.splitlines(), 0
46     while i < len(lines):
47         l = lines[i]
48         m = re.match(r"(#+)\s+(.*)", l)
49         if m:
50             lvl = min(len(m.group(1)) - 1 + base, 6)
51             if lvl < base:
52                 i += 1
53                 continue
54             a = prefix + slug(m.group(2))
55             if lvl <= base + 1:
56                 toc.append((lvl, a, m.group(2)))
57             out.append('<h%d id="%s">%s</h%d>' % (lvl, a, inline(m.group(2)), lvl))
58             i += 1
59         elif l.startswith("```"):
60             j = i + 1
61             while j < len(lines) and not lines[j].startswith("```"):
62                 j += 1
63             out.append("<pre>%s</pre>" % esc("\n".join(lines[i + 1:j])))
64             i = j + 1
65         elif l.startswith("    "):
66             j = i
67             while j < len(lines) and (lines[j].startswith("    ") or not lines[j].strip()):
68                 j += 1
69             out.append("<pre>%s</pre>" % esc("\n".join(x[4:] for x in lines[i:j]).rstrip()))
70             i = j
71         elif l.startswith("|"):
72             rows = []
73             while i < len(lines) and lines[i].startswith("|"):
74                 cells = [c.strip() for c in lines[i].strip().strip("|").split("|")]
75                 if not all(re.fullmatch(r":?-+:?", c) for c in cells):
76                     rows.append(cells)
77                 i += 1
78             h = "".join("<th>%s</th>" % inline(c) for c in rows[0])
79             b = "".join("<tr>%s</tr>" % "".join("<td>%s</td>" % inline(c) for c in r) for r in rows[1:])
80             out.append('<div class="tw"><table><tr>%s</tr>%s</table></div>' % (h, b))
81         elif re.match(r"- ", l):
82             items = []
83             while i < len(lines) and (lines[i].startswith("- ") or lines[i].startswith("  ")):
84                 if lines[i].startswith("- "):
85                     items.append(lines[i][2:])
86                 else:
87                     items[-1] += " " + lines[i].strip()
88                 i += 1
89             out.append("<ul>%s</ul>" % "".join("<li>%s</li>" % inline(x) for x in items))
90         elif not l.strip():
91             i += 1
92         else:
93             j = i
94             while j < len(lines) and lines[j].strip() and not re.match(r"(#|\||- |```|    )", lines[j]):
95                 j += 1
96             out.append("<p>%s</p>" % inline(" ".join(lines[i:j])))
97             i = j
98     return "\n".join(out)

101 def table(head, rows):
102     h = "".join("<th>%s</th>" % esc(x) for x in head)
103     b = "".join("<tr>%s</tr>" % "".join("<td>%s</td>" % c for c in r) for r in rows)
104     return '<div class="tw"><table><tr>%s</tr>%s</table></div>' % (h, b)

111 def prims_section():
112     src = read("prims.lisp")
113     if not src:
114         return ""
115     defs = {m.group(1): (m.group(2), m.group(3)) for m in re.finditer(
116         r"^\(defun\s+(\S+)\s+\(([^)]*)\)\s+(.*)\)\s*$", src, re.M)}
117     reg = re.search(r"\(defprims(.*?)\)\s*$", src, re.S)
118     pairs = re.findall(r"\((\S+)\s+(\S+)\)", reg.group(1)) if reg else []
119     rows = []
120     for name, word in pairs:
121         params, body = defs.get(word, ("", ""))
122         nargs = len(set(re.findall(r"\(car(?: \(cdr)* a", body)))
123         rows.append([code(name), str(nargs), code(word), code(body)])
124     fs = read("lisp.fs") or ""
125     for m in re.finditer(r"' (\S+) defprim (\S+)", fs):
126         rows.append([code(m.group(2)), "0", code(m.group(1)),
127                      "defined in Forth in lisp.fs"])
128     return ("<p>Generated from <code>prims.lisp</code> (the <code>defprims</code> table and "
129             "each <code>defun</code> body) and the <code>defprim</code> lines of "
130             "<code>lisp.fs</code>. Arity = distinct list positions the body reads. "
131             "Primitives check argument types but not arity.</p>" + table(["Lisp name", "Arity", "Forth word", "Semantics (host-dialect body)"], rows))

171 def words_section():
172     fs = read("lisp.fs")
173     if not fs:
174         return ""
175     rows, group = [], ""
176     for l in fs.splitlines():
177         g = re.match(r"\\ --- (.*) ---", l)
178         if g:
179             group = g.group(1)
180         m = re.match(r": (\S+)\s*(\([^)]*\))?\s*(.*)", l)
181         if m:
182             note = re.search(r"\\\s*(.*)$", l)
183             rows.append([esc(group), code(m.group(1)), code(m.group(2) or ""),
184                          esc(note.group(1) if note else "")])
185     return ("<p>Every colon definition in <code>lisp.fs</code>, in load order, with its stack "
186             "comment.</p>" + table(["Section", "Word", "Stack effect", "Comment"], rows))

189 def header_comment(name, mark):
190     s = read(name)
191     if not s:
192         return ""
193     lines = []
194     for l in s.splitlines():
195         if l.startswith("#!"):
196             continue
197         if l.startswith(mark):
198             lines.append(l[len(mark):].strip())
199         elif lines:
200             break
201     return " ".join(lines)
