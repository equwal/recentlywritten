2 ;;; host.lisp compiles each defun to a Mecrisp Forth word and sends it to the chip, where
3 ;;; lisp.fs registers it with defprim. A primitive takes its argument list (a chip list)
4 ;;; and returns one tagged value. car cdr cons fix num bool pair? sym? proc? fixnum? err
5 ;;; are words from lisp.fs; + - * / mod negate xor = < > are Forth.

7 ;; Generic arithmetic: add2 sub2 mul2 div2 lt2 eq2 (lisp.fs) handle fixnums, rationals and
8 ;; flonums. The accumulators here are tagged values: 1 is the fixnum 0, 3 is the fixnum 1.
9 (defun sum (l acc) (if (= l 0) acc (sum (cdr l) (add2 acc (car l)))))
10 (defun prod (l acc) (if (= l 0) acc (prod (cdr l) (mul2 acc (car l)))))
11 (defun diff (l acc) (if (= l 0) acc (diff (cdr l) (sub2 acc (car l)))))
12 (defun quo (l acc) (if (= l 0) acc (quo (cdr l) (div2 acc (car l)))))
13 (defun prim+ (a) (sum a 1))
14 (defun prim* (a) (prod a 3))
15 (defun prim- (a) (if (= (cdr a) 0) (sub2 1 (car a)) (diff (cdr a) (car a))))
16 (defun prim/ (a) (if (= (cdr a) 0) (div2 3 (car a)) (quo (cdr a) (car a))))
17 (defun divisor (a) (if (= (num (car (cdr a))) 0) (forth "s\" division by zero\" err") (num (car (cdr a)))))
18 (defun prim-quotient (a) (fix (/ (num (car a)) (divisor a))))
19 (defun prim-remainder (a) (fix (mod (num (car a)) (divisor a))))
20 (defun floored (r y) (if (if (= r 0) 0 (< (xor r y) 0)) (+ r y) r))
21 (defun prim-modulo (a) (fix (floored (mod (num (car a)) (divisor a)) (num (car (cdr a))))))
