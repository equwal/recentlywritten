37 (defun compile-tokens (tokens)
38   "Thread TOKENS, resolving control words at build time like a Forth compiler would."
39   (let ((cs '()))
40     (dolist (tk tokens)
41       (let ((s (and (symbolp tk) (symbol-name tk))))
42         (flet ((label () (let ((l (gensym "L"))) (emit l) l))
43                (fwd (op) (let ((l (gensym "F"))) (emit `(dw ,(wlabel op) ,l)) l)))
44           (cond ((integerp tk) (emit `(dw ,(wlabel "lit") ,(logand tk #xFFFF))))
45                 ((equal s "IF") (push (fwd "0branch") cs))
46                 ((equal s "ELSE") (let ((f (fwd "branch"))) (emit (pop cs)) (push f cs)))
47                 ((equal s "THEN") (emit (pop cs)))
48                 ((equal s "BEGIN") (push (label) cs))
49                 ((equal s "WHILE") (let ((f (fwd "0branch"))) (push f (cdr cs))))
50                 ((equal s "REPEAT") (emit `(dw ,(wlabel "branch") ,(pop cs))) (emit (pop cs)))
51                 ((equal s "UNTIL") (emit `(dw ,(wlabel "0branch") ,(pop cs))))
52                 ((equal s "AGAIN") (emit `(dw ,(wlabel "branch") ,(pop cs))))
53                 ((equal s "RECURSE") (emit `(dw ,(wlabel *current*))))
54                 ((and (consp tk) (eq (car tk) 'quote)) (emit `(dw ,(wlabel "lit") ,(wlabel (name-of (cadr tk))))))
55                 ((and (consp tk) (symbolp (car tk)) (equal (symbol-name (car tk)) "ADDR")) (emit `(dw ,(wlabel "lit") ,(cadr tk))))
56                 ((and (consp tk) (member (car tk) '("s\"" ".\"") :test #'equal))
57                  (emit `(dw ,(wlabel "(s\")")) `(db ,(length (cadr tk))) `(db ,(cadr tk)))
58                  (when (equal (car tk) ".\"") (emit `(dw ,(wlabel "type")))))
59                 (t (emit `(dw ,(wlabel (name-of tk)))))))))
60     (when cs (error "unbalanced control flow in ~a" *current*))))
