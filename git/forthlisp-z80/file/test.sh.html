1 #!/bin/sh
2 # test.sh -- run the three test layers: assembler, kernel words, Lisp conformance.
3 # Each kernel-level layer gets its own fresh emulator. PORT sets the first TCP port.
4 SRC=$(cd "$(dirname "$0")" && pwd)
5 SBCL=${SBCL:-sbcl}
6 PORT=${PORT:-4470}
7 rc=0
8 nice $SBCL --script "$SRC/test-asm.lisp" || rc=1
9 nice $SBCL --script "$SRC/kernel.lisp" "$SRC/kernel.mos"
10 for t in test-forth test; do
11   P=$(PORT=$PORT "$SRC/emu.sh" "$SRC/kernel.mos")
12   if [ $t = test ]; then arg=$SRC/r7rs-tests-z80.scm; else arg=; fi
13   PORT=$PORT $SBCL --script "$SRC/host.lisp" $t $arg || rc=1
14   kill $P 2>/dev/null
15   PORT=$((PORT + 1))
16 done
17 exit $rc
