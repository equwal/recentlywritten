1 ;;; asm.lisp -- layer 2: a two-pass Z80 assembler hosted in SBCL.
2 ;;; Source is a list of forms: a bare symbol is a label, a list is an
3 ;;; instruction or directive, e.g. (ld a (@ hl)), (jp nz loop), (dw docol).
4 ;;; (@ x) marks memory indirection: (@ hl), (@ ix 3) = (ix+3), (@ var) = (var).
5 ;;; Immediates are numbers, label symbols, or (+ ...) / (- ...) / (lo x) / (hi x).

29 (defun r8 (x)
30   "Register code 0-7 of X, (@ hl) being 6, or NIL."
31   (cond ((and (consp x) (is (car x) "@") (is (cadr x) "HL") (null (cddr x))) 6)

41 (defun value (x)
42   (cond ((integerp x) x)
43         ((characterp x) (char-code x))
44         ((symbolp x) (or (gethash x *labels*)
45                          (if *final* (error "undefined label ~s" x) 0)))
46         ((is (car x) "+") (reduce #'+ (mapcar #'value (cdr x))))
47         ((is (car x) "-") (apply #'- (mapcar #'value (cdr x))))
48         ((is (car x) "LO") (logand (value (cadr x)) 255))

61     (macrolet ((when-match (test &body body) `(when ,test ,@body (return-from insn))))

63         (when-match (and s (= n 0)) (apply #'b (cdr s))))

66         ((equal op "DW") (when-match t (mapc #'w (cdr form))))
67         ((equal op "LD")

72            (when-match (and (ixd x) (imm-p y)) (b #xDD #x36 (d8 (ixd x)) (value y)))
73            (when-match (and (is x "A") (consp y) (is (car y) "@") (is (cadr y) "BC")) (b #x0A))
74            (when-match (and (is x "A") (consp y) (is (cadr y) "DE") (is (car y) "@")) (b #x1A))
75            (when-match (and (is y "A") (consp x) (is (cadr x) "BC") (is (car x) "@")) (b #x02))

79            (when-match (and rx (imm-p y)) (b (+ #x06 (* 8 rx)) (value y)))

81            (when-match (and (is x "SP") (is y "IX")) (b #xDD #xF9))
82            (when-match (and (is x "IX") (imm-p y)) (b #xDD #x21) (w y))

84            (when-match (and (is y "IX") (mem x)) (b #xDD #x22) (w (mem x)))

86            (when-match (and (is y "HL") (mem x)) (b #x22) (w (mem x)))

88            (when-match (and (r16 y) (mem x)) (b #xED (+ #x43 (* 16 (r16 y)))) (w (mem x)))
89            (when-match (and (r16 x) (imm-p y)) (b (+ #x01 (* 16 (r16 x)))) (w y))))

91          (let ((base (if (equal op "PUSH") #xC5 #xC1)))
92            (when-match (is x "IX") (b #xDD (+ base #x20)))
93            (when-match t (b (+ base (* 16 (r16 x '("BC" "DE" "HL" "AF"))))))))
94         ((member op +alu+ :test #'equal)

97            (when-match (and (= n 2) (is x "HL") (r16 y))
98              (case k (0 (b (+ #x09 (* 16 (r16 y))))) (1 (b #xED (+ #x4A (* 16 (r16 y)))))
99                    (3 (b #xED (+ #x42 (* 16 (r16 y))))) (t (error "bad ~s" form))))

139 (defun assemble (forms)
140   "Assemble FORMS at address 0. Return the byte vector and the label table."
141   (let ((*labels* (make-hash-table)))
142     (dolist (*final* '(nil t))
143       (let ((*pc* 0) (*out* (make-array 0 :element-type '(unsigned-byte 8) :adjustable t :fill-pointer 0)))
144         (dolist (f forms)
145           (cond ((symbolp f)
146                  (when (and (not *final*) (gethash f *labels*)) (error "duplicate label ~s" f))
147                  (setf (gethash f *labels*) *pc*))
148                 ((is (car f) "EQU") (setf (gethash (cadr f) *labels*) (value (caddr f))))
149                 (t (insn f))))
150         (when *final* (return-from assemble (values (coerce *out* '(simple-array (unsigned-byte 8) (*))) *labels*)))))))

153 (defun opnd->text (x)
154   (cond ((and (consp x) (is (car x) "@"))
155          (if (and (is (cadr x) "IX") (cddr x)) (format nil "(ix+~d)" (caddr x)) (format nil "(~a)" (opnd->text (cadr x)))))
156         ((consp x) (format nil "~{~a~^~a~}" (cdr (loop for a in (cdr x) collect (car x) collect (opnd->text a)))))
157         ((stringp x) (format nil "'~a'" x))
158         ((is x "AF") "af")
159         ((symbolp x) (string-downcase (nm x)))
160         (t (princ-to-string x))))
161 (defun insn->text (form)
162   (let ((op (string-downcase (nm (car form)))))
163     (cond ((and (equal op "ex") (is (cadr form) "AF")) "ex af,af'")
164           ((member op '("add" "adc" "sbc") :test #'equal)
165            (format nil "~a ~{~a~^,~}" op (mapcar #'opnd->text (if (= (length form) 2) (list 'a (cadr form)) (cdr form)))))
166           ((equal op "dw") (format nil "defw ~{~a~^,~}" (mapcar #'opnd->text (cdr form))))
167           ((equal op "db") (format nil "defb ~{~a~^,~}" (mapcar #'opnd->text (cdr form))))
