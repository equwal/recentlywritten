15 (defun connect ()
16   (let ((port (parse-integer (or (sb-ext:posix-getenv "PORT") "4455"))))
17     (loop repeat 100
18           do (handler-case
19                  (let ((s (make-instance 'sb-bsd-sockets:inet-socket :type :stream :protocol :tcp)))
20                    (sb-bsd-sockets:socket-connect s #(127 0 0 1) port)
21                    (setf *io* (sb-bsd-sockets:socket-make-stream s :input t :output t :element-type 'character
22                                                                    :external-format :latin-1 :buffering :full))
23                    (return-from connect))
24                (error () (sleep 0.1))))
25     (error "no emulator on port ~d" port)))

27 (defun read-until (done-p)
28   "Read characters until (DONE-P text) is true; return the text."
29   (let ((buf (make-array 0 :element-type 'character :adjustable t :fill-pointer 0)))
30     (loop for c = (read-char *io*)
31           do (vector-push-extend c buf) (when *log* (write-char c) (finish-output))

47 (defun forth (line)
48   "Send one Forth line; return its output and whether it ended in ok."
49   (send line)
50   (let* ((out (read-until (lambda (b) (and (ends-with b (coerce '(#\Return #\Newline) 'string))
51                                           (let ((l (string-right-trim '(#\Return #\Newline) b)))
52                                             (or (ends-with l " ok") (ends-with l " ?") (search " error " l)))))))
53          (body (string-right-trim '(#\Return #\Newline) out)))
54     (values (string-trim " " (subseq body (min (length line) (length body))
55                                      (if (ends-with body " ok") (- (length body) 3) (length body))))
56             (ends-with body " ok"))))

70 (defun load-target ()
71   (read-until (lambda (b) (ends-with b (format nil "Forth~c~c" #\Return #\Newline))))
72   (dolist (l (file-lines "lisp.fs"))
73     (unless (or (string= (string-trim " " l) "") (eql 0 (search "\\ " l)))
74       (multiple-value-bind (out ok) (forth l)

106 (defun test-forth ()
107   (read-until (lambda (b) (ends-with b (format nil "Forth~c~c" #\Return #\Newline))))
108   (let ((pass 0) (fail 0))
109     (loop for (in want) in *forth-tests*
110           for got = (forth in)
111           do (if (string= got want) (incf pass)
112                  (progn (incf fail) (format t "FAIL ~a~%  want ~s~%  got  ~s~%" in want got))))
113     (format t "forth kernel: ~d passed, ~d failed, ~d total~%" pass fail (+ pass fail))
114     fail))

116 (defun conformance (file)
117   (let ((pass 0) (fail 0))
118     (dolist (l (with-open-file (s file) (loop for l = (read-line s nil) while l collect l)))
119       (unless (or (string= (string-trim " " l) "") (char= (char l 0) #\;))
120         (let* ((p (search "==>" l))
121                (expr (string-trim " " (if p (subseq l 0 p) l)))
122                (got (string-trim " " (lisp expr))))
123           (cond ((search "error:" got) (incf fail) (format t "FAIL ~a~%  got ~a~%" expr got))
124                 ((null p) (incf pass))
125                 ((string= got (string-trim " " (subseq l (+ p 3))))  (incf pass))
126                 (t (incf fail) (format t "FAIL ~a~%  want ~a~%  got  ~a~%" expr (string-trim " " (subseq l (+ p 3))) got))))))
127     (format t "conformance: ~d passed, ~d failed, ~d total~%" pass fail (+ pass fail))
128     fail))

130 (defun main (args)
131   (let ((cmd (first args)))
132     (connect)
133     (cond ((equal cmd "load") (load-target))
134           ((equal cmd "test-forth") (sb-ext:exit :code (if (zerop (test-forth)) 0 1)))
135           ((equal cmd "test")
136            (load-target)

138           ((equal cmd "eval") (load-target) (dolist (e (rest args)) (format t "~a~%" (lisp e))))
139           (t (format t "usage: host.lisp load | test-forth | test [FILE] | eval EXPR...~%")))))
