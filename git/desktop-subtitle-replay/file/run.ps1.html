1 # Runs livecap.py inside the venv. All arguments are passed straight through.
2 #   .\run.ps1                       -> Finnish captions, desktop audio
3 #   .\run.ps1 --selftest 12         -> 12s recording + speed check
4 #   .\run.ps1 --model medium --translate
5 $ErrorActionPreference = "Stop"
6 Set-Location $PSScriptRoot
7 if (-not (Test-Path ".\.venv\Scripts\python.exe")) { throw "Run .\setup.ps1 first." }
8 $env:KMP_DUPLICATE_LIB_OK = "TRUE"
9 & ".\.venv\Scripts\python.exe" ".\livecap.py" @args
