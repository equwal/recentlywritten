pashare (556 bytes)

1 #!/bin/sh
2 # Share the default output's monitor as raw 48k s16le stereo over TCP. No auth.
3 # Usage: pashare start [port]|stop   (port defaults to 8000; needs pactl >= 15)
4 case "$1" in
5   start)
6     pactl unload-module module-simple-protocol-tcp 2>/dev/null
7     pactl load-module module-simple-protocol-tcp rate=48000 format=s16le channels=2 source="$(pactl get-default-sink).monitor" record=true port="${2:-8000}"
8     ;;
9   stop)
10     pactl unload-module module-simple-protocol-tcp
11     ;;
12   *)
13     echo "Usage: $0 start [port]|stop" >&2
14     exit 1
15     ;;
16 esac
