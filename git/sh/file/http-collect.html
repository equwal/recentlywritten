66 def suffixify(basename, suff):
67     m = match(r'(.+)\.([^.]+)$', basename)
68     if not m:
69         return f'{basename} ({suff:03d})'
70 
71     return f'{m.group(1)} ({suff:03d}).{m.group(2)}'

74 def get_final_name(original_name):
75     candidate = base_name = sanitize_fs(original_name)
76     suff = 1
77     while lexists(candidate):
78         candidate = suffixify(base_name, suff)
79         suff += 1
80     return candidate

83 def determine_host_and_port(args):

96             rt = ''
97         for line in rt.splitlines():
98             m = match(r'default.* src ([0-9.]+)', line)
99             if m:
100                 host = m.group(1)
101 
102         if host is None:
103             print('Could not find my IPv4, using 0.0.0.0', file=stderr)
104             host = '0.0.0.0'
105 
106     success = False
107     for _ in range(5):
108         if port is None:
109             try_port = randint(1025, 65535)
110         else:
111             try_port = port
112 
113         try:
114             with socket(AF_INET, SOCK_STREAM) as s:
115                 s.bind((host, try_port))
116                 s.listen()
117         except Exception as e:
118             print(f'Failed to use {host}:{try_port}: {repr(e)}')
119             sleep(1)
120         else:
121             port = try_port
122             success = True
123             break
124 
125     if not success:
126         raise Exception('Could not find free port')
