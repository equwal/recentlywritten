26 def test(name, script, args, expect_key=None, expect_value=None):
27     code, stdout, stderr = run_script(script, *args)
28     passed = code == 0
29     detail = ""
30     if passed and expect_key:
31         try:
32             json_str = stdout
33             if "Polling for completion..." in stdout:
34                 json_str = stdout.split("Polling for completion...")[1].strip()
35             data = json.loads(json_str)
36             actual = data.get(expect_key)
37             if expect_value is not None and actual != expect_value:
38                 passed = False
39                 detail = f" (expected {expect_key}={expect_value!r}, got {actual!r})"
40             elif actual is None:
41                 passed = False
42                 detail = f" (key '{expect_key}' missing from response)"
43         except json.JSONDecodeError:
44             passed = False
45             detail = " (response was not valid JSON)"
46     status = PASS if passed else FAIL
47     print(f"{status} {name}{detail}")
48     if not passed:
49         if stderr:
50             print(f"       stderr: {stderr.strip()}")
51         if stdout:
52             print(f"       stdout: {stdout.strip()[:300]}")
53     results.append((name, passed))
54     return passed, stdout
