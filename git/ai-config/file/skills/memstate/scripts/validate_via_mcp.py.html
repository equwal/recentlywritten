19 def mcp(tool, args_dict):
20     """Call a Memstate MCP tool via manus-mcp-cli and return parsed JSON."""
21     cmd = [
22         "manus-mcp-cli", "tool", "call", tool,
23         "--server", "memstate",
24         "--input", json.dumps(args_dict)
25     ]
26     r = subprocess.run(cmd, capture_output=True, text=True)
27     # Extract JSON from the output (skip the "Tool execution result:" header line)
28     output = r.stdout
29     # Find the first '{' or '[' in output
30     for i, ch in enumerate(output):
31         if ch in ('{', '['):
32             try:
33                 return json.loads(output[i:])
34             except json.JSONDecodeError:
35                 pass
36     return None

39 def check(name, data, expect_key, expect_value=None):
40     """Validate a response and record the result."""
41     if data is None:
42         print(f"{FAIL} {name} (no JSON response)")
43         results.append((name, False))
44         return False
45 
46     # Navigate into data.data if present (MCP wraps responses)
47     inner = data.get("data", data)
48     actual = inner.get(expect_key)
49 
50     if actual is None:
51         print(f"{FAIL} {name} (key '{expect_key}' missing; keys: {list(inner.keys())})")
52         results.append((name, False))
53         return False
54 
55     if expect_value is not None and actual != expect_value:
56         print(f"{FAIL} {name} (expected {expect_key}={expect_value!r}, got {actual!r})")
57         results.append((name, False))
58         return False
59 
60     print(f"{PASS} {name}")
61     results.append((name, True))
62     return True
