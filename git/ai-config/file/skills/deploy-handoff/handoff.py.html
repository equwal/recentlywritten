1 """Open a web page and wait while the user does a step that only a human may do.
2 
3 A tool calls this script when it reaches a step that an AI agent must not do:
4 the final deploy, publish, or billing click, a sign-in, an OAuth consent, or
5 the creation of a GitHub pull request. The script opens the page in the Brave
6 profile that brave_profile names in the user config file, or else in the default
7 browser. It shows a small dialog with the steps. It never clicks for the user.
8 It prints the answer of the user as one JSON object on stdout.
9 
10 The command "terminal" shows the same dialog for a step at a prompt in a terminal,
11 for example the install question of a tool or a sign-in of a command line program.
12 It opens no page. The caller starts the program in a terminal that the user sees,
13 and stops at the prompt. The script never types the answer for the user.
14 
15 The command "browser --fresh" starts a separate browser with a fresh profile for
16 the browser-driver agent of this plugin. Use it only if the user asks for it.
17 Playwright MCP controls that browser through a local port.
18 
19 Exit codes: 0 done, started, or running, 1 error, 2 bad arguments, 3 not done,
20 4 timeout.
21 """

126 def check_url(url: str) -> None:
127     """Raise HandoffError if url is not an https URL with a plain host name.
128 
129     There is no list of allowed hosts: the user asks Claude to use only sites he trusts.
130     The dialog shows the host, so the user can compare it with the address bar.
131     """
132     if "\\" in url or not URL_CHARS_RE.fullmatch(url):
133         raise HandoffError(
134             f"The URL must be printable ASCII with no spaces or backslashes: {url!r}"
135         )
136     try:
137         parts = urlsplit(url)
138     except ValueError as exc:
139         raise HandoffError(f"The URL is not valid: {url}") from exc
140     host = parts.hostname or ""
141     # The netloc must be the host only: no user name, no password, and no port.
142     if parts.scheme != "https" or parts.netloc.lower() != host or not HOST_RE.fullmatch(host):
143         raise HandoffError(f"The URL must use https and a plain host name: {url}")

146 def check_text(title: str, steps: Iterable[str]) -> None:
147     """Raise HandoffError if the title or a step is not short Simplified Technical English."""
148     named = [("The title", title), *((f"Step {n}", step) for n, step in enumerate(steps, 1))]
149     for name, text in named:
150         words = len(text.split())
151         if not 0 < words <= MAX_STEP_WORDS or ";" in text:
152             raise HandoffError(
153                 f"{name} is not short Simplified Technical English ({words} words). "
154                 f"Give one action in 1 to {MAX_STEP_WORDS} words, with no semicolons."
155             )

199 def find_profile(local_state: Path, name: str) -> str:
200     """Return the folder of the Brave profile with the display name name. Ignore the case."""
201     try:
202         cache = json.loads(local_state.read_text(encoding="utf-8"))["profile"]["info_cache"]
203         folders = [
204             str(folder)
205             for folder, info in cache.items()
206             if str(info.get("name", "")).casefold() == name.casefold()
207         ]
208     except (OSError, ValueError, KeyError, TypeError, AttributeError) as exc:
209         raise HandoffError(f"Cannot read the Brave profiles in {local_state}: {exc!r}") from exc
210     if len(folders) != 1:
211         raise HandoffError(
212             f"Found {len(folders)} Brave profiles with the name {name!r} in {local_state}. "
213             f"Set brave_profile in {USER_CONFIG} to the name of one profile."
214         )
215     return folders[0]

292 def debug_version(port: int) -> dict[str, object] | None:
293     """Return the version data of the browser that listens on port, or None if none answers."""
294     try:
295         with LOCAL_OPENER.open(f"http://127.0.0.1:{port}/json/version", timeout=2) as response:
296             data = json.load(response)
297     except (OSError, ValueError, http.client.HTTPException):
298         return None
299     return data if isinstance(data, dict) else None

302 def start_browser(command: list[str]) -> None:
303     """Start the browser in a new process group, so that it runs after this script ends."""
304     devnull = subprocess.DEVNULL
305     try:
306         if sys.platform == "win32":
307             flags = subprocess.DETACHED_PROCESS | subprocess.CREATE_NEW_PROCESS_GROUP
308             subprocess.Popen(
309                 command, stdin=devnull, stdout=devnull, stderr=devnull, creationflags=flags
310             )
311         else:
312             subprocess.Popen(
313                 command, stdin=devnull, stdout=devnull, stderr=devnull, start_new_session=True
314             )
315     except OSError as exc:
316         raise HandoffError(f"Cannot start {command[0]}: {exc}") from exc

319 def wait_for_browser(port: int, seconds: float) -> dict[str, object]:
320     """Wait until a browser answers on port. Return its version data."""
321     deadline = time.monotonic() + seconds
322     while (version := debug_version(port)) is None:
323         if time.monotonic() > deadline:
324             raise HandoffError(
325                 f"The browser did not open port {port} in {seconds:g} seconds. If the browser "
326                 "of the agent runs without this port, close it. Then run this command again."
327             )
328         time.sleep(0.5)
329     return version

332 def run_browser(exe: Path | None) -> dict[str, str | None]:
333     """Start the browser of the browser-driver agent, or find it running."""
334     status = "running"
335     version = debug_version(DEBUG_PORT)
336     if version is None:
337         status = "started"
338         BROWSER_PROFILE.mkdir(parents=True, exist_ok=True)
339         exe = exe or find_browser(browser_candidates())
340         start_browser(browser_command(exe, BROWSER_PROFILE, DEBUG_PORT))
341         version = wait_for_browser(DEBUG_PORT, BROWSER_START_SECONDS)
342     endpoint = f"http://127.0.0.1:{DEBUG_PORT}"
343     return {"status": status, "browser": str(version.get("Browser")), "endpoint": endpoint}

354 def fork_owner(remote_url: str, repo: str) -> str | None:
355     """Return the owner of a GitHub remote that is not repo, for example a fork of repo, or None."""
356     match = GITHUB_REMOTE_RE.fullmatch(remote_url.strip())
357     if match is None or match["repo"].lower() == repo.lower():
358         return None
359     return match["repo"].split("/")[0]

362 def compare_url(repo: str, base: str | None, head: str, title: str, body: str) -> str:
363     """Return the GitHub page that shows a new pull request form with title and body filled in.
364 
365     head is BRANCH, or OWNER:BRANCH for a branch in a fork of repo.
366     """
367     # Without a base, GitHub compares head with the default branch. A branch name has no ":".
368     head_ref = quote(head, safe="/:")
369     refs = head_ref if base is None else f"{quote(base)}...{head_ref}"
370     query = urlencode({"expand": "1", "title": title, "body": body}, quote_via=quote)
371     url = f"https://github.com/{repo}/compare/{refs}?{query}"
372     if len(url) > MAX_PR_URL_LENGTH:
373         raise HandoffError(
374             f"The pull request URL has {len(url)} characters. The limit is {MAX_PR_URL_LENGTH}. "
375             "Make the body shorter."
376         )
377     return url

380 def git(repo_dir: Path, *args: str) -> str:
381     """Run git in repo_dir and return its output. Raise HandoffError if git fails."""
382     command = ["git", "-C", str(repo_dir), *args]
383     try:
384         result = subprocess.run(
385             command, capture_output=True, encoding="utf-8", errors="replace", timeout=60
386         )
387     except (OSError, subprocess.TimeoutExpired) as exc:
388         raise HandoffError(f"{' '.join(command)} failed: {exc}") from exc
389     if result.returncode != 0:
390         raise HandoffError(f"{' '.join(command)} failed: {result.stderr.strip()}")
391     return result.stdout.strip()

402 def check_pushed(repo_dir: Path, remote: str, branch: str) -> None:
403     """Raise HandoffError if remote does not have the commit of the local branch."""
404     ref = f"refs/heads/{branch}"
405     local = git(repo_dir, "rev-parse", "--verify", ref)
406     lines = git(repo_dir, "ls-remote", remote, ref).splitlines()
407     remote_commits = [line.split()[0] for line in lines if line.split()[1:] == [ref]]
408     if remote_commits != [local]:
409         raise HandoffError(
410             f"{remote} does not have the local commit of {branch}. Push {branch} first."
411         )

414 def find_open_pr(repo: str, head: str) -> str | None:
415     """Return the URL of the open pull request from head, or None if gh cannot find one.
416 
417     head is BRANCH, or OWNER:BRANCH for a branch in a fork of repo.
418     """
419     owner, _, branch = head.rpartition(":")
420     command = ["gh", "pr", "list", "--repo", repo, "--head", branch, "--state", "open"]
421     if owner:
422         # gh filters by the branch name only. The jq filter also checks the owner of the branch.
423         login = f'select(.headRepositoryOwner.login | ascii_downcase == "{owner.lower()}")'
424         command += ["--json", "url,headRepositoryOwner", "--jq", f"[.[] | {login}][0].url // empty"]
425     else:
426         command += ["--json", "url", "--jq", ".[0].url // empty"]
427     try:
428         result = subprocess.run(command, capture_output=True, encoding="utf-8", timeout=60)
429     except (OSError, subprocess.TimeoutExpired):
430         return None
431     url = result.stdout.strip()
432     return url if result.returncode == 0 and url else None

435 def pr_request(args: argparse.Namespace) -> Request:
436     """Make the request that opens the GitHub form for a new pull request."""
437     repo_dir: Path = args.repo_dir
438     remote_url = git(repo_dir, "remote", "get-url", args.remote)
439     repo: str = args.repo or github_repo(remote_url)
440     if not REPO_RE.fullmatch(repo):
441         raise HandoffError(f"{repo!r} is not a repository name of the form OWNER/NAME.")
442     head: str = args.head or current_branch(repo_dir)
443     check_pushed(repo_dir, args.remote, head)
444     # A pull request to another repository, for example from a fork to its upstream
445     # repository, names the branch as OWNER:BRANCH.
446     owner = fork_owner(remote_url, repo)
447     head_ref = head if owner is None else f"{owner}:{head}"
448     try:
449         body: str = args.body_file.read_text(encoding="utf-8") if args.body_file else args.body
450     except OSError as exc:
451         raise HandoffError(f"Cannot read {args.body_file}: {exc}") from exc
452     url = compare_url(repo, args.base, head_ref, args.title, body)
453     return Request(f"Create pull request: {args.title}", url, PR_STEPS, pr=(repo, head_ref))

456 class Dialog:
457     """A small window on top of the browser. It shows the steps and records the answer."""
458 
459     def __init__(self, request: Request, timeout_minutes: float, reopen: Callable[[], None]):
460         self.status = "not_done"
461         self.note_text = ""
462         if sys.platform == "win32":
463             # Draw sharp text on high-DPI screens, as IDLE does. The call fails if the
464             # process has already set the DPI awareness. That is not a problem.
465             with contextlib.suppress(OSError):
466                 ctypes.OleDLL("shcore").SetProcessDpiAwareness(1)
467         self.root = tk.Tk()
468         scale = self.root.winfo_fpixels("1i") / 96
469         wrap = round(420 * scale)
470         self.root.title(f"Claude handoff: {request.title}")
471         self.root.attributes("-topmost", True)
472         self.root.resizable(False, False)
473         # Put the window in the bottom-left corner, above the taskbar. Deploy pages
474         # usually put the final button on the right side.
475         self.root.geometry(f"+{round(24 * scale)}-{round(80 * scale)}")
476         self.root.protocol("WM_DELETE_WINDOW", lambda: self.finish("not_done"))
477 
478         steps = "\n".join(f"{number}. {step}" for number, step in enumerate(request.steps, 1))
479         # Tk deletes a font when its Python object goes away, so the dialog keeps it.
480         self.bold = tkfont.nametofont("TkDefaultFont").copy()
481         self.bold.configure(weight="bold")
482         frame = ttk.Frame(self.root, padding=round(12 * scale))
483         frame.grid()
484         # The user must see first what to do, and then where to do it.
485         ttk.Label(frame, text=request.title, font=self.bold, wraplength=wrap).grid(sticky="w")
486         heading = HEADING if request.url is not None else TERMINAL_HEADING
487         ttk.Label(frame, text=heading, font=self.bold).grid(sticky="w", pady=(8, 0))
488         ttk.Label(frame, text=steps, wraplength=wrap, justify="left").grid(sticky="w")
489         where = ttk.Label(frame, text=request.where(), font=self.bold, wraplength=wrap)
490         where.grid(sticky="w", pady=(8, 0))
491         warning = TERMINAL_WARNING
492         if request.url is not None:
493             # A step on a page: show the address, and open the page again on a click.
494             parts = urlsplit(request.url)
495             shown = f"https://{parts.netloc}{parts.path}"
496             if len(shown) > 80:
497                 shown = shown[:77] + "..."
498             link = ttk.Label(frame, text=shown, foreground="blue", cursor="hand2", wraplength=wrap)
499             link.grid(sticky="w")
500             link.bind("<Button-1>", lambda _event: reopen())
501             warning = WARNING
502         for text, color in ((FINISH, ""), (warning, "#b00020")):
503             label = ttk.Label(frame, text=text, foreground=color, wraplength=wrap, justify="left")
504             label.grid(sticky="w", pady=(8, 0))
505         ttk.Label(frame, text="Note for Claude (optional):").grid(sticky="w", pady=(8, 0))
506         self.note = ttk.Entry(frame)
507         self.note.grid(sticky="ew")
508         buttons = ttk.Frame(frame)
509         buttons.grid(sticky="e", pady=(12, 0))
510         self.done_button = ttk.Button(buttons, text="Done", command=lambda: self.finish("done"))
511         self.done_button.grid(row=0, column=0, padx=(0, 8))
512         self.not_done_button = ttk.Button(
513             buttons, text="Not done", command=lambda: self.finish("not_done")
514         )
515         self.not_done_button.grid(row=0, column=1)
516         self.timer = self.root.after(round(timeout_minutes * 60_000), self.finish, "timeout")
517 
518     def finish(self, status: str) -> None:
519         """Record the answer and close the window."""
520         self.root.after_cancel(self.timer)
521         self.status = status
522         self.note_text = self.note.get().strip()
523         self.root.destroy()
524 
525     def run(self) -> tuple[str, str]:
526         """Show the window until the user answers or the time ends."""
527         self.root.mainloop()
528         return self.status, self.note_text

531 class MenuDialog:
532     """The steps in a bemenu list, for a Wayland desktop that has no Tk."""
533 
534     DONE = "Done"
535     NOT_DONE = "Not done"
536 
537     def __init__(self, request: Request, timeout_minutes: float, reopen: Callable[[], None]):
538         if shutil.which("bemenu") is None:
539             raise HandoffError("Cannot show the steps: install Python Tk or bemenu.")
540         # Without a session, bemenu exits as if the user pressed Escape.
541         if not os.environ.get("WAYLAND_DISPLAY"):
542             raise HandoffError("Cannot show the steps: WAYLAND_DISPLAY is not set.")
543         self.title = request.title
544         self.deadline = time.monotonic() + timeout_minutes * 60
545         steps = [f"{number}. {step}" for number, step in enumerate(request.steps, 1)]
546         self.items = [
547             *steps,
548             request.where(),
549             "If you cannot finish, type the reason and press Enter.",
550             self.DONE,
551             self.NOT_DONE,
552         ]
553 
554     def run(self) -> tuple[str, str]:
555         """Show the list until the user selects an answer or the time ends."""
556         while (remaining := self.deadline - time.monotonic()) > 0:
557             command = ["bemenu", "--list", str(len(self.items)), "--prompt", self.title]
558             try:
559                 answer = subprocess.run(
560                     command,
561                     input="\n".join(self.items),
562                     capture_output=True,
563                     text=True,
564                     timeout=remaining,
565                 )
566             except subprocess.TimeoutExpired:
567                 break
568             choice = answer.stdout.strip()
569             # Escape makes bemenu exit with code 1.
570             if answer.returncode != 0 or choice == self.NOT_DONE:
571                 return "not_done", ""
572             if choice == self.DONE:
573                 return "done", ""
574             if choice not in self.items:
575                 return "not_done", choice
576             # The user selected a step or the site. Show the list again.
577         return "timeout", ""

633 def report(result: dict[str, str | None]) -> int:
634     """Print result as JSON. Return the exit code for its status."""
635     print(json.dumps(result))
636     return EXIT_CODES[str(result["status"])]

639 def main(argv: list[str] | None = None) -> int:
640     """Do the handoff. Return the exit code."""
641     args = parse_args(argv)
642     try:
643         if args.command == "browser":
644             if not args.fresh:
645                 raise HandoffError(
646                     "This command starts a browser with a fresh profile and no sign-in. Start it "
647                     "only if the user asked for a fresh browser. Then give --fresh. Otherwise "
648                     "use the Brave profile of the user."
649                 )
650             return report(run_browser(args.exe))
651         if args.command == "terminal":
652             # A step at a prompt in a terminal has no page: no host to check, nothing to open.
653             check_text(args.title, args.step)
654             check_place(args.where)
655             request = Request(args.title, None, tuple(args.step), place=args.where)
656         else:
657             if args.command == "pr":
658                 request = pr_request(args)
659             else:
660                 check_text(args.title, args.step)
661                 request = Request(args.title, args.url, tuple(args.step))
662             check_url(str(request.url))
663             # With --no-open, the caller drove a browser to the page already.
664             if args.command == "pr" or not args.no_open:
665                 open_page(str(request.url))
666         url = request.url
667         show = Dialog if HAVE_TK else MenuDialog
668         dialog = show(request, args.timeout, lambda: open_page(url) if url else None)
669     except HandoffError as exc:
670         return report({"status": "error", "error": str(exc)})
671     status, note = dialog.run()
672     result: dict[str, str | None] = {"status": status, "note": note}
673     if request.pr is not None and status == "done":
674         result["pr_url"] = find_open_pr(*request.pr)
675     return report(result)
