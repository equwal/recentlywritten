40 test('the cue for a moment is the one a slow search finds, for any cues and any moment', () => {
41   let seed = 11;
42   const random = () => (seed = (seed * 1103515245 + 12345) >>> 0) / 2 ** 32;
43   for (let round = 0; round < 300; round++) {
44     const cues = [];

46     for (let i = 0; i < Math.floor(random() * 40); i++) {
47       const start = t, end = start + 0.2 + random() * 5;
