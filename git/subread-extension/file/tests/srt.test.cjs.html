26 test('the cue for a moment is the one a slow search finds, for any cues and any moment', () => {
27   let seed = 11;
28   const random = () => (seed = (seed * 1103515245 + 12345) >>> 0) / 2 ** 32;
29   for (let round = 0; round < 300; round++) {
30     const cues = [];
31     let t = random() * 5;
32     for (let i = 0; i < Math.floor(random() * 40); i++) {
33       const start = t, end = start + 0.2 + random() * 5;
34       cues.push({ start, end, text: String(i) });
35       t = end + (random() < 0.5 ? 0 : random() * 3);        // back to back, or a gap
36     }
37     for (let k = 0; k < 60; k++) {
38       const at = random() * (t + 5) - 1;
39       const slow = cues.filter((c) => c.start <= at && at < c.end).pop() ?? null;
40       assert.equal(cueAt(cues, at), slow);
41     }
42   }
43   assert.equal(cueAt([], 3), null);
44 });
