9 // The links in the "More apps" section of the popup: the name, the address and the attributes of each.
10 function moreApps() {
11   const section = read('popup.html').match(/<details id="more-apps"[^>]*>([\s\S]*?)<\/details>/);
12   assert.ok(section, 'popup.html has no More apps section');
13   return [...section[1].matchAll(/<a ([^>]*)>([^<]*)<\/a>/g)].map(([, attrs, name]) => ({
14     name,
15     attrs,
16     href: (attrs.match(/href="([^"]*)"/) || [])[1],
17   }));
18 }

20 test('More apps links to the other apps with https, each in a new tab', () => {
21   const apps = moreApps();
22   assert.ok(apps.length > 1);
23   for (const app of apps) {
24     assert.match(app.href, /^https:\/\/[^?#]+$/, app.name);   // no tracking parameters

26     assert.match(app.attrs, /rel="noopener"/, app.name);
27   }
28   assert.equal(new Set(apps.map((app) => app.href)).size, apps.length);
29   assert.equal(apps.at(-1).href, 'https://recentlywritten.com/projects.html');
