15 # Reproducible build: F-Droid builds this library again, on Linux and at another
16 # path, and compares it with the signed APK. The maps remove the paths of the
17 # build machine from the __FILE__ strings of ggml. -ffile-reproducible writes "/"
18 # in them on Windows too. -fno-ident keeps the version line of the compiler out of
19 # the .comment section: the clang of the Windows NDK writes another line than the
20 # clang of the Linux NDK. An empty GIT_EXE stops ggml from recording the git state
21 # of the checkout, which the F-Droid scanner changes.
22 get_filename_component(SUBREP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
23 get_filename_component(SUBREP_TOOLCHAIN_BIN "${CMAKE_C_COMPILER}" DIRECTORY)
24 get_filename_component(SUBREP_TOOLCHAIN_ROOT "${SUBREP_TOOLCHAIN_BIN}" DIRECTORY)
25 add_compile_options(
26     -ffile-reproducible
27     -fno-ident
28     "-ffile-prefix-map=${SUBREP_ROOT}=."
29     "-ffile-prefix-map=${SUBREP_TOOLCHAIN_ROOT}=toolchain")
30 set(GIT_EXE "" CACHE FILEPATH "Not used: the build records no git state" FORCE)

40 # Half-precision and dot-product instructions are most of the speed on ARM. Each CPU since the
