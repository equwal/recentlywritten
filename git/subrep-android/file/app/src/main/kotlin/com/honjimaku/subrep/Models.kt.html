34      * the total (-1 when unknown). Returns null when done, else the reason it failed.

36     fun download(context: Context, model: Model, onProgress: (Long, Long) -> Unit): String? {
37         val target = model.file(context)
38         target.parentFile?.mkdirs()
39         // The whole file goes to a temporary name first: a half file must never pass as a model.
40         val temp = File(target.path + ".part")
41         try {
42             http.newCall(Request.Builder().url(model.url).build()).execute().use { response ->
43                 if (!response.isSuccessful) return "HTTP ${response.code}"
44                 val body = response.body ?: return "empty answer"
45                 val total = body.contentLength()
46                 var done = 0L
47                 var lastReport = 0L
48                 body.byteStream().use { input ->
49                     temp.outputStream().use { output ->
50                         val buffer = ByteArray(64 * 1024)
51                         while (true) {
52                             val n = input.read(buffer)
53                             if (n < 0) break
54                             output.write(buffer, 0, n)
55                             done += n
56                             if (done - lastReport >= 1024 * 1024) {
57                                 lastReport = done
58                                 onProgress(done, total)
59                             }
60                         }
61                     }
62                 }
63                 if (total > 0 && done != total) return "the download stopped at $done of $total bytes"
64                 if (!temp.renameTo(target)) return "cannot write ${target.name}"
65                 onProgress(done, total)
66                 return null
67             }
68         } catch (e: Exception) {
