260     private fun download() {
261         val model = chosenModel()
262         if (model.downloaded(this) || Feed.downloading != null) return
263         Feed.download(model, 0, -1)
264         val app = applicationContext
265         thread(name = "download") {
266             val error = Downloader(OkHttpClient()).download(app, model) { done, total -> Feed.download(model, done, total) }
267             Feed.download(null, 0, -1, error.orEmpty())
268         }

275     private fun startCapture() {
276         save()
277         if (store.cloud) {
278             if (Feed.cloudSecondsLeft == 0L) return toast(R.string.need_hours)
279         } else if (!chosenModel().downloaded(this)) {
280             return toast(R.string.need_model)
281         }
282         val missing = buildList {
283             if (checkSelfPermission(Manifest.permission.RECORD_AUDIO) != PackageManager.PERMISSION_GRANTED) add(Manifest.permission.RECORD_AUDIO)
284             if (Build.VERSION.SDK_INT >= 33 && checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED) {
285                 add(Manifest.permission.POST_NOTIFICATIONS)
286             }
287         }
288         if (missing.isNotEmpty()) return requestPermissions(missing.toTypedArray(), REQUEST_PERMISSIONS)
289         if (!store.screen) return CaptureService.start(this, screen = false)
290         if (Build.VERSION.SDK_INT < 29) return toast(R.string.need_android_10)
291         // Android asks the user for the consent. The answer comes to onActivityResult.
292         val manager = getSystemService(MediaProjectionManager::class.java)
293         startActivityForResult(manager.createScreenCaptureIntent(), REQUEST_PROJECTION)
294     }

321     private fun refreshCloud() {
322         val app = Cloud(http, store)
323         thread(name = "cloud") {
324             val state = runCatching { app.state() }.getOrNull()
325             runOnUiThread {
326                 cloud = state
327                 if (state == null) {
328                     cloudStatus.text = getString(R.string.cloud_offline)
329                     packButtons.removeAllViews()
330                 } else {
331                     Feed.cloudLeft(state.secondsLeft)
332                     showCloud(state)
333                 }
334             }
335         }
336     }

338     private fun showCloud(state: Cloud.State) {
339         if (!state.available) {
340             cloudStatus.text = getString(R.string.cloud_closed)
341             packButtons.removeAllViews()
342             return
343         }
344         cloudStatus.text = getString(R.string.cloud_left, Cloud.hours(state.secondsLeft), state.accountId)
345         if (packButtons.childCount == state.packs.size) return
346         packButtons.removeAllViews()
347         for (pack in state.packs) {
348             packButtons.addView(button(resources.getQuantityString(R.plurals.cloud_buy, pack.hours, pack.hours, pack.price)) { buy(pack) })
349         }
350     }

364     private fun status(): String {
365         if (!Feed.running) return Feed.error
366         val model = if (store.cloud) "Cloud" else "Whisper ${Model.byId(store.model).label}"
367         val engine = when {
368             Feed.engine == Engine.STATE_LOADING -> "loading…"
369             Feed.engine == Engine.STATE_READY && Feed.speed > 0f -> "ready, %.1f s for each second of speech".format(Feed.speed)
370             Feed.engine == Engine.STATE_READY -> "ready"
371             else -> Feed.engine
372         }
373         val late = if (Feed.late > 0) ", ${Feed.late} pieces dropped" else ""
374         val relay = when {
375             !Feed.relayOn -> "off"
376             Feed.relayUp -> "connected"
377             else -> "connecting…"
378         }
379         val overlay = when (Feed.overlayAnswer) {
380             "" -> if (store.overlay && Overlay.installed(this)) "waits for a line" else "off"
381             Overlay.OK -> "shows the line"
382             "panel_hidden" -> "press \"Show the subtitles\" in SubRead Overlay"
383             "no_overlay_permission", "no_notification_access" -> "allow steps 1 and 2 in SubRead Overlay"
384             Overlay.TOO_OLD -> "update SubRead Overlay: this version takes no lines"
385             else -> Feed.overlayAnswer
386         }
387         val bars = Feed.bars()
388         val level = "▮".repeat(bars) + "▯".repeat(10 - bars)
389         val error = if (Feed.error.isEmpty()) "" else "\n${Feed.error}"
390         return "$model: $engine$late\nShare link: $relay\nSubRead Overlay: $overlay\nSound: $level$error"
391     }
