121     fun level(peak: Int) {
122         val before = bars()
123         this.peak = peak
124         // Frames come ten times a second. The screen redraws only when the meter changes, and
125         // not more than twice a second: an e-ink screen cannot draw more.
126         val now = SystemClock.elapsedRealtime()
127         if (bars() == before || now - lastLevelAt < 500) return
128         lastLevelAt = now
129         changed()
130     }

132     fun caption(caption: Caption) {
133         synchronized(lock) {
134             when (caption.type) {
135                 "final" -> {
136                     finals.addLast(caption.text)
137                     while (finals.size > LINES) finals.removeFirst()
138                     partial = ""
139                 }
140                 "partial" -> partial = caption.text
141                 "clear" -> {
142                     finals.clear()
143                     partial = ""
144                 }
145             }
146         }
147         changed()
148     }
