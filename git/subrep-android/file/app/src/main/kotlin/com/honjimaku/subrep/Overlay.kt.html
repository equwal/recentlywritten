46 /**
47  * Sends the captions to the overlay from its own thread, because a provider call waits for
48  * the other app. Partial lines go at most every [PARTIAL_GAP_MS]: the overlay draws each line,
49  * and an e-ink screen cannot draw five times a second.
50  */
51 class OverlayFeed(private val context: Context, private val handler: Handler) {
52 
53     private var pendingPartial: String? = null
54     private var lastAt = 0L
55     private val flush = Runnable {
56         pendingPartial?.let { send(it, partial = true) }
57         pendingPartial = null
58     }
59 
60     fun caption(caption: Caption) {
61         handler.post {
62             when (caption.type) {
63                 "final" -> {
64                     handler.removeCallbacks(flush)
65                     pendingPartial = null
66                     send(caption.text, partial = false)
67                 }
68                 "partial" -> {
69                     pendingPartial = caption.text
70                     handler.removeCallbacks(flush)
71                     handler.postDelayed(flush, (PARTIAL_GAP_MS - (SystemClock.elapsedRealtime() - lastAt)).coerceAtLeast(0))
72                 }
73             }
74         }
75     }
76 
77     fun end() {
78         handler.removeCallbacks(flush)
79         handler.post { Overlay.end(context) }
80     }
81 
82     private fun send(text: String, partial: Boolean) {
83         lastAt = SystemClock.elapsedRealtime()
84         Feed.overlay(Overlay.line(context, text, partial))
85     }
