56     /** The text of [pcm], 16 kHz mono 16-bit little-endian PCM. */
57     fun transcribe(pcm: ByteArray, language: String): Answer {
58         val request = Request.Builder()
59             .url("$base/api/captions/transcribe".toHttpUrl().newBuilder().addQueryParameter("lang", language.ifEmpty { "auto" }).build())
60             .post(pcm.toRequestBody(PCM))
61             .build()
62         val response = try {
63             call(request)
64         } catch (e: IOException) {
65             throw PieceFailed("Cloud: no connection (${e.message}).")
66         }
67         return response.use {
68             val text = it.body!!.string()
69             when (it.code) {
70                 200 -> JSONObject(text).let { json -> Answer(json.getString("text"), json.getLong("seconds_left")) }
71                 402 -> throw OutOfHours()
72                 else -> throw PieceFailed("Cloud: ${detail(text) ?: "the server answered ${it.code}"}.")
73             }
74         }
75     }

77     private fun call(request: Request): Response {
78         val device = readDevice()
79         val response = http.newCall(
80             if (device.isEmpty()) request else request.newBuilder().header("Cookie", "$COOKIE=$device").build(),
81         ).execute()
82         deviceFrom(response.headers("Set-Cookie"))?.let { if (it != device) writeDevice(it) }
83         return response
84     }
