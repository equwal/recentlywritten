83 class PcmTest {
84 
85     private fun bytesOf(samples: List<Int>): ByteArray {
86         val bytes = ByteArray(samples.size * 2)
87         samples.forEachIndexed { i, s ->
88             bytes[2 * i] = (s and 0xFF).toByte()
89             bytes[2 * i + 1] = ((s shr 8) and 0xFF).toByte()
90         }
91         return bytes
92     }
93 
94     /** The peak of a frame is the largest sample size in it, with the sign taken away. */
95     @Test
96     fun thePeakIsTheLargestSampleSize(): Unit = runBlocking {
97         checkAll(Arb.list(Arb.int(-32768..32767), 0..400)) { samples ->
98             val bytes = bytesOf(samples)
99             val expected = samples.maxOfOrNull { if (it < 0) -it else it } ?: 0
100             assertEquals(expected, Pcm.peak(bytes, bytes.size))
101         }
102     }
103 
104     /** Each sample comes out as its value over 32768, in the same order. */
105     @Test
106     fun theFloatsAreTheSamplesOver32768(): Unit = runBlocking {
107         checkAll(Arb.list(Arb.int(-32768..32767), 0..400)) { samples ->
108             val bytes = bytesOf(samples)
109             val out = FloatArray(samples.size)
110             assertEquals(samples.size, Pcm.toFloat(bytes, bytes.size, out))
111             samples.forEachIndexed { i, s -> assertEquals(s / 32768f, out[i], 0f) }
112         }
113     }
