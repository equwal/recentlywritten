156     /** The engine stops asking the cloud once the hours are gone. */
157     @Test
158     fun theEngineStopsWhenTheHoursAreGone() {
159         val calls = CopyOnWriteArrayList<Int>()
160         val states = CopyOnWriteArrayList<String>()
161         val captions = CopyOnWriteArrayList<String>()
162         val recognizer = object : Recognizer {
163             override val detectedLanguage = ""
164             override fun transcribe(samples: FloatArray, language: String): String {
165                 calls += samples.size
166                 if (calls.size > 1) throw Cloud.OutOfHours()
167                 return "一つ目"
168             }
169             override fun close() = Unit
170         }
171         val engine = Engine({ recognizer }, "ja", { captions += it.text }, {}, { states += it })
172         engine.start()
173         val random = Random(7)
174         repeat(3) { feed(engine, speech(1_500, random) + FloatArray(16_000)) }
175         engine.stop()
176         assertEquals(listOf("一つ目"), captions)
177         assertEquals(2, calls.size)
178         assertTrue(states.toString(), states.last().startsWith(Engine.STATE_ERROR + "No cloud hours left"))
179     }

181     private fun speech(ms: Int, random: Random) = FloatArray(16 * ms) { random.nextFloat() * 0.6f - 0.3f }
182 
183     /** Feeds [sound] in frames of 100 ms, and waits until the worker took each piece. */
184     private fun feed(engine: Engine, sound: FloatArray) {
185         val bytes = Pcm.toBytes(sound)
186         var at = 0
187         while (at < bytes.size) {
188             val n = minOf(3_200, bytes.size - at)
189             engine.feed(bytes.copyOfRange(at, at + n), n)
190             at += n
191         }
192         Thread.sleep(300)
193     }
