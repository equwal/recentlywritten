19 (defun provide-greet (available)
20   "Define GREET in each AVAILABLE fork, and in no other fork."
21   (dolist (fork *forks*)
22     (let ((greet (intern "GREET" fork)))
23       (if (member fork available)
24           (setf (symbol-function greet) (constantly fork))
25           (fmakunbound greet)))))

27 (test first-available-fork-wins
28   "For each subset of forks that define GREET, DEFSL takes the first one."
29   (dotimes (mask (expt 2 (length *forks*)))
30     (let ((available (loop for fork in *forks*
31                            for bit from 0
32                            when (logbitp bit mask) collect fork)))
33       (provide-greet available)
34       (fmakunbound 'greet)
35       (if available
36           (progn (defsl greet :fn :eq sl-test.a sl-test.b sl-test.c)
37                  (is (eq (first available) (funcall 'greet))))
38           (signals error
39             (defsl greet :fn :eq sl-test.a sl-test.b sl-test.c))))))
