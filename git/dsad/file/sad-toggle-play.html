sad-toggle-play (442 bytes)

1 #!/bin/sh
2 # Overload the play button to also pause but never stop/replay
3 
4 SAD_SOCK=${SAD_SOCK:-"$HOME"/.socket}
5 # Long repetitive lines of code are really good style, they said.
6 cmd () {
7     echo "$1" | nc -NU "$SAD_SOCK"
8 }
9 playback () {
10     echo 'status playback' | nc -NU $SAD_SOCK | grep -Fq "playback: $1"
11 }
12 playback 'play' && cmd 'pause' && exit 0
13 playback 'pause' && cmd 'pause' && exit 0
14 playback 'stop' && cmd 'play' && exit 0
15 exit 1
