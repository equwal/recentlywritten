15 /**
16  * Writes the video and its subtitles into a folder the user picked. The two
17  * files have the same name, which is how a video player finds the subtitles.
18  */
19 object VideoExport {
20     private val state = MutableStateFlow(VideoStatus())
21     val status: StateFlow<VideoStatus> = state
22 
23     /** Blocks until the files are written; call from a worker thread. */
24     fun run(context: Context, audio: Uri, book: Uri?, srt: File, folder: Uri, size: VideoSize, fps: Int) {
25         state.value = VideoStatus(running = true)
26         val resolver = context.contentResolver
27         var video: Uri? = null
28         try {
29             val parent = DocumentsContract.buildDocumentUriUsingTree(folder, DocumentsContract.getTreeDocumentId(folder))
30             val base = TranscriptStore.describe(context, audio).first.substringBeforeLast('.')
31             video = DocumentsContract.createDocument(resolver, parent, "video/mp4", "$base.mp4")
32                 ?: throw IllegalStateException("The folder did not accept a new file.")
33             // The folder can change the name (a file with this name is there already). The subtitles follow it.
34             val written = TranscriptStore.describe(context, video).first.substringBeforeLast('.')
35 
36             val cover = book?.let { uri -> runCatching { resolver.openInputStream(uri)?.use(BookText::cover) }.getOrNull() }
37             VideoMaker.make(context, audio, cover, video, size, fps) { state.value = VideoStatus(true, it) }
38 
39             val subtitles = DocumentsContract.createDocument(resolver, parent, "application/x-subrip", "$written.srt")
40                 ?: throw IllegalStateException("The folder did not accept the subtitles file.")
41             resolver.openOutputStream(subtitles)!!.use { out -> srt.inputStream().use { it.copyTo(out) } }
42 
43             state.value = VideoStatus(message = "Saved $written.mp4 and $written.srt.")
44         } catch (e: Throwable) {
45             Log.w("SubRead", "video export failed", e)
46             // Half a video is of no use to anyone.
47             video?.let { runCatching { DocumentsContract.deleteDocument(resolver, it) } }
48             state.value = VideoStatus(message = "The video failed: ${e.message ?: e.javaClass.simpleName}")
49         }
50     }
