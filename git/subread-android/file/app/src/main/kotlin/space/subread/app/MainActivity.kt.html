119     /** Read the ask of another app and, if it is good, run the job for it. */
120     private fun take(from: Intent) {
121         if (from.action != AlignContract.ACTION) return
122         val audio = IntentCompat.getParcelableExtra(from, AlignContract.EXTRA_AUDIO, Uri::class.java)
123         val book = IntentCompat.getParcelableExtra(from, AlignContract.EXTRA_BOOK, Uri::class.java)
124         val state = Job.status.value
125         val read = AlignRequests.read(
126             audio?.toString(), book?.toString(), from.getStringExtra(AlignContract.EXTRA_LANGUAGE),
127             busy = !started && state.running,
128         )
129         if (read !is AlignRequest.Accepted) {
130             refuse((read as AlignRequest.Refused).error)
131             return
132         }
133         ask = read
134         keep(from, audio!!)
135         keep(from, book!!)
136         // Start at once: the other app sent the user here for this one job, the
137         // screen names the asker and the two files, and Stop is on it. After a
138         // rotation the job is already there, so do not start a second one.
139         if (!state.running && (!started || state.phase == Phase.IDLE)) {
140             started = true
141             takenAt = SystemClock.elapsedRealtime()
142             Job.clear()
143             val app = applicationContext
144             val language = read.language
145             thread(name = "subread-job") { Job.run(app, audio, book, language) }
146         }
147     }

163     /** Give the asking app its answer. [asked] is the language it asked for. */
164     private fun answer(status: JobStatus, asked: String) {
165         val srt = status.srt
166         if (status.phase != Phase.DONE || srt == null) {
167             // Back sends this. The screen stays, so the user can read the error.
168             val error = status.detail.ifEmpty { "The job failed." }
169             setResult(RESULT_CANCELED, Intent().putExtra(AlignContract.EXTRA_ERROR, error))
170             return
171         }
172         val uri = FileProvider.getUriForFile(this, "$packageName.files", srt)
173         setResult(
174             RESULT_OK,
175             Intent().setData(uri)
176                 .addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
177                 .putExtra(AlignContract.EXTRA_CUES, status.cues)
178                 .putExtra(AlignContract.EXTRA_MATCH_RATE, status.matchRate ?: 0.0)
179                 .putExtra(AlignContract.EXTRA_LANGUAGE, AlignRequests.languageOf(srt.name, asked)),
180         )
181         if (ready || SystemClock.elapsedRealtime() - takenAt < QUICK_ANSWER_MS) ready = true else finish()
182     }

184     /** The name of the app that asked, for the screen. */
185     private fun asker(): String {
186         val who = callingPackage ?: return "Another app"
187         return runCatching {
188             val info = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
189                 packageManager.getApplicationInfo(who, PackageManager.ApplicationInfoFlags.of(0))
190             } else {
191                 @Suppress("DEPRECATION") // The flags class is Android 13 and later.
192                 packageManager.getApplicationInfo(who, 0)
193             }
194             packageManager.getApplicationLabel(info).toString()
195         }.getOrDefault(who)
196     }

514 @Composable
515 private fun Outcome(status: JobStatus) {
516     when (status.phase) {
517         Phase.DONE -> Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
518             val rate = ((status.matchRate ?: 0.0) * 100).toInt()
519             Text("Done: ${status.cues} lines, $rate% found in the book.", fontWeight = FontWeight.Bold)
520             if (rate < 80) {
521                 Text("That is low. Usually it means a different edition or translation of the book, " +
522                     "or the wrong language.", style = MaterialTheme.typography.bodySmall)
523             }
524             if (status.paragraphsDropped > 0) {
525                 Text("${status.paragraphsDropped} paragraphs of the book were never narrated " +
526                     "(front matter, notes) and were left out.", style = MaterialTheme.typography.bodySmall)
527             }
528         }
529         Phase.FAILED -> Text("Failed: ${status.detail}", fontWeight = FontWeight.Bold)
530         Phase.CANCELLED -> Text(status.detail)
531         else -> {}
532     }

596 private fun share(context: Context, srt: File) {
597     val uri = FileProvider.getUriForFile(context, "${context.packageName}.files", srt)
598     val send = Intent(Intent.ACTION_SEND)
599         .setType("application/x-subrip")
600         .putExtra(Intent.EXTRA_STREAM, uri)
601         .addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
602     context.startActivity(Intent.createChooser(send, srt.name))
603 }
