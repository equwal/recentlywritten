51     /**
52      * Transcribe (or resume transcribing) [audio], align it to [book], write
53      * the subtitles. Blocks for as long as that takes; call from a worker thread.
54      *
55      * @param language a whisper language code, or "auto"
56      */
57     fun run(context: Context, audio: Uri, book: Uri, language: String) {
58         cancelled = false
59         state.value = JobStatus(Phase.PREPARING, "Loading the speech model")
60         try {
61             val store = TranscriptStore.forAudio(context, audio)
62             var lang = if (language == "auto") store.language ?: "auto" else language
63 
64             if (!store.complete) transcribe(context, audio, store, lang) { lang = it }
65             if (lang == "auto") lang = store.language ?: "en"
66 
67             state.value = state.value.copy(phase = Phase.ALIGNING, detail = "Matching the book to the narration",
68                 fraction = 0.97f, etaSeconds = null)
69             val (bookName, _) = TranscriptStore.describe(context, book)
70             val paragraphs = context.contentResolver.openInputStream(book)!!.use { BookText.read(it, bookName) }
71             if (paragraphs.isEmpty()) throw IllegalStateException("No text could be read from $bookName.")
72 
73             val result = BookAligner.align(store.segments(), paragraphs, Language.of(lang))
74 
75             val (audioName, _) = TranscriptStore.describe(context, audio)
76             val out = File(context.filesDir, "subtitles").apply { mkdirs() }
77                 .resolve(audioName.substringBeforeLast('.') + ".$lang.srt")
78             out.writeText(Srt.write(result.cues))
79 
80             state.value = JobStatus(
81                 Phase.DONE, "Done", 1f, srt = out, cues = result.cues.size,
82                 matchRate = BookAligner.matchRate(result.cues), paragraphsDropped = result.paragraphsDropped,
83             )
84         } catch (e: TranscriptionCancelled) {
85             state.value = JobStatus(Phase.CANCELLED, "Stopped. Progress is saved; start again to continue.")
86         } catch (e: Throwable) {
87             state.value = JobStatus(Phase.FAILED, e.message ?: e.javaClass.simpleName)
88         }
89     }

91     private fun transcribe(
92         context: Context, audio: Uri, store: TranscriptStore, language: String, onLanguage: (String) -> Unit,
93     ) {
94         var lang = language
95         Whisper.open(context).use { whisper ->
96             AudioDecoder(context, audio, startAtSeconds = store.doneUntil).use { decoder ->
97                 val total = decoder.durationSeconds
98                 val resumedAt = store.doneUntil
99                 val started = SystemClock.elapsedRealtime()
100 
101                 while (true) {
102                     if (cancelled) throw TranscriptionCancelled()
103                     val chunk = decoder.next() ?: break
104                     report(chunk.startSeconds, total, resumedAt, started)
105 
106                     val segments = whisper.transcribe(chunk.samples, lang, chunk.startSeconds)
107                     // Detect once, then hold it: a stretch of music must not be
108                     // allowed to flip the rest of the book into another language.
109                     if (lang == "auto") {
110                         lang = whisper.detectedLanguage.ifEmpty { "en" }
111                         onLanguage(lang)
112                     }
113                     store.append(segments, chunk.endSeconds, lang)
114                     report(chunk.endSeconds, total, resumedAt, started)
115                 }
116                 store.finish()
117             }
118         }
119     }
120 
121     private fun report(at: Double, total: Double, resumedAt: Double, startedMs: Long) {
122         val elapsed = (SystemClock.elapsedRealtime() - startedMs) / 1000.0
123         val done = at - resumedAt
124         val speed = if (elapsed > 5 && done > 0) done / elapsed else null
125         val eta = if (speed != null && total > 0) ((total - at) / speed).toLong().coerceAtLeast(0) else null
126         state.value = JobStatus(
127             Phase.TRANSCRIBING,
128             "Listening: ${clock(at)}" + if (total > 0) " of ${clock(total)}" else "",
129             // Alignment takes seconds; nearly all of the bar is transcription.
130             fraction = if (total > 0) (at / total * 0.96).toFloat().coerceIn(0f, 0.96f) else 0f,
131             etaSeconds = eta, speed = speed,
132         )
133     }
134 
135     fun clock(seconds: Double): String {
136         val s = seconds.toLong()
137         return if (s >= 3600) "%d:%02d:%02d".format(s / 3600, s / 60 % 60, s % 60)
138         else "%d:%02d".format(s / 60, s % 60)
139     }
