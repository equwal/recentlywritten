21 /**
22  * A video of the book: a still picture over the audiobook's own audio, as an
23  * MP4 that any player opens and YouTube accepts. The subtitles go beside it as
24  * an .srt with the same name, which players load by themselves.
25  *
26  * A still picture is cheap if nothing is encoded twice. The encoder makes one
27  * key frame. The frames after it say "no change" in about ten bytes each; they
28  * are written by hand (see H264Still), because an encoder spends kilobytes on
29  * the same frame. That minute is written again and again with new time stamps.
30  * So the frame rate costs almost nothing, and the picture size costs only in
31  * the one key frame of each minute.
32  * AAC audio is copied as it is; other audio is encoded to AAC once, which the
33  * MP4 container needs.
34  */
35 class VideoSize(val width: Int, val height: Int) {
36     val label: String get() = "${height}p"
37 }

55     /**
56      * @param cover image bytes, or null for a plain card
57      * @param size one of [sizes]
58      * @param fps frames in a second, 1 to 60
59      */
60     fun make(
61         context: Context, audio: Uri, cover: ByteArray?, dest: Uri,
62         size: VideoSize = SIZES[0], fps: Int = 1, onProgress: (Float) -> Unit,
63     ) {
64         require(fps in 1..60) { "fps $fps" }
65         val yuv = Yuv(picture(cover, size), size)
66         var rate = fps
67         var (videoFormat, gop) = encodeStill(yuv, size, 1)
68         try {
69             val skips = H264Still.skipFrames(bytes(videoFormat, "csd-0"), bytes(videoFormat, "csd-1"), GOP_SECONDS * fps - 1)
70             gop = gop + skips.map { Frame(it, false) }
71         } catch (e: H264Still.Unsupported) {
72             // Bigger and at one frame a second, but correct: the encoder's own frames.
73             Log.w(TAG, "still: no hand-made frames for this stream (${e.message})")
74             rate = 1
75             encodeStill(yuv, size, GOP_SECONDS).let { videoFormat = it.first; gop = it.second }
76         }
77         val level = H264Still.level(size.width, size.height, rate)
78         videoFormat.setByteBuffer("csd-0", ByteBuffer.wrap(H264Still.withLevel(bytes(videoFormat, "csd-0"), level)))
79         Log.i(TAG, "still: ${size.width}x${size.height} at $rate fps, level $level, ${gop.size} frames, " +
80             "key ${gop[0].bytes.size} B, the rest ${gop.drop(1).sumOf { it.bytes.size }} B")
81 
82         val extractor = MediaExtractor()
83         extractor.setDataSource(context, audio, null)
84         val track = (0 until extractor.trackCount).firstOrNull {
85             extractor.getTrackFormat(it).getString(MediaFormat.KEY_MIME).orEmpty().startsWith("audio/")
86         } ?: throw IllegalArgumentException("No audio track in this file.")
87         extractor.selectTrack(track)
88         val source = extractor.getTrackFormat(track)
89         val durationUs = if (source.containsKey(MediaFormat.KEY_DURATION)) source.getLong(MediaFormat.KEY_DURATION) else 0L
90 
91         val sound: AudioSource =
92             if (source.getString(MediaFormat.KEY_MIME) == MediaFormat.MIMETYPE_AUDIO_AAC) Copy(extractor, source)
93             else Transcode(extractor, source)
94 
95         context.contentResolver.openFileDescriptor(dest, "rw")!!.use { pfd ->
96             val muxer = MediaMuxer(pfd.fileDescriptor, MediaMuxer.OutputFormat.MUXER_OUTPUT_MPEG_4)
97             try {
98                 val v = muxer.addTrack(videoFormat)
99                 val a = muxer.addTrack(sound.format())
100                 muxer.start()
101 
102                 val info = MediaCodec.BufferInfo()
103                 var frame = 0L
104                 var frameUs = 0L
105                 fun picture() {
106                     val f = gop[(frame % gop.size).toInt()]
107                     info.set(0, f.bytes.size, frameUs, if (f.key) MediaCodec.BUFFER_FLAG_KEY_FRAME else 0)
108                     muxer.writeSampleData(v, ByteBuffer.wrap(f.bytes), info)
109                     frame++
110                     frameUs = frame * 1_000_000 / rate
111                 }
112 
113                 var lastUs = 0L
114                 while (true) {
115                     val s = sound.next() ?: break
116                     // Keep the two tracks side by side in the file, so a player
117                     // does not have to jump about to find the next of either.
118                     while (frameUs <= s.ptsUs) picture()
119                     info.set(0, s.bytes.size, s.ptsUs, MediaCodec.BUFFER_FLAG_KEY_FRAME)
120                     muxer.writeSampleData(a, ByteBuffer.wrap(s.bytes), info)
121                     lastUs = s.ptsUs
122                     if (durationUs > 0) onProgress((s.ptsUs.toFloat() / durationUs).coerceIn(0f, 1f))
123                 }
124                 while (frameUs <= lastUs) picture()
125                 muxer.stop()
126             } finally {
127                 runCatching { muxer.release() }
128                 sound.close()
129                 extractor.release()
130             }
131         }

157     private class Copy(val extractor: MediaExtractor, val source: MediaFormat) : AudioSource {
158         private val buffer = ByteBuffer.allocate(1 shl 18)
159         override fun format() = source
160         override fun next(): Sample? {
161             val n = extractor.readSampleData(buffer, 0)
162             if (n < 0) return null
163             val bytes = ByteArray(n)

172     /**
173      * MP3, Opus and the rest: decoded and encoded to AAC, once.
174      *
175      * Each call into a codec costs about a millisecond on a slow device, and
176      * that cost, not the arithmetic, sets the speed. So the decoder has its own
177      * thread and works while the encoder works, and the encoder gets the sound
178      * in large pieces: one call for many frames.
179      */
180     private class Transcode(val extractor: MediaExtractor, source: MediaFormat) : AudioSource {
181         private val decoder = MediaCodec.createDecoderByType(source.getString(MediaFormat.KEY_MIME)!!).apply {
182             configure(source, null, null, 0); start()
183         }
184         @Volatile private var rate = source.getInteger(MediaFormat.KEY_SAMPLE_RATE)
185         @Volatile private var channels = source.getInteger(MediaFormat.KEY_CHANNEL_COUNT)
186         @Volatile private var failure: Throwable? = null
187         @Volatile private var closed = false
188         private val pieces = java.util.concurrent.ArrayBlockingQueue<ByteArray>(8)
189         private val decoding = kotlin.concurrent.thread(name = "subread-audio-decode") {
190             try { decode() } catch (e: Throwable) { if (!closed) failure = e }
191             runCatching { pieces.put(END) }
192         }
193 
194         private var encoder: MediaCodec? = null
195         private var outFormat: MediaFormat? = null
196         private val info = MediaCodec.BufferInfo()
197         private var piece: ByteArray? = null
198         private var pieceAt = 0
199         private var pcmEnded = false
200         private var encoderEnded = false
201         private var framesIn = 0L
202         private val ready = ArrayDeque<Sample>()
203         private var finished = false
204 
205         override fun format(): MediaFormat {
206             while (outFormat == null && !finished) pump()
207             return outFormat ?: throw IllegalStateException("The audio could not be encoded.")
208         }
209 
210         override fun next(): Sample? {
211             while (ready.isEmpty() && !finished) pump()
212             return ready.removeFirstOrNull()
213         }
214 
215         /** The decoder's thread: compressed audio in, PCM out in pieces of about [PIECE] bytes. */
216         private fun decode() {
217             val info = MediaCodec.BufferInfo()
218             val pcm = java.io.ByteArrayOutputStream(PIECE + (1 shl 14))
219             var extractorDone = false
220             while (!closed) {
221                 // Each free input buffer gets a frame, so the decoder never waits for this loop.
222                 while (!extractorDone) {
223                     val i = decoder.dequeueInputBuffer(0)
224                     if (i < 0) break
225                     val n = extractor.readSampleData(decoder.getInputBuffer(i)!!, 0)
226                     if (n < 0) {
227                         decoder.queueInputBuffer(i, 0, 0, 0, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
228                         extractorDone = true
229                     } else {
230                         decoder.queueInputBuffer(i, 0, n, 0, 0)
231                         extractor.advance()
232                     }
233                 }
234                 val o = decoder.dequeueOutputBuffer(info, 10_000)
235                 if (o == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED) {
236                     // The container can be wrong about these; the decoder is not.
237                     rate = decoder.outputFormat.getInteger(MediaFormat.KEY_SAMPLE_RATE)
238                     channels = decoder.outputFormat.getInteger(MediaFormat.KEY_CHANNEL_COUNT)
239                 }
240                 if (o < 0) continue
241                 if (info.size > 0) {
242                     val bytes = ByteArray(info.size)
243                     decoder.getOutputBuffer(o)!!.apply { position(info.offset); get(bytes) }
244                     pcm.write(bytes)
245                 }
246                 val end = info.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM != 0
247                 decoder.releaseOutputBuffer(o, false)
248                 if (pcm.size() >= PIECE || (end && pcm.size() > 0)) {
249                     pieces.put(pcm.toByteArray())
250                     pcm.reset()
251                 }
252                 if (end) return
253             }
254         }

256         private fun startEncoder(): MediaCodec {
257             val f = MediaFormat.createAudioFormat(MediaFormat.MIMETYPE_AUDIO_AAC, rate, channels).apply {
258                 setInteger(MediaFormat.KEY_AAC_PROFILE, MediaCodecInfo.CodecProfileLevel.AACObjectLC)
259                 setInteger(MediaFormat.KEY_BIT_RATE, if (channels == 1) 80_000 else 128_000)
260                 setInteger(MediaFormat.KEY_MAX_INPUT_SIZE, PIECE)
261             }
262             return MediaCodec.createEncoderByType(MediaFormat.MIMETYPE_AUDIO_AAC).apply {
263                 configure(f, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE); start()
264             }.also { encoder = it }
265         }
266 
267         /** One step of the encoder's side: take encoded audio out, or put PCM in. */
268         private fun pump() {
269             encoder?.let { enc ->
270                 // Wait only when there is nothing to put in.
271                 val o = enc.dequeueOutputBuffer(info, if (piece != null || pieces.isNotEmpty()) 0 else 10_000)
272                 if (o == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED) outFormat = enc.outputFormat
273                 else if (o >= 0) {
274                     if (info.size > 0 && info.flags and MediaCodec.BUFFER_FLAG_CODEC_CONFIG == 0) {
275                         val bytes = ByteArray(info.size)
276                         enc.getOutputBuffer(o)!!.apply { position(info.offset); get(bytes) }
277                         ready += Sample(bytes, info.presentationTimeUs)
278                     }
279                     val end = info.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM != 0
280                     enc.releaseOutputBuffer(o, false)
281                     if (end) finished = true
282                     return
283                 }
284             }
285 
286             if (piece == null && !pcmEnded) {
287                 val next = pieces.poll(10, java.util.concurrent.TimeUnit.MILLISECONDS) ?: return
288                 if (next === END) {
289                     pcmEnded = true
290                     failure?.let { throw it }
291                 } else {
292                     piece = next
293                     pieceAt = 0
294                 }
295             }
296             if (encoderEnded) return
297 
298             val enc = encoder ?: startEncoder()
299             val i = enc.dequeueInputBuffer(10_000)
300             if (i < 0) return
301             val ptsUs = framesIn * 1_000_000 / rate
302             val pcm = piece
303             if (pcm != null) {
304                 val buf = enc.getInputBuffer(i)!!
305                 val frame = 2 * channels
306                 val n = minOf(buf.capacity() / frame * frame, pcm.size - pieceAt)
307                 buf.clear()
308                 buf.put(pcm, pieceAt, n)
309                 enc.queueInputBuffer(i, 0, n, ptsUs, 0)
310                 pieceAt += n
311                 framesIn += n / frame
312                 if (pieceAt >= pcm.size) piece = null
313             } else {
314                 enc.queueInputBuffer(i, 0, 0, ptsUs, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
315                 encoderEnded = true
316             }
317         }
318 
319         override fun close() {
320             closed = true
321             pieces.clear()              // so a decoder that waits to put a piece can go on, and stop
322             decoding.join(2_000)
323             runCatching { decoder.stop() }; decoder.release()
324             encoder?.let { runCatching { it.stop() }; it.release() }
325         }

330             val END = ByteArray(0)
331         }

336     /** The cover, letterboxed on a black card; a plain dark card when there is none. */
337     internal fun picture(cover: ByteArray?, size: VideoSize): Bitmap {
338         val width = size.width
339         val height = size.height
340         val card = createBitmap(width, height)
341         val canvas = android.graphics.Canvas(card)
342         canvas.drawColor(Color.rgb(0x16, 0x16, 0x1a))
343         val image = cover?.let { runCatching { BitmapFactory.decodeByteArray(it, 0, it.size) }.getOrNull() }
344         if (image != null) {
345             canvas.drawColor(Color.BLACK)
346             val scale = minOf(width.toFloat() / image.width, height.toFloat() / image.height)
347             val w = image.width * scale
348             val h = image.height * scale
349             canvas.drawBitmap(image, null, RectF((width - w) / 2, (height - h) / 2, (width + w) / 2, (height + h) / 2),
350                 Paint(Paint.FILTER_BITMAP_FLAG))
351         }
352         return card
353     }

359     private fun encoderFor(size: VideoSize): String? =

362             .filter { c ->

365                 }.getOrDefault(false)

370     /**
371      * The first [count] seconds of the picture at one frame a second. The
372      * frames go in as YUV buffers: that is the input each encoder accepts
373      * (the MediaTek encoder fails on frames drawn with a Canvas).
374      */
375     private fun encodeStill(yuv: Yuv, size: VideoSize, count: Int): Pair<MediaFormat, List<Frame>> {
376         val width = size.width
377         val height = size.height
378         val name = encoderFor(size) ?: throw IllegalStateException("This device has no H.264 encoder for ${size.label} video.")
379 
380         val format = MediaFormat.createVideoFormat(MediaFormat.MIMETYPE_VIDEO_AVC, width, height).apply {
381             setInteger(MediaFormat.KEY_COLOR_FORMAT, MediaCodecInfo.CodecCapabilities.COLOR_FormatYUV420Flexible)
382             setInteger(MediaFormat.KEY_BIT_RATE, 600_000)
383             setInteger(MediaFormat.KEY_FRAME_RATE, 1)
384             setInteger("video-qp-i-min", KEY_FRAME_QP)
385             setInteger("video-qp-i-max", KEY_FRAME_QP)
386             // Far longer than the clip: only the first frame is a key frame.
387             setInteger(MediaFormat.KEY_I_FRAME_INTERVAL, 3600)
388             // Baseline: each player decodes it, and H264Still can write frames for it.
389             setInteger(MediaFormat.KEY_PROFILE, MediaCodecInfo.CodecProfileLevel.AVCProfileBaseline)
390             // The level for one frame a second. make() writes the true level afterwards.
391             setInteger(MediaFormat.KEY_LEVEL, AVC_LEVELS.getValue(H264Still.level(width, height, 1)))
392         }
393         val codec = MediaCodec.createByCodecName(name)
394         codec.configure(format, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE)
395         codec.start()
396         val info = MediaCodec.BufferInfo()
397         val frames = ArrayList<Frame>()
398         var outFormat: MediaFormat? = null
399         var queued = 0
400         var ended = false
401         try {
402             while (!ended) {
403                 if (queued <= count) {
404                     val i = codec.dequeueInputBuffer(10_000)
405                     if (i >= 0) {
406                         if (queued == count) {
407                             codec.queueInputBuffer(i, 0, 0, queued * 1_000_000L, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
408                         } else {
409                             yuv.fill(codec.getInputImage(i)!!)
410                             codec.queueInputBuffer(i, 0, width * height * 3 / 2, queued * 1_000_000L, 0)
411                         }
412                         queued++
413                     }
414                 }
415                 val o = codec.dequeueOutputBuffer(info, 10_000)
416                 if (o == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED) outFormat = codec.outputFormat
417                 if (o < 0) continue
418                 if (info.size > 0 && info.flags and MediaCodec.BUFFER_FLAG_CODEC_CONFIG == 0) {
419                     val bytes = ByteArray(info.size)
420                     codec.getOutputBuffer(o)!!.apply { position(info.offset); get(bytes) }
421                     frames += Frame(bytes, info.flags and MediaCodec.BUFFER_FLAG_KEY_FRAME != 0)
422                 }
423                 ended = info.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM != 0
424                 codec.releaseOutputBuffer(o, false)
425             }
426         } finally {
427             runCatching { codec.stop() }
428             codec.release()
429         }
430         check(frames.size == count && frames[0].key) { "The video encoder made ${frames.size} of $count frames." }
431         check(frames.drop(1).none { it.key }) { "The video encoder made more than one key frame." }
432         return (outFormat ?: throw IllegalStateException("The video encoder gave no format.")) to frames
433     }

435     /** The picture as Y, U and V planes (BT.601, video range), made once and copied into each frame. */
436     internal class Yuv(picture: Bitmap, size: VideoSize) {
437         private val width = size.width
438         private val height = size.height
439         val y = ByteArray(width * height)
440         private val u = ByteArray(width * height / 4)
441         private val v = ByteArray(width * height / 4)
442 
443         init {
444             val argb = IntArray(width * height)
445             picture.getPixels(argb, 0, width, 0, 0, width, height)
446             for (row in 0 until height) for (col in 0 until width) {
447                 val c = argb[row * width + col]
448                 val r = (c shr 16) and 0xff
449                 val g = (c shr 8) and 0xff
450                 val b = c and 0xff
451                 y[row * width + col] = (((66 * r + 129 * g + 25 * b + 128) shr 8) + 16).toByte()
452                 if (row % 2 == 0 && col % 2 == 0) {
453                     val k = (row / 2) * (width / 2) + col / 2
454                     u[k] = (((-38 * r - 74 * g + 112 * b + 128) shr 8) + 128).toByte()
455                     v[k] = (((112 * r - 94 * g - 18 * b + 128) shr 8) + 128).toByte()
456                 }
457             }
458         }
459 
460         /** Planar or semi-planar, padded or not: the Image says, and this follows. */
461         fun fill(image: android.media.Image) {
462             copy(y, width, height, image.planes[0])
463             copy(u, width / 2, height / 2, image.planes[1])
464             copy(v, width / 2, height / 2, image.planes[2])
465         }
466 
467         private fun copy(src: ByteArray, w: Int, h: Int, plane: android.media.Image.Plane) {
468             val buffer = plane.buffer
469             if (plane.pixelStride == 1 && plane.rowStride == w) {
470                 buffer.position(0)
471                 buffer.put(src, 0, w * h)
472                 return
473             }
474             for (row in 0 until h) for (col in 0 until w) {
475                 buffer.put(row * plane.rowStride + col * plane.pixelStride, src[row * w + col])
476             }
477         }
478     }
