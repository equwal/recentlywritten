31     /**
32      * @param samples 16 kHz mono, -1..1
33      * @param offset where this buffer starts in the whole recording, seconds
34      */
35     fun transcribe(samples: FloatArray, language: String, offset: Double): List<TranscriptSegment> {
36         when (val rc = WhisperLib.transcribe(ptr, samples, threads, language)) {
37             0 -> {}
38             1000 -> throw TranscriptionCancelled()
39             else -> throw WhisperUnavailable("The speech model failed on this audio (code $rc).")
40         }
41         return (0 until WhisperLib.segmentCount(ptr)).mapNotNull { i ->
42             // Lenient decode: the tiny model can end a segment mid-character.
43             val text = String(WhisperLib.segmentText(ptr, i), Charsets.UTF_8).replace("�", "").trim()
44             if (text.isEmpty()) null
45             else TranscriptSegment(
46                 text,
47                 offset + WhisperLib.segmentStart(ptr, i) / 100.0,
48                 offset + WhisperLib.segmentEnd(ptr, i) / 100.0,
49             )
50         }
51     }

70         fun open(context: Context): Whisper {
71             requireCpu()
72             if (!loaded) {
73                 System.loadLibrary("subread_whisper")
74                 loaded = true
75             }
76             val model = File(context.filesDir, "ggml-tiny-q8_0.bin")
77             val asset = context.assets.openFd(MODEL_ASSET)
78             // whisper.cpp wants a path, and an asset inside the APK does not have one.
79             if (!model.exists() || model.length() != asset.length) {
80                 asset.createInputStream().use { src -> model.outputStream().use { src.copyTo(it) } }
81             }
82             asset.close()
83             val ptr = WhisperLib.initContext(model.absolutePath)
84             if (ptr == 0L) throw WhisperUnavailable("The speech model could not be loaded.")
85             return Whisper(ptr)
86         }

88         /**
89          * The native library is built for ARMv8.2 with half-precision and
90          * dot-product instructions. Loading it without them is a crash with no
91          * message, so look first.
92          */
93         private fun requireCpu() {
94             val features = runCatching {
95                 File("/proc/cpuinfo").readLines()
96                     .firstOrNull { it.startsWith("Features") }.orEmpty()
97                     .substringAfter(':').trim().split(' ').toSet()
98             }.getOrDefault(emptySet())
99             if (features.isEmpty()) return   // cannot tell; let it try
100             val missing = listOf("asimdhp", "asimddp").filter { it !in features }
101             if (missing.isNotEmpty()) {
102                 throw WhisperUnavailable(
103                     "This phone's processor is too old for on-device transcription " +
104                         "(needs ARMv8.2 half-precision and dot-product support)."
105                 )
106             }
107         }
108 
109         /**
110          * Threads to use: the fast cores only. whisper waits for its slowest
111          * thread at every layer, so adding little cores makes it slower, not
112          * faster. "Fast" is anything clocked above the slowest cluster.
113          */
114         private fun fastCores(): Int {
115             val max = (0 until Runtime.getRuntime().availableProcessors()).mapNotNull { cpu ->
116                 runCatching {
117                     File("/sys/devices/system/cpu/cpu$cpu/cpufreq/cpuinfo_max_freq").readText().trim().toLong()
118                 }.getOrNull()
119             }
120             if (max.isEmpty()) return 4
121             val slowest = max.min()
122             val fast = max.count { it > slowest }
123             // All cores alike (or unreadable clusters): use half, at most four.
124             return if (fast == 0) (max.size / 2).coerceIn(2, 4) else fast.coerceIn(2, 6)
125         }
