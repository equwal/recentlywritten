14 /**
15  * Streams any audio file the platform can play as 16 kHz mono float chunks,
16  * which is the only thing the speech model accepts.
17  *
18  * A ten-hour book is over two gigabytes as PCM, so it is never held whole:
19  * each call to [next] decodes a couple of minutes, cuts at the quietest moment
20  * near the end (so no word is split across two transcriptions) and carries the
21  * remainder into the next chunk.
22  */
23 class AudioDecoder(context: Context, uri: Uri, startAtSeconds: Double = 0.0) : Closeable {

73     /** The next chunk, or null at the end of the file. */
74     fun next(targetSeconds: Int = CHUNK_SECONDS): Chunk? {
75         // Re-read the rate each time round: HE-AAC files announce half of what
76         // the decoder actually produces, and only say so once decoding starts.
77         while (pending.size < targetSeconds.toLong() * sourceRate && !outputDone) pump()
78         if (pending.size == 0) return null
79         val want = targetSeconds.toLong() * sourceRate
80 
81         val cut = if (outputDone && pending.size <= want) pending.size
82         else quietestPoint(pending, sourceRate, min(want, pending.size.toLong()).toInt())
83 
84         val head = pending.take(cut)
85         val start = pendingStart
86         pending = pending.drop(cut)
87         pendingStart = start + cut / sourceRate.toDouble()
88         return Chunk(resample(head, sourceRate), start)
89     }

91     private fun pump() {
92         if (!inputDone) {
93             val i = codec.dequeueInputBuffer(10_000)
94             if (i >= 0) {
95                 val buffer = codec.getInputBuffer(i)!!
96                 val n = extractor.readSampleData(buffer, 0)
97                 if (n < 0) {
98                     codec.queueInputBuffer(i, 0, 0, 0, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
99                     inputDone = true
100                 } else {
101                     codec.queueInputBuffer(i, 0, n, extractor.sampleTime, 0)
102                     extractor.advance()
103                 }
104             }
105         }
106 
107         when (val o = codec.dequeueOutputBuffer(info, 10_000)) {
108             MediaCodec.INFO_OUTPUT_FORMAT_CHANGED -> {
109                 // The container can be wrong about these; the decoder is not.
110                 val f = codec.outputFormat
111                 sourceRate = f.getInteger(MediaFormat.KEY_SAMPLE_RATE)
112                 channels = f.getInteger(MediaFormat.KEY_CHANNEL_COUNT)
113             }
114             in 0..Int.MAX_VALUE -> {
115                 if (info.size > 0) {
116                     val bufferStart = info.presentationTimeUs / 1e6
117                     val shorts = codec.getOutputBuffer(o)!!.order(ByteOrder.LITTLE_ENDIAN).asShortBuffer()
118                     val frames = shorts.remaining() / channels
119                     val skip = if (bufferStart >= skipUntil) 0
120                     else min(frames, ((skipUntil - bufferStart) * sourceRate).toInt())
121                     shorts.position(shorts.position() + skip * channels)
122                     if (frames > skip && pendingStart < 0) {
123                         pendingStart = bufferStart + skip / sourceRate.toDouble()
124                     }
125                     pending.ensure(frames - skip)
126                     for (f in skip until frames) {
127                         var sum = 0
128                         for (c in 0 until channels) sum += shorts.get()
129                         pending.add(sum / (32768f * channels))
130                     }
131                 }
132                 codec.releaseOutputBuffer(o, false)
133                 if (info.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM != 0) outputDone = true
134             }
135         }
136     }

144     companion object {
145         const val TARGET_RATE = 16_000
146         const val CHUNK_SECONDS = 120
147 
148         /** How far back from the end of a chunk to look for a pause. */
149         private const val SEARCH_SECONDS = 12
150         private const val WINDOW_SECONDS = 0.25
151 
152         /** Index at which to cut: the middle of the quietest quarter second near [limit]. */
153         internal fun quietestPoint(buf: FloatBuffer, rate: Int, limit: Int): Int {
154             val window = (WINDOW_SECONDS * rate).toInt()
155             val from = max(0, limit - SEARCH_SECONDS * rate)
156             if (limit - from <= window * 2) return limit
157 
158             var energy = 0.0
159             for (i in from until from + window) energy += abs(buf[i])
160             var best = energy
161             var bestAt = from
162             var i = from
163             while (i + window < limit) {
164                 energy += abs(buf[i + window]) - abs(buf[i])
165                 i++
166                 if (energy < best) { best = energy; bestAt = i }
167             }
168             return bestAt + window / 2
169         }
170 
171         /**
172          * To 16 kHz with a triangular kernel as wide as the rate ratio: a
173          * low-pass and an interpolator in one pass. Plenty for speech
174          * recognition, and it costs a handful of multiplies per sample.
175          */
176         internal fun resample(src: FloatArray, rate: Int): FloatArray {
177             if (rate == TARGET_RATE) return src
178             val step = rate / TARGET_RATE.toDouble()
179             val half = max(step, 1.0)
180             val out = FloatArray((src.size / step).toInt())
181             for (o in out.indices) {
182                 val centre = o * step
183                 val lo = max(0, (centre - half).toInt() + 1)
184                 val hi = min(src.size - 1, (centre + half).toInt())
185                 var sum = 0.0
186                 var weight = 0.0
187                 for (k in lo..hi) {
188                     val w = 1.0 - abs(k - centre) / half
189                     if (w > 0) { sum += w * src[k]; weight += w }
190                 }
191                 out[o] = if (weight > 0) (sum / weight).toFloat() else 0f
192             }
193             return out
194         }
195     }

199 internal class FloatBuffer(capacity: Int = 1 shl 16) {
200     private var data = FloatArray(capacity)
201     var size = 0
202         private set
203 
204     operator fun get(i: Int) = data[i]

206     fun ensure(extra: Int) {
207         if (size + extra > data.size) data = data.copyOf(max(data.size * 2, size + extra))
208     }
209 
210     fun add(v: Float) { data[size++] = v }
211
