25 /**
26  * The decisions of the intent API, with no Android type in them, so that a JVM
27  * test covers them without a device.
28  */
29 object AlignRequests {
30     /** A Whisper language code is two or three lower-case letters, as in "ja" or "yue". */
31     private val CODE = Regex("[a-z]{2,3}")
32 
33     /** The default language: the speech model listens and decides. */
34     const val AUTO = "auto"
35 
36     /**
37      * Read the ask. [audio] and [book] are the Uris of the two extras, as text.
38      * [busy] is true when a job already runs, because SubRead runs one job at a
39      * time and must not stop that job for a new ask.
40      */
41     fun read(audio: String?, book: String?, language: String?, busy: Boolean): AlignRequest {
42         if (busy) {
43             return AlignRequest.Refused(
44                 "SubRead is busy with another job. Ask again when that job is done.",
45             )
46         }
47         val code = language?.trim().orEmpty().ifEmpty { AUTO }
48         val problem = badUri("audio file", AlignContract.EXTRA_AUDIO, audio)
49             ?: badUri("book", AlignContract.EXTRA_BOOK, book)
50             ?: badLanguage(code)
51         return if (problem != null) AlignRequest.Refused(problem)
52         else AlignRequest.Accepted(audio!!.trim(), book!!.trim(), code)
53     }
54 
55     /**
56      * The language the job used, read from the name of the file it wrote
57      * ("book.ja.srt" gives "ja"). [asked] is the answer when the name holds no
58      * code, which happens only if the naming changes.
59      */
60     fun languageOf(srtName: String, asked: String): String {
61         val code = srtName.removeSuffix(".srt").substringAfterLast('.', "")
62         return if (CODE.matches(code)) code else asked
63     }
64 
65     private fun badUri(what: String, extra: String, uri: String?): String? {
66         val text = uri?.trim().orEmpty()
67         return when {
68             text.isEmpty() -> "The ask gives no $what. Put its Uri in $extra."
69             // Only a content Uri is granted by the caller. A file Uri would make
70             // SubRead read its own private files for a stranger.
71             !text.startsWith("content://") ->
72                 "The $what must be a content:// Uri that the asking app grants, not $text."
73 
74             else -> null
75         }
76     }
77 
78     private fun badLanguage(code: String): String? =
79         if (code == AUTO || CODE.matches(code)) null
80         else "$code is not a language. Use a Whisper code such as ja, or $AUTO."
