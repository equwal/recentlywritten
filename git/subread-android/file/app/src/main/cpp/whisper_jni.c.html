1 // The narrowest possible bridge to whisper.cpp: load a model, transcribe one
2 // buffer of 16 kHz mono float PCM, read the segments back.
3 //
4 // One transcription runs at a time (the service guarantees it), which is what
5 // lets progress and cancellation be plain process-wide atomics rather than
6 // callbacks into the JVM from whisper's worker threads.

54 // Returns 0 on success, 1000 if cancelled, anything else is whisper's own error.
55 JNIEXPORT jint JNICALL
56 JNI_FN(transcribe)(JNIEnv *env, jobject thiz, jlong ptr, jfloatArray samples,
57                    jint n_threads, jstring language) {
58     (void) thiz;
59     struct whisper_context *ctx = (struct whisper_context *) ptr;
60     if (ctx == NULL) return -2;
61 
62     const jsize n = (*env)->GetArrayLength(env, samples);
63     jfloat *pcm = (*env)->GetFloatArrayElements(env, samples, NULL);
64     const char *lang = (*env)->GetStringUTFChars(env, language, NULL);
65 
66     struct whisper_full_params p = whisper_full_default_params(WHISPER_SAMPLING_GREEDY);
67     p.print_realtime   = false;
68     p.print_progress   = false;
69     p.print_timestamps = false;
70     p.print_special    = false;
71     p.translate        = false;
72     p.language         = lang;         // "auto" detects; the caller pins it after the first chunk
73     p.n_threads        = n_threads;
74     // Each chunk stands alone. Carrying text over as a prompt is how one
75     // hallucinated line turns into a page of the same line.
76     p.no_context       = true;
77     p.single_segment   = false;
78     p.suppress_blank   = true;
79     p.suppress_nst     = true;
80     p.progress_callback = on_progress;
81     p.abort_callback    = should_abort;
82 
83     atomic_store(&g_progress, 0);
84     atomic_store(&g_abort, false);
85 
86     int rc = whisper_full(ctx, p, pcm, n);
87 
88     (*env)->ReleaseStringUTFChars(env, language, lang);
89     (*env)->ReleaseFloatArrayElements(env, samples, pcm, JNI_ABORT);
90 
91     if (atomic_load(&g_abort)) return 1000;
92     if (rc != 0) LOGW("whisper_full failed: %d", rc);
93     return rc;
94 }

114 // Text as raw UTF-8 bytes. NewStringUTF wants *modified* UTF-8 and mangles or
115 // aborts on anything outside the BMP, and the tiny model will happily emit a
116 // truncated multi-byte sequence at a segment edge. Let Kotlin decode leniently.
117 JNIEXPORT jbyteArray JNICALL
118 JNI_FN(segmentText)(JNIEnv *env, jobject thiz, jlong ptr, jint index) {
119     (void) thiz;
120     const char *text = whisper_full_get_segment_text((struct whisper_context *) ptr, index);
121     if (text == NULL) text = "";
122     const jsize len = (jsize) strlen(text);
123     jbyteArray out = (*env)->NewByteArray(env, len);
124     if (out != NULL) (*env)->SetByteArrayRegion(env, out, 0, len, (const jbyte *) text);
125     return out;
126 }
