23 /**
24  * The job runs from the open screen. There is no background service.
25  * Start must begin the job, the display must stay on while the job runs, and

32     @get:Rule

35     @Test
36     fun startRunsTheJobFromTheScreenAndKeepsTheDisplayOn() {
37         val instrumentation = InstrumentationRegistry.getInstrumentation()
38         val names = instrumentation.context.assets.list("local").orEmpty()
39         val audioName = names.firstOrNull { it.endsWith(".mp3") || it.endsWith(".m4b") || it.endsWith(".m4a") }
40         val bookName = names.firstOrNull { it.endsWith(".epub") || it.endsWith(".txt") }
41         assumeTrue("no local test media", audioName != null && bookName != null)
42 
43         val app = instrumentation.targetContext
44         fun stage(name: String): Uri {
45             val out = File(app.cacheDir, name)
46             instrumentation.context.assets.open("local/$name").use { src -> out.outputStream().use { src.copyTo(it) } }
47             return Uri.fromFile(out)
48         }
49         val audio = stage(audioName!!)
50         val book = stage(bookName!!)
51         TranscriptStore.forAudio(app, audio).reset()
52         // The screen reads its picks from here, as it does after a restart.
53         app.getSharedPreferences("picks", Context.MODE_PRIVATE).edit()
54             .putString("audio", audio.toString()).putString("book", book.toString())
55             .putString("language", "auto").commit()
56 
57         ActivityScenario.launch(MainActivity::class.java).use { scenario ->
58             fun keepScreenOn(): Boolean {
59                 var on = false
60                 scenario.onActivity { on = it.findViewById<android.view.View>(android.R.id.content).let { root ->
61                     generateSequence(listOf(root)) { level ->
62                         level.flatMap { v -> (v as? android.view.ViewGroup)?.let { g -> (0 until g.childCount).map(g::getChildAt) } ?: emptyList() }
63                             .takeIf { it.isNotEmpty() }
64                     }.flatten().any { it.keepScreenOn }
65                 } }
66                 return on
67             }
