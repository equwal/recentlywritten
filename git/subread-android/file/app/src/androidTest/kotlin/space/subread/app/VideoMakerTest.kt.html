55         }
56         val audio = stage(audioName!!)

103     private fun trackBytes(file: File, mime: String): Long {
104         val ex = MediaExtractor()
105         ex.setDataSource(file.absolutePath)
106         ex.selectTrack((0 until ex.trackCount).first { ex.getTrackFormat(it).getString(MediaFormat.KEY_MIME) == mime })
107         var total = 0L
108         while (ex.sampleSize >= 0) { total += ex.sampleSize; ex.advance() }
109         ex.release()
110         return total
111     }

113     private fun decodeLuma(file: File): Luma {
114         val ex = MediaExtractor()
115         ex.setDataSource(file.absolutePath)
116         val track = (0 until ex.trackCount).first { ex.getTrackFormat(it).getString(MediaFormat.KEY_MIME) == "video/avc" }
117         ex.selectTrack(track)
118         val format = ex.getTrackFormat(track)
119         val width = format.getInteger(MediaFormat.KEY_WIDTH)
120         val height = format.getInteger(MediaFormat.KEY_HEIGHT)
121         val duration = format.getLong(MediaFormat.KEY_DURATION)
122         // Needed to read frames as an Image.
123         format.setInteger(MediaFormat.KEY_COLOR_FORMAT, MediaCodecInfo.CodecCapabilities.COLOR_FormatYUV420Flexible)
124         val codec = MediaCodec.createDecoderByType("video/avc")
125         codec.configure(format, null, null, 0)
126         codec.start()
127         val info = MediaCodec.BufferInfo()
128         var count = 0
129         var first: ByteArray? = null
130         var last = ByteArray(0)
131         var fed = false
132         var done = false
133         while (!done) {
134             if (!fed) {
135                 val i = codec.dequeueInputBuffer(10_000)
136                 if (i >= 0) {
137                     val n = ex.readSampleData(codec.getInputBuffer(i)!!, 0)
138                     if (n < 0) { codec.queueInputBuffer(i, 0, 0, 0, MediaCodec.BUFFER_FLAG_END_OF_STREAM); fed = true }
139                     else { codec.queueInputBuffer(i, 0, n, ex.sampleTime, 0); ex.advance() }
140                 }
141             }
142             val o = try { codec.dequeueOutputBuffer(info, 10_000) } catch (e: IllegalStateException) {
143                 throw AssertionError("the decoder failed after $count frames, fed up to ${ex.sampleTime / 1_000_000}s", e)
144             }
145             if (o < 0) continue
146             // Copy the first frame and the frames of the last second only: a copy of each frame takes minutes.
147             if (info.size > 0 && first != null && info.presentationTimeUs < duration - 1_000_000) count++
148             else if (info.size > 0) {
149                 val plane = codec.getOutputImage(o)!!.planes[0]
150                 val y = ByteArray(width * height)
151                 for (row in 0 until height) {
152                     plane.buffer.position(row * plane.rowStride)
153                     plane.buffer.get(y, row * width, width)
154                 }
155                 if (first == null) first = y
156                 last = y
157                 count++
158             }
159             done = info.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM != 0
160             codec.releaseOutputBuffer(o, false)
161         }
162         codec.stop(); codec.release(); ex.release()
163         return Luma(count, width, height, first!!, last)
164     }

166     private fun psnr(a: ByteArray, b: ByteArray): Double {
167         var sum = 0.0
168         for (i in a.indices) { val d = (a[i].toInt() and 0xff) - (b[i].toInt() and 0xff); sum += d * d }
169         return 10 * log10(255.0 * 255.0 / (sum / a.size).coerceAtLeast(1e-9))
170     }
