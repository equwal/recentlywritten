18 /**
19  * The whole job on real hardware: decode, transcribe with the native speech
20  * model, align, write the subtitles. No screen is needed.
21  *
22  * The media is not in the repository. Put an audio file and its book in
23  * app/src/androidTest/assets/local/ (gitignored); the test skips without them.
24  */
25 @RunWith(AndroidJUnit4::class)
26 class OnDeviceJobTest {
27 
28     @Test
29     fun anAudiobookAndItsBookBecomeSubtitles() {
30         val instrumentation = InstrumentationRegistry.getInstrumentation()
31         val names = instrumentation.context.assets.list("local").orEmpty()
32         val audioName = names.firstOrNull { it.endsWith(".mp3") || it.endsWith(".m4b") || it.endsWith(".m4a") }
33         val bookName = names.firstOrNull { it.endsWith(".epub") || it.endsWith(".txt") }
34         assumeTrue("no local test media", audioName != null && bookName != null)
35 
36         val app = instrumentation.targetContext
37         fun stage(name: String): Uri {
38             val out = File(app.cacheDir, name)
39             instrumentation.context.assets.open("local/$name").use { src -> out.outputStream().use { src.copyTo(it) } }
40             return Uri.fromFile(out)
41         }
42         val audio = stage(audioName!!)
43         val book = stage(bookName!!)
44         TranscriptStore.forAudio(app, audio).reset()   // measure a cold run, not a cached one
45 
46         val started = System.nanoTime()
47         Job.run(app, audio, book, "auto")
48         val seconds = (System.nanoTime() - started) / 1e9
49 
50         val status = Job.status.value
51         Log.i("SubReadTest", "phase=${status.phase} detail=${status.detail} cues=${status.cues} " +
52             "match=${status.matchRate} dropped=${status.paragraphsDropped} seconds=%.1f".format(seconds))
53         assertEquals(status.detail, Phase.DONE, status.phase)
54         assertNotNull(status.srt)
55         val srt = status.srt!!.readText()
56         Log.i("SubReadTest", "language=${TranscriptStore.forAudio(app, audio).language} srt head:\n" +
57             srt.lineSequence().take(12).joinToString("\n"))
58         // Side by side: what the model heard, and the book text given to that cue.
59         val heard = TranscriptStore.forAudio(app, audio).segments()
60         val cues = srt.trim().split("\n\n").map { it.lines().getOrElse(2) { "" } }
61         heard.forEach { s ->
62             Log.i("SubReadSeg", org.json.JSONObject().put("text", s.text).put("start", s.start).put("end", s.end).toString())
63         }
64         heard.take(14).forEachIndexed { i, s ->
65             Log.i("SubReadTest", "%6.2f-%6.2f heard: %s | book: %s".format(s.start, s.end, s.text, cues.getOrElse(i) { "" }))
66         }
67         assertTrue("only ${status.cues} cues", status.cues >= 5)
68         assertTrue("match rate ${status.matchRate}", (status.matchRate ?: 0.0) > 0.8)
69         assertTrue(srt.startsWith("1\n00:00:"))
70 
71         // A second run reuses the saved transcript: seconds, not minutes.
72         val again = System.nanoTime()
73         Job.run(app, audio, book, "auto")
74         val cached = (System.nanoTime() - again) / 1e9
75         Log.i("SubReadTest", "cached rerun seconds=%.1f".format(cached))
76         assertEquals(Phase.DONE, Job.status.value.phase)
77         assertTrue("cached rerun took ${cached}s", cached < seconds / 2 + 5)
78     }
