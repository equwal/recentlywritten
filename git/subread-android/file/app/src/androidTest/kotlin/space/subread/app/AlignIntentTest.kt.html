48     @Test
49     fun anotherAppAsksForSubtitlesAndGetsTheFileBack() {
50         val names = instrumentation.context.assets.list("local").orEmpty()
51         val audioName = names.firstOrNull { it.endsWith(".mp3") || it.endsWith(".m4b") || it.endsWith(".m4a") }
52         val bookName = names.firstOrNull { it.endsWith(".epub") || it.endsWith(".txt") }
53         assumeTrue("no local test media", audioName != null && bookName != null)
54 
55         val audio = stage(audioName!!)
56         val book = stage(bookName!!)
57         TranscriptStore.forAudio(app, audio).reset()
58         val picks = savePicks()
59 
60         ActivityScenario.launchActivityForResult<MainActivity>(ask(audio, book, "auto")).use { scenario ->
61             // The job starts by itself, and the screen says who asked for it.
62             compose.waitUntil(30_000) { Job.status.value.running }
63             compose.onNodeWithText("asked for subtitles", substring = true).assertExists()
64             // The files belong to the ask: the user cannot change them under it.
65             compose.onAllNodesWithText("Change")[0].assertIsNotEnabled()
66 
67             compose.waitUntil(1_800_000) {
68                 Job.status.value.phase == Phase.DONE || Job.status.value.phase == Phase.FAILED
69             }
70             // A failed job stays on the screen for the user to read, so there is no answer to wait for.
71             assertEquals(Job.status.value.detail, Phase.DONE, Job.status.value.phase)
72             // The screen answers from its own update loop, and in a Compose test that loop runs
73             // only while the test asks the rule to wait. scenario.result alone would block it.
74             compose.waitUntil(60_000) { scenario.state == Lifecycle.State.DESTROYED }
75             val result = scenario.result
76             assertEquals(Job.status.value.detail, Activity.RESULT_OK, result.resultCode)
77 
78             val answer = result.resultData
79             val srt = answer.data
80             assertNotNull("no Uri in the answer", srt)
81             assertEquals("content", srt!!.scheme)
82             assertEquals(
83                 Intent.FLAG_GRANT_READ_URI_PERMISSION,
84                 answer.flags and Intent.FLAG_GRANT_READ_URI_PERMISSION,
85             )
86 
87             val text = app.contentResolver.openInputStream(srt)!!.use { it.readBytes().decodeToString() }
88             assertTrue(text.take(20), text.startsWith("1\n00:00:"))
89             assertEquals(text.trim().split("\n\n").size, answer.getIntExtra(AlignContract.EXTRA_CUES, 0))
90             assertTrue(
91                 "match rate ${answer.getDoubleExtra(AlignContract.EXTRA_MATCH_RATE, 0.0)}",
92                 answer.getDoubleExtra(AlignContract.EXTRA_MATCH_RATE, 0.0) > 0.8,
93             )
94             val language = answer.getStringExtra(AlignContract.EXTRA_LANGUAGE)
95             assertTrue("language $language", language != null && language.matches(Regex("[a-z]{2,3}")))
96         }
97         assertEquals("the ask must not change what the user picked", picks, savePicks())
98 
99         // The same ask again. SubRead has the work of the first job, so the subtitles are ready in
100         // less than a second. On a tablet the screen opened and closed at once, and the user
101         // took it for a fault. Now the screen stays, says why, and has a button for the way back.
102         ActivityScenario.launchActivityForResult<MainActivity>(ask(audio, book, "auto")).use { scenario ->
103             compose.waitUntil(60_000) { Job.status.value.phase == Phase.DONE }
104             compose.waitUntil(10_000) {
105                 compose.onAllNodesWithText("ready at once", substring = true).fetchSemanticsNodes().isNotEmpty()
106             }
107             assertTrue("the screen closed by itself", scenario.state != Lifecycle.State.DESTROYED)
108             compose.onNodeWithText("Back to", substring = true).performClick()
109             compose.waitUntil(10_000) { scenario.state == Lifecycle.State.DESTROYED }
110             assertEquals(Activity.RESULT_OK, scenario.result.resultCode)
111             assertNotNull("no Uri in the answer", scenario.result.resultData.data)
112         }
113     }

136     private fun ask(audio: Uri, book: Uri, language: String) =
137         Intent(AlignContract.ACTION)
138             .setClass(app, MainActivity::class.java)
139             .putExtra(AlignContract.EXTRA_AUDIO, audio)
140             .putExtra(AlignContract.EXTRA_BOOK, book)
141             .putExtra(AlignContract.EXTRA_LANGUAGE, language)
142             .addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
143 
144     /**
145      * A content Uri of the test media, as the Uri of another app would be.
146      * The files go where the FileProvider serves from (see res/xml/file_paths.xml).
147      */
148     private fun stage(name: String): Uri {
149         val out = File(app.filesDir, "subtitles/asked").apply { mkdirs() }.resolve(name)
150         instrumentation.context.assets.open("local/$name").use { src ->
151             out.outputStream().use { src.copyTo(it) }
152         }
153         return FileProvider.getUriForFile(app, "${app.packageName}.files", out)
154     }
155 
156     /** What the user picked by hand. The ask of another app must leave it alone. */
157     private fun savePicks(): Map<String, Any?> {
158         val prefs = app.getSharedPreferences("picks", Context.MODE_PRIVATE)
159         if (!prefs.contains("audio")) {
160             prefs.edit().putString("audio", "content://user/audio")
161                 .putString("book", "content://user/book").putString("language", "en").commit()
162         }
163         return HashMap(prefs.all)
164     }
