34             for (words in listOf(
35                 "Subtitles (.srt)", "Video + subtitles", "upload it to YouTube", "Size", "Frames a second",
36                 "What it takes", "m4b, m4a, mp3", "epub, plain text", "Japanese, Spanish, Portuguese, Russian, Finnish",

38             )) compose.onNodeWithText(words, substring = true).assertExists()
39             // The outputs are real controls, not only words. The options work now;
40             // the save buttons wait for a job.
41             for (button in listOf("Save .srt", "Share", "Save video + .srt")) {
42                 compose.onNodeWithText(button).assertExists().assertIsNotEnabled()
43             }
44             compose.onNodeWithText(VideoMaker.sizes().first().label).performScrollTo().performClick()
45             compose.waitForIdle()
46             // The menu holds the sizes of this device.
47             assertTrue(compose.onAllNodesWithText(VideoMaker.sizes().last().label).fetchSemanticsNodes().isNotEmpty())
48             compose.onNodeWithText("Start").assertExists().assertIsNotEnabled()      // nothing is picked
49             // The Ko-fi link is in each build but the one for Google Play (-PplayStore=true).
50             val link = compose.onAllNodesWithText("Support on Ko-fi").fetchSemanticsNodes().size
51             assertTrue(link == if (BuildConfig.DONATE_LINK) 1 else 0)
52             val library = compose.onAllNodesWithText("Open honjimaku.com").fetchSemanticsNodes().size
