1 """Generate golden fixtures for languages that have no recorded transcript.
2 
3 make_golden.py needs a real Whisper transcript. This script makes one: it
4 takes a public-domain text, cuts it into segments the size of Whisper's, and
5 adds the kinds of error a small speech model makes (a dropped word, a wrong
6 word, lost punctuation, a hallucinated line). The reference implementation
7 then aligns that transcript, and every stage is recorded as make_golden.py
8 records it. The same fixture file is read by the Kotlin tests and by the
9 browser engine tests of subplz-web.
10 
11 The transcript is deterministic: the seed is derived from the fixture name.
12 
13 Needs the subplz environment, like make_golden.py:
14 
15     SubPlz/venv311/Scripts/python tools/make_synthetic_golden.py \
16         --text dom_casmurro.txt --language pt --name pt_casmurro \
17         --skip 40 --chars 5000 --out core/src/test/resources/golden \
18         --out ../subplz-web/tests/engine/golden
19 
20 The text is a Project Gutenberg plain-text file: the header and the licence
21 are cut at the START and END markers.
22 """

43 def gutenberg_paragraphs(path: Path) -> list[str]:
44     raw = path.read_text(encoding="utf-8-sig")
45     lines = raw.replace("\r\n", "\n").split("\n")
46     start = next((i for i, l in enumerate(lines) if l.startswith("*** START")), -1) + 1
47     end = next((i for i, l in enumerate(lines) if l.startswith("*** END")), len(lines))
48     out, current = [], []
49     for line in lines[start:end]:
50         if line.strip():
51             current.append(line.strip())
52         elif current:
53             out.append(" ".join(current))
54             current = []
55     if current:
56         out.append(" ".join(current))
57     return out

60 def prose_only(paragraphs: list[str]) -> list[str]:
61     """Drop headings and tables of contents: short lines in capitals, or without a full stop."""
62     return [p for p in paragraphs if len(p) > 60 and not p.isupper()]

65 def garble(word: str, rng: random.Random, vocabulary: list[str]) -> str:
66     kind = rng.random()
67     letters = [c for c in word if c.isalpha()]
68     if kind < 0.4 and len(letters) > 3:
69         # One letter wrong, as in a mishearing.
70         at = rng.randrange(len(word))
71         return word[:at] + rng.choice("aeiou") + word[at + 1:]
72     if kind < 0.7 and len(word) > 3:
73         # A letter lost.
74         at = rng.randrange(len(word))
75         return word[:at] + word[at + 1:]
76     # A different word of the same text.
77     return rng.choice(vocabulary)

80 def transcribe(paragraphs: list[str], rng: random.Random) -> list[dict]:
81     """Segments the size of Whisper's, with its errors, and times at a reading speed."""
82     words = " ".join(paragraphs).split()
83     vocabulary = [w.strip(_SILENT + _SOMETIMES) for w in words if w.strip(_SILENT + _SOMETIMES)]
84     segments = []
85     at = 0.0
86     i = 0
87     while i < len(words):
88         want = rng.randint(25, 75)
89         chunk = []
90         length = 0
91         while i < len(words) and (length < want or not chunk):
92             w = words[i]
93             i += 1
94             length += len(w) + 1
95             r = rng.random()
96             if r < 0.03:
97                 continue                          # a word lost
98             if r < 0.10:
99                 w = garble(w, rng, vocabulary)     # a word wrong
100             w = "".join(c for c in w if c not in _SILENT)
101             if w and w[-1] in _SOMETIMES and rng.random() < 0.5:
102                 w = w.rstrip(_SOMETIMES)
103             w = w.lstrip("¿¡")
104             if w:
105                 chunk.append(w)
106         text = " ".join(chunk)
107         if not text:
108             continue
109         seconds = len(text) / rng.uniform(12.0, 18.0)
110         segments.append({"text": text, "start": round(at, 3), "end": round(at + seconds, 3)})
111         at += seconds + rng.uniform(0.0, 0.6)
112     # A hallucinated line at the start, as Whisper makes on silence.
113     lead = {"text": rng.choice(vocabulary).capitalize() + ".", "start": 0.0, "end": 1.2}
114     return [lead] + [{**s, "start": round(s["start"] + 1.5, 3), "end": round(s["end"] + 1.5, 3)}
115                      for s in segments]
