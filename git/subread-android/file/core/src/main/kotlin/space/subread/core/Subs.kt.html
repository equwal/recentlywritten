27     /**
28      * One cue per transcript segment, timed by the recogniser and worded by
29      * the book. A segment that claimed no text keeps its own words, flagged.
30      */
31     fun toSubs(
32         text: List<String>,
33         subs: List<TranscriptSegment>,
34         alignment: List<List<Span>>,
35         offset: Double = 0.0,
36     ): MutableList<Cue> {
37         class Flat(val span: Span, val line: Int)
38         val flat = alignment.flatMapIndexed { i, spans -> spans.map { Flat(it, i) } }
39         val lines = text.map { it.toCodePoints() }
40 
41         val out = ArrayList<Cue>(subs.size)
42         var start = 0
43         var end = 0
44         for ((si, s) in subs.withIndex()) {
45             while (end < flat.size && flat[end].span.sub == si) end++
46             val r = StringBuilder()
47             for (k in start until end) {
48                 val a = flat[k]
49                 r.append(pySlice(lines[a.line], a.span.start, a.span.end))
50             }
51             val body = r.toString()
52             out += if (body.isNotBlank()) Cue(body, s.start + offset, s.end + offset)
53             else Cue(UNMATCHED + s.text, s.start + offset, s.end + offset)
54             start = end
55         }
56         return out
57     }

59     /** `seq[a:b]` with Python's rules: negatives wrap, out of range clamps. */
60     private fun pySlice(cps: IntArray, a: Int, b: Int): String {
61         val n = cps.size
62         val lo = (if (a < 0) a + n else a).coerceIn(0, n)
63         val hi = (if (b < 0) b + n else b).coerceIn(0, n)
64         return cps.asString(lo, hi)
65     }

71     private fun punctuationIndices(s: String) = s.indices.filter { s[it] in punctuation }
72     private fun countNonPunctuation(s: String) = s.count { it !in punctuation }

76     /**
77      * Move a stranded character or two across a cue boundary so that cues
78      * start and end on punctuation where they nearly did already.
79      */
80     fun shiftAlign(segments: List<Cue>): List<Cue> {
81         val fresh = ArrayList<Cue>(segments.size)
82         var startIndex = 0  // read by the second pass too, as in the original
83 
84         for ((i, segment) in segments.withIndex()) {
85             var text = segment.text
86             val indices = punctuationIndices(text)
87             if (indices.isEmpty()) {
88                 fresh += segment
89                 continue
90             }
91             startIndex = indices[0]
92             val nonPunc = countNonPunctuation(text.substring(0, startIndex))
93             if (nonPunc == 0 || countNonPunctuation(text.substring(startIndex + 1)) == 0) {
94                 fresh += segment
95                 continue
96             }
97             if (nonPunc <= 2 && i > 0 && fresh.isNotEmpty() &&
98                 !hasEndingPunctuation(fresh.last().text.last()) &&
99                 !hasDoubleComma(fresh.last().text, text.substring(0, startIndex + 1))
100             ) {
101                 // A word or less before the first mark: it belongs to the previous cue.
102                 fresh.last().text += text.substring(0, startIndex + 1)
103                 text = text.substring(startIndex + 1)
104             }
105             fresh += Cue(text, segment.start, segment.end)
106         }
107 
108         val final = ArrayList<Cue>(fresh.size)
109         for (i in fresh.indices) {
110             val segment = fresh[i]
111             var text = segment.text
112             val indices = punctuationIndices(text)
113             if (indices.isNotEmpty()) {
114                 val lastIndex = indices.last()
115                 val tail = text.substring(lastIndex)
116                 if (countNonPunctuation(tail) == 0 || tail.length == text.length) {
117                     final += segment
118                     continue
119                 }
120                 if (countNonPunctuation(tail) <= 2 && i + 1 < fresh.size &&
121                     !hasEndingPunctuation(fresh[i + 1].text.first()) &&
122                     !hasDoubleComma(
123                         fresh[i + 1].text.substring(0, 1),
124                         text.substring(0, minOf(startIndex + 1, text.length)),
125                     )
126                 ) {
127                     // A word or less after the last mark: it belongs to the next cue.
128                     // The original reaches into the *input* list here, not the
129                     // first pass's output. Kept: the fixtures depend on it.
130                     val next = segments[i + 1]
131                     next.text = text.substring(lastIndex + 1) + next.text
132                     text = text.substring(0, lastIndex + 1)
133                     final += Cue(text, segment.start, segment.end)
134                     fresh[i + 1] = next
135                     continue
136                 }
137             }
138             final += Cue(text, segment.start, segment.end)
139         }
140 
141         return doubleCheckMisalignedPairs(final).map { Cue(it.text.trim(), it.start, it.end) }
142     }

144     private val strayOpening = Regex("(」「(.{1,2})、$|」「(.{1})、$)")
145 
146     private fun doubleCheckMisalignedPairs(segments: MutableList<Cue>): List<Cue> {
147         if (segments.size < 2) return segments
148         val adjusted = ArrayList<Cue>(segments.size)
149         for ((i, segment) in segments.withIndex()) {
150             // 」「ばか、 at the very end is the start of the next line of dialogue.
151             strayOpening.find(segment.text)?.let { m ->
152                 if (i < segments.size - 1) {
153                     segments[i + 1].text = m.value + segments[i + 1].text
154                     segment.text = segment.text.substring(0, m.range.first)
155                 }
156             }
157             if (segment.text.isNotEmpty() && segment.text.first() in endPunc && i > 0) {
158                 adjusted.last().text += segment.text.first()
159                 segment.text = segment.text.substring(1)
160             }
161             if (segment.text.isNotEmpty() && segment.text.last() in startPunc && i < segments.size - 1) {
162                 segments[i + 1].text = segment.text.last() + segments[i + 1].text
163                 segment.text = segment.text.substring(0, segment.text.length - 1)
164             }
165             adjusted += segment
166         }
167         return adjusted
168     }
