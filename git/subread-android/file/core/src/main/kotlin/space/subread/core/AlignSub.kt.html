15     /**
16      * @param text cleaned paragraphs, as code points
17      * @param subs cleaned transcript segments, as code points
18      * @return for each paragraph, the spans of it claimed by transcript segments
19      */
20     fun alignSub(
21         coordinates: Coordinates,
22         text: List<IntArray>,
23         subs: List<IntArray>,
24         thing: Int = 2,
25     ): List<MutableList<Span>> {
26         var line = 0            // current[0]
27         var sub = 0             // current[1]
28         val pos = intArrayOf(0, 0)
29         var toff = 0
30         val p = intArrayOf(0, 0)
31         val gaps = intArrayOf(0, 0)
32         val segments = mutableListOf<MutableList<Span>>(mutableListOf())
33         var off = 0
34 
35         for (i in 0 until coordinates.size) {
36             val c0 = coordinates.target[i]
37             val c1 = coordinates.query[i]
38             // A column where either side stood still is a gap in the other.
39             val isGap = (c0 - p[0] == 0) || (c1 - p[1] == 0)
40 
41             while (sub < subs.size && pos[1] + subs[sub].size <= c1) {
42                 if (line >= text.size) return segments.take(text.size)
43                 val subLen = subs[sub].size
44                 pos[1] += subLen
45                 if (isGap) {
46                     // np.clip(pos - p, 0, None)[::-1]
47                     gaps[0] += maxOf(pos[1] - p[1], 0)
48                     gaps[1] += maxOf(pos[0] - p[0], 0)
49                 }
50 
51                 val diff = subLen + gaps[1] - gaps[0]
52                 if (diff > subLen / 4) {
53                     var target = toff + diff + off
54                     off = 0
55                     while (line < text.size && target >= text[line].size) {
56                         val start = toff
57                         val end = text[line].size
58                         if (end - start != 0) {
59                             val prev = segments.last()
60                             if (end - start < thing || distinctExceptFullStop(text[line], start, end) < thing) {
61                                 // Too small to stand alone: glue it on.
62                                 if (prev.isNotEmpty()) prev.last().end = end
63                                 else prev.add(Span(start, end, sub))
64                             } else {
65                                 prev.add(Span(start, end, sub))
66                             }
67                         }
68                         segments.add(mutableListOf())
69                         pos[0] += end - start
70                         target -= text[line].size
71                         line++
72                         toff = 0
73                     }
74                     pos[0] += target - toff
75                     segments.last().add(Span(toff, target, sub))
76                     toff = target
77                 } else {
78                     // The recogniser heard something the book does not have.
79                     val prev = segments.last()
80                     if (toff >= text[line].size / 2 && prev.isNotEmpty()) {
81                         prev.last().end += diff
82                         toff += diff
83                     } else {
84                         off += diff
85                     }
86                 }
87 
88                 sub++
89                 gaps[0] = 0
90                 gaps[1] = 0
91                 p[0] = maxOf(pos[0], p[0])
92                 p[1] = maxOf(pos[1], p[1])
93             }
94 
95             if (isGap) {
96                 // gaps += (c - p)[::-1]
97                 gaps[0] += c1 - p[1]
98                 gaps[1] += c0 - p[0]
99             }
100             p[0] = c0
101             p[1] = c1
102         }
103         return segments.take(text.size)
104     }

106     private fun distinctExceptFullStop(cps: IntArray, start: Int, end: Int): Int {
107         val seen = HashSet<Int>()
108         for (k in start until minOf(end, cps.size)) if (cps[k] != '。'.code) seen.add(cps[k])
109         return seen.size
110     }

112     /**
113      * Spans were computed on cleaned text. Map them back onto the original.
114      *
115      * The cleaned string is the translated one with characters removed, so it
116      * is a subsequence of it, and a two-finger walk recovers each offset.
117      */
118     fun fix(lang: Language, original: List<String>, edited: List<IntArray>, segments: List<MutableList<Span>>) {
119         for ((l, spans) in segments.withIndex()) {
120             val o = lang.translate(original[l]).toCodePoints()
121             val e = edited[l]
122             val m = IntArray(e.size + 1) { -1 }
123             var ei = 0
124             for (oi in o.indices) {
125                 if (ei < e.size && o[oi] == e[ei]) {
126                     m[ei] = oi
127                     ei++
128                 }
129             }
130             m[ei] = o.size  // snap to end
131             m[0] = 0        // snap to zero
132             var last = 0
133             for (i in 0 until e.size) {
134                 if (m[i] != -1) last = i else m[i] = m[last]
135             }
136             if (m[e.size] == -1) m[e.size] = o.size
137 
138             for (f in spans) {
139                 f.start = m[f.start.coerceIn(0, e.size)]
140                 f.end = m[f.end.coerceIn(0, e.size)]
141             }
142         }
143     }

145     /**
146      * Nudge span ends so punctuation stays with the words it belongs to: a
147      * closing quote goes with what it closes, an opening one with what follows,
148      * and a small kana or ellipsis is never stranded at the head of a cue.
149      */
150     fun fixPunc(
151         text: List<String>,
152         segments: List<MutableList<Span>>,
153         prepend: Set<Int>,
154         append: Set<Int>,
155         nopend: Set<Int>,
156     ) {
157         for ((l, s) in segments.withIndex()) {
158             if (s.isEmpty()) continue
159             val t = text[l].toCodePoints()
160             for (k in s.indices) {
161                 val p = s[k]
162                 // The last span is paired with itself, as in the original's
163                 // zip(s, s[1:] + [s[-1]]).
164                 val f = if (k + 1 < s.size) s[k + 1] else s[k]
165                 val connected = f.start == p.end
166                 var loop = 0
167                 while (loop <= 20) {
168                     if (p.end < t.size && at(t, p.end) in append) {
169                         p.end += 1
170                     } else if (at(t, p.end - 1) in prepend) {
171                         p.end -= 1
172                     } else if (
173                         (p.end > 0 && at(t, p.end - 1) in nopend) ||
174                         (p.end < t.size && at(t, p.end) in nopend) ||
175                         (p.end < t.size - 1 && at(t, p.end + 1) in nopend)
176                     ) {
177                         var start = p.end - 1
178                         var end = p.end
179                         if (p.end < t.size - 1) {
180                             val here = at(t, p.end)
181                             // Do not end a cue on a kanji whose okurigana follows.
182                             if ((at(t, p.end + 1) in nopend && 0x4E00 > here) || here > 0x9FAF) end += 1
183                         }
184                         while (start > 0 && at(t, start) in nopend) start -= 1
185                         while (end < t.size - 1 && at(t, end) in nopend) end += 1
186 
187                         if (at(t, start) in prepend) {
188                             if (p.end == start) break
189                             p.end = start
190                         } else if (at(t, start) in append) {
191                             if (p.end == start + 1) break
192                             p.end = start + 1
193                         } else if (end < t.size && at(t, end) in prepend) {
194                             if (p.end == end) break
195                             p.end = end
196                         } else if (end < t.size && at(t, end) in append) {
197                             if (p.end == end + 1) break
198                             p.end = end + 1
199                         } else {
200                             break
201                         }
202                     } else {
203                         break
204                     }
205                     loop++
206                 }
207                 if (connected) f.start = p.end
208             }
209         }
210     }

212     /** Python indexing: negatives count from the end. Out of range is "no character". */
213     private fun at(t: IntArray, i: Int): Int {
214         val k = if (i < 0) i + t.size else i
215         return if (k in t.indices) t[k] else -1
216     }
