4  * The whole job: a transcript with times and a book without them go in,

9     /**
10      * Above this many traceback cells the exact aligner is not attempted and
11      * the anchored one takes over. 64M cells is 64 MB, comfortable on a phone.
12      */
13     const val EXACT_CELL_LIMIT = 64_000_000L
14 
15     fun align(
16         transcript: List<TranscriptSegment>,
17         paragraphs: List<String>,
18         language: Language,
19         offset: Double = 0.0,
20     ): List<Cue> {
21         if (transcript.isEmpty()) return emptyList()
22 
23         val subsClean = transcript.map { language.clean(it.text).toCodePoints() }
24         val textClean = paragraphs.map { language.clean(it).toCodePoints() }
25         val query = concat(subsClean)
26         val target = concat(textClean)
27 
28         if (target.isEmpty() || query.isEmpty()) {
29             return transcript.map { Cue(Subs.UNMATCHED + it.text, it.start + offset, it.end + offset) }
30         }
31 
32         val coordinates =
33             if (Gotoh.cells(target.size, query.size) <= EXACT_CELL_LIMIT) Gotoh.align(target, query).coordinates
34             else AnchoredAligner.align(target, query).coordinates
35 
36         val spans = AlignSub.alignSub(coordinates, textClean, subsClean)
37         AlignSub.fix(language, paragraphs, textClean, spans)
38         AlignSub.fixPunc(paragraphs, spans, Subs.PREPEND_SET, Subs.APPEND_SET, Subs.NOPEND_SET)
39 
40         return Subs.shiftAlign(Subs.toSubs(paragraphs, transcript, spans, offset))
41     }
42 
43     internal fun concat(parts: List<IntArray>): IntArray {
44         val out = IntArray(parts.sumOf { it.size })
45         var at = 0
46         for (p in parts) {
47             p.copyInto(out, at)
48             at += p.size
49         }
50         return out
51     }
