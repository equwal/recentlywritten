5 /**
6  * "No change" frames for an H.264 video of a still picture.
7  *
8  * An encoder is given the picture once and makes one key frame. Each frame
9  * after it is written here by hand: a P slice in which each macroblock is
10  * skipped. A decoder shows the key frame again, bit for bit, and the frame
11  * costs about ten bytes. An encoder asked for the same frame spends kilobytes
12  * on it, because its rate control has bits to use.
13  *
14  * Only streams with CAVLC entropy coding, one slice group and frame
15  * macroblocks are handled (Baseline profile is always like that). For another
16  * stream, [skipFrames] throws [Unsupported] and the caller must use the
17  * encoder's own frames.
18  */
19 object H264Still {
20     class Unsupported(message: String) : Exception(message)

22     private val HIGH_PROFILES = setOf(100, 110, 122, 244, 44, 83, 86, 118, 128, 138, 139, 134, 135)

24     /**
25      * The frames that follow a key frame, in order. Frame `i` of the result is
26      * frame `i + 1` of the group of pictures. Each is one NAL unit with a
27      * four-byte start code, as MediaCodec gives and MediaMuxer takes.
28      *
29      * @param sps the sequence parameter set, with or without a start code
30      * @param pps the picture parameter set, with or without a start code
31      */
32     fun skipFrames(sps: ByteArray, pps: ByteArray, count: Int): List<ByteArray> {
33         try {
34             val s = Sps(BitReader(unescape(payload(sps, 7))))
35             val p = Pps(BitReader(unescape(payload(pps, 8))))
36             return (1..count).map { slice(s, p, it) }
37         } catch (e: IndexOutOfBoundsException) {
38             throw Unsupported("The parameter sets are cut short.")
39         }
40     }

46         Triple(52, 36_864, 2_073_600), Triple(60, 139_264, 4_177_920), Triple(61, 139_264, 8_355_840),

50     /** The lowest level that allows this picture size at this frame rate. */
51     fun level(width: Int, height: Int, fps: Int): Int {
52         val macroblocks = ((width + 15) / 16) * ((height + 15) / 16)
53         return LEVELS.firstOrNull { (_, frame, second) -> macroblocks <= frame && macroblocks.toLong() * fps <= second }?.first
54             ?: throw Unsupported("no H.264 level allows ${width}x$height at $fps frames a second")
55     }

57     /**
58      * The sequence parameter set with its level raised to [level]. The encoder
59      * sees one frame and declares a level for that; the file holds more frames
60      * in a second, and a decoder must be told the truth. Never lowers a level.
61      */
62     fun withLevel(sps: ByteArray, level: Int): ByteArray {
63         val at = sps.size - payload(sps, 7).size + 2      // profile_idc, constraint flags, level_idc
64         val out = sps.copyOf()
65         if (out[at].toInt() and 0xff < level) out[at] = level.toByte()
66         return out
67     }

69     private class Sps(r: BitReader) {
70         val log2MaxFrameNum: Int
71         val pocType: Int
72         var log2MaxPocLsb = 0
73         var deltaPocAlwaysZero = false
74         val macroblocks: Int
75 
76         init {
77             val profile = r.bits(8)
78             r.bits(16)                      // constraint flags, level
79             r.ue()                          // seq_parameter_set_id
80             if (profile in HIGH_PROFILES) {
81                 val chroma = r.ue()
82                 if (chroma == 3 && r.bit() == 1) throw Unsupported("separate colour planes")
83                 r.ue(); r.ue(); r.bit()     // bit depths, transform bypass
84                 if (r.bit() == 1) throw Unsupported("scaling matrix")
85             }
86             log2MaxFrameNum = r.ue() + 4
87             pocType = r.ue()
88             when (pocType) {
89                 0 -> log2MaxPocLsb = r.ue() + 4
90                 1 -> {
91                     deltaPocAlwaysZero = r.bit() == 1
92                     r.se(); r.se()
93                     repeat(r.ue()) { r.se() }
94                 }
95             }
96             r.ue(); r.bit()                 // max_num_ref_frames, gaps allowed
97             val width = r.ue() + 1
98             val height = r.ue() + 1
99             if (r.bit() == 0) throw Unsupported("field macroblocks")
100             macroblocks = width * height
101         }
102     }

104     private class Pps(r: BitReader) {
105         val id = r.ue()
106         val bottomFieldPoc: Boolean
107         val deblockingControl: Boolean
108         val redundantPicCnt: Boolean
109 
110         init {
111             r.ue()                          // seq_parameter_set_id
112             if (r.bit() == 1) throw Unsupported("CABAC")
113             bottomFieldPoc = r.bit() == 1
114             if (r.ue() != 0) throw Unsupported("slice groups")
115             r.ue(); r.ue()                  // default reference counts
116             if (r.bit() == 1) throw Unsupported("weighted prediction")
117             r.bits(2)
118             r.se(); r.se(); r.se()          // initial QP, QS, chroma QP offset
119             deblockingControl = r.bit() == 1
120             r.bit()
121             redundantPicCnt = r.bit() == 1
122         }
123     }

125     /** Frame [n] after the key frame: slice header, one run of skipped macroblocks, end. */
126     private fun slice(s: Sps, p: Pps, n: Int): ByteArray {
127         val w = BitWriter()
128         w.ue(0)                             // first_mb_in_slice
129         w.ue(5)                             // slice_type: P, and so is every slice of the picture
130         w.ue(p.id)
131         w.bits(n % (1 shl s.log2MaxFrameNum), s.log2MaxFrameNum)
132         if (s.pocType == 0) {
133             w.bits((2 * n) % (1 shl s.log2MaxPocLsb), s.log2MaxPocLsb)
134             if (p.bottomFieldPoc) w.se(0)
135         }
136         if (s.pocType == 1 && !s.deltaPocAlwaysZero) {
137             w.se(0)
138             if (p.bottomFieldPoc) w.se(0)
139         }
140         if (p.redundantPicCnt) w.ue(0)
141         w.bit(1); w.ue(0)                   // one reference picture: the frame before
142         w.bit(0)                            // no reference list modification
143         w.bit(0)                            // sliding-window reference marking
144         w.se(0)                             // slice_qp_delta
145         if (p.deblockingControl) w.ue(1)    // no deblocking: there is nothing to smooth
146         w.ue(s.macroblocks)                 // mb_skip_run: all of them
147         val rbsp = w.trailing()
148 
149         val out = ByteArrayOutputStream()
150         out.write(byteArrayOf(0, 0, 0, 1, 0x41))    // start code; reference picture, non-IDR slice
151         out.write(escape(rbsp))
152         return out.toByteArray()
153     }

155     /** The bytes after the NAL header, which must be of [type]. */
156     private fun payload(nal: ByteArray, type: Int): ByteArray {
157         var at = 0
158         while (at < nal.size && nal[at].toInt() == 0) at++
159         if (at > 0 && at < nal.size && nal[at].toInt() == 1) at++ else at = 0
160         if (at >= nal.size || nal[at].toInt() and 0x1f != type) throw Unsupported("not a NAL unit of type $type")
161         return nal.copyOfRange(at + 1, nal.size)
162     }

164     /** Puts in the 0x03 bytes that keep a start code from showing up inside a NAL unit. */
165     internal fun escape(rbsp: ByteArray): ByteArray {
166         val out = ByteArrayOutputStream(rbsp.size + 4)
167         var zeros = 0
168         for (b in rbsp) {
169             if (zeros >= 2 && b.toInt() and 0xff <= 3) { out.write(3); zeros = 0 }
170             out.write(b.toInt())
171             zeros = if (b.toInt() == 0) zeros + 1 else 0
172         }
173         return out.toByteArray()
174     }

176     internal fun unescape(nal: ByteArray): ByteArray {
177         val out = ByteArrayOutputStream(nal.size)
178         var zeros = 0
179         for (b in nal) {
180             if (zeros >= 2 && b.toInt() == 3) { zeros = 0; continue }
181             out.write(b.toInt())
182             zeros = if (b.toInt() == 0) zeros + 1 else 0
183         }
184         return out.toByteArray()
185     }

187     internal class BitReader(private val data: ByteArray) {
188         private var at = 0
189         fun bit(): Int {
190             if (at shr 3 >= data.size) throw IndexOutOfBoundsException()
191             val b = (data[at shr 3].toInt() shr (7 - (at and 7))) and 1
192             at++
193             return b
194         }
195         fun bits(n: Int): Int { var v = 0; repeat(n) { v = (v shl 1) or bit() }; return v }
196         fun ue(): Int {
197             var zeros = 0
198             while (bit() == 0) if (++zeros > 31) throw Unsupported("bad Exp-Golomb code")
199             return (1 shl zeros) - 1 + bits(zeros)
200         }
201         fun se(): Int { val k = ue(); return if (k and 1 == 1) (k + 1) / 2 else -(k / 2) }
202     }
203 
204     internal class BitWriter {
205         private val out = ByteArrayOutputStream()
206         private var current = 0
207         private var filled = 0
208         fun bit(b: Int) {
209             current = (current shl 1) or (b and 1)
210             if (++filled == 8) { out.write(current); current = 0; filled = 0 }
211         }
212         fun bits(v: Int, count: Int) { for (i in count - 1 downTo 0) bit(v shr i) }
213         fun ue(v: Int) {
214             val x = v + 1
215             val length = 32 - Integer.numberOfLeadingZeros(x)
216             bits(0, length - 1)
217             bits(x, length)
218         }
219         fun se(v: Int) = ue(if (v > 0) 2 * v - 1 else -2 * v)
220         /** The stop bit, then zeros to the end of the byte. */
221         fun trailing(): ByteArray {
222             bit(1)
223             while (filled != 0) bit(0)
224             return out.toByteArray()
225         }
226     }
