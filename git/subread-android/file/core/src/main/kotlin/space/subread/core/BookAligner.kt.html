3 /**
4  * Aligning a whole book at once, rather than chapter by matched chapter.
5  *
6  * The reference implementation gives every character of the text it is handed
7  * to some cue. That is right when the text is exactly what was read aloud, and
8  * wrong for a whole epub: the title page, the contents, the copyright notice,
9  * a translator's afterword nobody narrated - all of it would be glued onto
10  * whichever cue happens to sit next to it. subplz avoids that by matching
11  * chapters up first and dropping the ones that match nothing, which needs
12  * chaptered audio and a similarity threshold that does not hold across scripts.
13  *
14  * Here the alignment itself says what was narrated. A first pass lines up
15  * everything; stretches of text where almost nothing agrees with the
16  * transcript were evidently not read, and are left out of the second pass.
17  */
18 object BookAligner {
19 
20     /** Share of a paragraph's characters that must agree before it counts as heard. */
21     private const val HEARD = 0.20
22 
23     /**
24      * Unheard paragraphs are only dropped in runs at least this long (cleaned
25      * characters). A single heading the recogniser mangled completely is still
26      * a heading that was read; a page of nothing is not.
27      */
28     private const val MIN_UNREAD_RUN = 240
29 
30     class Result(val cues: List<Cue>, val paragraphsUsed: Int, val paragraphsDropped: Int)
31 
32     fun align(
33         transcript: List<TranscriptSegment>,
34         paragraphs: List<String>,
35         language: Language,
36     ): Result {
37         val narrated = narratedOnly(transcript, paragraphs, language)
38         val cues = Aligner.align(transcript, narrated, language)
39         return Result(cues, narrated.size, paragraphs.size - narrated.size)
40     }
41 
42     internal fun narratedOnly(
43         transcript: List<TranscriptSegment>,
44         paragraphs: List<String>,
45         language: Language,
46     ): List<String> {
47         val textClean = paragraphs.map { language.clean(it).toCodePoints() }
48         val target = Aligner.concat(textClean)
49         val query = Aligner.concat(transcript.map { language.clean(it.text).toCodePoints() })
50         if (target.isEmpty() || query.isEmpty()) return paragraphs
51 
52         val coordinates = AnchoredAligner.align(target, query).coordinates
53 
54         // Characters of the text that met the same character in the transcript.
55         val agreed = BooleanArray(target.size)
56         for (k in 1 until coordinates.size) {
57             val t0 = coordinates.target[k - 1]
58             val q0 = coordinates.query[k - 1]
59             val dt = coordinates.target[k] - t0
60             if (dt > 0 && coordinates.query[k] - q0 > 0) {
61                 for (d in 0 until dt) agreed[t0 + d] = target[t0 + d] == query[q0 + d]
62             }
63         }
64 
65         val heard = BooleanArray(paragraphs.size)
66         var at = 0
67         for ((i, cps) in textClean.withIndex()) {
68             val hits = (at until at + cps.size).count { agreed[it] }
69             // Nothing to compare (a paragraph of pure punctuation) is not evidence either way.
70             heard[i] = cps.isEmpty() || hits >= HEARD * cps.size
71             at += cps.size
72         }
73 
74         // Drop unheard paragraphs only where they form a long enough run.
75         val keep = BooleanArray(paragraphs.size) { true }
76         var i = 0
77         while (i < paragraphs.size) {
78             if (heard[i]) { i++; continue }
79             var j = i
80             var chars = 0
81             while (j < paragraphs.size && (!heard[j] || textClean[j].isEmpty())) {
82                 chars += textClean[j].size
83                 j++
84             }
85             if (chars >= MIN_UNREAD_RUN) for (k in i until j) keep[k] = false
86             i = j
87         }
88 
89         val out = paragraphs.filterIndexed { k, _ -> keep[k] }
90         // A book with nothing recognisably narrated is a mismatched book; let
91         // the caller see that in the cues rather than returning nothing at all.
92         return out.ifEmpty { paragraphs }
93     }
94 
95     /** How much of the transcript found a home in the book, 0..1. For "is this the right book?". */
96     fun matchRate(cues: List<Cue>): Double =
97         if (cues.isEmpty()) 0.0 else cues.count { !it.text.startsWith(Subs.UNMATCHED) }.toDouble() / cues.size
