3 /**
4  * How text is reduced to something comparable before alignment.
5  *
6  * Whisper-tiny gets a great deal wrong - kana for kanji, は for わ, numerals
7  * either way round - and the book and the transcript disagree on punctuation
8  * and spacing everywhere. Alignment happens on a cleaned copy of both, and the
9  * result is mapped back onto the original text afterwards.
10  *
11  * Ported from ats/lang.py. Behaviour has to match character for character, or
12  * offsets computed here would not line up with the reference implementation.
13  */
14 sealed class Language {
15     protected open val translations: Map<Int, Int> = emptyMap()
16 
17     /** One-pass, one-to-one character substitution, then lowercased. */
18     fun translate(s: String): String {
19         if (translations.isEmpty()) return s.lowercase()
20         val out = StringBuilder(s.length)
21         s.codePoints().forEach { cp -> out.appendCodePoint(translations[cp] ?: cp) }
22         return out.toString().lowercase()
23     }
24 
25     open fun clean(s: String): String = translate(s)
26 
27     companion object {
28         /** Only Japanese has rules of its own; everything else is case-folded. */
29         fun of(code: String?): Language = if (code == "ja") Japanese else Plain
30     }
31 }
32 
33 object Plain : Language()
34 
35 object Japanese : Language() {
36     override val translations: Map<Int, Int> = buildMap {
37         // Katakana to hiragana: the recogniser picks between them at random.
38         for (i in 0 until 0x56) put(0x30A1 + i, 0x3041 + i)
39 
40         // Kanji numerals to digits. The digit string is shorter than the kanji
41         // one and wraps, which is how 十 and 拾 both come out as １.
42         val kansuu = "一二三四五六七八九十〇零壱弐参肆伍陸漆捌玖拾".toCodePoints()
43         val arabic = "１２３４５６７８９１００".toCodePoints()
44         kansuu.forEachIndexed { i, cp -> put(cp, arabic[i % arabic.size]) }
45 
46         // ASCII to full width, so "A" in one text meets "Ａ" in the other.
47         for (cp in 0x21 until 0x7F) put(cp, cp + 0xFEE0)
48 
49         // Particles written one way and pronounced another.
50         put('は'.code, 'わ'.code)
51         put('あ'.code, 'わ'.code)
52         put('お'.code, 'を'.code)
53         put('へ'.code, 'え'.code)
54     }
55 
56     // Everything that is not a letter or digit goes, except 。 when it leads a
57     // run - sentence ends are worth keeping as alignment landmarks.
58     private val noise = Regex("(?![。])[\\p{C}\\p{M}\\p{P}\\p{S}\\p{Z}\\sー々ゝ]+")
59 
60     // Collapse repeats to a single character: long vowels and stutters are
61     // exactly where the recogniser and the book disagree on length.
62     private val repeats = Regex("(.)(?=\\1+)")
63 
64     override fun clean(s: String): String =
65         repeats.replace(noise.replace(translate(s), ""), "")
66 }
