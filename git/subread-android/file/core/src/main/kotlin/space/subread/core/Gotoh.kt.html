19 /**
20  * Exact global alignment with affine gaps (Gotoh).
21  *
22  * Scores are the reference's, times ten so the arithmetic stays in integers
23  * and ties are exact: match 1, mismatch -0.6, gap open -0.8, gap extend -0.5.
24  *
25  * Costs rows * columns bytes for the traceback. That is fine for a chapter and
26  * hopeless for a book, which is why whole books go through [AnchoredAligner]
27  * and only the stretches between anchors end up here.
28  */
29 object Gotoh {
30     const val MATCH = 10
31     const val MISMATCH = -6
32     const val OPEN = -8
33     const val EXTEND = -5
34 
35     private const val NEG = Int.MIN_VALUE / 4
36 
37     // Traceback, packed per cell: which state each of the three states came from.
38     private const val M_FROM_X = 1       // bits 0-1: predecessor of M (0 = M)
39     private const val M_FROM_Y = 2
40     private const val X_FROM_X = 1 shl 2 // bits 2-3: predecessor of X (0 = M)
41     private const val X_FROM_Y = 2 shl 2
42     private const val Y_FROM_X = 1 shl 4 // bits 4-5: predecessor of Y (0 = M)
43     private const val Y_FROM_Y = 2 shl 4
44 
45     fun cells(n: Int, m: Int): Long = (n + 1L) * (m + 1L)
46 
47     fun align(target: IntArray, query: IntArray): Alignment {
48         val n = target.size
49         val m = query.size
50         val width = m + 1
51         val trace = ByteArray(Math.toIntExact(cells(n, m)))
52 
53         // M: target[i-1] against query[j-1]. X: target[i-1] against a gap.
54         // Y: query[j-1] against a gap. Two rows of each are enough for scores.
55         var pm = IntArray(width) { NEG }
56         var px = IntArray(width) { NEG }
57         var py = IntArray(width) { NEG }
58         var cm = IntArray(width)
59         var cx = IntArray(width)
60         var cy = IntArray(width)
61 
62         pm[0] = 0
63         for (j in 1..m) {
64             py[j] = OPEN + (j - 1) * EXTEND
65             trace[j] = (if (j == 1) 0 else Y_FROM_Y).toByte()
66         }
67 
68         for (i in 1..n) {
69             val row = i * width
70             cm[0] = NEG
71             cy[0] = NEG
72             cx[0] = OPEN + (i - 1) * EXTEND
73             trace[row] = (if (i == 1) 0 else X_FROM_X).toByte()
74 
75             val t = target[i - 1]
76             for (j in 1..m) {
77                 var bits = 0
78 
79                 // M: diagonal step out of whichever state was best.
80                 var best = pm[j - 1]
81                 if (px[j - 1] > best) { best = px[j - 1]; bits = M_FROM_X }
82                 if (py[j - 1] > best) { best = py[j - 1]; bits = M_FROM_Y }
83                 cm[j] = best + if (t == query[j - 1]) MATCH else MISMATCH
84 
85                 // X: step down. Extending is cheaper than reopening.
86                 var bx = pm[j] + OPEN
87                 var xb = 0
88                 if (px[j] + EXTEND > bx) { bx = px[j] + EXTEND; xb = X_FROM_X }
89                 if (py[j] + OPEN > bx) { bx = py[j] + OPEN; xb = X_FROM_Y }
90                 cx[j] = bx
91 
92                 // Y: step right.
93                 var by = cm[j - 1] + OPEN
94                 var yb = 0
95                 if (cx[j - 1] + OPEN > by) { by = cx[j - 1] + OPEN; yb = Y_FROM_X }
96                 if (cy[j - 1] + EXTEND > by) { by = cy[j - 1] + EXTEND; yb = Y_FROM_Y }
97                 cy[j] = by
98 
99                 trace[row + j] = (bits or xb or yb).toByte()
100             }
101 
102             val tm = pm; pm = cm; cm = tm
103             val tx = px; px = cx; cx = tx
104             val ty = py; py = cy; cy = ty
105         }
106 
107         var state = 0 // 0 = M, 1 = X, 2 = Y
108         var score = pm[m]
109         if (px[m] > score) { score = px[m]; state = 1 }
110         if (py[m] > score) { score = py[m]; state = 2 }
111 
112         // Walk back, noting a breakpoint whenever the kind of step changes.
113         val ti = IntList()
114         val qi = IntList()
115         var i = n
116         var j = m
117         ti.add(i); qi.add(j)
118         var last = -1
119         while (i > 0 || j > 0) {
120             if (last != -1 && last != state) { ti.add(i); qi.add(j) }
121             last = state
122             val bits = trace[i * width + j].toInt()
123             when (state) {
124                 0 -> { state = bits and 3; i--; j-- }
125                 1 -> { state = (bits shr 2) and 3; i-- }
126                 else -> { state = (bits shr 4) and 3; j-- }
127             }
128         }
129         if (ti.last() != 0 || qi.last() != 0) { ti.add(0); qi.add(0) }
130 
131         return Alignment(score, Coordinates(ti.reversed(), qi.reversed()))
132     }

134     /** What [coordinates] is worth, for checking an alignment made some other way. */
135     fun score(target: IntArray, query: IntArray, coordinates: Coordinates): Int {
136         var total = 0
137         for (k in 1 until coordinates.size) {
138             val dt = coordinates.target[k] - coordinates.target[k - 1]
139             val dq = coordinates.query[k] - coordinates.query[k - 1]
140             when {
141                 dt > 0 && dq > 0 -> {
142                     require(dt == dq) { "ragged aligned run at column $k" }
143                     val t0 = coordinates.target[k - 1]
144                     val q0 = coordinates.query[k - 1]
145                     for (d in 0 until dt) {
146                         total += if (target[t0 + d] == query[q0 + d]) MATCH else MISMATCH
147                     }
148                 }
149                 dt > 0 || dq > 0 -> total += OPEN + (maxOf(dt, dq) - 1) * EXTEND
150             }
151         }
152         return total
153     }
154 }

157 internal class IntList(capacity: Int = 64) {
158     private var data = IntArray(capacity)
159     var size = 0
160         private set
161 
162     fun add(v: Int) {
163         if (size == data.size) data = data.copyOf(size * 2)
164         data[size++] = v
165     }
166 
167     operator fun get(i: Int) = data[i]
168     fun last() = data[size - 1]
169     fun setLast(v: Int) { data[size - 1] = v }
170     fun toArray(): IntArray = data.copyOf(size)
171     fun reversed(): IntArray = IntArray(size) { data[size - 1 - it] }
