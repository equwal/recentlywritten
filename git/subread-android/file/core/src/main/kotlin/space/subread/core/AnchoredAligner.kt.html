3 /**
4  * Global alignment for inputs far too big for a full dynamic-programming table.
5  *
6  * The reference implementation hands Biopython one chapter at a time and needs
7  * a couple of gigabytes to do it, which is why it depends on first guessing
8  * which chapter of the book each chapter of audio is. A phone has neither the
9  * memory nor, often, chaptered audio. So this aligns the whole transcript
10  * against the whole book instead, the way diff tools handle large files:
11  *
12  *  1. Find stretches that occur exactly once in both sequences. In two copies
13  *     of the same book, those are overwhelmingly the same words in the same
14  *     place - anchors.
15  *  2. Keep the longest chain of anchors that is in order in both.
16  *  3. Between neighbouring anchors, recurse with shorter stretches; when what
17  *     is left is small, solve it exactly with [Gotoh].
18  *
19  * Text the narrator skipped, or an introduction the book lacks, simply has no
20  * anchors in it and comes out as one long gap. No chapter matching needed.
21  */
22 object AnchoredAligner {
23 
24     /** A stretch identical in both sequences. */
25     private class Run(var t: Int, var q: Int, var len: Int)
26 
27     fun align(
28         target: IntArray,
29         query: IntArray,
30         exactCells: Long = 4_000_000L,
31         firstK: Int = 14,
32         lastK: Int = 5,
33     ): Alignment {
34         val path = Path()
35         Solver(target, query, exactCells, lastK, path).solve(0, target.size, 0, query.size, firstK)
36         val coordinates = path.build(target.size, query.size)
37         return Alignment(Gotoh.score(target, query, coordinates), coordinates)
38     }

40     private class Solver(
41         val target: IntArray,
42         val query: IntArray,
43         val exactCells: Long,
44         val lastK: Int,
45         val path: Path,
46     ) {
47         fun solve(t0: Int, t1: Int, q0: Int, q1: Int, k: Int) {
48             val n = t1 - t0
49             val m = q1 - q0
50             if (n == 0 || m == 0) {
51                 path.moveTo(t1, q1)
52                 return
53             }
54             if (Gotoh.cells(n, m) <= exactCells) {
55                 val sub = Gotoh.align(target.copyOfRange(t0, t1), query.copyOfRange(q0, q1)).coordinates
56                 for (i in 0 until sub.size) path.moveTo(t0 + sub.target[i], q0 + sub.query[i])
57                 return
58             }
59 
60             var kk = k
61             var chain = emptyList<Run>()
62             while (kk >= lastK) {
63                 chain = chainOf(anchors(t0, t1, q0, q1, kk))
64                 if (chain.isNotEmpty()) break
65                 kk -= 3
66             }
67 
68             if (chain.isEmpty()) {
69                 // Two long stretches with nothing in common. There is no good
70                 // answer here, so take a cheap one: halve both and carry on.
71                 val tm = t0 + n / 2
72                 val qm = q0 + m / 2
73                 solve(t0, tm, q0, qm, lastK)
74                 solve(tm, t1, qm, q1, lastK)
75                 return
76             }
77 
78             var ct = t0
79             var cq = q0
80             for (run in chain) {
81                 solve(ct, run.t, cq, run.q, kk)
82                 path.moveTo(run.t + run.len, run.q + run.len)
83                 ct = run.t + run.len
84                 cq = run.q + run.len
85             }
86             solve(ct, t1, cq, q1, kk)
87         }

89         /** Every k-gram occurring exactly once on each side, joined into runs. */
90         private fun anchors(t0: Int, t1: Int, q0: Int, q1: Int, k: Int): List<Run> {
91             if (t1 - t0 < k || q1 - q0 < k) return emptyList()
92 
93             val inTarget = uniqueGrams(target, t0, t1, k)
94             val runs = ArrayList<Run>()
95             var current: Run? = null
96 
97             val inQuery = uniqueGrams(query, q0, q1, k)
98             // Walk the query in order so that consecutive hits extend a run.
99             for (qi in q0..q1 - k) {
100                 val h = hash(query, qi, k)
101                 if (inQuery.get(h) != qi) { current = null; continue }
102                 val ti = inTarget.get(h)
103                 if (ti < 0 || !same(ti, qi, k)) { current = null; continue }
104 
105                 val c = current
106                 if (c != null && ti == c.t + (qi - c.q)) {
107                     c.len = qi - c.q + k
108                 } else {
109                     current = Run(ti, qi, k).also { runs += it }
110                 }
111             }
112             return runs
113         }
114 
115         private fun same(ti: Int, qi: Int, k: Int): Boolean {
116             for (d in 0 until k) if (target[ti + d] != query[qi + d]) return false
117             return true
118         }
119 
120         /** k-gram hash to its position, or to DUPLICATE when it occurs more than once. */
121         private fun uniqueGrams(seq: IntArray, from: Int, to: Int, k: Int): LongIntMap {
122             val map = LongIntMap(to - from)
123             for (i in from..to - k) {
124                 val h = hash(seq, i, k)
125                 map.put(h, if (map.get(h) == LongIntMap.ABSENT) i else LongIntMap.DUPLICATE)
126             }
127             return map
128         }

130         private fun hash(seq: IntArray, at: Int, k: Int): Long {
131             var h = 1469598103934665603L
132             for (d in 0 until k) h = (h xor seq[at + d].toLong()) * 1099511628211L
133             return h
134         }

136         /** Longest chain of runs in order on both sides, overlaps trimmed. */
137         private fun chainOf(runs: List<Run>): List<Run> {
138             if (runs.isEmpty()) return runs
139             val byTarget = runs.sortedBy { it.t }
140 
141             // Longest increasing subsequence on query position (patience sort).
142             val tails = IntArray(byTarget.size)
143             val prev = IntArray(byTarget.size) { -1 }
144             var size = 0
145             for (i in byTarget.indices) {
146                 val q = byTarget[i].q
147                 var lo = 0
148                 var hi = size
149                 while (lo < hi) {
150                     val mid = (lo + hi) ushr 1
151                     if (byTarget[tails[mid]].q < q) lo = mid + 1 else hi = mid
152                 }
153                 if (lo > 0) prev[i] = tails[lo - 1]
154                 tails[lo] = i
155                 if (lo == size) size++
156             }
157             val picked = ArrayList<Run>(size)
158             var at = tails[size - 1]
159             while (at >= 0) { picked += byTarget[at]; at = prev[at] }
160             picked.reverse()
161 
162             // Neighbouring runs on different diagonals can overlap by a few
163             // characters; give the overlap to the earlier one.
164             val out = ArrayList<Run>(picked.size)
165             var endT = 0
166             var endQ = 0
167             for (r in picked) {
168                 val cut = maxOf(endT - r.t, endQ - r.q, 0)
169                 if (cut >= r.len) continue
170                 r.t += cut; r.q += cut; r.len -= cut
171                 out += r
172                 endT = r.t + r.len
173                 endQ = r.q + r.len
174             }
175             return out
176         }
177     }

179     /**
180      * Collects breakpoints, merging consecutive steps of one kind so the result
181      * has the same canonical shape Biopython produces.
182      */
183     private class Path {
184         private val ts = IntList()
185         private val qs = IntList()
186         private var lastKind = 0
187 
188         init { ts.add(0); qs.add(0) }
189 
190         fun moveTo(t: Int, q: Int) {
191             val pt = ts.last()
192             val pq = qs.last()
193             if (t == pt && q == pq) return
194             val dt = t - pt
195             val dq = q - pq
196             require(dt >= 0 && dq >= 0) { "path went backwards: ($pt,$pq) -> ($t,$q)" }
197             if (dt > 0 && dq > 0 && dt != dq) {
198                 // A diagonal and a gap in one step: split at the corner.
199                 val d = minOf(dt, dq)
200                 moveTo(pt + d, pq + d)
201                 moveTo(t, q)
202                 return
203             }
204             val kind = if (dt > 0 && dq > 0) 1 else if (dt > 0) 2 else 3
205             if (kind == lastKind && ts.size > 1) { ts.setLast(t); qs.setLast(q) }
206             else { ts.add(t); qs.add(q) }
207             lastKind = kind
208         }
209 
210         fun build(n: Int, m: Int): Coordinates {
211             moveTo(n, m)
212             return Coordinates(ts.toArray(), qs.toArray())
213         }
214     }

217 /** Open-addressing Long -> Int map. A HashMap<Long, Int> of a whole book boxes millions of objects. */
218 internal class LongIntMap(expected: Int) {
219     private val capacity = Integer.highestOneBit(maxOf(16, expected * 2) - 1) shl 1
220     private val mask = capacity - 1
221     private val keys = LongArray(capacity)
222     private val values = IntArray(capacity) { ABSENT }
223 
224     private fun slot(key: Long): Int {
225         var i = (key xor (key ushr 32)).toInt() and mask
226         while (values[i] != ABSENT && keys[i] != key) i = (i + 1) and mask
227         return i
228     }
229 
230     fun get(key: Long): Int = values[slot(key)]
231 
232     fun put(key: Long, value: Int) {
233         val i = slot(key)
234         keys[i] = key
235         values[i] = value
236     }
237 
238     companion object {
239         const val ABSENT = -1
240         const val DUPLICATE = -2
241     }
