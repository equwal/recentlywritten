31     fun epub(input: InputStream): List<String> {
32         val entries = LinkedHashMap<String, ByteArray>()
33         ZipInputStream(input).use { zip ->
34             while (true) {
35                 val e = zip.nextEntry ?: break
36                 if (!e.isDirectory) entries[e.name] = zip.readBytes()
37             }
38         }
39 
40         val order = spine(entries) ?: entries.keys.filter(::isPage).sorted()
41         return order.flatMap { path -> entries[path]?.let(::page) ?: emptyList() }
42     }

44     /**
45      * The cover of an epub: the image with "cover" in its name, or else the
46      * largest image. This works on malformed epubs too, where the package file
47      * does not name a cover. Null for a book that is not an epub.
48      */
49     fun cover(input: InputStream): ByteArray? {
50         var best: ByteArray? = null
51         var named: ByteArray? = null
52         ZipInputStream(input).use { zip ->
53             while (true) {
54                 val e = zip.nextEntry ?: break
55                 val name = e.name.lowercase()
56                 if (e.isDirectory || !(name.endsWith(".jpg") || name.endsWith(".jpeg") || name.endsWith(".png") || name.endsWith(".webp"))) continue
57                 val bytes = zip.readBytes()
58                 if (bytes.size > (best?.size ?: 1023)) best = bytes
59                 if ("cover" in name.substringAfterLast('/') && bytes.size > (named?.size ?: 1023)) named = bytes
60             }
61         }
62         return named ?: best
63     }

68     /** Document paths in spine order, or null when the package file cannot be followed. */
69     private fun spine(entries: Map<String, ByteArray>): List<String>? {
70         val container = entries["META-INF/container.xml"] ?: return null
71         val opfPath = Jsoup.parse(String(container), "", Parser.xmlParser())
72             .selectFirst("rootfile")?.attr("full-path")?.takeIf { it.isNotEmpty() } ?: return null
73         val opf = Jsoup.parse(String(entries[opfPath] ?: return null), "", Parser.xmlParser())
74         val base = opfPath.substringBeforeLast('/', "")
75 
76         val hrefs = opf.select("manifest > item").associate { it.attr("id") to it.attr("href") }
77         val paths = opf.select("spine > itemref")
78             // linear="no" is auxiliary content (notes, answers) outside the reading order.
79             .filter { it.attr("linear") != "no" }
80             .mapNotNull { hrefs[it.attr("idref")] }
81             .map { resolve(base, java.net.URLDecoder.decode(it.substringBefore('#'), "UTF-8")) }
82             .filter { it in entries }
83         return paths.ifEmpty { null }
84     }

86     private fun resolve(base: String, href: String): String {
87         val parts = ArrayDeque(if (base.isEmpty()) emptyList() else base.split('/'))
88         for (seg in href.split('/')) when (seg) {
89             "", "." -> {}
90             ".." -> parts.removeLastOrNull()
91             else -> parts.addLast(seg)
92         }
93         return parts.joinToString("/")
94     }

96     private fun page(bytes: ByteArray): List<String> {
97         // An epub page is XHTML, so read it as XML first. To an HTML parser a
98         // self-closing <title/> never closes, and it takes the whole page as the
99         // title text. The HTML parser stays as the fallback for malformed pages.
100         val source = String(bytes, Charsets.UTF_8)
101         val xml = Jsoup.parse(source, "", Parser.xmlParser())
102         val doc = if (xml.selectFirst("body") != null) xml else Jsoup.parse(source)
103         // Furigana would otherwise be read twice: once as kanji, once as kana.
104         doc.select("rt, rp").remove()
105         val body = doc.selectFirst("body") ?: doc
106         val blocks = body.select(BLOCKS)
107             // A quote holding paragraphs would yield its text twice; keep the leaves.
108             .filter { it.select(BLOCKS).size == 1 }
109         val parts = if (blocks.isEmpty()) listOf(body) else blocks
110         return parts.map { it.text().trim() }.filter { it.isNotEmpty() }
111     }

115     /** An Aozora Bunko download: a zip holding one Shift_JIS text with ruby markup. */
116     fun aozoraZip(input: InputStream): List<String> {
117         ZipInputStream(input).use { zip ->
118             while (true) {
119                 val e = zip.nextEntry ?: break
120                 if (e.name.endsWith(".txt", true)) return aozora(decode(zip.readBytes()))
121             }
122         }
123         return emptyList()
124     }

126     private val annotation = Regex("［＃[^］]*］")
127     private val ruby = Regex("《[^》]*》")
128 
129     fun aozora(raw: String): List<String> {
130         var lines = raw.replace("\r\n", "\n").split('\n')
131         // Title block, then a legend fenced by two dashed rules; then the text.
132         val rules = lines.indices.filter { lines[it].startsWith("-----") }
133         if (rules.size >= 2) lines = lines.drop(rules[1] + 1)
134         // The colophon.
135         val colophon = lines.indexOfFirst { it.startsWith("底本：") }
136         if (colophon >= 0) lines = lines.take(colophon)
137 
138         return lines.map { ruby.replace(annotation.replace(it, ""), "").replace("｜", "").trim() }
139             .filter { it.isNotEmpty() }
140     }

144     fun plain(text: String): List<String> {
145         if ("《" in text && "底本：" in text) return aozora(text)
146         return text.replace("\r\n", "\n").split('\n').map { it.trim() }.filter { it.isNotEmpty() }
147     }

149     /** UTF-8 unless it plainly is not, then Shift_JIS - the two a text file here will be. */
150     private fun decode(bytes: ByteArray): String {
151         val utf8 = String(bytes, Charsets.UTF_8)
152         if ('�' !in utf8) return utf8.removePrefix("﻿")
153         return String(bytes, Charset.forName("Shift_JIS"))
154     }
