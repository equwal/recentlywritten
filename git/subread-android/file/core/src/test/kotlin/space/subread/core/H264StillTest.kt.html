65             assertEquals(71, hashes.size)

113     /** One key frame of a test card. Skips the test on a machine with no ffmpeg that can encode H.264. */
114     private fun encodeKeyFrame(dir: File, profile: String): ByteArray {
115         val cabac = if (profile == "main") arrayOf("-coder", "cabac") else emptyArray()
116         val made = listOf("libx264" to profile, "libopenh264" to profile.replace("baseline", "constrained_baseline")).any { (encoder, name) ->
117             runCatching {
118                 run(dir, "ffmpeg", "-v", "error", "-y", "-f", "lavfi", "-i", "testsrc=size=320x240:rate=1", "-frames:v", "1",
119                     "-c:v", encoder, "-profile:v", name, *cabac, "-pix_fmt", "yuv420p", "-f", "h264", "key.h264").first == 0
120             }.getOrDefault(false)
121         }
122         assumeTrue("no ffmpeg with an H.264 encoder on the PATH", made)
123         return File(dir, "key.h264").readBytes()
124     }

126     private fun run(dir: File, vararg command: String): Triple<Int, String, String> {
127         val out = File(dir, "stdout.txt")
128         val err = File(dir, "stderr.txt")
129         val process = ProcessBuilder(*command).directory(dir).redirectOutput(out).redirectError(err).start()
130         return Triple(process.waitFor(), out.readText(), err.readText())
131     }

133     /** NAL units of an Annex B stream, without their start codes. */
134     private fun split(stream: ByteArray): List<ByteArray> {
135         val starts = (0..stream.size - 3).filter { stream[it].toInt() == 0 && stream[it + 1].toInt() == 0 && stream[it + 2].toInt() == 1 }
136         return starts.mapIndexed { k, at ->
137             var end = if (k + 1 < starts.size) starts[k + 1] else stream.size
138             while (end > at + 3 && stream[end - 1].toInt() == 0) end--
139             stream.copyOfRange(at + 3, end)
140         }
141     }
