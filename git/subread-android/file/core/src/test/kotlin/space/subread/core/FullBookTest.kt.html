15  * A whole audiobook against a whole book, in one pass - what the phone does.

18 class FullBookTest {
19     @Test
20     fun aNineteenHourBookAlignsInOneGo() {
21         val dir = javaClass.classLoader.getResource("golden-local/full")?.let { File(it.toURI()) }
22         val files = dir?.listFiles { f -> f.extension == "json" }.orEmpty()
23         assumeTrue("no whole-book fixtures on this machine", files.isNotEmpty())
24 
25         for (file in files) {
26             val json = Json.parseToJsonElement(file.readText()).jsonObject
27             val language = Language.of(json.getValue("language").jsonPrimitive.content)
28             val transcript = json.getValue("transcript").jsonArray.map {
29                 val o = it.jsonObject
30                 TranscriptSegment(
31                     o.getValue("text").jsonPrimitive.content,
32                     o.getValue("start").jsonPrimitive.double,
33                     o.getValue("end").jsonPrimitive.double,
34                 )
35             }
36             val paragraphs = json.getValue("paragraphs").jsonArray.map { it.jsonPrimitive.content }
37 
38             System.gc()
39             val before = Runtime.getRuntime().let { it.totalMemory() - it.freeMemory() }
40             val started = System.nanoTime()
41             val result = BookAligner.align(transcript, paragraphs, language)
42             val seconds = (System.nanoTime() - started) / 1e9
43             val after = Runtime.getRuntime().let { it.totalMemory() - it.freeMemory() }
44 
45             val rate = BookAligner.matchRate(result.cues)
46             val longest = result.cues.maxOf { it.text.length }
47             println("%s: %d cues in %.1fs, +%d MB; %.1f%% matched; %d of %d paragraphs dropped; longest cue %d chars"
48                 .format(file.nameWithoutExtension, result.cues.size, seconds, (after - before) shr 20,
49                     rate * 100, result.paragraphsDropped, paragraphs.size, longest))
50 
51             assertEquals(transcript.size, result.cues.size)
52             assertTrue("match rate $rate", rate > 0.95)
53             // A cue swallowing pages of text is the signature of unread text glued on.
54             assertTrue("a $longest-character cue", longest < 400)
55             assertTrue("took ${seconds}s", seconds < 120)
56         }
57     }
