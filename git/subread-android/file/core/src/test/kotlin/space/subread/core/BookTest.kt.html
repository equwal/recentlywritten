55     fun textThatWasNotNarratedIsLeftOut() {
56         val c = case("neko_003_006")
57         val other = case("neko_137_140").paragraphs      // same author, different chapter
58         val front = other.take(6)
59         // Long enough to count as a passage; a stray line or two is kept on purpose.
60         val insert = other.drop(40).take(12)
61         assertTrue(insert.sumOf { c.language.clean(it).length } > 400)
62         val back = other.takeLast(8)
63         assertTrue(front.sumOf { it.length } > 500 && back.sumOf { it.length } > 300)
64 
65         val clean = BookAligner.align(c.transcript, c.paragraphs, c.language)
66         val mid = c.paragraphs.size / 2
67         val padded = front + c.paragraphs.take(mid) + insert + c.paragraphs.drop(mid) + back
68         val got = BookAligner.align(c.transcript, padded, c.language)
69 
70         assertEquals(clean.paragraphsUsed, got.paragraphsUsed)
71         assertEquals(front.size + insert.size + back.size + clean.paragraphsDropped, got.paragraphsDropped)
72         assertEquals(clean.cues.map { it.text }, got.cues.map { it.text })
73         val junk = (front + insert + back).map { it.take(12) }
74         got.cues.forEach { cue -> junk.forEach { assertFalse("junk in: ${cue.text}", it in cue.text) } }

92     private fun epub(vararg files: Pair<String, String>): ByteArray {
93         val out = ByteArrayOutputStream()
94         ZipOutputStream(out).use { zip ->
95             for ((name, body) in files) {
96                 zip.putNextEntry(ZipEntry(name))
97                 zip.write(body.toByteArray())
98                 zip.closeEntry()
99             }
100         }
101         return out.toByteArray()
102     }

133     /**
134      * Found on a real device: two thirds of a real epub disappeared, and the
135      * subtitles got text from the wrong passage. An HTML parser takes a
136      * self-closing <title/> as an open title, reads the start of the page as
137      * its text, and the first paragraphs are lost. The page must be large for
138      * the defect to show: a small page parses correctly.
139      */

141     fun aSelfClosingTitleDoesNotSwallowTheStartOfALargePage() {
142         val paragraphs = (1..120).map { n -> "Paragraph $n. " + "word ".repeat(40).trim() }
143         val bytes = epub("a.xhtml" to
144             "<?xml version='1.0' encoding='UTF-8'?><html xmlns='http://www.w3.org/1999/xhtml'><head><title/>" +
145             "<link rel='stylesheet' href='s.css' type='text/css'/></head><body><span id='x'>" +
146             paragraphs.joinToString("") { "<p class='p1'>$it</p>" } + "</span></body></html>")
147         assertEquals(paragraphs, BookText.read(ByteArrayInputStream(bytes), "x.epub"))
148     }
