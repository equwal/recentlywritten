111     /**
112      * The whole-book aligner never sees the full table, so it cannot promise
113      * the optimum - but on two copies of the same text it should land on it,
114      * or within a whisker. Forced through tiny blocks here so that even these
115      * chapter-sized fixtures exercise anchoring, chaining and stitching.
116      */
117     @Test
118     fun anchoredAlignmentIsAsGoodAsExact() = cases.forEach { c ->
119         val target = c.paragraphsClean.joinToString("").toCodePoints()
120         val query = c.transcriptClean.joinToString("").toCodePoints()
121         val exact = Math.round(c.score * 10).toInt()
122         // Production block size must be essentially exact. The tiny sizes are a
123         // stress test of the stitching: every join that lands inside a long gap
124         // pays to reopen it, so they are allowed to fall a little short.
125         for ((cells, floor) in listOf(4_000_000L to 0.999, 40_000L to 0.98, 2_500L to 0.97)) {
126             val got = AnchoredAligner.align(target, query, exactCells = cells)
127             assertEquals("$c: path must span both sequences", target.size, got.coordinates.target.last())
128             assertEquals(query.size, got.coordinates.query.last())
129             val ratio = got.score.toDouble() / exact
130             println("%-14s blocks<=%-8d anchored %6d / exact %6d (%.2f%%)".format(c.name, cells, got.score, exact, ratio * 100))
131             assertTrue("$c: anchored score ${got.score} vs exact $exact", ratio >= floor)
132         }
133     }

150     }
