14 def parse_snippet(body: str):
15     """Reference parser for VSCode snippet text.
16 
17     Return the source text with placeholders restored, and the list of tab
18     stop numbers in order. This is the inverse of convert_placeholders.
19     """
20     out, stops, i = [], [], 0
21     while i < len(body):
22         c = body[i]
23         if c == "\\" and i + 1 < len(body) and body[i + 1] in "\\$}":
24             out.append(body[i + 1])
25             i += 2
26         elif body.startswith("${", i):
27             i += 2
28             inner = []
29             while body[i] != "}":
30                 if body[i] == "\\" and body[i + 1] in "\\$}":
31                     inner.append(body[i + 1])
32                     i += 2
33                 else:
34                     inner.append(body[i])
35                     i += 1
36             i += 1
37             num, _, name = "".join(inner).partition(":")
38             stops.append(num)
39             out.append(f"<+{name}+>" if name else PLACEHOLDER)
40         else:
41             assert c not in "$}", f"unescaped {c!r} at {i} in {body!r}"
42             out.append(c)
43             i += 1
44     return "".join(out), stops
