+static void
+pipeuri(char *cmd, char *msg, char *uri)
+{
+	FILE *sel;
+
+	if ((sel = popen(cmd, "w")) == NULL) {
+		diag("URI not %s\n", msg);
+		return;
+	}
+
+	fputs(uri, sel);
+	pclose(sel);
+	diag("%s \"%s\"", msg, uri);
+}
