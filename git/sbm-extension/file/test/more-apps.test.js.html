9 // The links in the "More apps" section of the options page: the name, the
10 // address and the attributes of each.
11 function moreApps() {
12   const section = read("options.html").match(/<section id="more-apps">([\s\S]*?)<\/section>/);
13   assert.ok(section, "options.html has no More apps section");
14   return [...section[1].matchAll(/<a ([^>]*)>([^<]*)<\/a>/g)].map(([, attrs, name]) => ({
15     name,
16     attrs,
17     href: (attrs.match(/href="([^"]*)"/) || [])[1],
18   }));
19 }

21 test("More apps links to the other apps with https, each in a new tab", () => {
22   const apps = moreApps();
23   assert.ok(apps.length > 1);
24   for (const app of apps) {
25     assert.match(app.href, /^https:\/\/[^?#]+$/, app.name); // no tracking parameters
26     assert.match(app.attrs, /target="_blank"/, app.name);
27     assert.match(app.attrs, /rel="noopener"/, app.name);
28   }
29   assert.equal(new Set(apps.map((app) => app.href)).size, apps.length);
30   assert.equal(apps.at(-1).href, "https://recentlywritten.com/projects.html");
31 });
