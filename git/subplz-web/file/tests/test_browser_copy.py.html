tests/test_browser_copy.py (3288 bytes)

1 """During the beta, a job in the browser also sends a copy of its audio and its
2 book to the server, as the privacy and user data policy says (terms.html,
3 "Input data kept for debugging"). The server keeps the copy with the
4 subtitles, in the folder of the job. A job in the browser never spends a
5 credit, with or without the copy.
6 """

15 BOOK = {
16     "audio_filename": "copy.m4b", "audio_bytes": 11,
17     "text_filename": "copy.epub", "language": "en",
18 }
19 SRT = "1\n00:00:00,000 --> 00:00:01,000\nhello\n\n"

22 def begin(client) -> dict:
23     r = client.post("/api/local/jobs", json=BOOK)
24     assert r.status_code == 200
25     return r.json()

28 def send_copy(client, job_id: str, files: list[tuple[str, bytes]]):
29     return client.post(
30         f"/api/local/jobs/{job_id}/files",
31         files=[("files", (name, data, "application/octet-stream")) for name, data in files],
32     )

39 def test_a_browser_job_keeps_audio_book_and_subtitles_together(client):
40     job = begin(client)
41     r = send_copy(client, job["id"], [("copy.m4b", b"audio"), ("copy.epub", b"book")])
42     assert r.status_code == 200 and r.json() == {"saved": 2}
43 
44     done = client.post(f"/api/local/jobs/{job['id']}/finish",
45                        json={"srt": SRT, "filename": "copy.en.srt"})
46     assert done.status_code == 200
47 
48     root = Paths.for_job(job["id"]).root
49     assert (root / "input" / "copy.m4b").read_bytes() == b"audio"
50     assert (root / "input" / "copy.epub").read_bytes() == b"book"
51     assert (root / "copy.en.srt").read_bytes() == SRT.encode("utf-8")

54 def test_a_browser_job_and_its_copy_spend_no_credit(client, monkeypatch):

67 def test_a_file_name_cannot_leave_the_folder_of_the_job(client):
