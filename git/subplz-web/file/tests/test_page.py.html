8 def test_the_javascript_switch_is_on_the_first_screen_and_on(client):
9     page = client.get("/").text
10     switch = re.search(r'<input[^>]*\bid="in-browser"[^>]*>', page)
11     assert switch, "the page has no in-browser switch"
12     tag = switch.group(0)
13     # On when the page opens. autocomplete="off" stops a browser from
14     # bringing back an "off" from an earlier visit.
15     assert re.search(r"\schecked\b", tag) and 'autocomplete="off"' in tag
16 
17     # Above the drop zone, so that it shows before a file is chosen.
18     dropzone = page.index('id="dropzone"')
19     assert switch.start() < dropzone
20     assert "JavaScript" in page[switch.start():dropzone]
