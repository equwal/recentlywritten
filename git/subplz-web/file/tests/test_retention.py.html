15 def age(job_id: str, **delta) -> None:
16     with SessionLocal() as s:
17         job = s.get(Job, job_id)
18         job.created_at = utcnow() - timedelta(**delta)
19         job.finished_at = utcnow() - timedelta(**delta)
20         s.commit()

23 def stage_inputs(job_id: str):
24     paths = Paths.for_job(job_id)
25     paths.create()
26     (paths.inp / "book.m4b").write_bytes(b"audio")
27     return paths

35 def test_a_failed_job_keeps_its_upload_for_a_day_and_no_longer(client):
36     job_id = make_job(client, JobStatus.failed)
37     paths = stage_inputs(job_id)

39     age(job_id, hours=23)
