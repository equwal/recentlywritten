1 """Subrep Pro: the plan of the Subrep desktop app, sold with the accounts of this site.
2 
3   yearly   $9.99 a year (first on the page)
4   monthly  $0.99 a month
5 
6 A Stripe subscription makes the account Pro. The desktop app does not have the
7 cookie of the browser: the page gives a link command with the account id and a
8 refresh key, and the app sends the two to /api/subrep/refresh. The answer is a
9 licence token that the app checks offline with the public key in its build.
10 A Subrep plan is not a book plan: an account can have both, or one of them.
11 """

45 def b64d(text: str) -> bytes:
46     return base64.urlsafe_b64decode(text + "=" * (-len(text) % 4))

49 def verify(token: str) -> dict:
50     """The check of the desktop app (verify_token in subrep/licensing.py)."""
51     payload, sig = token.split(".")
52     PUBLIC_KEY.verify(b64d(sig), b64d(payload))
53     return json.loads(b64d(payload))

56 def sub_object(account, plan_id, status="active", period_end=None, sub_id=None,
57                price_key=None) -> dict:
58     item: dict = {"current_period_end": period_end or int(time.time()) + 30 * DAY}
59     if price_key:
60         item["price"] = {"id": f"price_{price_key}", "lookup_key": price_key}
61     return {
62         "id": sub_id or f"sub_subrep_{account}", "object": "subscription",
63         "status": status, "customer": f"cus_{account}",
64         "metadata": {"account_id": account, "plan_id": plan_id},
65         "items": {"data": [item]},
66     }

359 @hsettings(max_examples=60, deadline=None)
360 @given(claims=claims_st, where=st.integers(min_value=0))
361 def test_a_changed_token_fails_the_check(claims, where):
362     payload, sig = subrep.sign(claims).split(".")
363     raw = bytearray(b64d(payload))
364     raw[where % len(raw)] ^= 1
365     changed = base64.urlsafe_b64encode(bytes(raw)).rstrip(b"=").decode() + "." + sig
366     with pytest.raises(InvalidSignature):
367         verify(changed)

374 def test_the_key_tool_makes_one_key_and_never_replaces_it(tmp_path, monkeypatch, capsys):
375     import importlib.util
376     from pathlib import Path
377 
378     spec = importlib.util.spec_from_file_location(
379         "subrep_key", Path(__file__).resolve().parent.parent / "tools" / "subrep_key.py")
380     tool = importlib.util.module_from_spec(spec)
381     spec.loader.exec_module(tool)
382     env = tmp_path / ".env"
383     env.write_text("SUBPLZ_WEB_OTHER=1", encoding="utf-8")  # no newline at the end
384     monkeypatch.setattr(tool, "ROOT", tmp_path)
385     monkeypatch.setattr(tool, "ENV", env)
386 
387     assert tool.main() == 0
388     public = capsys.readouterr().out.strip()
389     text = env.read_text(encoding="utf-8")
390     assert text.startswith("SUBPLZ_WEB_OTHER=1\nSUBPLZ_WEB_SUBREP_LICENSE_KEY=")
391     saved = text.split("SUBPLZ_WEB_SUBREP_LICENSE_KEY=")[1].strip()
392     monkeypatch.setattr(settings, "subrep_license_key", saved)
393     assert subrep.public_key() == public  # the printed key belongs to the saved key
394 
395     assert tool.main() == 0
396     assert capsys.readouterr().out.strip() == public
397     assert env.read_text(encoding="utf-8") == text
