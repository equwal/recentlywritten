1 """Subscriptions: a plan renews each month until the customer cancels it.
2 
3   capped     a number of books a month. The books of one month do not carry
4              over to the next. A server job spends the daily free credit
5              first (it ends sooner), then the books of the plan, then free
6              and bought credits.
7   unlimited  no credit is spent while the plan is active.
8 
9 Stripe tells the server about a plan with customer.subscription.* events. A
10 new month (a new current_period_end) starts the count again. The lookup key
11 of the price names the plan, so a switch in the customer portal changes the
12 plan. The Stripe account also sells other products: a subscription that is
13 not for a plan of this site is ignored.
14 """

48 def subscription_event(kind, account, plan_id, status="active", period_end=None,
49                        price_key=None):
50     """`plan_id` goes in the checkout metadata; `price_key` is the lookup key
51     of the price, which the portal changes when the customer switches plans."""
52     end = period_end or int(time.time()) + 30 * 86400
53     item = {"current_period_end": end}
54     if price_key is not None:
55         item["price"] = {"id": f"price_{price_key}", "lookup_key": price_key}
56     return {
57         "id": f"evt_{kind}_{time.time_ns()}",
58         "type": f"customer.subscription.{kind}",
59         "data": {"object": {
60             "id": f"sub_{account}", "object": "subscription", "status": status,
61             "customer": f"cus_{account}",
62             "metadata": {"account_id": account, "plan_id": plan_id},
63             "items": {"data": [item]},
64         }},
65     }

123 def test_the_books_of_a_month_do_not_carry_over(client):
124     acct = account_id(client)
125     end = int(time.time()) + 30 * 86400
126     post_webhook(client, subscription_event("created", acct, "month10", period_end=end))
127     for _ in range(10):
128         assert start(client, make_job(client)).status_code == 200
129     assert start(client, make_job(client)).status_code == 402

132     post_webhook(client, subscription_event("updated", acct, "month10", period_end=end))
133     assert account(client)["plan_credits"] == 0

137                                             period_end=end + 30 * 86400))
138     assert account(client)["plan_credits"] == 10

200 def test_the_plan_prices_are_made_once(monkeypatch):

202 
203     prices = []  # what Stripe has
204 
205     def price_list(lookup_keys, **kw):
206         return FakeStripeObject({"data": [p for p in prices if p["lookup_key"] in lookup_keys]})

208     def price_create(**kw):
209         price = {"id": f"price_{kw['lookup_key']}", "lookup_key": kw["lookup_key"],
210                  "unit_amount": kw["unit_amount"], "currency": kw["currency"],
211                  "recurring": kw["recurring"]}

216     monkeypatch.setattr(stripe.Price, "create", price_create)

221     assert [p["unit_amount"] for p in prices] == [499, 999, 5000]
