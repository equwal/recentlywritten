173 @contextmanager
174 def before_the_first_write(action):
175     """Run `action` in another session, and commit it, just before the first
176     INSERT or UPDATE of caption_balances.
177 
178     Postgres lets another transaction commit between a SELECT and a later
179     write of a transaction (READ COMMITTED). SQLite lets it only while the
180     transaction has not written yet, so a test must call add in a new
181     transaction to see that race on SQLite.
182     """
183     engine = SessionLocal.kw["bind"]
184     results = []
185 
186     def before(conn, cursor, statement, parameters, context, executemany):
187         if not results and statement.startswith(
188                 ("INSERT INTO caption_balances", "UPDATE caption_balances")):
189             results.append(None)  # the other session writes too
190             with SessionLocal() as other:
191                 results[0] = action(other)
192                 other.commit()
193 
194     event.listen(engine, "before_cursor_execute", before)
195     try:
196         yield results
197     finally:
198         event.remove(engine, "before_cursor_execute", before)

203        theirs=st.integers(1, 100), spend=st.booleans())
204 def test_a_credit_loses_no_credit_or_spend_of_another_session(start, ours, theirs, spend):
205     """Another session gives or spends `theirs` seconds while add gives `ours`.
206     With `start` None, the account has no row yet."""
207     with SessionLocal() as s:
208         acct = accounts.new_account(s).id
209         s.commit()
210         if start is not None:
211             captions.add(s, acct, start)
212             s.commit()
213 
214     def other(session):
215         if spend:
216             return captions.spend(session, acct, theirs)
217         return captions.add(session, acct, theirs)
218 
219     with SessionLocal() as s, before_the_first_write(other) as done:
220         captions.add(s, acct, ours)
221         s.commit()
222 
223     assert done
224     spent = theirs if spend and done == [True] else 0
225     credited = (start or 0) + ours + (0 if spend else theirs)
226     with SessionLocal() as s:
227         row = s.get(captions.CaptionBalance, acct)
228         assert (row.seconds_left, row.seconds_used) == (credited - spent, spent)

245 def test_the_request_is_the_openai_form(monkeypatch):
246     """transcribe() against a local server that reads the form as the service does."""
247     import json
248     import threading
249     from email.parser import BytesParser
250     from email.policy import HTTP
251     from http.server import BaseHTTPRequestHandler, HTTPServer
252 
253     seen = {}
254 
255     class Service(BaseHTTPRequestHandler):
256         def do_POST(self):
257             body = self.rfile.read(int(self.headers["Content-Length"]))
258             head = f"Content-Type: {self.headers['Content-Type']}\r\n\r\n".encode()
259             form = BytesParser(policy=HTTP).parsebytes(head + body)
260             for part in form.iter_parts():
261                 seen[part.get_param("name", header="content-disposition")] = part.get_payload(decode=True)
262             seen["auth"] = self.headers["Authorization"]
263             out = json.dumps({"text": " 今日は "}).encode()
264             self.send_response(200)
265             self.send_header("Content-Length", str(len(out)))
266             self.end_headers()
267             self.wfile.write(out)
268 
269         def log_message(self, *args):
270             pass
271 
272     server = HTTPServer(("127.0.0.1", 0), Service)
273     threading.Thread(target=server.handle_request, daemon=True).start()
274     monkeypatch.setattr(settings, "caption_api_url", f"http://127.0.0.1:{server.server_port}/")
275     monkeypatch.setattr(settings, "caption_api_key", "gsk_test")
276     pcm = bytes(range(256)) * 10
277 
278     assert captions.transcribe(pcm, "ja") == "今日は"
279     server.server_close()
280     assert seen["auth"] == "Bearer gsk_test"
281     assert seen["model"] == b"openai/whisper-large-v3-turbo"
282     assert seen["language"] == b"ja"
283     assert seen["file"] == captions.wav(pcm)
