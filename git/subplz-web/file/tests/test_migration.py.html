46 PROBE = """
47 from backend.db import init_db, SessionLocal, Account, Job
48 init_db(); init_db()  # twice: a restart must be a no-op
49 with SessionLocal() as s:
50     job = s.get(Job, "job_old")
51     acct = s.get(Account, "acct_old")
52     print(job.tier, job.credit_spent, job.status.value, acct.merged_into, acct.signed_in)

56 def test_old_database_is_upgraded_in_place(tmp_path):
57     db = tmp_path / "subplz.db"
58     con = sqlite3.connect(db)
59     con.executescript(OLD_SCHEMA)
60     con.commit()
61     con.close()
62 
63     env = {**os.environ, "SUBPLZ_WEB_DATA_DIR": str(tmp_path)}
64     root = Path(__file__).resolve().parent.parent
65     out = subprocess.run([sys.executable, "-c", PROBE], cwd=root, env=env,
66                          capture_output=True, text=True, timeout=120)
67     assert out.returncode == 0, out.stderr
68     # Old rows read back with the new columns defaulted, nothing lost.
69     assert out.stdout.split() == ["free", "0", "succeeded", "None", "False"]
70 
71     con = sqlite3.connect(db)
72     cols = lambda t: {r[1] for r in con.execute(f"PRAGMA table_info({t})")}  # noqa: E731
73     assert {"tier", "credit_spent"} <= cols("jobs")
74     assert {"merged_into", "subscription_id", "subscription_status",
75             "subscription_period_end"} <= cols("accounts")
76     tables = {r[0] for r in con.execute("SELECT name FROM sqlite_master WHERE type='table'")}
77     assert {"login_tokens", "purchases"} <= tables
78     # The unique email index has to exist, or two accounts could share one.
79     indexes = {r[1]: r[2] for r in con.execute("PRAGMA index_list(accounts)")}
80     assert indexes.get("ix_accounts_email") == 1
81     con.close()
