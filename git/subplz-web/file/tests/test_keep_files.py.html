19 def stage_inputs(job_id: str) -> tuple[Paths, list]:
20     """Put an audio file and a book where an upload puts them."""
21     paths = Paths.for_job(job_id)
22     paths.create()
23     audio = runner.staged_audio_path(job_id, "book.m4b")
24     book = runner.staged_text_path(job_id, "book.epub")
25     audio.write_bytes(b"audio")
26     book.write_bytes(b"book")
27     return paths, [audio, book]

30 def age(job_id: str, **delta) -> None:
31     with SessionLocal() as s:
32         job = s.get(Job, job_id)
33         job.created_at = utcnow() - timedelta(**delta)
34         job.finished_at = utcnow() - timedelta(**delta)
35         s.commit()
