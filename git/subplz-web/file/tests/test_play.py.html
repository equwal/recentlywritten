36 class FakeGooglePlay:
37     """Google Play as this server sees it: the purchases of the app, by token."""
38 
39     def __init__(self):
40         self.purchases: dict[str, dict] = {}
41         # The consumes that came after the server committed the ledger row.
42         self.consumed: list[str] = []
43         # The consumes that came before that commit. The fixture `google`
44         # fails each test that makes one: if the commit fails after the
45         # consume, the buyer paid, and Google closed the purchase, but the
46         # account has no hours.
47         self.early: list[str] = []
48         self.gets: list[tuple[str, str]] = []
49         self.down = False
50         self.fail_consume = 0
51         # The tokens of the purchases that Google refunded or charged back.
52         self.voided_tokens: list[str] = []
53 
54     def buy(self, token: str, account: str | None, product: str = "captions20",
55             state: int = 0, quantity: int | None = None, test: bool = False) -> dict:
56         """Add a purchase, as purchases.products.get returns it.
57 
58         `account` None is a purchase without an account id, for example with
59         a promo code in the Play Store app.
60         """
61         purchase = {"kind": "androidpublisher#productPurchase", "productId": product,
62                     "purchaseState": state, "consumptionState": 0,
63                     "orderId": f"GPA.3300-{next(_orders):04d}"}
64         if account is not None:
65             purchase["obfuscatedExternalAccountId"] = account
66         if quantity is not None:
67             purchase["quantity"] = quantity
68         if test:
69             purchase["purchaseType"] = 0
70         self.purchases[token] = purchase
71         return purchase
72 
73     def get(self, product_id: str, token: str) -> dict:
74         self.gets.append((product_id, token))
75         if self.down:
76             raise play.PlayError(502, play.NO_ANSWER)
77         if token not in self.purchases:
78             raise play.PlayError(400, UNKNOWN)
79         return dict(self.purchases[token])
80 
81     def consume(self, product_id: str, token: str) -> None:
82         if self.fail_consume:
83             self.fail_consume -= 1
84             raise play.PlayError(502, play.NO_ANSWER)
85         (self.consumed if ledger(token) else self.early).append(token)
86         self.purchases[token]["consumptionState"] = 1
87 
88     def voided(self) -> list[str]:
89         return list(self.voided_tokens)

167 def test_a_key_folder_that_the_server_cannot_open_stops_only_the_play_sales(
168         client, google, monkeypatch, tmp_path):
169     # The service user cannot open the folder of the key, so stat of the key
170     # fails with EACCES. The server runs Python 3.11. There, Path.is_file
171     # raises PermissionError for this error, and each GET /api/captions
172     # answered 500. The tests can run on a newer Python, so this test makes
173     # Path.is_file act as on Python 3.11. The test does not make the key file:
174     # os.path.isfile gives False for each stat error, EACCES or ENOENT.
175     key = str(tmp_path / "locked" / "play-key.json")
176     is_file = pathlib.Path.is_file
177 
178     def is_file_on_python_3_11(self, *args, **kwargs):
179         if str(self) == key:
180             raise PermissionError(errno.EACCES, "Permission denied", key)
181         return is_file(self, *args, **kwargs)
182 
183     monkeypatch.setattr(pathlib.Path, "is_file", is_file_on_python_3_11)
184     monkeypatch.setattr(settings, "play_service_account_file", key)
185     monkeypatch.setattr(settings, "caption_api_key", "dg_test")
186     assert_only_the_play_sales_stop(client, google)
187     main.housekeeping()  # The hourly pass does not fail.

416 class FakeHttp:
417     """Stands in for the signed session: records each request and gives the next answer."""
418 
419     def __init__(self):
420         self.answers: list = []
421         self.seen: list[tuple[str, str]] = []
422 
423     def __enter__(self):
424         return self
425 
426     def __exit__(self, *exc):
427         return False

429     def request(self, method: str, url: str, timeout: float | None = None):
430         assert timeout  # A request without a timeout can hang a worker.
431         self.seen.append((method, url))
432         answer = self.answers.pop(0)
433         if isinstance(answer, Exception):
434             raise answer
435         return answer

642 @hsettings(max_examples=50, deadline=None)
643 @given(buys=st.lists(st.tuples(st.sampled_from(sorted(captions.HOURS)), st.integers(1, 3),
644                                st.sampled_from(["me", "other", "none"])),
645                      min_size=1, max_size=4),
646        calls=st.lists(st.tuples(st.integers(0, 3), st.sampled_from([0, 1, 2]),
647                                 st.sampled_from([0, 1])),
648                       max_size=25))
649 def test_a_token_gives_its_hours_once_and_only_when_paid_for_this_account(buys, calls):
650     run = secrets.token_hex(6)  # The database lives across examples.
651     with SessionLocal() as s:
652         me, other = accounts.new_account(s), accounts.new_account(s)
653         s.commit()
654         ids = {"me": me.id, "other": other.id, "none": None}
655         expected, added = {}, {}
656         for i, state, consumed in calls:
657             if i >= len(buys):
658                 continue
659             product, quantity, owner = buys[i]
660             token = f"{run}.{i}"
661             google = {"purchaseState": state, "consumptionState": consumed,
662                       "productId": product, "quantity": quantity}
663             if ids[owner]:
664                 google["obfuscatedExternalAccountId"] = ids[owner]
665             try:
666                 if play.credit(s, me, product, token, google):
667                     added[token] = added.get(token, 0) + 1
668             except play.PlayError:
669                 pass
670             # Hours come only from a call that saw the purchase paid and not consumed.
671             if state == 0 and consumed == 0 and owner != "other":
672                 expected[token] = captions.HOURS[product] * HOUR * quantity
673         assert captions.seconds_left(s, ids["me"]) == sum(expected.values())
674         assert captions.seconds_left(s, ids["other"]) == 0
675         assert added == {token: 1 for token in expected}
676         assert s.query(play.PlayPurchase).filter_by(account_id=ids["me"]).count() == len(expected)

679 @hsettings(max_examples=50, deadline=None)
680 @given(bought=st.lists(st.sampled_from(sorted(captions.HOURS)), min_size=1, max_size=4),
681        spent=st.integers(0, 300 * HOUR),
682        voids=st.lists(st.lists(st.integers(0, 5), max_size=6), max_size=4))
683 def test_a_refund_takes_back_the_hours_once_and_never_below_zero(bought, spent, voids):
684     run = secrets.token_hex(6)  # The database lives across examples.
685     with SessionLocal() as s:
686         me = accounts.new_account(s)
687         s.commit()
688         for i, product in enumerate(bought):
689             assert play.credit(s, me, product, f"{run}.{i}", paid(product, me.id))
690         total = sum(captions.HOURS[product] * HOUR for product in bought)
691         used = min(spent, total)
692         assert captions.spend(s, me.id, used)
693         voided: set[int] = set()
694         taken = 0
695         for indexes in voids:  # An index past the purchases is a token that this server never saw.
696             taken += play.take_back(s, [f"{run}.{i}" for i in indexes])
697             voided |= {i for i in indexes if i < len(bought)}
698         back = sum(captions.HOURS[bought[i]] * HOUR for i in voided)
699         assert captions.seconds_left(s, me.id) == max(0, total - used - back)
700         assert taken == len(voided)
701         assert s.query(play.PlayPurchase).filter(
702             play.PlayPurchase.account_id == me.id,
703             play.PlayPurchase.voided_at.is_not(None)).count() == len(voided)
