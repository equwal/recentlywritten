66 def make_job(
67     client: TestClient, status: JobStatus = JobStatus.draft, with_files: bool = False
68 ) -> str:
69     """A job row owned by `client`, skipping the upload (which needs real audio)."""
70     acct = account_id(client)
71     with SessionLocal() as s:
72         job = Job(
73             account_id=acct, status=status, language="en", splitter="pysbd",
74             model="tiny", audio_filename="book.m4b", text_filename="book.epub",
75         )
76         s.add(job)
77         s.commit()
78         job_id = job.id

80         if with_files:
81             for kind, name in [("srt", "book.en.srt"), ("video", "book.en.mp4"),
82                                ("video_embedded", "book.en.mkv")]:
83                 src = Path(_TMP) / f"{job_id}-{name}"

88                                storage_key=key, size_bytes=10))
89             s.commit()
90     return job_id

103 def signed(payload: dict, secret: str = WEBHOOK_SECRET) -> tuple[bytes, dict]:
104     """A webhook body plus the header Stripe would have sent with it."""
105     body = json.dumps(payload).encode()
106     stamp = int(time.time())
107     digest = hmac.new(
108         secret.encode(), f"{stamp}.".encode() + body, hashlib.sha256
109     ).hexdigest()
110     return body, {
111         "stripe-signature": f"t={stamp},v1={digest}",
112         "content-type": "application/json",
113     }
