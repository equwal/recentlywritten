113 def signed(payload: dict, secret: str = WEBHOOK_SECRET) -> tuple[bytes, dict]:
114     """A webhook body plus the header Stripe would have sent with it."""
115     body = json.dumps(payload).encode()
116     stamp = int(time.time())
117     digest = hmac.new(
118         secret.encode(), f"{stamp}.".encode() + body, hashlib.sha256
119     ).hexdigest()
120     return body, {
121         "stripe-signature": f"t={stamp},v1={digest}",
122         "content-type": "application/json",
123     }
