50 @st.composite
51 def states(draw):
52     pool = draw(st.sets(names, max_size=8))
53     workers = {n: worker(draw(st.floats(0, 60 * MIN)), draw(st.sampled_from(["old", REF])),
54                          draw(st.sampled_from([AWS, HETZNER]))) for n in pool}
55     live = {n: draw(st.sampled_from(["busy", "idle"])) for n in pool if draw(st.booleans())}
56     registered = {n for n in pool if n in live or draw(st.booleans())}
57     draining = {n for n in registered if draw(st.booleans())}
58     idle_since = {n: NOW - draw(st.floats(0, 30 * MIN)) for n in live if live[n] == "idle"}
59     memory = Memory(workers=workers, registered=registered, idle_since=idle_since,
60                     draining=draining, cooldown=draw(st.sampled_from([{}, {"aws": NOW + 60}])))
61     offers = draw(st.lists(offer_st, max_size=4, unique=True))
62     cap = draw(st.sampled_from([math.inf, 0.0, 0.5, 5.0]))
63     return (draw(st.integers(0, 12)), live, memory, Config(max_workers=draw(st.integers(0, 6))),
64             offers, cap)

67 @settings(max_examples=500)
68 @given(states())
69 def test_invariants(state):
70     waiting, live, memory, config, offers, cap = state
71     d, after = decide(NOW, waiting, live, memory, config, REF, offers, cap)
72     added = [n for n, _ in d.add]
73 
74     # A worker that rq knows is never destroyed: it must stop by itself first.
75     assert not d.remove & set(live)
76     assert set(after.workers) == (set(memory.workers) - d.remove) | set(added)
77     assert all(n.startswith(NAME_PREFIX) and n not in memory.workers and n not in live for n in added)
78     # The cap on workers holds, and no more new workers than waiting jobs.
79     assert len([n for n in after.workers if n not in after.draining]) <= config.max_workers
80     assert len(added) <= waiting
81     # Only a worker that rq knows gets the shutdown command.
82     assert d.drain <= set(live)
83     # New workers come from offers that have room, from a cloud that is not resting.
84     for offer in {o for _, o in d.add}:
85         assert offer in offers and memory.cooldown.get(offer.provider, 0) <= NOW
86         assert sum(1 for _, o in d.add if o == offer) <= offer.capacity
87     # Each machine that ends goes to the ledger, once.
88     assert sorted(e["name"] for e in after.ledger if e["end"] == NOW) == sorted(d.remove)

91 @settings(max_examples=500)
92 @given(states())
93 def test_a_new_worker_goes_to_the_cheapest_offer_that_fits(state):
94     waiting, live, memory, config, offers, cap = state
95     d, _ = decide(NOW, waiting, live, memory, config, REF, offers, cap)
96     if not d.add:
97         return
98     first = d.add[0][1]
99     fitting = [o for o in offers if o.capacity > 0 and memory.cooldown.get(o.provider, 0) <= NOW]
100     cheaper = [o for o in fitting if o.usd_per_book(config) < first.usd_per_book(config)]
101     # A cheaper offer is left out only when the spend cap has no room for its
102     # reserve, which is its hourly price: so it must cost more by the hour.
103     # (A counterexample of 2026-09-22: 8 vCPU at $0.30/h is cheaper by the book
104     # than 2 vCPU at $0.125/h, and a small cap takes only the second.)
105     for o in cheaper:
106         assert o.usd_per_hour > first.usd_per_hour
107     if cap == math.inf:
108         assert not cheaper
