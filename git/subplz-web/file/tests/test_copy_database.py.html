55 def rows(url: str) -> dict[str, list[dict]]:
56     engine = create_engine(url)
57     with engine.connect() as c:
58         return {t.name: sorted((dict(r._mapping) for r in c.execute(select(t))), key=repr)
59                 for t in Base.metadata.sorted_tables}
