126   async function post(path, body) {
127     const res = await fetch(path, {
128       method: 'POST', credentials: 'same-origin',
129       headers: { 'Content-Type': 'application/json' },
130       body: JSON.stringify(body || {}),
131     });
132     const data = await res.json().catch(() => ({}));
133     if (!res.ok) throw new Error(data.detail || data.error || `Error ${res.status}`);
134     return data;
135   }

137   function planCard(plan, state, first) {
138     const card = document.createElement('div');
139     card.className = 'plan' + (first ? ' featured' : '');
140     const name = document.createElement('div');
141     name.className = 'plan-name';
142     name.textContent = plan.name;
143     const price = document.createElement('div');
144     price.className = 'plan-price';
145     price.textContent = plan.price_display + ' ';
146     const per = document.createElement('small');
147     per.textContent = plan.interval === 'year' ? 'billed yearly' : 'billed monthly';
148     price.append(per);
149     const blurb = document.createElement('p');
150     blurb.className = 'plan-blurb';
151     blurb.textContent = plan.blurb;
152     const buy = document.createElement('button');
153     buy.className = first ? 'primary' : 'ghost';
154     buy.type = 'button';
155     buy.textContent = `Pay ${plan.price_display}`;
156     buy.disabled = !state.available || state.needs_sign_in;
157     buy.addEventListener('click', async () => {
158       buy.disabled = true;
159       try {
160         location.href = (await post('/api/subrep/checkout', { plan_id: plan.id })).url;
161       } catch (err) {
162         say(err.message);
163         buy.disabled = false;
164       }
165     });
166     card.append(name, price, blurb, buy);
167     return card;
168   }

170   async function render() {
171     const state = await (await fetch('/api/subrep', { credentials: 'same-origin' })).json();
172     $('plans').replaceChildren(...state.plans.map((p, i) => planCard(p, state, i === 0)));
173     $('sign-in-note').hidden = !state.needs_sign_in;
174     $('buy').hidden = state.pro;
175     $('pro').hidden = !state.pro;
176     if (!state.available) say('Subrep Pro is not for sale yet. Check again soon.');
177     if (state.pro) {
178       const ends = new Date(state.period_end).toLocaleDateString();
179       const plan = state.plans.find((p) => p.id === state.plan_id);
180       $('pro-status').textContent = `${plan ? plan.name : 'Subrep Pro'}: paid until ${ends}.`;
181       $('link-email').textContent = `Email: ${state.link.email || '(none)'}`;
182       $('link-key').textContent = state.link.refresh_key;
183       $('command').textContent = state.link.command;
184     }
185     return state;
186   }
