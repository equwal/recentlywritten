103       #!/bin/bash
104       # A safety net for the burst controller: if the controller is gone, this
105       # machine still stops. After ${idle_checks} idle checks in a row (10 minutes
106       # each), or when Redis cannot be reached, power off. On AWS a power-off
107       # ends the machine and its bill; on Hetzner the controller then deletes it.
108       state_file=/var/lib/subplz-idle-count
109       state=$(/opt/subplz-web/.venv/bin/python - <<'PY' 2>/dev/null || echo idle
110       import os, socket
111       from redis import Redis
112       from rq import Worker
113       conn = Redis.from_url(os.environ["SUBPLZ_WEB_REDIS_URL"], socket_timeout=10)
114       host = socket.gethostname()
115       states = [w.get_state() for w in Worker.all(connection=conn) if w.hostname == host]
116       print("busy" if "busy" in states else "idle")
117       PY
118       )
119       if [ "$state" = "busy" ]; then
120         echo 0 > "$state_file"
121         exit 0
122       fi
123       count=$(( $(cat "$state_file" 2>/dev/null || echo 0) + 1 ))
124       echo "$count" > "$state_file"
125       if [ "$count" -ge ${idle_checks} ]; then
126         systemctl poweroff
127       fi
