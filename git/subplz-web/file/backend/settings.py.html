12 def _find_subplz() -> str:
13     """Locate the alignment backend on PATH.
14 
15     subplz is an ordinary dependency of this project, installed into whatever
16     environment is running it - not a sibling checkout. Falling back to the bare
17     name keeps the error useful when it is genuinely missing.
18     """
19     return shutil.which("subplz") or "subplz"
