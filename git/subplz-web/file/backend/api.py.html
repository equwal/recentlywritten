1 """HTTP API.
2 
3 Flow: drop files -> POST /api/uploads (stages, pairs, detects language) ->
4 POST /api/jobs/{id}/start (entitlement check, enqueue) -> poll GET /api/jobs/{id}
5 -> download from /api/jobs/{id}/files/{kind}.
6 
7 Upload and start are separate so a wrong language guess costs a click rather
8 than a re-upload and a wasted multi-hour run.
9 
10 Around that: /api/auth/* (email sign-in links) and /api/billing/* (Stripe
11 checkout, its return trip and its webhook).
12 """

57 def get_account(
58     response: Response,
59     session: Annotated[Session, Depends(get_session)],
60     subplz_device: Annotated[str | None, Cookie()] = None,
61 ) -> Account:
62     """Identify the caller.
63 
64     A cookie is the whole identity check for the free tier, on purpose. Anyone
65     who clears it gets another free book, and that is an accepted cost: a
66     signup wall does not belong in front of a tool whose pitch is "drop two
67     files in". The real protection against abuse is capacity - the queue and
68     per-worker limits - not identity.
69 
70     Signing in does not replace the cookie, it re-points it: the cookie then
71     names the signed-in account, on every device that has signed in.
72     """
73     account = None
74     if subplz_device:
75         account = (
76             session.query(Account)
77             .filter(Account.device_token == subplz_device)
78             .first()
79         )
80 
81     if account is None:
82         account = accounts.new_account(session)
83         session.commit()
84         _set_identity(response, account)
85         return account
86 
87     # This device's anonymous row was folded into a real account while no
88     # browser was attached (a payment landing by webhook). Follow it.
89     survivor = accounts.resolve(session, account)
90     if survivor.id != account.id:
91         _set_identity(response, survivor)
92     return survivor

352 @router.post("/auth/request")
353 def request_sign_in(
354     body: EmailIn,
355     account: Annotated[Account, Depends(get_account)],
356     session: Annotated[Session, Depends(get_session)],
357 ):
358     """Mail a one-time sign-in link."""
359     try:
360         email = accounts.normalize_email(body.email)
361     except accounts.InvalidEmail as exc:
362         raise HTTPException(400, str(exc)) from exc
363 
364     # A server that cannot send mail has no safe way to do this: the only
365     # fallback is showing the link, which would sign anyone in as anyone.
366     if not settings.sign_in_available:
367         raise HTTPException(
368             503, "Email sign-in is not set up on this server yet."
369         )
370 
371     try:
372         link = auth.issue(session, account, email)
373     except auth.TooManyRequests as exc:
374         raise HTTPException(429, str(exc)) from exc
375 
376     try:
377         sent = mailer.send_login_link(email, link)
378     except mailer.MailError as exc:
379         raise HTTPException(502, str(exc)) from exc
380 
381     out = {"sent": sent, "email": email}
382     if not sent and settings.dev_login_links:
383         out["dev_link"] = link
384     return out

490 @router.post("/uploads", response_model=UploadOut)
491 async def create_upload(
492     account: Annotated[Account, Depends(get_account)],
493     session: Annotated[Session, Depends(get_session)],
494     files: Annotated[list[UploadFile], File()],
495 ):
496     """Stage a dropped pair, work out which is which, and guess the language."""
497     names = [f.filename or "" for f in files]
498     try:
499         pairing = detect.classify(names)
500     except detect.DetectionError as exc:
501         raise HTTPException(400, str(exc)) from exc
502 
503     job_id = new_id("job")
504     paths = Paths.for_job(job_id)
505     paths.create()
506 
507     try:
508         order = {name: i for i, name in enumerate(pairing.audio_names)}
509         by_name = {(f.filename or ""): f for f in files}
510 
511         text_path = staged_text_path(job_id, pairing.text_name)
512         await _save(by_name[pairing.text_name], text_path)
513 
514         # fb2/mobi/azw3 become something the aligner can read. Do it now, not at
515         # run time, so a book we cannot open fails while the user is watching.
516         if convert.needs_conversion(pairing.text_name):
517             try:
518                 converted = convert.to_readable(
519                     text_path, text_path.with_suffix("")
520                 )
521             except convert.ConversionError as exc:
522                 raise HTTPException(400, str(exc)) from exc
523             if converted != text_path:
524                 text_path.unlink(missing_ok=True)
525                 text_path = converted
526 
527         audio_paths: list[Path] = []
528         for name in pairing.audio_names:
529             upload = by_name.get(name)
530             if upload is None:
531                 raise HTTPException(400, f"Upload did not include {name}.")
532             dest = (
533                 staged_part_path(job_id, order[name] + 1, name)
534                 if pairing.is_multipart
535                 else staged_audio_path(job_id, name)
536             )
537             # The staged stem is shared between audio and text, so a single
538             # audio file must not land on the text file's path.
539             if dest == text_path:
540                 raise HTTPException(
541                     400, "The audiobook and the book must be different formats."
542                 )
543             await _save(upload, dest)
544             audio_paths.append(dest)
545 
546         if not audio_paths:
547             raise HTTPException(400, "Upload did not include any audio.")
548 
549         # Custom cover art is a signed-in feature. Say so plainly rather than
550         # accepting the file and quietly ignoring it: the UI labels the drop
551         # zone before anyone picks an image, and this is the backstop.
552         cover_name = None
553         cover_blocked = False
554         if pairing.cover_name:
555             if account.signed_in:
556                 upload = by_name.get(pairing.cover_name)
557                 if upload is not None:
558                     await _save(upload, staged_cover_path(job_id, pairing.cover_name))
559                     cover_name = pairing.cover_name
560             else:
561                 cover_blocked = True
562 
563         detection = detect.detect_language(detect.extract_text_sample(text_path))
564         # Parts are merged at run time, so total the durations here.
565         durations = [probe_duration(p) for p in audio_paths]
566         duration = sum(d for d in durations if d) if any(durations) else None
567 
568         # Does this text actually belong to this audio? Answering now costs a
569         # few seconds; finding out during the run costs the whole run.
570         match = matching.check(
571             audio_paths[0],
572             text_path,
573             detection.code if detection.supported else "en",
574             aligner,
575         )
576 
577         language = detection.code if detection.supported else "en"
578         lang = languages.require(language)
579 
580         job = Job(
581             id=job_id,
582             account_id=account.id,
583             status=JobStatus.draft,
584             language=lang.code,
585             splitter=lang.splitter,
586             model=settings.model,
587             audio_filename=pairing.display_name,
588             text_filename=pairing.text_name,
589             audio_parts=len(audio_paths),
590             cover_filename=cover_name,
591             audio_bytes=sum(p.stat().st_size for p in audio_paths),
592             audio_duration_seconds=duration,
593             stage="Ready to start",
594         )
595         session.add(job)
596         session.commit()
597 
598     # A refused upload keeps what arrived, for debugging. The operator deletes
599     # it by hand.
600     except HTTPException:
601         raise
602     except detect.DetectionError as exc:
603         raise HTTPException(400, str(exc)) from exc
604     except Exception as exc:  # noqa: BLE001
605         raise HTTPException(500, f"Upload failed: {exc}") from exc
606 
607     return UploadOut(
608         job=_job_out(job, []),
609         detected=DetectionOut(
610             code=detection.code, name=detection.name,
611             confidence=round(detection.confidence, 4),
612             supported=detection.supported,
613         ),
614         match=match.as_dict(),
615         cover_requires_sign_in=cover_blocked,
616     )

619 async def _save(upload: UploadFile, dest: Path) -> None:
620     """Stream to disk in chunks - these files run to hundreds of megabytes."""
621     dest.parent.mkdir(parents=True, exist_ok=True)
622     written = 0
623     with dest.open("wb") as out:
624         while chunk := await upload.read(_CHUNK):
625             written += len(chunk)
626             if written > settings.max_upload_bytes:
627                 raise HTTPException(
628                     413,
629                     f"{upload.filename} exceeds the "
630                     f"{settings.max_upload_bytes // (1024**3)} GiB upload limit.",
631                 )
632             out.write(chunk)

635 @router.post("/jobs/{job_id}/start", response_model=JobOut)
636 def start_job(
637     job_id: str,
638     body: StartIn,
639     account: Annotated[Account, Depends(get_account)],
640     session: Annotated[Session, Depends(get_session)],
641 ):
642     job = _load(session, account, job_id)
643     # A failed job keeps its staged inputs, so it can be retried in place -
644     # usually after correcting the language.
645     if job.status not in (JobStatus.draft, JobStatus.failed):
646         raise HTTPException(409, f"Job is already {job.status.value}.")
647     if job.status == JobStatus.failed and not Paths.for_job(job.id).inp.exists():
648         raise HTTPException(
649             409, "The uploaded files for this job are gone. Upload them again."
650         )
651 
652     if body.language:
653         try:
654             lang = languages.require(body.language)
655         except languages.UnsupportedLanguage as exc:
656             raise HTTPException(400, str(exc)) from exc
657         job.language, job.splitter = lang.code, lang.splitter
658     if body.model:
659         job.model = body.model
660 
661     try:
662         billing.authorize_start(session, account, job)
663     except billing.PaymentRequired as exc:
664         session.rollback()
665         raise HTTPException(402, str(exc)) from exc
666 
667     # With an external queue the worker is probably not this machine, so the
668     # staged inputs have to go somewhere both sides can reach before enqueuing.
669     if settings.queue_backend != "memory":
670         paths = Paths.for_job(job.id)
671         prefix = input_prefix(job.id)
672         for local in sorted(paths.inp.rglob("*")):
673             if local.is_file():
674                 rel = local.relative_to(paths.inp).as_posix()
675                 storage.put_file(f"{prefix}/{rel}", local)
676 
677     job.status = JobStatus.queued
678     job.stage = "Queued"
679     job.progress = 0.0
680     job.error = None
681     session.commit()
682 
683     queue.enqueue(job.id, paid=billing.is_paid(job))
684     return _job_out(job, [])

699 @router.post("/local/jobs", response_model=JobOut)

705     # The same book started again - a closed tab, a reload - carries on under

708         session.query(Job)
709         .filter(
710             Job.account_id == account.id, Job.local == 1,
711             Job.status == JobStatus.running,
712             Job.audio_filename == body.audio_filename,
713             Job.audio_bytes == body.audio_bytes,
714         )
715         .order_by(Job.created_at.desc())
716         .first()
717     )
718     if running is not None:
719         running.language = body.language
720         session.commit()
721         return _job_out(running, [])
722 
723     job = Job(
724         account_id=account.id, status=JobStatus.running, local=1,
725         language=body.language, splitter="browser", model="whisper-tiny",
726         audio_filename=body.audio_filename, text_filename=body.text_filename,
727         audio_parts=body.audio_parts, audio_bytes=body.audio_bytes,
728         audio_duration_seconds=body.audio_duration_seconds,
729         stage="Running in your browser", started_at=utcnow(),
730     )
731     session.add(job)
732     try:
733         billing.authorize_start(session, account, job)
734     except billing.PaymentRequired as exc:
735         session.rollback()
736         raise HTTPException(402, str(exc)) from exc
737     session.commit()
738     return _job_out(job, [])

741 @router.post("/local/jobs/{job_id}/finish", response_model=JobOut)
742 def finish_local_job(
743     job_id: str,
744     body: LocalFinishIn,
745     account: Annotated[Account, Depends(get_account)],
746     session: Annotated[Session, Depends(get_session)],
747 ):
748     job = _load(session, account, job_id)
749     if not job.local or job.status != JobStatus.running:
750         raise HTTPException(409, f"Job is already {job.status.value}.")
751 
752     import json
753     import tempfile
754 
755     name = Path(body.filename).name or "subtitles.srt"
756     with tempfile.TemporaryDirectory() as tmp:
757         for kind, filename, content in (
758             ("srt", name, body.srt),
759             ("metadata", "metadata.json",
760              json.dumps({"job_id": job.id, **body.metadata}, ensure_ascii=False, indent=2)),
761         ):
762             src = Path(tmp) / filename
763             # Bytes, not text: on Windows write_text would turn every line ending into CRLF.
764             src.write_bytes(content.encode("utf-8"))
765             key = f"{job.id}/{filename}"
766             size = storage.put_file(key, src)
767             session.add(Artifact(job_id=job.id, kind=kind, filename=filename,
768                                  storage_key=key, size_bytes=size))
769 
770     if settings.browser_copy:
771         # The subtitles go next to the copy of the audio and the book.
772         root = Paths.for_job(job.id).root
773         root.mkdir(parents=True, exist_ok=True)
774         (root / (_plain_name(body.filename) or "subtitles.srt")).write_bytes(
775             body.srt.encode("utf-8"))
776 
777     job.status = JobStatus.succeeded
778     job.stage, job.progress, job.finished_at = "Done", 1.0, utcnow()
779     session.commit()
780     return _job_out(job, _artifacts(session, job.id))

783 def _plain_name(raw: str) -> str:
784     """The last part of a file name from a browser, with no folder in it."""
785     name = re.split(r"[\\/]", raw or "")[-1].strip()
786     return "" if name in ("", ".", "..") else name

789 @router.post("/local/jobs/{job_id}/files")

791     job_id: str,
792     account: Annotated[Account, Depends(get_account)],
793     session: Annotated[Session, Depends(get_session)],
794     files: Annotated[list[UploadFile], File()],
795 ):
796     """Keep a copy of the audio and the book of a job in the browser.
797 
798     Only during the beta (settings.browser_copy), for debugging, as the
799     privacy and user data policy says. The copy goes with the other files of

835 def expire_stale_local_jobs(session: Session) -> int:
836     """A tab that was closed for good never reports back. Release what it held."""

838 
839     stale = (

852 @router.post("/convert")
853 async def convert_book(file: Annotated[UploadFile, File()]):
854     """mobi / azw3 to epub. The one thing the browser cannot do for itself:
855     those formats need a real parser, and a book is small enough to send."""
856     name = Path(file.filename or "book").name
857     if not convert.needs_conversion(name):
858         raise HTTPException(400, "That format does not need converting.")
859     import tempfile
860 
861     with tempfile.TemporaryDirectory() as tmp:
862         src = Path(tmp) / name
863         await _save(file, src)
864         try:
865             out = convert.to_readable(src, Path(tmp) / "converted")
866         except convert.ConversionError as exc:
867             raise HTTPException(400, str(exc)) from exc
868         data = out.read_bytes()
869     return Response(data, media_type="application/epub+zip",
870                     headers={"X-Filename": Path(name).stem + ".epub"})
