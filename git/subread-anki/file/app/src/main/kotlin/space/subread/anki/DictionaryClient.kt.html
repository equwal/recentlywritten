8 /**
9  * Asks SubRead Dictionary for the terms of a text, through its content provider
10  * `content://space.subread.dictionary.lookup` (see docs/provider-api.md of that app).
11  *
12  * `terms?text=…` gives one row for each term of each dictionary, the longest term first. The
13  * rows of one term and reading make one [Entry]. `audio?expression=…&reading=…` gives one row
14  * with the file of the first source that has the audio, and `audio/<file>` its bytes.
15  */

52     /**
53      * The terms that start at [offset] in [text], the longest first. Empty when the dictionary
54      * is not there or has no term there.
55      */
56     fun lookup(context: Context, text: String, offset: Int = 0): List<Entry> {
57         val authority = authority(context)
58         if (text.isEmpty() || offset !in text.indices || authority == null) return emptyList()
59         val uri = "content://$authority/terms".toUri().buildUpon()
60             .appendQueryParameter("text", text).appendQueryParameter("offset", offset.toString()).build()
61         val cursor = runCatching { context.contentResolver.query(uri, null, null, null, null) }.getOrNull() ?: return emptyList()
62         data class Row(
63             val expression: String, val reading: String, val length: Int, val reasons: String,
64             val dictionary: String, val glossary: String, val frequency: String, val pitch: String,
65         )
66         val rows = cursor.use { c ->
67             buildList {
68                 while (c.moveToNext()) add(
69                     Row(
70                         c.text("expression") ?: continue, c.text("reading").orEmpty(), c.long("length")?.toInt() ?: 0,
71                         c.text("reasons").orEmpty(), c.text("dictionary").orEmpty(),
72                         c.text("definition_html") ?: c.text("glossary").orEmpty(),
73                         c.text("frequency").orEmpty(), c.text("pitch").orEmpty(),
74                     ),
75                 )
76             }
77         }
78         return rows.groupBy { it.expression to it.reading }.map { (key, group) ->
79             val first = group.first()
80             Entry(
81                 expression = key.first,
82                 reading = key.second,
83                 length = first.length,
84                 reasons = first.reasons,
85                 glossary = group.joinToString("<br>") { row ->
86                     if (row.dictionary.isEmpty()) row.glossary else "<i>${escape(row.dictionary)}</i><br>${row.glossary}"
87                 },
88                 frequency = group.map { it.frequency }.firstOrNull { it.isNotEmpty() }.orEmpty(),
89                 pitch = group.map { it.pitch }.firstOrNull { it.isNotEmpty() }.orEmpty(),
90                 audio = "content://$authority/audio".toUri().buildUpon()
91                     .appendQueryParameter("expression", key.first).appendQueryParameter("reading", key.second).build(),
92             )
93         }
94     }

96     /**
97      * The bytes of the audio of a term, or null when no source has it. The provider names the
98      * file in a row first, then gives the bytes at `audio/<file>`.
99      */
100     fun audio(context: Context, ask: Uri): ByteArray? = runCatching {
101         val name = context.contentResolver.query(ask, null, null, null, null)?.use { c ->
102             if (c.moveToFirst()) c.text("file") else null
103         } ?: return null
104         val file = "content://${ask.authority}/audio/$name".toUri()
105         context.contentResolver.openInputStream(file)?.use { it.readBytes() }?.takeIf { it.isNotEmpty() }
106     }.getOrNull()
