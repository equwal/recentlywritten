41  * The pop-up. It takes a text the way a dictionary does: from the text selection menu of any
42  * app, the share sheet, another app, or a link. It shows the text, the terms of SubRead
43  * Dictionary at the word, and a "+ Anki" button for each term. A tap on a character of the

153     private fun addAtOnce(scan: Scan) {
154         work.execute {
155             val result = runCatching {
156                 val word = scan.word
157                 // Where the word is in the text, also in another form: for the bold and the sentence.
158                 val found = if (scan.offset >= 0 || word == null) null else locate(scan.text, word)
159                 val at = if (scan.offset >= 0) scan.offset else found?.first ?: -1
160                 val card = if (request.definition != null) {
161                     when {
162                         found != null -> Miner.card(request, scan.text, found.first, found.count(), null)
163                         word != null && at >= 0 -> Miner.card(request, scan.text, at, word.length, null)
164                         else -> Miner.card(request, scan.text, -1, 0, null)
165                     }
166                 } else if (at >= 0) {
167                     val entry = DictionaryClient.lookup(this, scan.text, at).firstOrNull()
168                     if (entry != null) {
169                         Miner.card(request, scan.text, at, entry.length, entry)
170                     } else {
171                         val range = Miner.wordAt(scan.text, at, selected = isSelection(scan.text, at))
172                         Miner.card(request, scan.text, range.first, range.count(), null)
173                     }
174                 } else {
175                     Miner.card(request, scan.text, -1, 0, DictionaryClient.lookup(this, word.orEmpty()).firstOrNull())
176                 }
177                 miner.add(grab!!.get(), card)
178             }.getOrElse { Miner.Result.Failed(it.message ?: it.toString()) }
179             finishAfter(result)
180         }
181     }

319     /** One block for each term. [at] is where the terms start in the text, -1 when not in it. */
320     private fun showTerms(found: List<DictionaryClient.Entry>, at: Int) {
321         results.removeAllViews()
322         addButtons.clear()
323         if (found.isEmpty()) {
324             // No dictionary, or no term here: the word at the tap goes on the card as it is.
325             val range = if (at >= 0) Miner.wordAt(text, at, selected = isSelection(text, at)) else IntRange.EMPTY
326             if (at >= 0) mark(range)
327             note(getString(if (DictionaryClient.answers(this)) R.string.no_term else R.string.no_dictionary), Color.DKGRAY, TEXT_SP - 3)
328             val word = if (at >= 0) text.substring(range.first, range.last + 1) else request.expression.orEmpty()
329             term(word, null, if (at >= 0) range.first else -1, range.count())
330             return
331         }
332         if (at >= 0) mark(at until at + found.first().length)
333         for (entry in found) term(entry.headword, entry, at, entry.length)
334         markInAnki(found)
335     }

427     private fun mark(range: IntRange?) {
428         val shown = SpannableString(text)
429         if (range != null && !range.isEmpty() && range.last < text.length) {
430             shown.setSpan(BackgroundColorSpan(0xFFFFE082.toInt()), range.first, range.last + 1, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
431             // The underline stays visible on an e-ink screen, where the colour is a light grey.
432             shown.setSpan(UnderlineSpan(), range.first, range.last + 1, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
433         }
434         textView.text = shown

438     private fun statusLine(grabbed: Miner.Grab, deck: String): String {
439         val parts = ArrayList<String>()
440         val captureOff = CaptureService.instance == null
441         if (captureOff && grabbed.picture == null && grabbed.sentenceAudio == null) {
442             parts += getString(R.string.status_capture_off)
443         } else {
444             parts += getString(if (grabbed.picture != null) R.string.status_picture else R.string.status_no_picture)
445             val ms = grabbed.soundMs
446             parts += when {
447                 grabbed.sentenceAudio == null -> getString(R.string.status_no_sound)
448                 ms != null -> getString(R.string.status_sound_seconds, "%.1f".format(ms / 1000.0))
449                 else -> getString(R.string.status_sound)
450             }
451         }
452         parts += getString(R.string.deck_current, deck)
453         return parts.joinToString("   ")
454     }

456     /**
457      * Plays the audio of the word from the dictionary. The file is written off the UI thread,
458      * and the player prepares on its own thread. A pop-up that closed in the meantime plays
459      * nothing: its onDestroy has already run, and no later player would be released.
460      */
461     private fun play(entry: DictionaryClient.Entry) {
462         val ask = entry.audio ?: return
463         thread {
464             val file = DictionaryClient.audio(this, ask)?.let { bytes -> File(cacheDir, "play.tmp").apply { writeBytes(bytes) } }
465             runOnUiThread {
466                 if (isFinishing || isDestroyed) return@runOnUiThread
467                 if (file == null) {
468                     toast(getString(R.string.no_audio))
469                     return@runOnUiThread
470                 }
471                 player?.release()
472                 player = runCatching {
473                     MediaPlayer().apply {
474                         FileInputStream(file).use { setDataSource(it.fd) }
475                         setOnPreparedListener { it.start() }
476                         prepareAsync()
477                     }
478                 }.getOrNull()
479             }
480         }
481     }
