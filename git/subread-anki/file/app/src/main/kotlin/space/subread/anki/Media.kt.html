58     /** The picture as a JPEG in [out], at most [MAX_SIDE] pixels on the long side. Null when the write fails. */
59     fun writeJpeg(bitmap: Bitmap, out: File): File? {
60         val longest = maxOf(bitmap.width, bitmap.height)
61         val scaled = if (longest <= MAX_SIDE) bitmap else {
62             val scale = MAX_SIDE.toFloat() / longest
63             bitmap.scale((bitmap.width * scale).toInt(), (bitmap.height * scale).toInt())
64         }
65         return runCatching {
66             out.outputStream().use { scaled.compress(Bitmap.CompressFormat.JPEG, 85, it) }
67             out
68         }.getOrNull()
69     }

71     /** The bytes of a source. */
72     fun bytes(context: Context, source: String): ByteArray? {
73         val uri = source.toUri()
74         return when (uri.scheme?.lowercase()) {
75             "content" -> context.contentResolver.openInputStream(uri)?.use { it.readBytes(MAX_BYTES) }
76             "http", "https" -> {
77                 val connection = URL(source).openConnection() as HttpURLConnection
78                 connection.connectTimeout = 15_000
79                 connection.readTimeout = 30_000
80                 try {
81                     if (connection.responseCode != 200) null else connection.inputStream.use { it.readBytes(MAX_BYTES) }
82                 } finally {
83                     connection.disconnect()
84                 }
85             }
86             "data" -> {
87                 // data:audio/mpeg;base64,SUQz...
88                 val comma = source.indexOf(',')
89                 if (comma < 0) return null
90                 val head = source.substring(0, comma)
91                 val body = source.substring(comma + 1)
92                 if (head.endsWith(";base64")) Base64.decode(body, Base64.DEFAULT) else Uri.decode(body).toByteArray()
93             }
94             // A file path would read the private files of this app for the sender.
95             else -> null
96         }
97     }

99     private fun java.io.InputStream.readBytes(limit: Long): ByteArray? {
100         val out = java.io.ByteArrayOutputStream()
101         val buffer = ByteArray(64 * 1024)
102         var total = 0L
103         while (true) {
104             val n = read(buffer)
105             if (n < 0) break
106             total += n
107             if (total > limit) return null
108             out.write(buffer, 0, n)
109         }
110         return out.toByteArray()
111     }

113     /** The extension for the bytes of a file of [kind], from its first bytes. Null for an unknown file. */
114     fun extension(bytes: ByteArray, kind: Kind): String? {
115         fun at(offset: Int, text: String) =
116             bytes.size >= offset + text.length && text.indices.all { bytes[offset + it] == text[it].code.toByte() }
117         return when (kind) {
118             Kind.IMAGE -> when {
119                 bytes.size > 2 && bytes[0] == 0xFF.toByte() && bytes[1] == 0xD8.toByte() -> "jpg"
120                 at(1, "PNG") -> "png"
121                 at(0, "RIFF") && at(8, "WEBP") -> "webp"
122                 at(0, "GIF8") -> "gif"
123                 else -> null
124             }
125             Kind.AUDIO -> when {
126                 at(0, "ID3") -> "mp3"
127                 bytes.size > 2 && bytes[0] == 0xFF.toByte() && (bytes[1].toInt() and 0xE0) == 0xE0 -> "mp3"
128                 at(0, "OggS") -> "ogg"
129                 at(0, "RIFF") && at(8, "WAVE") -> "wav"
130                 at(0, "fLaC") -> "flac"
131                 at(4, "ftyp") -> "m4a"
132                 else -> null
133             }
134         }
135     }
