58     fun ensureDeck(): Long {
59         val decks = decks()

82     /**
83      * Makes the note type of the app: first the note type with its fields, then the template
84      * of each card. Returns its id, or null when AnkiDroid refused.
85      */
86     private fun addModel(deckId: Long): Long? {
87         val values = ContentValues().apply {
88             put("name", NoteType.NAME)
89             put("field_names", NoteType.FIELDS.joinToString(SEPARATOR))
90             put("num_cards", NoteType.CARD_NAMES.size)
91             put("css", NoteType.CSS)
92             put("deck_id", deckId)
93             put("sort_field_index", 0)
94         }
95         val model = resolver.insert(MODELS, values) ?: return null
96         val templates = Uri.withAppendedPath(model, "templates")
97         NoteType.CARD_NAMES.forEachIndexed { i, card ->
98             val template = ContentValues().apply {
99                 put("card_template_name", card)
100                 put("question_format", NoteType.QUESTION_FORMATS[i])
101                 put("answer_format", NoteType.ANSWER_FORMATS[i])
102             }
103             resolver.update(Uri.withAppendedPath(templates, i.toString()), template, null, null)
104         }
105         return model.lastPathSegment?.toLongOrNull()
106     }

119     /**
120      * True when a note of the note type has [expression] in its first field, and that field
121      * holds the word. False when the note type does not exist yet.
122      */
123     fun inAnki(expression: String): Boolean {
124         val modelId = existingModel() ?: return false
125         val fields = fields(modelId)
126         if (fields.isEmpty() || mapping(modelId, fields)[fields.first()] != Source.EXPRESSION) return false
127         return isDuplicate(modelId, expression)
128     }

131     fun mapping(modelId: Long, fields: List<String>): Map<String, Source> {
132         val set = store.mapping
133         return if (set.isNotEmpty() && set.keys.containsAll(fields)) set else Fields.guess(fields)

136     /**
137      * Copies a file into the media folder of AnkiDroid and returns its name there. [kind] is
138      * `audio` or `image`; AnkiDroid takes the kind from the file. AnkiDroid reads the file
139      * through [MediaProvider], so the read is granted to it for the copy.
140      */
141     @Suppress("UNUSED_PARAMETER")
142     fun addMedia(file: File, kind: String): String? {
143         val uri = Media.uriFor(context, file)
144         val anki = packageName(context) ?: return null
145         context.grantUriPermission(anki, uri, Intent.FLAG_GRANT_READ_URI_PERMISSION)
146         try {
147             val values = ContentValues().apply {
148                 put("file_uri", uri.toString())
149                 put("preferred_name", file.nameWithoutExtension.replace(' ', '_'))
150             }
151             val stored = runCatching { resolver.insert(MEDIA, values) }.getOrNull() ?: return null
152             // The path of the answer is the name of the file in the media folder, after a slash.
153             return stored.path?.trimStart('/')?.takeIf { it.isNotEmpty() }
154         } finally {
155             context.revokeUriPermission(uri, Intent.FLAG_GRANT_READ_URI_PERMISSION)
156         }
157     }

159     /** True when a note of [modelId] has [key] as its first field. */
160     fun isDuplicate(modelId: Long, key: String): Boolean {
161         if (key.isBlank()) return false
162         return runCatching {
163             val model = models()[modelId] ?: return@runCatching false
164             val first = fields(modelId).firstOrNull() ?: return@runCatching false
165             // The selection of the notes path is an Anki search. The quotes make one term of
166             // "field:text", and the escapes stop Anki from reading _ and * as wildcards.
167             val search = "\"${escape(first)}:${escape(key)}\" \"note:${escape(model)}\""
168             resolver.query(NOTES, arrayOf("_id"), search, null, null)?.use { it.count > 0 } ?: false
169         }.getOrDefault(false)
170     }

172     /** Adds a note and puts its cards in the deck. Null when AnkiDroid refused it. */
173     fun addNote(modelId: Long, deckId: Long, fields: Array<String>, tags: Set<String>): Long? {
174         val values = ContentValues().apply {
175             put("mid", modelId)
176             put("flds", fields.joinToString(SEPARATOR))
177             put("tags", tags.joinToString(" ") { it.replace(' ', '_') })
178         }
179         val note = resolver.insert(NOTES, values) ?: return null
180         val cards = Uri.withAppendedPath(note, "cards")
181         resolver.query(cards, arrayOf("ord"), null, null, null)?.use { c ->
182             while (c.moveToNext()) {
183                 val deck = ContentValues().apply { put("deck_id", deckId) }
184                 resolver.update(Uri.withAppendedPath(cards, c.getString(0)), deck, null, null)
185             }
186         }
187         return note.lastPathSegment?.toLongOrNull()
188     }

190     fun note(noteId: Long): Note? = runCatching {
191         resolver.query(Uri.withAppendedPath(NOTES, noteId.toString()), arrayOf("_id", "flds", "tags"), null, null, null)?.use { c ->
192             if (!c.moveToFirst()) {
193                 null
194             } else {
195                 Note(
196                     c.getLong(0),
197                     c.getString(1).split(SEPARATOR).toTypedArray(),
198                     c.getString(2).orEmpty().split(' ').filter { it.isNotBlank() }.toSet(),
199                 )
200             }
201         }
202     }.getOrNull()
