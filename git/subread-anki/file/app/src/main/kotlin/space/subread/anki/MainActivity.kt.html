subread-anki

One tap makes an Anki card on Android: word, definition, sentence, screenshot, audio. Works with any app, with SubRead Dictionary and SubRead Overlay.

170     private fun chooseDeck() {
171         thread {
172             val decks = AnkiClient(this).decks().entries.sortedBy { it.value }
173             runOnUiThread {
174                 val names = decks.map { it.value } + getString(R.string.deck_new)
175                 AlertDialog.Builder(this).setItems(names.toTypedArray()) { _, which ->
176                     val deck = decks.getOrNull(which)
177                     if (deck != null) {
178                         store.deckId = deck.key
179                         draw()
180                     } else {
181                         newDeck()
182                     }
183                 }.show()
184             }
185         }
186     }

188     private fun newDeck() {
189         val field = EditText(this).apply { hint = getString(R.string.deck_new_name) }
190         AlertDialog.Builder(this).setView(field).setPositiveButton(R.string.save) { _, _ ->
191             val name = field.text.toString().trim()
192             if (name.isEmpty()) return@setPositiveButton
193             thread {
194                 val id = runCatching { AnkiClient(this).run { decks().entries.firstOrNull { it.value == name }?.key ?: addDeck(name) } }.getOrNull()
195                 runOnUiThread {
196                     if (id != null) store.deckId = id else toast(getString(R.string.failed, name))
197                     draw()
198                 }
199             }
200         }.setNegativeButton(R.string.cancel, null).show()
201     }

203     private fun chooseModel() {
204         thread {
205             val models = AnkiClient(this).models().entries.sortedBy { it.value }
206             runOnUiThread {
207                 val names = listOf(getString(R.string.model_own_choice)) + models.map { it.value }
208                 AlertDialog.Builder(this).setItems(names.toTypedArray()) { _, which ->
209                     val model = models.getOrNull(which - 1)
210                     store.modelId = model?.key ?: Store.NONE
211                     // A new note type gets a fresh guess of its fields.
212                     store.mapping = if (model == null) NoteType.MAPPING else emptyMap()
213                     draw()
214                 }.show()
215             }
216         }

219     /** One dialog with the fields; a tap on a field opens the list of what can fill it. */
220     private fun editFields() {
221         thread {
222             val anki = AnkiClient(this)
223             val modelId = runCatching { anki.ensureModel(anki.ensureDeck()) }.getOrNull()
224             val fields = modelId?.let { anki.fields(it) }.orEmpty()
225             runOnUiThread {
226                 if (modelId == null || fields.isEmpty()) {
227                     toast(getString(R.string.anki_no_permission))
228                     return@runOnUiThread
229                 }
230                 val mapping = LinkedHashMap(anki.mapping(modelId, fields).let { m -> fields.associateWith { m[it] ?: Source.NONE } })
231                 showFields(fields, mapping)
232             }
233         }
234     }

236     private fun showFields(fields: List<String>, mapping: LinkedHashMap<String, Source>) {
237         val labels = fields.map { getString(R.string.field_source, it, mapping[it]!!.name) }
238         AlertDialog.Builder(this)
239             .setTitle(R.string.fields)
240             .setItems(labels.toTypedArray()) { _, which ->
241                 val field = fields[which]
242                 val sources = Source.entries
243                 AlertDialog.Builder(this).setTitle(field).setItems(sources.map { it.name }.toTypedArray()) { _, i ->
244                     mapping[field] = sources[i]
245                     showFields(fields, mapping)
246                 }.show()
247             }
248             .setPositiveButton(R.string.save) { _, _ ->
249                 store.mapping = mapping
250                 toast(getString(R.string.saved))
251             }
252             .setNegativeButton(R.string.cancel, null)
253             .show()
254     }
