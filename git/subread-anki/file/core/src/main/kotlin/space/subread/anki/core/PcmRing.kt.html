3 /**
4  * The last seconds of captured sound, 16-bit PCM, with a wall clock. The capture writes
5  * what it records; a request reads the frames of a wall-clock range back out.
6  *
7  * The clock: each write says when its last frame was recorded. Frames before it are
8  * spaced by the sample rate. The capture is continuous, so the end of one write is the
9  * start of the next.
10  *
11  * Not thread-safe: the caller locks.
12  */
13 class PcmRing(val sampleRate: Int, val channels: Int, capacityFrames: Int) {
14     init {
15         require(sampleRate > 0) { "sampleRate must be above zero" }
16         require(channels > 0) { "channels must be above zero" }
17         require(capacityFrames > 0) { "capacityFrames must be above zero" }
18     }
19 
20     /** How many frames the ring holds. */
21     val capacityFrames: Int = capacityFrames
22 
23     private val data = ShortArray(capacityFrames * channels)
24 
25     /** Frames written since the start, in total. */
26     var framesWritten: Long = 0
27         private set
28 
29     /** The wall clock (nanoseconds) of the end of the last frame written. */
30     var endNanos: Long = 0
31         private set
32 
33     /** Frames in the ring now. */
34     val framesHeld: Long get() = minOf(framesWritten, capacityFrames.toLong())
35 
36     /** The wall clock of the start of the oldest frame held. */
37     val startNanos: Long get() = endNanos - framesToNanos(framesHeld)
38 
39     /** True when nothing was written yet. */
40     val isEmpty: Boolean get() = framesWritten == 0L
41 
42     /**
43      * Appends [frames] frames from [samples] (interleaved, `frames * channels` values).
44      * [endNanos] is the wall clock at the end of the last of them.
45      */
46     fun write(samples: ShortArray, frames: Int, endNanos: Long) {
47         require(frames >= 0 && frames * channels <= samples.size) { "frames $frames do not fit the samples" }
48         var offset = 0
49         var count = frames
50         if (count > capacityFrames) {
51             // More than the ring holds: the head is as good as written and gone.
52             val skip = count - capacityFrames
53             offset = skip * channels
54             framesWritten += skip
55             count = capacityFrames
56         }
57         val values = count * channels
58         val at = ((framesWritten % capacityFrames) * channels).toInt()
59         val first = minOf(values, data.size - at)
60         System.arraycopy(samples, offset, data, at, first)
61         if (values > first) System.arraycopy(samples, offset + first, data, 0, values - first)
62         framesWritten += count
63         this.endNanos = endNanos
64     }
65 
66     /**
67      * The frames recorded from [fromNanos] to [toNanos], interleaved. The range is cut to
68      * what the ring holds. An empty array when nothing of the range is held.
69      */
70     fun read(fromNanos: Long, toNanos: Long): ShortArray {
71         if (isEmpty || toNanos <= fromNanos) return ShortArray(0)
72         val from = maxOf(fromNanos, startNanos)
73         val to = minOf(toNanos, endNanos)
74         if (to <= from) return ShortArray(0)
75         val firstFrame = framesWritten - nanosToFrames(endNanos - from)
76         val lastFrame = framesWritten - nanosToFrames(endNanos - to)
77         val count = (lastFrame - firstFrame).toInt()
78         if (count <= 0) return ShortArray(0)
79         val out = ShortArray(count * channels)
80         val at = ((firstFrame % capacityFrames) * channels).toInt()
81         val first = minOf(out.size, data.size - at)
82         System.arraycopy(data, at, out, 0, first)
83         if (out.size > first) System.arraycopy(data, 0, out, first, out.size - first)
84         return out
85     }

87     /** Nanoseconds for a count of frames. */
88     fun framesToNanos(frames: Long): Long = frames * 1_000_000_000L / sampleRate
89 
90     /** Frames for a count of nanoseconds, rounded down. */
91     fun nanosToFrames(nanos: Long): Long = nanos * sampleRate / 1_000_000_000L
