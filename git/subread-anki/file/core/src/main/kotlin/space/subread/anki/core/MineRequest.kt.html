subread-anki

One tap makes an Anki card on Android: word, definition, sentence, screenshot, audio. Works with any app, with SubRead Dictionary and SubRead Overlay.

151 
152         /** The pairs of a percent-encoded query, decoded. The last value of a repeated key wins. */
153         fun parseQuery(query: String?): Map<String, String> {
154             if (query.isNullOrEmpty()) return emptyMap()
155             val out = LinkedHashMap<String, String>()
156             for (pair in query.split('&')) {
157                 if (pair.isEmpty()) continue
158                 val eq = pair.indexOf('=')
159                 val key = if (eq < 0) pair else pair.substring(0, eq)
160                 val value = if (eq < 0) "" else pair.substring(eq + 1)
161                 out[decode(key)] = decode(value)
162             }
163             return out
164         }
