50 /** Fills the fields of a note type from a [Note]. */
51 object Fields {
52     /** The value of one [Source] for a note. */
53     fun value(note: Note, source: Source): String = when (source) {
54         Source.NONE -> ""
55         Source.EXPRESSION -> note.expression
56         Source.READING -> note.reading
57         Source.FURIGANA -> furigana(note.expression, note.reading)
58         Source.DEFINITION -> note.definition
59         Source.SENTENCE -> Sentences.escape(note.sentence)
60         Source.SENTENCE_BOLD -> Sentences.emphasize(note.sentence, note.selection.ifEmpty { note.expression }.takeIf { it.isNotEmpty() })
61         Source.SENTENCE_AUDIO -> sound(note.sentenceAudioFile)
62         Source.WORD_AUDIO -> sound(note.wordAudioFile)
63         Source.IMAGE -> image(note.imageFile)
64         Source.SOURCE -> Sentences.escape(note.source)
65         Source.PITCH -> note.pitch
66         Source.FREQUENCY -> note.frequency
67         Source.MARK -> "x"
68     }

70     /** One value for each of [fieldNames], from [mapping]. A field with no mapping is empty. */
71     fun render(note: Note, fieldNames: List<String>, mapping: Map<String, Source>): Array<String> =
72         Array(fieldNames.size) { i -> value(note, mapping[fieldNames[i]] ?: Source.NONE) }

74     /** `[sound:name]`, or empty for no file. */
75     fun sound(file: String): String = if (file.isEmpty()) "" else "[sound:$file]"

77     /** `<img src="name">`, or empty for no file. */
78     fun image(file: String): String = if (file.isEmpty()) "" else "<img src=\"${Sentences.escape(file)}\">"

80     /**
81      * The expression with its reading in the furigana form of Anki: ` 漢字[かんじ]`. The
82      * expression alone when the reading is empty or the same as the expression.
83      */
84     fun furigana(expression: String, reading: String): String =
85         if (reading.isEmpty() || reading == expression || expression.isEmpty()) expression
86         else " $expression[$reading]"
87 
88     /**
89      * A mapping for a note type from the names of its fields. Names are compared without
90      * case and without anything but letters and digits, so `Sentence Audio`,
91      * `sentence_audio` and `SentenceAudio` are the same field. Unknown fields get [Source.NONE].
92      */
93     fun guess(fieldNames: List<String>): Map<String, Source> {
94         val out = LinkedHashMap<String, Source>()
95         val taken = HashSet<Source>()
96         for (name in fieldNames) {
97             val source = BY_NAME[normalize(name)] ?: Source.NONE
98             // The first field of a kind wins: two "Definition" fields do not both get it.
99             out[name] = if (source != Source.NONE && source != Source.MARK && !taken.add(source)) Source.NONE else source
100         }
101         // A note type with no named expression field is a basic one: its first field gets
102         // the expression, its second the definition.
103         if (Source.EXPRESSION !in out.values && fieldNames.isNotEmpty() && out[fieldNames[0]] == Source.NONE) {
104             out[fieldNames[0]] = Source.EXPRESSION
105             if (Source.DEFINITION !in out.values && fieldNames.size > 1 && out[fieldNames[1]] == Source.NONE) {
106                 out[fieldNames[1]] = Source.DEFINITION
107             }
108         }
109         return out
110     }
111 
112     /** The name of a field with case, spaces and punctuation removed. */
113     fun normalize(name: String): String = name.lowercase().filter { it.isLetterOrDigit() }
