3 /** Finds the sentence around a word, and marks the word in it. */
4 object Sentences {
5     /** Characters that end a sentence. */
6     private const val TERMINATORS = "。｡．！？!?…‥\n\r"
7 
8     /** Quotes and brackets that close a sentence after its terminator: `「猫だ。」` */
9     private const val CLOSERS = "」』）)】〉》\"'’”"
10 
11     /** True for a character that ends a sentence. An ASCII period counts only before a space, so `3.5` stays one. */
12     fun isTerminator(c: Char): Boolean = c in TERMINATORS
13 
14     /** True when the character at [index] ends a sentence, with the ASCII period rule applied. */
15     fun endsAt(text: String, index: Int): Boolean {
16         val c = text[index]
17         if (c in TERMINATORS) return true
18         return c == '.' && (index + 1 >= text.length || text[index + 1].isWhitespace())
19     }
20 
21     private fun isLineBreak(c: Char) = c == '\n' || c == '\r'
22 
23     /**
24      * The range of the sentence that holds the character at [index]. The sentence runs from
25      * the character after the previous terminator (and its closers) to the next terminator
26      * and its closers, both inclusive. A line break ends a sentence but is not part of it.
27      * Spaces at the ends are not trimmed here.
28      */
29     fun rangeAround(text: String, index: Int): IntRange {
30         require(index in text.indices) { "index $index is not in the text" }
31         var start = index
32         while (start > 0 && !endsAt(text, start - 1)) start--
33         // A closer right after the previous terminator belongs to the previous sentence.
34         while (start < index && text[start] in CLOSERS) start++
35         var end = index
36         while (end < text.length && !endsAt(text, end)) end++
37         if (end < text.length && !isLineBreak(text[end])) {
38             // Take the terminator, and every terminator that follows it: `…。` or `！？`, then the closers.
39             while (end < text.length && endsAt(text, end) && !isLineBreak(text[end])) end++
40             while (end < text.length && text[end] in CLOSERS) end++
41         }
42         return start until end
43     }
44 
45     /**
46      * The sentence as HTML, with the first occurrence of [word] in `<b>`. When the word is
47      * not in the sentence, the sentence is only escaped.
48      */
49     fun emphasize(sentence: String, word: String?): String {
50         if (word.isNullOrEmpty()) return escape(sentence)
51         val at = sentence.indexOf(word)
52         if (at < 0) return escape(sentence)
53         return escape(sentence.substring(0, at)) + "<b>" + escape(word) + "</b>" +
54             escape(sentence.substring(at + word.length))
55     }
56 
57     /** Text made safe for HTML. */
58     fun escape(text: String): String = buildString(text.length) {
59         for (c in text) {
60             when (c) {
61                 '&' -> append("&amp;")
62                 '<' -> append("&lt;")
63                 '>' -> append("&gt;")
64                 '"' -> append("&quot;")
65                 else -> append(c)
66             }
67         }
68     }
