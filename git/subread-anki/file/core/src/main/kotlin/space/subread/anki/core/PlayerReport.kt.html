3 /**
4  * What a player said about its position: the position, the wall-clock moment of that
5  * report, and the speed. The same three numbers as the media session of Android.
6  *
7  * The report maps media time to wall-clock time in both directions, also when the player
8  * is paused: the pause is the anchor, and the media before it ran at [speed].
9  */
10 data class PlayerReport(
11     /** The position in the media at [atNanos], in milliseconds. */
12     val positionMs: Long,
13     /** The wall clock of the report, `System.nanoTime()`. */
14     val atNanos: Long,
15     /** The playback speed. 1.0 is normal. Must be above zero. */
16     val speed: Double = 1.0,
17     /** True while the player plays. */
18     val playing: Boolean = true,
19 ) {
20     init {
21         require(speed > 0.0) { "speed must be above zero, was $speed" }
22     }
23 
24     /** The wall-clock moment when the media was, or will be, at [mediaMs]. */
25     fun wallAt(mediaMs: Long): Long =
26         atNanos + Math.round((mediaMs - positionMs) * NANOS_PER_MILLI / speed)
27 
28     /** The position of the media at the wall-clock moment [nanos]. */
29     fun mediaAt(nanos: Long): Long =
30         positionMs + Math.round((nanos - atNanos) * speed / NANOS_PER_MILLI)
31 
32     /**
33      * The wall-clock range in which the media from [startMs] to [endMs] played, with
34      * [padMs] of media added on each side.
35      */
36     fun wallRange(startMs: Long, endMs: Long, padMs: Long = 0): LongRange {
37         require(endMs >= startMs) { "end $endMs is before start $startMs" }
38         return wallAt(startMs - padMs)..wallAt(endMs + padMs)
39     }
40 
41     companion object {
42         const val NANOS_PER_MILLI = 1_000_000.0
43 
44         /**
45          * Reads the `state` row of the SubRead Overlay player provider:
46          * `playing=1;position=96153;speed=1.0;package=de.ph1b.audiobook`. Null for an error
47          * row (`error=no_player`) or a row without a position.
48          */
49         fun parseOverlayState(state: String?, atNanos: Long): PlayerReport? {
50             if (state.isNullOrBlank()) return null
51             val pairs = state.split(';').mapNotNull { pair ->
52                 val eq = pair.indexOf('=')
53                 if (eq < 0) null else pair.substring(0, eq).trim() to pair.substring(eq + 1).trim()
54             }.toMap()
55             if (pairs.containsKey("error")) return null
56             val position = pairs["position"]?.toLongOrNull() ?: return null
57             val speed = pairs["speed"]?.toDoubleOrNull()?.takeIf { it > 0.0 } ?: 1.0
58             val playing = pairs["playing"] == "1" || pairs["playing"] == "true"
59             return PlayerReport(position, atNanos, speed, playing)
60         }
61 
62         /** The package of the player in an overlay `state` row, or null. */
63         fun overlayPackage(state: String?): String? =
64             state?.split(';')?.firstOrNull { it.startsWith("package=") }?.substringAfter('=')?.takeIf { it.isNotBlank() }
65     }
