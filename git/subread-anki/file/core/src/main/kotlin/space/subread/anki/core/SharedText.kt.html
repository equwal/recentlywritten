3 /** The text that an app shares. A browser adds the link of the page to the selection. */
4 object SharedText {
5 
6     private val LINK = Regex("""https?://\S+""")
7 
8     /** Quotes that a browser puts around the selection that it shares. */
9     private val QUOTES = listOf('"' to '"', '“' to '”')
10 
11     /**
12      * The selection in a shared text. Chrome and Brave share `"selection"` and the link of the
13      * page: the link goes, and then the quotes around the rest. A text without a link stays as
14      * it is, quotes and all: the user selected them.
15      */
16     fun clean(text: String): String {
17         if (!LINK.containsMatchIn(text)) return text.trim()
18         val rest = LINK.replace(text, "").trim()
19         for ((open, close) in QUOTES) {
20             if (rest.length >= 2 && rest.first() == open && rest.last() == close) return rest.substring(1, rest.length - 1).trim()
21         }
22         return rest
23     }
