3 /** Names for the files that go into the media folder of Anki. */
4 object MediaNames {
5     /** The longest part of a name taken from the expression. */
6     const val MAX_STEM = 40
7 
8     /**
9      * `subread_<word>_<stamp>`: the letters and digits of [expression] in any script; the
10      * rest becomes `_`. An empty word becomes `card`. The stamp makes the name unique. The
11      * files of one card share the stem and differ in the extension or a suffix.
12      */
13     fun stem(expression: String, stampMillis: Long): String {
14         var word = buildString {
15             for (c in expression) append(if (c.isLetterOrDigit()) c else '_')
16         }.trim('_').replace(Regex("_+"), "_").take(MAX_STEM).trim('_')
17         if (word.isEmpty()) word = "card"
18         return "subread_${word}_$stampMillis"
19     }
20 
21     /** [stem] with an extension. */
22     fun name(expression: String, stampMillis: Long, extension: String): String {
23         require(extension.isNotEmpty() && extension.all { it.isLetterOrDigit() }) { "bad extension $extension" }
24         return "${stem(expression, stampMillis)}.$extension"
25     }
