17 /** Chooses the text of the pop-up, and the sentence of a card in it. */
18 object Scans {
19 
20     /**
21      * The text of the pop-up for a request:
22      * - a sentence from the sender, at the word when the word is in it;
23      * - else the subtitle line of the overlay that holds the selection, at the selection;
24      * - else the selection itself, at its start.
25      *
26      * Null when there is neither a selection nor a sentence.
27      */
28     fun resolve(selection: String?, sentence: String?, line: String?): Scan? {
29         val word = selection?.trim().orEmpty()
30         val given = sentence?.trim().orEmpty()
31         if (given.isNotEmpty()) {
32             return if (word.isEmpty()) Scan(given, 0, null, false) else Scan(given, given.indexOf(word), word, false)
33         }
34         if (word.isEmpty()) return null
35         val subtitle = line?.trim().orEmpty()
36         val at = if (subtitle.isEmpty()) -1 else subtitle.indexOf(word)
37         if (at >= 0) return Scan(subtitle, at, word, true)
38         return Scan(word, 0, word, false)
39     }

41     /**
42      * Where the first scan of the pop-up looks up: the first letter or digit of [text] at or
43      * after [from]. A quote or a bracket is no word. [from] when no letter follows.
44      */
45     fun firstLetter(text: String, from: Int): Int {
46         for (at in from until text.length) if (text[at].isLetterOrDigit()) return at
47         return from
48     }

50     /**
51      * Where [expression] is in [text], also in another form. A sender gives the dictionary form
52      * (`食べる`) and a sentence with the form of the text (`食べた`). [termsAt] gives the terms that
53      * start at a position of the text, with the characters that each covers: the dictionary finds
54      * `食べる` at `食べた`. Null when the text holds the expression in no form.
55      */
56     fun locate(text: String, expression: String, termsAt: (Int) -> List<Pair<String, Int>>): IntRange? {
57         if (expression.isEmpty()) return null
58         val direct = text.indexOf(expression)
59         if (direct >= 0) return direct until direct + expression.length
60         for (at in text.indices) {
61             if (text[at].isWhitespace()) continue
62             val length = termsAt(at).firstOrNull { it.first == expression }?.second ?: continue
63             if (length > 0) return at until minOf(text.length, at + length)
64         }
65         return null
66     }

68     /**
69      * The sentence of the card for a word that starts at [start] and is [length] characters
70      * long: the sentence of [text] around it. Empty when the text is the word alone: a word is
71      * no sentence.
72      */
73     fun sentence(text: String, start: Int, length: Int): String {
74         if (text.isBlank()) return ""
75         val at = start.coerceIn(0, text.length - 1)
76         val found = text.substring(Sentences.rangeAround(text, at)).trim()
77         return if (found.length <= length) "" else found
78     }
