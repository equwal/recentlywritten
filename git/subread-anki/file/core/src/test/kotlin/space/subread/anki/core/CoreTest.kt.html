131     /** The sentence of [text] that holds the first [word], or null when the word is not in the text. */
132     private fun around(text: String, word: String): String? {
133         val at = text.indexOf(word)
134         return if (word.isEmpty() || at < 0) null else text.substring(Sentences.rangeAround(text, at)).trim()
135     }
136 
137     /** The text of HTML that [Sentences.emphasize] and [Sentences.escape] wrote. */
138     private fun plain(html: String): String = html.replace(Regex("<[^>]*>"), "")
139         .replace("&lt;", "<").replace("&gt;", ">").replace("&quot;", "\"").replace("&amp;", "&")

141     private val sentenceBody: Arb<String> = text(12, listOf("a", "b", "c", "d", "猫", "ね", "こ", " ", "「", "」"), least = 1)
142     private val terminator: Arb<String> = Arb.element(listOf("。", "！", "？", "!", "?", "…", "\n", "。」", "！？"))

144     @Test
145     fun theSentenceOfAWordIsTheSentenceItWasBuiltFrom() = property {
146         checkAll(Arb.list(sentenceBody, 1..5), Arb.list(terminator, 5..5), Arb.int(0..4)) { bodies, ends, pick ->
147             val sentences = bodies.mapIndexed { i, body -> body + ends[i] }
148             val text = sentences.joinToString("")
149             val i = pick % sentences.size
150             val body = bodies[i]
151             // A body that starts with a closer belongs to the sentence before it, by design.
152             if (bodies.any { it.startsWith("」") }) return@checkAll
153             val word = body.trim().takeIf { it.isNotEmpty() && '」' !in it && '「' !in it } ?: return@checkAll
154             // The first occurrence of the word counts: skip a word that is in an earlier sentence.
155             if (sentences.take(i).any { it.contains(word) }) return@checkAll
156             val found = around(text, word)
157             assertEquals(sentences[i].trim(), found)
158         }
159     }

252     @Test
253     fun theLastFramesComeBackInOrder() = property {
254         checkAll(Arb.int(1..50), Arb.list(Arb.int(0..40), 1..30), Arb.int(1..2)) { capacity, chunks, channels ->
255             val rate = 100
256             val ring = PcmRing(rate, channels, capacity)
257             val all = ArrayList<Short>()
258             var frames = 0L
259             for (chunk in chunks) {
260                 val samples = ShortArray(chunk * channels) { (all.size + it).toShort() }
261                 samples.forEach { all.add(it) }
262                 frames += chunk
263                 ring.write(samples, chunk, ring.framesToNanos(frames))
264             }
265             val got = ring.read(Long.MIN_VALUE, Long.MAX_VALUE)
266             val held = minOf(frames, capacity.toLong()).toInt()
267             assertEquals(held * channels, got.size)
268             assertArrayEquals(all.takeLast(held * channels).toShortArray(), got)
269             assertEquals(ring.endNanos - ring.framesToNanos(held.toLong()), ring.startNanos)
270         }
271     }

273     @Test
274     fun aRangeIsTheSliceOfTheWhole() = property {
275         checkAll(Arb.int(10..200), Arb.int(0..300), Arb.int(0..300)) { capacity, a, b ->
276             val rate = 1000
277             val ring = PcmRing(rate, 1, capacity)
278             val total = capacity + 37
279             // Written in one piece: frame i ends at (i + 1) ms.
280             ring.write(ShortArray(total) { it.toShort() }, total, ring.framesToNanos(total.toLong()))
281             val from = minOf(a, b).toLong() * 1_000_000L
282             val to = maxOf(a, b).toLong() * 1_000_000L

286             val lo = (maxOf(from, ring.startNanos) / 1_000_000L).toInt() - start
287             val hi = (minOf(to, ring.endNanos) / 1_000_000L).toInt() - start
288             val expected = if (hi > lo) whole.copyOfRange(lo, hi) else ShortArray(0)
289             assertArrayEquals(expected, got)
290         }
291     }
