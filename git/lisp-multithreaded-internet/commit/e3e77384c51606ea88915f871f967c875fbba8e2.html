+(defun multiuri (urilist)
+  (let ((lock (gensym))
+	(list (gensym))
+	(uri (gensym)))
+    (setf lock (make-lock))		;a gensym
+    (setf list nil)
+    (let ((thread-list (collect (var urilist)
+			 (setf uri var)
+			 (make-thread (lambda ()
+					(let ((result (getpage uri)))
+					  (with-lock-held (lock)
+					    (setf list (cons result list)))))
+				      :name uri))))
+      (dolist (thread thread-list list)
+	(join-thread thread)))))

+		 internal-time-units-per-second)))))
