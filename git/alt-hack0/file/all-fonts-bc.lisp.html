16 (defun verify (spec)
17   "Ensure that the spec is not bogus."
18   (let ((group (not (set-difference spec *groups*)))
19         (exclusion (not (some #'identity
20                               (mapcar #'(lambda (ex)
21                                           (<= 2 (length (intersection spec ex))))
22                                       *exclusions*)))))
23     (values (and group exclusion)
24             (if group (if exclusion 'group-and-exclusion 'group)
25                 (if exclusion 'exclusion 'unknown)))))
26 (defun gen-all-fonts ()
27   (loop for x from 1 to (length *groups*)
28         do (map-combinations
29              (lambda (f)
30                (when (verify f)
31                  (run-program (format nil "./generator-bc.ros --ttf --woff --woff2 ~(~{ ~A~}~)" f) :output t)
