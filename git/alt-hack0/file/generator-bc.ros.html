70 (defun prune (test tree)
71   (labels ((rec (tree acc)
72              (cond
73                ((null tree) (nreverse acc))
74                ((consp (car tree))
75                 (rec (cdr tree)
76                      (cons (rec (car tree) nil) acc)))
77                (t (rec (cdr tree)
78                        (if (funcall test (car tree))
79                            acc
80                            (cons (car tree) acc)))))))
81     (rec tree nil)))

165 (defun verify (spec)
166   "Ensure that the spec is not bogus."
167   (let ((group (not (set-difference spec (mapcar #'car *groups*))))
168         (exclusion (not (some #'identity
169                     (mapcar #'(lambda (ex)
170                                 (<= 2 (length (intersection spec ex))))
171                             *exclusions*)))))
172     (values (and group exclusion)
173             (if group (if exclusion 'group-and-exclusion 'group)
174                 (if exclusion 'exclusion 'unknown)))))

193 (defun format-parser (argv)
194   (when argv
195     (let ((c (car argv)))
196       (when (and (< 2 (length c))
197                  (string= "--" (subseq c 0 2)))
198        (cons (subseq c 2) (format-parser (cdr argv)))))))
199 (defun should-help (argv)
200   (or (null argv)
201       (dolist (a argv)
202         (when (or (search "-?" a)
203                   (search "-h" a)
204                   (search "help" a))
205           (return t)))))

219 (defun main (&rest argv)
220   (declare (ignorable argv))
221   (if (should-help argv)
222       help
223       (if (should-test argv)
224           test
225           (let ((formats (format-parser argv)))
226             (let ((argv (mapcar (lambda (g) (intern (string-upcase g)))
227                                 (nthcdr (length formats) argv)))
228                   (formats (or formats "ttf woff woff2")))
229               (if (and argv (string= "SHOW" (string-upcase (car argv))))
230                   (show-characters)
231                   (multiple-value-bind (bool name)
232                       (verify argv)
233                     (if bool
234                         (progn (remove-hints)
235                                (mapc #'put-glyphs!
236                                      (mapcar #'folders argv))
237                                ;; Can't get woff2 on my system.
238                                (mapc #'build formats))
239                         (progn (format t "Invalid ~S in spec: ~{~A ~}~%"
240                                        name
241                                        argv))))))))))
