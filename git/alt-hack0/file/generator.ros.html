104 (defun remove-hints ()
105   (let ((files (directory-files #P"tt-hinting")))
106     (dolist (f files)
107       (with-open-file (si f :direction :input)
108         (with-open-file (so f :direction :output
109                               :if-exists :supersede)
110           (let ((line (read-line si nil nil)))
111             (when (and line (char/= #\#(elt line 0)))
112               (write-line (format nil "# ~S~%" line) so))))))))
